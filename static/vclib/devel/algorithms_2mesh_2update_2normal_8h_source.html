19template<ElemId ELEM_ID, LoggerConcept LogType = NullLogger>
20void normalizeNoThrow(auto elem, LogType& log = nullLogger)
21{
22 using PointType = std::remove_cvref_t<decltype(elem.normal())>;
23 using Scalar = typename PointType::Scalar;
24 const Scalar sqrNorm = elem.normal().squaredNorm();
25 if (sqrNorm > Scalar(0)) [[likely]] {
26 elem.normal() /= std::sqrt(sqrNorm);
27 }
28 else [[unlikely]] {
29 log.log(
30 log.WARNING_LOG,
31 "{} {} : normal is zero",
32 elementEnumString<ELEM_ID>(),
33 elem.index());
34 }
35}

53template<ElemId ELEM_ID, LoggerConcept LogType = NullLogger>
54void clearElementNormals(MeshConcept auto& mesh, LogType& log = nullLogger)
55{
56 requireElementComponent<ELEM_ID, CompId::NORMAL>(mesh);
57
58 log.log(0, "Clearing per-{} normals...", elementEnumString<ELEM_ID>());
59
60 parallelFor(mesh.template elements<ELEM_ID>().normals(), [](auto& n) {
61 n.setZero();
62 });
63
64 log.log(100, "Per-{} normals cleared.", elementEnumString<ELEM_ID>());
65}

80template<ElemId ELEM_ID, LoggerConcept LogType = NullLogger>
81void normalizeElementNormals(MeshConcept auto& mesh, LogType& log = nullLogger)
82{
83 requireElementComponent<ELEM_ID, CompId::NORMAL>(mesh);
84
85 log.log(0, "Normalizing per-{} normals...", elementEnumString<ELEM_ID>());
86
87 // define a lambda that, for each element, normalizes the normal
88 auto normalize = [&](auto elem) {
89 detail::normalizeNoThrow<ELEM_ID>(elem, log);
90 };
91
92 parallelFor(mesh.template elements<ELEM_ID>(), normalize);
93
94 log.log(100, "Per-{} normals normalized.", elementEnumString<ELEM_ID>());
95}

198template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
199void clearReferencedVertexNormals(MeshType& mesh, LogType& log = nullLogger)
200{
201 requireVertexNormal(mesh);
202
203 // function that is called for each container Cont of the mesh
204 auto f = [&]<mesh::ElementContainerConcept Cont>() {
205 using Elem = Cont::ElementType; // the Element type of Cont
206 if constexpr (comp::HasVertexReferences<Elem>) { // if Elem has vertices
207 for (auto e : mesh.template elements<Elem::ELEMENT_ID>()) {
208 for (auto vi : e.vertexIndices()) {
209 // normal().setZero() does not compile on gcc (Release) with
210 // Eigen 5.0.1
211 mesh.vertex(vi).normal() = {0., 0., 0.};
212 }
213 }
214 }
215 };
216
217 log.log(0, "Clearing per-Vertex normals...");
218
219 // apply the function f to all the containers of the mesh
220 ForEachType<typename MeshType::ElementContainers>::apply(f);
221
222 log.log(100, "Per-Vertex normals cleared.");
223}

351template<LoggerConcept LogType = NullLogger>
352void updateFaceNormals(
353 FaceMeshConcept auto& mesh,
354 bool normalize = true,
355 LogType& log = nullLogger)
356{
357 requireFaceNormal(mesh);
358
359 using FaceType = FaceType<decltype(mesh)>;
360 using ScalarType = NormalType<FaceType>::ScalarType;
361
362 log.log(0, "Updating per-Face normals...");
363
364 parallelFor(mesh.faces(), [](auto f) {
365 f.normal() = faceNormal(f).template cast<ScalarType>();
366 });
367
368 if (normalize) {
369 log.startNewTask(50, 100, "Normalizing per-Face normals...");
370 normalizeFaceNormals(mesh, log);
371 log.endTask("Normalizing per-Face normals...");
372 }
373
374 log.log(100, "Per-Face normals updated.");
375}

394template<LoggerConcept LogType = NullLogger>
395void updateVertexNormals(
396 FaceMeshConcept auto& mesh,
397 bool normalize = true,
398 LogType& log = nullLogger)
399{
400 using VertexType = VertexType<decltype(mesh)>;
401 using NScalar = NormalType<VertexType>::ScalarType;
402
403 log.log(0, "Updating per-Vertex normals...");
404
405 log.startNewTask(0, 20, "Clearing per-Vertex normals...");
406 clearReferencedVertexNormals(mesh, log);
407 log.endTask("Clearing per-Vertex normals...");
408
409 log.log(20, "Updating per-Vertex normals...");
410
411 for (auto f : mesh.faces()) {
412 const auto fn = faceNormal(f).template cast<NScalar>();
413 for (auto vi : f.vertexIndices()) {
414 mesh.vertex(vi).normal() += fn;
415 }
416 }
417
418 if (normalize) {
419 log.startNewTask(80, 100, "Normalizing per-Vertex normals...");
420 normalizeReferencedVertexNormals(mesh, log);
421 log.endTask("Normalizing per-Vertex normals...");
422 }
423
424 log.log(100, "Per-Vertex normals updated.");
425}

551template<LoggerConcept LogType = NullLogger>
552void updateVertexNormalsAngleWeighted(
553 FaceMeshConcept auto& mesh,
554 bool normalize = true,
555 LogType& log = nullLogger)
556{
557 using VertexType = VertexType<decltype(mesh)>;
558 using NormalType = NormalType<VertexType>;
559 using NScalarType = NormalType::ScalarType;
560
561 log.log(0, "Updating per-Vertex normals...");
562
563 log.startNewTask(0, 5, "Clearing per-Vertex normals...");
564 clearReferencedVertexNormals(mesh, log);
565 log.endTask("Clearing per-Vertex normals...");
566
567 log.log(5, "Updating per-Vertex normals...");
568
569 for (auto f : mesh.faces()) {
570 auto n = faceNormal(f).template cast<NScalarType>();
571
572 for (uint i = 0; i < f.vertexCount(); ++i) {
573 NormalType vec1 =
574 (f.vertexMod(i - 1).position() - f.vertexMod(i).position())
575 .normalized()
576 .template cast<NScalarType>();
577 NormalType vec2 =
578 (f.vertexMod(i + 1).position() - f.vertexMod(i).position())
579 .normalized()
580 .template cast<NScalarType>();
581
582 f.vertex(i).normal() += n * vec1.angle(vec2);
583 }
584 }
585
586 if (normalize) {
587 log.startNewTask(95, 100, "Normalizing per-Vertex normals...");
588 normalizeReferencedVertexNormals(mesh, log);
589 log.endTask("Normalizing per-Vertex normals...");
590 }
591
592 log.log(100, "Per-Vertex normals updated.");
593}

625template<LoggerConcept LogType = NullLogger>
626void updateVertexNormalsNelsonMaxWeighted(
627 FaceMeshConcept auto& mesh,
628 bool normalize = true,
629 LogType& log = nullLogger)
630{
631 using VertexType = VertexType<decltype(mesh)>;
632 using NScalarType = NormalType<VertexType>::ScalarType;
633
634 log.log(0, "Updating per-Vertex normals...");
635
636 log.startNewTask(0, 5, "Clearing per-Vertex normals...");
637 clearReferencedVertexNormals(mesh, log);
638 log.endTask("Clearing per-Vertex normals...");
639
640 log.log(5, "Updating per-Vertex normals...");
641
642 for (auto f : mesh.faces()) {
643 auto n = faceNormal(f).template cast<NScalarType>();
644
645 for (uint i = 0; i < f.vertexCount(); ++i) {
646 NScalarType e1 =
647 (f.vertexMod(i - 1).position() - f.vertexMod(i).position())
648 .squaredNorm();
649 NScalarType e2 =
650 (f.vertexMod(i + 1).position() - f.vertexMod(i).position())
651 .squaredNorm();
652
653 f.vertex(i).normal() += n / (e1 * e2);
654 }
655 }
656
657 if (normalize) {
658 log.startNewTask(95, 100, "Normalizing per-Vertex normals...");
659 normalizeReferencedVertexNormals(mesh, log);
660 log.endTask("Normalizing per-Vertex normals...");
661 }
662
663 log.log(100, "Per-Vertex normals updated.");
664}
