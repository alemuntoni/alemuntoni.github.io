101 decltype(auto) parentMesh(this auto&& self)
102 requires (!std::is_void_v<ParentMeshType>)
103 {
104 assert(self.isValid() && "Element proxy is invalid");
105 using SelfType = std::remove_reference_t<decltype(self)>;
106 if constexpr (std::is_const_v<SelfType> || std::is_const_v<ContainerType>) {
107 return std::as_const(*self.mContainer).parentMesh();
108 }
109 else {
110 return self.mContainer->parentMesh();
111 }
112 }

117 IndexType index() const { return mIndex; }

125 bool isValid() const
126 {
127 return mContainer && mIndex != NULL_INDEX;
128 }
129
133 explicit operator bool() const { return isValid(); }

140 template<typename OtherContainer>
141 requires std::same_as<
142 std::remove_const_t<ContainerType>,
143 std::remove_const_t<OtherContainer>>
144 bool operator==(
145 const ElementProxy<ELEM_ID, OtherContainer, vcl::TypeWrapper<Comps...>>&
146 other) const
147 {
148 if (!isValid() && !other.isValid()) {
149 return true;
150 }
151 return mContainer == other.mContainer && mIndex == other.mIndex;
152 }
153
158 template<typename OtherContainer>
159 requires std::same_as<
160 std::remove_const_t<ContainerType>,
161 std::remove_const_t<OtherContainer>>
162 auto operator<=>(
163 const ElementProxy<ELEM_ID, OtherContainer, vcl::TypeWrapper<Comps...>>&
164 other) const
165 {
166 if (!isValid() && !other.isValid()) {
167 return std::strong_ordering::equal;
168 }
169 if (!isValid()) {
170 return std::strong_ordering::less;
171 }
172 if (!other.isValid()) {
173 return std::strong_ordering::greater;
174 }
175 if (mContainer != other.mContainer) {
176 return mContainer <=> other.mContainer;
177 }
178 return mIndex <=> other.mIndex;
179 }
180
185 bool operator==(std::nullptr_t) const { return !isValid(); }
186
193 void swap(ElementProxy& other) noexcept
194 {
195 std::swap(mContainer, other.mContainer);
196 std::swap(mIndex, other.mIndex);
197 }

254 template<typename OtherElement>
255 void importFrom(const OtherElement& other, bool importRefs = true)
256 requires (!std::is_const_v<ContainerType>)
257 {
258 static_assert(
259 ELEMENT_ID == OtherElement::ELEMENT_ID,
260 "Cannot import from an element of a different type.");
261
262 auto importSingleComponent = [&]<typename Comp>() {
263 if constexpr (requires { this->Comp::importFrom(other, importRefs); }) {
264 if constexpr (requires { Comp::COMPONENT_ID; }) {
265 if (this->template isComponentAvailable<Comp::COMPONENT_ID>()) {
266 this->Comp::importFrom(other, importRefs);
267 }
268 }
269 else {
270 this->Comp::importFrom(other, importRefs);
271 }
272 }
273 };
274
275 (importSingleComponent.template operator()<Comps>(), ...);
276 }

372template<typename T>
373concept ElementConcept = requires (T&& obj) {
374 { std::remove_cvref_t<T>::ELEMENT_ID } -> std::convertible_to<ElemId>;
375 typename std::remove_cvref_t<T>::Components;
376 typename std::remove_cvref_t<T>::IndexType;
377 { obj.index() } -> std::same_as<typename std::remove_cvref_t<T>::IndexType>;
378 { obj.isValid() } -> std::same_as<bool>;
379};

385template<vcl::ElementConcept Element>
386struct hash<Element>
387{
388 std::size_t operator()(const Element& ep) const
389 {
390 if (!ep.isValid()) {
391 return 0;
392 }
393 std::size_t seed = 0;
394 vcl::hashCombine(seed, ep.index());
395 return seed;
396 }
397};
