188 uint addElement()
189 {
190 mVerticalCompVecTuple.resize(mElemVec.size() + 1);
191 if constexpr (comp::HasCustomComponents<T>)
192 mCustomCompVecMap.resize(mElemVec.size() + 1);
193
194 T* oldB = mElemVec.data();
195 mElemVec.emplace_back();
196 T* newB = mElemVec.data();
197 mElemCount++;
198
199 mElemVec.back().setParentMesh(mParentMesh);
200 mElemVec.back().initVerticalComponents();
201
202 if (oldB != newB) {
203 setParentMeshPointers(mParentMesh);
204 mParentMesh->updateAllReferences(oldB);
205 }
206
207 return mElemVec.size() - 1;
208 }

218 uint addElements(uint size)
219 {
220 mVerticalCompVecTuple.resize(mElemVec.size() + size);
221 if constexpr (comp::HasCustomComponents<T>)
222 mCustomCompVecMap.resize(mElemVec.size() + size);
223
224 uint baseId = mElemVec.size();
225 T* oldB = mElemVec.data();
226 mElemVec.resize(mElemVec.size() + size);
227 T* newB = mElemVec.data();
228 mElemCount += size;
229
230 for (uint i = baseId; i < mElemVec.size(); ++i) {
231 mElemVec[i].setParentMesh(mParentMesh);
232 mElemVec[i].initVerticalComponents();
233 }
234
235 if (oldB != newB) {
236 setParentMeshPointers(mParentMesh);
237 mParentMesh->updateAllReferences(oldB);
238 }
239
240 return baseId;
241 }

283 void resizeElements(uint size)
284 {
285 if (size > mElemCount) {
286 addElements(size - mElemCount);
287 }
288 else if (size < mElemCount) {
289 uint nToDelete = mElemCount - size;
290 for (uint i = mElemVec.size() - 1; nToDelete > 0; --i) {
291 if (!mElemVec[i].deleted()) {
292 deleteElement(i);
293 }
294 }
295 }
296 }

306 mCustomCompVecMap.reserve(size);
307
308 if (oldB != newB) {
309 setParentMeshPointers(mParentMesh);
310 mParentMesh->updateAllReferences(oldB);
311 }

445 void append(const ElementContainer<T>& other)
446 {
447 using Comps = T::Components;
448
449 uint on = other.elementContainerSize();
450 uint n = elementContainerSize();
451 addElements(on);
452 for (uint i = 0; i < on; ++i) {
453 // copy everything from the other elements, also the (not updated)
454 // pointers:
455 if (other.element(i).deleted()) {
456 deleteElement(n + i);
457 }
458 element(n + i).rawCopyFrom(other.element(i));
459 element(n + i).setParentMesh(mParentMesh);
460 }
461 // importing also optional, vertical and custom components:
462 appendVerticalComponents(other, VertComps());
463 appendCustomComponents(other);
464 }

793 template<typename C>
794 void enableOptionalComponent()
795 {
796 if (!isOptionalComponentEnabled<C>()) {
797 mVerticalCompVecTuple.template enableComponent<C>();
798 // first call init on all the just enabled components
799 if constexpr (comp::HasInitMemberFunction<C>) {
800 for (auto& e : elements()) {
801 e.C::init();
802 }
803 }
804 // then resize the component containers with tied size to number of
805 // vertices of the element, if needed
806 if constexpr (comp::IsTiedToVertexCount<C>) {
807 static const int N = T::VERTEX_COUNT;
808 if constexpr (N < 0) {
809 for (auto& e : elements()) {
810 e.C::resize(e.vertexCount());
811 }
812 }
813 }
814 }

1003 template<typename OtherMesh>
1004 void importFrom(const OtherMesh& m)
1005 {
1006 if constexpr (OtherMesh::template hasContainerOf<T>()) {
1007 // get the container type of the other mesh for T - used to upcast
1008 // othMesh
1009 using Container = OtherMesh::template ContainerOf<T>::type;
1010
1011 const Container& c = (const Container&) m;
1012
1013 clearElements();
1014
1015 // recreate a container having the same number of elements as the
1016 // other mesh. this call will also update the parentMesh pointers
1017 // and sets the vertical components vectors in vcVecTuple
1018 addElements(c.elementContainerSize());
1019 unsigned int eid = 0;
1020 for (const typename Container::ElementType& e : c.elements(false)) {
1021 // import the components of the e, when they are available.
1022 // note that will set also the deleted flag of e, therefore if e
1023 // is deleted, then also element(eid) will be deleted.
1024 element(eid).importFrom(e);
1025 ++eid;
1026 }
1027 // set the number of elements (different from the container size)
1028 mElemCount = c.mElemCount;
1029 if constexpr (
1030 comp::HasCustomComponents<T> &&
1031 comp::HasCustomComponents<typename Container::ElementType>) {
1032 mCustomCompVecMap = c.mCustomCompVecMap;
1033 }
1034 }

1045 template<typename ElPtr, typename... Comps>
1046 void updateReferencesOnComponents(
1047 const ElPtr* oldBase,
1048 TypeWrapper<Comps...>,
1049 uint firstElementToProcess = 0,
1050 uint offset = 0)
1051 {
1052 (updateReferencesOnComponent<Comps>(
1053 oldBase, firstElementToProcess, offset),
1054 ...);

1057 template<typename ElPtr, typename... Comps>
1058 void updateReferencesOnComponents(
1059 const std::vector<uint>& newIndices,
1060 TypeWrapper<Comps...>)
1061 {
1062 (updateReferencesOnComponent<Comps, ElPtr>(newIndices), ...);
1063 }
1064
1065 /*
1066 * This function is called for each component of the element.
1067 *
1068 * Only if a component has references of the type ElPtr, then the
1069 * updateReferences on each element will be executed.
1070 *
1071 * firstElementToProcess and offset are used only when an append operation
1072 * has been executed. In this case, the firstElementToProcess is the index
1073 * of the first element that has been appended, and offset is the number of
1074 * ElPtr elements that have been appended (the offset that must be added
1075 * to the newBase w.r.t. the oldBase that was the other container from which
1076 * the elements have been copied).
1077 */
1078 template<typename Comp, typename ElPtr>
1079 void updateReferencesOnComponent(
1080 const ElPtr* oldBase,
1081 uint firstElementToProcess = 0,
1082 uint offset = 0)
1083 {
1084 if constexpr (comp::HasReferencesOfType<Comp, ElPtr>) {
1085 // lambda to avoid code duplication
1086 auto loop = [&]() {
1087 for (uint i = firstElementToProcess; i < elementContainerSize();
1088 i++) {
1089 T& e = element(i);
1090 if (!e.deleted()) {
1091 e.Comp::updateReferences(oldBase, offset);
1092 }
1093 }
1094 };
1095
1096 if constexpr (comp::HasOptionalReferencesOfType<Comp, ElPtr>) {
1097 if (isOptionalComponentEnabled<Comp>()) {
1098 loop();
1099 }
1100 }
1101 else {
1102 loop();
1103 }
1104 }
1105 }
1106
1107 template<typename Comp, typename ElPtr>
1108 void updateReferencesOnComponent(const std::vector<uint>& newIndices)
1109 {
1110 if constexpr (comp::HasReferencesOfType<Comp, ElPtr>) {
1111 if constexpr (comp::HasOptionalReferencesOfType<Comp, ElPtr>) {
1112 if (isOptionalComponentEnabled<Comp>()) {
1113 for (T& e : elements()) {
1114 e.Comp::updateReferences(newIndices);
1115 }
1116 }
1117 }
1118 else {
1119 for (T& e : elements()) {
1120 e.Comp::updateReferences(newIndices);
1121 }
1122 }
1123 }
1124 }

1134 template<typename Comp>
1135 void appendVerticalComponent(const ElementContainer<T>& other)
1136 {
1137 uint on = other.elementContainerSize();
1138 uint n = elementContainerSize() - on;
1139
1140 if (mVerticalCompVecTuple.template isComponentEnabled<Comp>() &&
1141 other.mVerticalCompVecTuple.template isComponentEnabled<Comp>()) {
1142 auto& vc = mVerticalCompVecTuple.template vector<Comp>();
1143 const auto& ovc =
1144 other.mVerticalCompVecTuple.template vector<Comp>();
1145
1146 for (uint i = 0; i < on; ++i) {
1147 vc[n + i] = ovc[i];
1148 }
1149 }
1150 }

1152 void appendCustomComponents(const ElementContainer<T>& other)
1153 {
1154 if constexpr (comp::HasCustomComponents<T>) {
1155 uint on = other.elementContainerSize();
1156 uint n = elementContainerSize() - on;
1157
1158 std::vector<std::string> ccNames =
1159 mCustomCompVecMap.allComponentNames();
1160
1161 for (const std::string& name : ccNames) {
1162 for (uint i = 0; i < on; ++i) {
1163 mCustomCompVecMap.importSameCustomComponentFrom(
1164 n + i, i, name, other.mCustomCompVecMap);
1165 }
1166 }
