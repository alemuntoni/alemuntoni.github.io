89template<MeshConcept MeshType>
90auto covarianceMatrixOfPointCloud(const MeshType& m)
91{
92 using PositionType = PositionType<MeshType>;
93 using ScalarType = PositionType::ScalarType;
94
95 PositionType bar = barycenter(m);
96
97 Matrix33<ScalarType> mm;
98 mm.setZero();
99 // compute covariance matrix
100 for (auto&& v : m.vertices()) {
101 PositionType e = v.position() - bar;
102 mm += e.outerProduct(e);
103 }
104 return mm;
105}

115template<FaceMeshConcept MeshType>
116auto covarianceMatrixOfMesh(const MeshType& m)
117{
118 using PositionType = PositionType<MeshType>;
119 using ScalarType = PositionType::ScalarType;
120
121 PositionType bar = shellBarycenter(m);
122 Matrix33<ScalarType> C;
123 C.setZero();
124 Matrix33<ScalarType> C0;
125 C0.setZero();
126 C0(0, 0) = C0(1, 1) = 2.0;
127 C0(0, 1) = C0(1, 0) = 1.0;
128 C0 *= 1 / 24.0;
129 // integral of (x,y,0) in the same triangle
130 Eigen::Vector3<ScalarType> x;
131 x << 1 / 6.0, 1 / 6.0, 0;
132 Matrix33<ScalarType> A; // matrix that bring the vertices to (v1-v0,v2-v0,n)
133 Matrix33<ScalarType> DC;
134
135 for (auto&& f : m.faces()) {
136 const PositionType& p0 = f.vertex(0).position();
137 const PositionType& p1 = f.vertex(1).position();
138 const PositionType& p2 = f.vertex(2).position();
139 PositionType n = (p1 - p0).cross(p2 - p0);
140 double da = n.norm();
141 n /= da * da;
142
143 PositionType tmpp = p1 - p0;
144 for (uint j = 0; j < 3; j++)
145 A(j, 0) = tmpp(j);
146 tmpp = p2 - p0;
147 for (uint j = 0; j < 3; j++)
148 A(j, 1) = tmpp(j);
149 for (uint j = 0; j < 3; j++)
150 A(j, 2) = n(j);
151 Eigen::Vector3<ScalarType> delta;
152 tmpp = p0 - bar;
153 for (uint j = 0; j < 3; j++)
154 delta(j) = tmpp(j);
155
156 /* DC is calculated as integral of (A*x+delta) * (A*x+delta)^T over the
157 * triangle, where delta = v0-bary */
158 DC.setZero();
159 DC += A * C0 * A.transpose();
160 Matrix33<ScalarType> tmp = (A * x) * delta.transpose();
161 DC += tmp + tmp.transpose();
162 DC += tmp;
163 tmp = delta * delta.transpose();
164 DC += tmp * 0.5;
165 DC *= da; // the determinant of A is also the double area of f
166 C += DC;
167 }
168 return C;
169}

186template<MeshConcept MeshType>
187auto vertexRadiusFromWeights(
188 const MeshType& m,
189 Range auto&& weights,
190 double diskRadius,
191 double radiusVariance,
192 bool invert = false)
193{
194 using ScalarType = ScalarType<MeshType>;
195
196 assert(std::ranges::size(weights) == m.vertexContainerSize());
197
198 std::vector<ScalarType> radius(m.vertexContainerSize());
199 if (m.vertexCount() == 0) {
200 return radius;
201 }
202
203 auto it = std::ranges::begin(weights);
204 auto activeWeights =
205 m.vertices() | std::views::transform([&](const auto& v) {
206 return it[v.index()];
207 });
208 const auto [min, max] = std::ranges::minmax(activeWeights);
209
210 double minRad = diskRadius;
211 double maxRad = diskRadius * radiusVariance;
212 double deltaQ = max - min;
213 double deltaRad = maxRad - minRad;
214
215 for (const auto& v : m.vertices()) {
216 if (deltaQ == 0.0) {
217 radius[v.index()] = minRad;
218 }
219 else {
220 auto w = it[v.index()];
221 double num = invert ? (max - w) : (w - min);
222 radius[v.index()] = minRad + deltaRad * (num / deltaQ);
223 }
224 }
225
226 return radius;
227}

244template<FaceMeshConcept MeshType>
245std::vector<std::pair<IndexType<MeshType>, uint>> creaseFaceEdges(
246 const MeshType& m,
247 double angleRadNeg,
248 double angleRadPos,
249 bool alsoBorderEdges = false)
250{
251 requireFaceAdjacentFaces(m);
252
253 using IndexType = IndexType<MeshType>;
254
255 std::vector<std::pair<IndexType, uint>> creaseEdges;
256
257 for (auto&& f : m.faces()) {
258 for (uint i = 0; i < f.vertexCount(); ++i) {
259 if (!f.adjFace(i)) {
260 // border edge
261 if (alsoBorderEdges) {
262 creaseEdges.push_back({f.index(), i});
263 }
264 }
265 else {
266 // internal edge
267 double angle = faceDihedralAngleOnEdge(f, i);
268 if (angle < angleRadNeg || angle > angleRadPos) {
269 creaseEdges.push_back({f.index(), i});
270 }
271 }
272 }
273 }
274 return creaseEdges;
275}
