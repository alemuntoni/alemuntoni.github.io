85 // indices of the bits
86 enum {
87 DELETED = 0, // bit 0
88 SELECTED = 1, // bit 1
89 VISITED = 2, // bit 2
90 // Faux edges: when representing polygonal meshes on triangle meshes,
91 // some triangle edges can be marked as "faux", meaning that they are

93 FAUX0 = 3, // bits [3, 5]
94 // Edge border
95 BORDER0 = 6, // bits [6, 8]
96 // Edge selection
97 EDGESEL0 = 9, // bits [9, 11]
98 EDGEVIS0 = 12, // bits [12, 14]
99 };

352 void importFlagsFromVCGFormat(int f)
353 {
354 if (f & 0x00000010)
355 visited() = true;
356 if (f & 0x00000020)
357 selected() = true;
358 if (f & 0x00000040)
359 edgeOnBorder(0) = true;
360 if (f & 0x00000080)
361 edgeOnBorder(1) = true;
362 if (f & 0x00000100)
363 edgeOnBorder(2) = true;
364 if (f & 0x00008000)
365 edgeSelected(0) = true;
366 if (f & 0x00010000)
367 edgeSelected(1) = true;
368 if (f & 0x00020000)
369 edgeSelected(2) = true;
370 if (f & 0x00040000)
371 edgeFaux(0) = true;
372 if (f & 0x00080000)
373 edgeFaux(1) = true;
374 if (f & 0x00100000)
375 edgeFaux(2) = true;
376 }

385 int exportFlagsToVCGFormat() const
386 {
387 int f = 0;
388 if (visited())
389 f |= 0x00000010;
390 if (selected())
391 f |= 0x00000020;
392 if (edgeOnBorder(0))
393 f |= 0x00000040;
394 if (edgeOnBorder(1))
395 f |= 0x00000080;
396 if (edgeOnBorder(2))
397 f |= 0x00000100;
398 if (edgeSelected(0))
399 f |= 0x00008000;
400 if (edgeSelected(1))
401 f |= 0x00010000;
402 if (edgeSelected(2))
403 f |= 0x00020000;
404 if (edgeFaux(0))
405 f |= 0x00040000;
406 if (edgeFaux(1))
407 f |= 0x00080000;
408 if (edgeFaux(2))
409 f |= 0x00100000;
410 return f;
411 }

416 // Component interface functions
417 template<typename Element>
418 void importFrom(const Element& e, bool = true)
419 {
420 if constexpr (HasBitFlags<Element>) {
421 resetBitFlags();
422 if constexpr (HasTriangleBitFlags<Element>) {
423 flags() = e.flags();
424 }
425 else {
426 // BitFlags
427 deletedBit() = e.deleted();
428 selected() = e.selected();
429 visited() = e.visited();
430 const uint UM = std::min(USER_BIT_COUNT, e.USER_BIT_COUNT);
431 for (uint i = 0; i < UM; ++i)
432 userBit(i) = e.userBit(i);
433 if constexpr (HasPolygonBitFlags<Element>) {
434 // PolygonBitFlags
435 for (uint i = 0; i < 3; ++i) {
436 edgeOnBorder(i) = e.edgeOnBorder(i);
437 edgeSelected(i) = e.edgeSelected(i);
438 edgeFaux(i) = e.edgeFaux(i);
439 }
440 }
441 }
442 }
443 }
