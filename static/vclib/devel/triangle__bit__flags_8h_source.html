59class TriangleBitFlags : public Component<CompId::BIT_FLAGS, BitSet<uint>>

63 // indices of the bits
64 enum {
65 DELETED = 0, // bit 0
66 SELECTED = 1, // bit 1
67 VISITED = 2, // bit 2
68 // Faux edges: when representing polygonal meshes on triangle meshes,
69 // some triangle edges can be marked as "faux", meaning that they are
70 // internal on the polygon
71 FAUX0 = 3, // bits [3, 5]
72 // Edge border
73 BORDER0 = 6, // bits [6, 8]
74 // Edge selection
75 EDGESEL0 = 9, // bits [9, 11]
76 EDGEVIS0 = 12, // bits [12, 14]
77 };

237 void importFlagsFromVCGFormat(this auto&& self, int f)
238 {
239 self.resetBitFlags();

242 if (f & 0x00000020)
243 self.selected() = true;
244 if (f & 0x00000040)

246 if (f & 0x00000080)

251 self.edgeSelected(0) = true;

255 self.edgeSelected(2) = true;
256 if (f & 0x00040000)
257 self.edgeFaux(0) = true;
258 if (f & 0x00080000)
259 self.edgeFaux(1) = true;

271 int exportFlagsToVCGFormat(this auto&& self)
272 {
273 int f = 0;
274 if (self.visited())
275 f |= 0x00000010;
276 if (self.selected())
277 f |= 0x00000020;
278 if (self.edgeOnBorder(0))
279 f |= 0x00000040;
280 if (self.edgeOnBorder(1))
281 f |= 0x00000080;
282 if (self.edgeOnBorder(2))
283 f |= 0x00000100;
284 if (self.edgeSelected(0))
285 f |= 0x00008000;
286 if (self.edgeSelected(1))
287 f |= 0x00010000;
288 if (self.edgeSelected(2))
289 f |= 0x00020000;
290 if (self.edgeFaux(0))
291 f |= 0x00040000;
292 if (self.edgeFaux(1))
293 f |= 0x00080000;
294 if (self.edgeFaux(2))
295 f |= 0x00100000;
296 return f;
297 }

314template<typename OtherElement>
315void TriangleBitFlags::importFrom(this auto&& self, const OtherElement& other, bool)
316{
317 if constexpr (HasBitFlags<OtherElement>) {
318 const bool wasDeleted = self.deleted();
319 const bool isDeleted = other.deleted();
320 self.resetBitFlags();
321 if constexpr (HasTriangleBitFlags<OtherElement>) {
322 self.setUnderlyingBitFlags(other.underlyingBitFlags());
323 self.setDeleted(isDeleted);
324 }
325 else {
326 self.setDeleted(isDeleted);
327 self.selected() = other.selected();
328 self.visited() = other.visited();
329
330 const uint um =
331 std::min(USER_BIT_COUNT, (uint) OtherElement::USER_BIT_COUNT);
332 for (uint i = 0; i < um; ++i) {
333 self.userBit(i) = other.userBit(i);
334 }
335 if constexpr (HasPolygonBitFlags<OtherElement>) {
336 const uint numEdges = std::min(3u, (uint) other.edgeCount());
337 for (uint i = 0; i < numEdges; ++i) {
338 self.edgeOnBorder(i) = other.edgeOnBorder(i);
339 self.edgeSelected(i) = other.edgeSelected(i);
340 self.edgeVisited(i) = other.edgeVisited(i);
341 self.edgeFaux(i) = other.edgeFaux(i);
342 }
343 }
344 }
345 if constexpr (CanAdjustContainerElemCount<decltype(self)>) {
346 if (self.isValid()) {
347 self.parentContainer().adjustElemCountOnDeletedChange(
348 wasDeleted, isDeleted);
349 }
350 }
351 }
352}
