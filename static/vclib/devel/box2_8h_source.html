65template<Box2Concept BoxType>
66auto boxVertex(const BoxType& box, uint i) -> BoxType::PointType
67{
68 using PointType = BoxType::PointType;
69
70 switch (i) {
71 case 0: return box.min();
72 case 1: return PointType(box.max().x(), box.min().y());
73 case 2: return box.max();
74 case 3: return PointType(box.min().x(), box.max().y());
75 default:
76 throw std::out_of_range("Invalid vertex index");
77 return PointType();
78 }
79}

105template<Box2Concept BoxType>
106auto boxEdge(const BoxType& box, uint i) -> Segment<typename BoxType::PointType>
107{
108 using PointType = BoxType::PointType;
109
110 switch (i) {
111 case 0: return Segment<PointType>(boxVertex(box, 0), boxVertex(box, 1));
112 case 1: return Segment<PointType>(boxVertex(box, 1), boxVertex(box, 2));
113 case 2: return Segment<PointType>(boxVertex(box, 2), boxVertex(box, 3));
114 case 3: return Segment<PointType>(boxVertex(box, 3), boxVertex(box, 0));
115 default:
116 throw std::out_of_range("Invalid edge index");
117 return Segment<PointType>();
118 }
119}
