118 void setFieldOfViewAdaptingEyeDistance(const Scalar& fovDeg)
119 {
120 mFovDeg = fovDeg;
121 PointType targetToEye = (mEye - mCenter).normalized();
122 mEye =
123 mCenter + targetToEye * ((mVerticalHeight / 2.0) /
124 std::tan((fovDeg / 2.0) / 180.0 * M_PI));
125 }

169 MatrixType viewMatrix() const
170 {
171 MatrixType res;
172 detail::lookAtMatrix(res.data(), mEye, mCenter, mUp);
173 return res;
174 }
175
176 MatrixType projectionMatrix() const
177 {
178 MatrixType res;
179 switch (mProjectionMode) {
180 case ProjectionMode::ORTHO: {
181 const Scalar h = mVerticalHeight / 2.0;
182 const Scalar w = h * mAspect;
183 detail::orthoProjectionMatrix(
184 res.data(), -w, w, h, -h, mNear, mFar, false);
185 break;
186 }
187 case ProjectionMode::PERSPECTIVE: {
188 detail::projectionMatrix(
189 res.data(), mFovDeg, mAspect, mNear, mFar, false);
190 break;
191 }
192 default: assert(false); return MatrixType::Identity();
193 }
194 return res;
195 }
