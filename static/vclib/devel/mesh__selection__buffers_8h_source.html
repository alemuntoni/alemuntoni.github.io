148 template<MeshConcept MeshType>
149 void setVertexSelectionFromMesh(const MeshType& mesh)
150 {
151 const uint numVerts = mesh.vertexCount();
152 if (numVerts == 0)
153 return;
154
155 // Compute number of bits needed to store vertex selection
156 vcl::BitVector<true> vertexBackup(numVerts, false);
157
158 // Build bitfield from mesh vertex selection flags
159 // Note: For duplicated vertices (indices >= numVerts), they will
160 // remain unselected unless explicitly set elsewhere
161 uint vidx = 0;
162 for (const auto& v : mesh.vertices()) {
163 vertexBackup[vidx++] = v.selected();
164 }
165
166 mVertexSelection.setFromCPUBuffer(vertexBackup);
167 }
168
169 template<MeshConcept MeshType>
170 void setFaceSelectionFromMesh(
171 const MeshType& mesh,
172 const TriPolyIndexBiMap& indexMap)
173 {
174 const uint numFaces = mesh.faceCount();
175 if (numFaces == 0 || !hasFaceSelectionBuffer())
176 return;
177
178 // Compute number of bits needed to store face selection
179 vcl::BitVector<true> faceBackup(indexMap.triangleCount(), false);
180
181 // For each triangle, get its polygon index and read the selection from
182 // the mesh
183 for (uint tIdx = 0; tIdx < indexMap.triangleCount(); ++tIdx) {
184 uint faceIdx = indexMap.polygon(tIdx);
185 faceBackup[tIdx] = mesh.face(faceIdx).selected();
186 }
187
188 mFaceSelection.setFromCPUBuffer(faceBackup);
189 }

191 // ---- Selection operations -------------------------------------------
192
193 void computeSelection(
194 const SelectionParameters& params,
195 const Matrix44f& model,
196 const VertexBuffer& vertPosBuf,
197 const IndexBuffer& triIdxBuf,
198 const MeshPolyMappingBuffers& polyMapping)
199 {
200 bool toCompute = false;
201
202 if (params.mode.isFaceSelection()) {
203 if (params.mode.isVisibleSelection()) {
204 toCompute = faceSelectionVisible(
205 params, model, vertPosBuf, triIdxBuf, polyMapping);
206 }
207 else {
208 toCompute = faceSelection(
209 params, model, vertPosBuf, triIdxBuf, polyMapping);
210 }
211 }
212 else if (params.mode.isVertexSelection()) {
213 toCompute = vertexSelection(params, model, vertPosBuf);
214 }
215
216 if (toCompute && !params.isTemporary) {
217 // If no readback is in progress, start one. Otherwise queue the
218 // most recent mode so both vertex and face backups stay
219 // synchronized even when two tools trigger selection of both
220 // vertices and faces.
221 if (mBufToTexRemainingFrames == UINT_NULL) {
222 mLastReadbackMode = params.mode;
223 mBufToTexRemainingFrames =
224 requestCPUCopyOfSelectionBuffer(params.mode);
225 }
226 else {
227 // Queue the most recent mode so both vertex and face backups
228 // stay synchronized even when two tools trigger selection in
229 // sequence.
230 mPendingReadbackMode = params.mode;
231 mHasPendingReadback = true;
232 }
233 }
234 }

249 bool vertexSelection(
250 const SelectionParameters& params,
251 const Matrix44f& model,
252 const VertexBuffer& vertPosBuf)
253 {
254 using enum SelectionDragAction;
255
256 assert(params.mode.primitive == SelectionPrimitive::VERTEX);
257 if (params.isTemporary &&
258 (params.mode.isAction(ADD) || params.mode.isAction(SUBTRACT))) {
259 mVertexSelection.setFromCPUBuffer(mVertexSelection.cpuBackup());
260 }
261 if (params.mode.isAtomicAction())
262 return vertexSelectionAtomic(params);
263 else
264 return vertexSelectionNonAtomic(params, model, vertPosBuf);
265 }

318 bool faceSelectionVisible(
319 const SelectionParameters& params,
320 const Matrix44f& model,
321 const VertexBuffer& vertPosBuf,
322 const IndexBuffer& triIdxBuf,
323 const MeshPolyMappingBuffers& polyMapping)
324 {
325 using enum SelectionDragAction;
326
327 assert(params.mode.primitive == SelectionPrimitive::FACE);
328 if (params.isTemporary && params.mode.visible &&
329 (params.mode.isAction(ADD) || params.mode.isAction(SUBTRACT))) {
330 mFaceSelection.setFromCPUBuffer(mFaceSelection.cpuBackup());
331 }
332
333 if (params.mode.primitive == SelectionPrimitive::FACE &&
334 params.mode.isAction(REGULAR)) {
335 SelectionParameters params2(params);
336 params2.drawViewId = params2.pass1ViewId;
337 params2.mode.action = SelectionAtomicAction::NONE;
338 faceSelectionAtomic(params2);
339 }
340
341 ProgramManager& pm = Context::instance().programManager();
342
343 bgfx::ProgramHandle passProgram = pm.getProgram<
344 VertFragProgram::SELECTION_FACE_VISIBLE_RENDER_PASS>();
345 bgfx::ProgramHandle computeProg =
346 getComputeProgramFromSelectionMode(pm, params.mode);
347
348 std::array<uint, 3> workGroupSize = workGroupSizesFrom1DSize(
349 params.textureSize[0] * params.textureSize[1]);
350
351 uint64_t state = 0 | BGFX_STATE_WRITE_RGB | BGFX_STATE_WRITE_A |
352 BGFX_STATE_WRITE_Z | BGFX_STATE_DEPTH_TEST_LEQUAL;
353 bgfx::setState(state);
354 SelectionUniforms::setMeshIdForSelection(params.meshId);
355 SelectionUniforms::setSelectionWorkgroupSize(workGroupSize);
356 SelectionUniforms::bind();
357 vertPosBuf.bind(0);
358 triIdxBuf.bind();
359 bgfx::setTransform(model.data());
360 bgfx::submit(params.pass1ViewId, passProgram);
361
362 SelectionUniforms::setSelectionAction(
363 std::get<SelectionDragAction>(params.mode.action));
364 SelectionUniforms::setMeshIdForSelection(params.meshId);
365 SelectionUniforms::setSelectionWorkgroupSize(workGroupSize);
366 SelectionUniforms::bind();
367 bgfx::setImage(
368 0,
369 params.primIdTex,
370 0,
371 bgfx::Access::Read,
372 bgfx::TextureFormat::RGBA8);
373 bgfx::setImage(
374 1,
375 params.meshIdTex,
376 0,
377 bgfx::Access::Read,
378 bgfx::TextureFormat::RGBA8);
379 mFaceSelection.bind(
380 VCL_MRB_PRIMITIVE_SELECTION_BUFFER, bgfx::Access::ReadWrite);
381 polyMapping.bindTriToPolyBuffer();
382 polyMapping.bindPolyToTriBeginBuffer();
383 polyMapping.bindPolyToTriCountBuffer();
384 bgfx::setTransform(model.data());
385 bgfx::dispatch(
386 params.pass2ViewId,
387 computeProg,
388 workGroupSize[0],
389 workGroupSize[1],
390 workGroupSize[2]);
391 return true;
392 }

394 // ---- CPU readback ---------------------------------------------------
395
396 // called on draw
397 template<MeshConcept MeshType>
398 bool selectionReadback(MeshType& m, const TriPolyIndexBiMap& indexMap)
399 {
400 bool readbackCompletedThisFrame = false;
401 switch (mBufToTexRemainingFrames) {
402 case 0: {
403 if (mLastReadbackMode.isVertexSelection()) {
404 mVertexSelection.getResultsCopy();
405 }
406 else if (mLastReadbackMode.isFaceSelection()) {
407 mFaceSelection.getResultsCopy();
408 }
409
410 // Update CPU-side selection flags from GPU readback
411 updateCPUSelectionFromGPU(m, indexMap);
412 readbackCompletedThisFrame = true;
413
414 // TODO: right now, only one additional readback can be queued. If
415 // multiple selection actions are performed in a single frame.
416 // This is fine now because we can select at the same time both
417 // vertices and faces. If this changes, we need to implement a queue
418 // of pending readbacks.
419 if (mHasPendingReadback) {
420 mLastReadbackMode = mPendingReadbackMode;
421 mHasPendingReadback = false;
422 mBufToTexRemainingFrames =
423 requestCPUCopyOfSelectionBuffer(mLastReadbackMode);
424 }
425 else {
426 mBufToTexRemainingFrames = UINT_NULL;
427 }
428 } break;
429 case UINT_NULL: break;
430 default: mBufToTexRemainingFrames--;
431 }
432 return readbackCompletedThisFrame;
433 }
434
435 bool isSelectionReadbackPending() const
436 {
437 return mBufToTexRemainingFrames != UINT_NULL || mHasPendingReadback;
438 }

451 template<MeshConcept MeshType>
452 void updateCPUSelectionFromGPU(
453 MeshType& m,
454 const TriPolyIndexBiMap& indexMap)
455 {
456 // Update vertex selection
457 if (mLastReadbackMode.isVertexSelection() &&
458 !mVertexSelection.cpuBackup().empty()) {
459 const auto& cpuBackup = mVertexSelection.cpuBackup();
460
461 uint numSelectedVertices = 0;
462 uint vidx = 0;
463
464 vcl::BitSet<uint8_t, true> flags;
465
466 for (auto v : m.vertices()) {
467 const uint byteIdx = vidx / 8;
468 if (byteIdx < cpuBackup.bytes().size()) {
469 if (vidx % 8 == 0)
470 flags.setUnderlying(cpuBackup.bytes()[byteIdx]);
471
472 bool selected = flags[vidx % 8];
473 v.selected() = selected;
474 if (selected)
475 ++numSelectedVertices;
476 }
477 ++vidx;
478 }
479 mNumSelectedVertices = numSelectedVertices;
480 }
481
482 // Update face selection
483 if constexpr (HasFaces<MeshType>) {
484 if (mLastReadbackMode.isFaceSelection() &&
485 !mFaceSelection.cpuBackup().empty()) {
486 const auto& cpuBackup = mFaceSelection.cpuBackup();
487
488 const bool isMappingTrivial = indexMap.isMappingTrivial();
489
490 uint numSelectedFaces = 0;
491 // For each face, check if its first triangle is selected
492 // (compute shaders ensure all triangles of a face have the same
493 // selection state)
494 vcl::BitSet<uint8_t, true> flags;
495 uint lastByteIdx = UINT_NULL;
496
497 for (auto f : m.faces()) {
498 const uint faceIdx = f.index();
499 const uint firstTriIdx =
500 isMappingTrivial ? faceIdx :
501 indexMap.triangleBegin(faceIdx);
502 const uint byteIdx = firstTriIdx / 8;
503
504 if (byteIdx < cpuBackup.bytes().size()) {
505 if (byteIdx != lastByteIdx) {
506 flags.setUnderlying(cpuBackup.bytes()[byteIdx]);
507 lastByteIdx = byteIdx;
508 }
509
510 bool selected = flags[firstTriIdx % 8];
511 f.selected() = selected;
512 if (selected)
513 ++numSelectedFaces;
514 }
515 }
516 mNumSelectedFaces = numSelectedFaces;
517 }
518 }
519 }

764 bgfx::ProgramHandle getComputeProgramFromSelectionMode(
765 ProgramManager& pm,
766 SelectionMode mode) const
767 {
768 using enum SelectionPrimitive;
769 using enum ComputeProgram;
770
771 if (mode.isAtomicAction()) {
772 using enum SelectionAtomicAction;
773
774 SelectionAtomicAction action =
775 std::get<SelectionAtomicAction>(mode.action);
776 switch (action) {
777 case ALL: return pm.getComputeProgram<SELECTION_ALL>();
778 case NONE: return pm.getComputeProgram<SELECTION_NONE>();
779 case INVERT: return pm.getComputeProgram<SELECTION_INVERT>();
780 }
781 }
782 else {
783 switch (mode.primitive) {
784 case VERTEX: return pm.getComputeProgram<SELECTION_VERTEX>();
785 case FACE:
786 if (mode.visible)
787 return pm.getComputeProgram<SELECTION_FACE_VISIBLE>();
788 else
789 return pm.getComputeProgram<SELECTION_FACE>();
790 }
791 }
792 assert(false && "Invalid selection mode");
793 return BGFX_INVALID_HANDLE;
794 }

805 static std::array<uint, 3> workGroupSizesFrom1DSize(uint size)
806 {
807 std::array<uint, 3> sizes = {1, 1, 1};
808 if (size == 0)
809 return sizes;
810
811 // cut the size considering the compute thread counts
812 size = static_cast<uint>(
813 std::ceil(double(size) / VCL_COMPUTE_THREAD_COUNT_X));
814 // At the moment we assume Y and Z compute thread count == 1
815
816 const uint dim = uint(std::ceil(std::cbrt(double(size))));
817
818 sizes[0] = std::min(dim, MAX_COMPUTE_WORKGROUP_SIZE);
819
820 // Cap at sizes[0] to keep the grid roughly cubic rather than letting
821 // one dimension grow much larger than the others.
822 const uint remainingWork = uint(std::ceil(double(size) / sizes[0]));
823 sizes[1] = std::min(remainingWork, sizes[0]);
824 sizes[2] = uint(std::ceil(double(size) / (sizes[0] * sizes[1])));
825
826 return sizes;
827 }
