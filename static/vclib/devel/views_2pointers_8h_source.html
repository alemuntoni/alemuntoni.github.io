17struct NotNullView

29struct DereferenceView
30{
31 constexpr DereferenceView() = default;
32
33 template<std::ranges::range R>
34 friend constexpr auto operator|(R&& r, DereferenceView)
35 {
36 return std::views::transform(
37 r, [](auto p) -> decltype(auto) { return *p; });
38 }
39};
