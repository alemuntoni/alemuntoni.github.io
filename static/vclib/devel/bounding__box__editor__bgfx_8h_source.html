20template<typename ViewerDrawer>
21class BoundingBoxEditorBGFX : public Editor<ViewerDrawer>
22{
23 using Base = Editor<ViewerDrawer>;
24
25 std::vector<DrawableBox3> mBoxes;
26
27 BoundingBoxEditorSettings mSettings;
28
29public:
30 BoundingBoxEditorBGFX() = default;
31
32 std::string name() const override { return "Bounding Box"; }
33
34 void onViewerSet() override
35 {
36 Base::viewerRegisterGlobalAction(
37 "Toggle Bounding Box",
38 {Key::B, {KeyModifier::NO_MODIFIER}},
39 [this]() {
40 this->setActive(!this->isActive());
41 });
42 }
43
44 BoundingBoxEditorSettings& settings() override { return mSettings; }
45
46 const BoundingBoxEditorSettings& settings() const override
47 {
48 return mSettings;
49 }
50
51 void loadSettings(const nlohmann::json& j) override
52 {
53 mSettings.loadSettings(j);
54 }
55
56 void saveSettings(nlohmann::json& j) const override
57 {
58 mSettings.saveSettings(j);
59 }
60
61 void setActive(bool active) override
62 {
63 Base::setActive(active);
64 Base::viewerUpdate();
65 }
66
67 void refresh() override
68 {
69 mBoxes.clear();
70
71 Color c = mSettings.color;
72 float thickness = mSettings.thickness;
73
74 for (const auto& drawable : *Base::drawList()) {
75 const AbstractDrawableMesh* m =
76 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
77 if (m) {
78 // note: the bounding box is already trasformed if the mesh has
79 // a transform matrix
80 mBoxes.push_back(DrawableBox3(m->boundingBox(), c, thickness));
81 }
82 else {
83 mBoxes.push_back(DrawableBox3(Box3f()));
84 }
85 }
86 Base::viewerUpdate();
87 }
88
89 void refreshSettings() override
90 {
91 Color c = mSettings.color;
92 float thickness = mSettings.thickness;
93
94 for (auto& b : mBoxes) {
95 b.setColor(c);
96 b.setThickness(thickness);
97 }
98 Base::viewerUpdate();
99 }
100
101 void drawContent(uint viewId) override
102 {
103 using enum EditorSettings::EditMode;
104
105 DrawObjectSettings settings;
106 settings.viewId = viewId;
107
108 if (mSettings.editMode == NONE)
109 return;
110
111 if (mSettings.editMode == CURRENT_OBJECT) {
112 uint id = Base::drawList()->selectedObjectId();
113 if (id < mBoxes.size()) {
114 mBoxes[id].draw(settings);
115 }
116 }
117 else {
118 for (uint i = 0; i < mBoxes.size(); ++i) {
119 bool show = (mSettings.editMode == VISIBLE_OBJECTS) ?
120 Base::drawList()->at(i)->isVisible() :
121 true;
122
123 if (show) {
124 mBoxes[i].draw(settings);
125 }
126 }
127 }
128 }
129};
130
