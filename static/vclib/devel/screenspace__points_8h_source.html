134 template<Range R>
135 requires Point2Concept<std::ranges::range_value_t<R>>
136 void setVertices(R&& verts)
137 {
138 mVerPosCount = std::ranges::size(verts);
139
140 // Compute padding to ensure the buffer size is a multiple of 4 floats
141 // (16 bytes). This is required because the vertex shader reads the
142 // buffer as vec4 elements.
143 uint padding = (4 - (mVerPosCount % 4)) % 4;
144 uint nv = mVerPosCount + padding;
145
146 VertexBuffer vertBuff;
147 auto [buffer, releaseFn] =
148 Context::getAllocatedBufferAndReleaseFn<float>(nv * 2);
149
150 for (size_t i = 0; const auto& v : verts) {
151 buffer[i * 2 + 0] = v.x();
152 buffer[i * 2 + 1] = v.y();
153 ++i;
154 }
155
156 vertBuff.create(
157 buffer,
158 nv,
159 bgfx::Attrib::Position,
160 2,
161 PrimitiveType::FLOAT,
162 releaseFn);
163 mVertexPositions.setOwned(std::move(vertBuff));
164 mIsUpdateProgramNeeded = true;
165 }
166
174 template<Range R>
175 requires ColorConcept<std::ranges::range_value_t<R>>
176 void setVertexColors(R&& vertColors)
177 {
178 assert(std::ranges::size(vertColors) == mVerPosCount);
179
180 // Compute padding to ensure the buffer size is a multiple of 16 bytes.
181 uint padding = (4 - (mVerPosCount % 4)) % 4;
182 uint nv = mVerPosCount + padding;
183
184 VertexBuffer vColsBuff;
185
186 auto [buffer, releaseFn] =
187 Context::getAllocatedBufferAndReleaseFn<uint>(nv);
188
189 for (uint i = 0; const auto& c : vertColors) {
190 buffer[i] = c.abgr();
191 ++i;
192 }
193
194 vColsBuff.create(
195 buffer,
196 nv,
197 bgfx::Attrib::Color0,
198 4,
199 PrimitiveType::UCHAR,
200 true,
201 releaseFn);
202 mVertexColors.setOwned(std::move(vColsBuff));
203 mIsUpdateProgramNeeded = true;
204 }
