398 static PointT circumcenter(
399 const PointT& p0,
400 const PointT& p1,
401 const PointT& p2)
402 {
403 ScalarType a2 = p1.squaredDist(p2);
404 ScalarType b2 = p2.squaredDist(p0);
405 ScalarType c2 = p0.squaredDist(p1);
406
407 PointType c = p0 * a2 * (-a2 + b2 + c2) + p1 * b2 * (a2 - b2 + c2) +
408 p2 * c2 * (a2 + b2 - c2);
409
410 c /= 2 * (a2 * b2 + a2 * c2 + b2 * c2) - a2 * a2 - b2 * b2 - c2 * c2;
411
412 return c;
413 }

444 static ScalarType area(const PointT& p0, const PointT& p1, const PointT& p2)
445 {
446 if constexpr (DIM == 2) {
447 return ((p1[0] - p0[0]) * (p2[1] - p0[1]) -
448 (p2[0] - p0[0]) * (p1[1] - p0[1])) /
449 2;
450 }
451 if constexpr (DIM == 3) {
452 return normal(p0, p1, p2).norm() / 2;
453 }
454 else {
455 // heron's formula
456 ScalarType s = perimeter() / 2;
457 return std::sqrt(
458 s * (s - p0.dist(p1)) * (s - p1.dist(p2)) * (s - p2.dist(p0)));
459 }
460 }

479 static ScalarType quality(
480 const PointT& p0,
481 const PointT& p1,
482 const PointT& p2)
483 {
484 ScalarType a = area();
485 if (a == 0)
486 return 0; // Area zero triangles have surely quality==0;
487
488 PointType d10 = p1 - p0;
489 PointType d20 = p2 - p0;
490 PointType d12 = p1 - p2;
491 double b =
492 std::min({d10.squaredNorm(), d20.squaredNorm(), d12.squaredNorm()});
493 if (b == 0)
494 return 0; // Again: area zero triangles have surely quality==0;
495 return (2 * a) / b;
496 }

515 static ScalarType qualityRadii(
516 const PointT& p0,
517 const PointT& p1,
518 const PointT& p2)
519 {
520 double a = p0.dist(p1);
521 double b = p2.dist(p0);
522 double c = p1.dist(p2);
523
524 double sum = (a + b + c) * 0.5;
525 double area2 = sum * (a + b - sum) * (a + c - sum) * (b + c - sum);
526
527 if (area2 <= 0)
528 return 0;
529
530 return (8 * area2) / (a * b * c * sum);
531 }
532
553 static ScalarType qualityMeanRatio(
554 const PointT& p0,
555 const PointT& p1,
556 const PointT& p2)
557 {
558 double a = p0.dist(p1);
559 double b = p2.dist(p0);
560 double c = p1.dist(p2);
561
562 double sum = (a + b + c) * 0.5; // semiperimeter
563 double area2 = sum * (a + b - sum) * (a + c - sum) * (b + c - sum);
564 if (area2 <= 0)
565 return 0;
566 return (4.0 * std::sqrt(3.0) * std::sqrt(area2)) /
567 (a * a + b * b + c * c);
568 }
