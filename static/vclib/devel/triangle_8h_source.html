405 static Point3<ScalarType> barycentricCoordinates(
406 const PointT& p0,
407 const PointT& p1,
408 const PointT& p2,
409 const PointT& p)
410 {
411 PointT v0 = p1 - p0;
412 PointT v1 = p2 - p0;
413 PointT v2 = p - p0;
414
415 ScalarType d00 = v0.dot(v0);
416 ScalarType d01 = v0.dot(v1);
417 ScalarType d11 = v1.dot(v1);
418 ScalarType d20 = v2.dot(v0);
419 ScalarType d21 = v2.dot(v1);
420
421 ScalarType denom = d00 * d11 - d01 * d01;
422
423 if (denom == 0) {
424 return Point3<ScalarType>(1.0 / 3.0, 1.0 / 3.0, 1.0 / 3.0);
425 }
426
427 ScalarType v = (d11 * d20 - d01 * d21) / denom;
428 ScalarType w = (d00 * d21 - d01 * d20) / denom;
429 ScalarType u = 1 - v - w;
430
431 return Point3<ScalarType>(u, v, w);
432 }

453 static PointT circumcenter(
454 const PointT& p0,
455 const PointT& p1,
456 const PointT& p2)
457 {
458 ScalarType a2 = p1.squaredDist(p2);
459 ScalarType b2 = p2.squaredDist(p0);
460 ScalarType c2 = p0.squaredDist(p1);
461
462 PointType c = p0 * a2 * (-a2 + b2 + c2) + p1 * b2 * (a2 - b2 + c2) +
463 p2 * c2 * (a2 + b2 - c2);
464
465 c /= 2 * (a2 * b2 + a2 * c2 + b2 * c2) - a2 * a2 - b2 * b2 - c2 * c2;
466
467 return c;
468 }

499 static ScalarType area(const PointT& p0, const PointT& p1, const PointT& p2)
500 {
501 if constexpr (DIM == 2) {
502 return ((p1[0] - p0[0]) * (p2[1] - p0[1]) -
503 (p2[0] - p0[0]) * (p1[1] - p0[1])) /
504 2;
505 }
506 if constexpr (DIM == 3) {
507 return normal(p0, p1, p2).norm() / 2;
508 }
509 else {
510 // heron's formula
511 ScalarType s = perimeter() / 2;
512 return std::sqrt(
513 s * (s - p0.dist(p1)) * (s - p1.dist(p2)) * (s - p2.dist(p0)));
514 }
515 }
516
534 static ScalarType quality(
535 const PointT& p0,
536 const PointT& p1,
537 const PointT& p2)
538 {
539 ScalarType a = area();
540 if (a == 0)
541 return 0; // Area zero triangles have surely quality==0;
542
543 PointType d10 = p1 - p0;
544 PointType d20 = p2 - p0;
545 PointType d12 = p1 - p2;
546 double b =
547 std::min({d10.squaredNorm(), d20.squaredNorm(), d12.squaredNorm()});
548 if (b == 0)
549 return 0; // Again: area zero triangles have surely quality==0;
550 return (2 * a) / b;
551 }
552
570 static ScalarType qualityRadii(
571 const PointT& p0,
572 const PointT& p1,
573 const PointT& p2)
574 {
575 double a = p0.dist(p1);
576 double b = p2.dist(p0);
577 double c = p1.dist(p2);
578
579 double sum = (a + b + c) * 0.5;
580 double area2 = sum * (a + b - sum) * (a + c - sum) * (b + c - sum);
581
582 if (area2 <= 0)
583 return 0;
584
585 return (8 * area2) / (a * b * c * sum);
586 }
587
608 static ScalarType qualityMeanRatio(
609 const PointT& p0,
610 const PointT& p1,
611 const PointT& p2)
612 {
613 double a = p0.dist(p1);
614 double b = p2.dist(p0);
615 double c = p1.dist(p2);
616
617 double sum = (a + b + c) * 0.5; // semiperimeter
618 double area2 = sum * (a + b - sum) * (a + c - sum) * (b + c - sum);
619 if (area2 <= 0)
620 return 0;
621 return (4.0 * std::sqrt(3.0) * std::sqrt(area2)) /
622 (a * a + b * b + c * c);
623 }
