15template<int N>
16class CellIterator
17{
18 Point<uint, N> mIt;
19 Point<uint, N> mFirst, mEnd;
20
21public:
22 using difference_type = ptrdiff_t;
23 using value_type = Point<uint, N>;
24 using reference = const Point<uint, N>&;
25 using pointer = const Point<uint, N>*;
26 using iterator_category = std::forward_iterator_tag;
27
28 CellIterator()
29 {
30 mIt.setConstant(UINT_NULL);
31 mFirst = mEnd = mIt;
32 }
33
34 CellIterator(const Point<uint, N>& first, const Point<uint, N>& end) :
35 mIt(first), mFirst(first), mEnd(end)
36 {
37 }
38
39 reference operator*() const { return mIt; }
40
41 pointer operator->() const { return &mIt; }
42
43 bool operator==(const CellIterator& oi) const { return (mIt == oi.mIt); }
44
45 bool operator!=(const CellIterator& oi) const { return (mIt != oi.mIt); }
46
47 CellIterator operator++()
48 {
49 uint d = N - 1;
50 while (d != UINT_NULL && mIt(d) == mEnd(d) - 1) {
51 mIt(d) = mFirst(d);
52 d--;
53 }
54 if (d != -1)
55 mIt(d)++;
56 else
57 mIt.setConstant(UINT_NULL);
58 return *this;
59 }
60
61 CellIterator operator++(int)
62 {
63 CellIterator<N> oit = mIt;
64 ++(*this);
65 return oit;
66 }
67};
