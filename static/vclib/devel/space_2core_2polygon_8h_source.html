252 template<typename Iterator>
253 static PointT normal(Iterator begin, Iterator end) requires (
254 std::is_same_v<typename Iterator::value_type, PointT> &&
255 PointT::DIM == 3)
256 {
257 if (begin == end) {
258 return PointT::Zero();
259 }
260
261 const auto first = *begin;
262 auto prev = first;
263 auto it = begin;
264 ++it;
265
266 if (it == end) {
267 return PointT::Zero();
268 }
269
270 const auto second = *it;
271 ++it;
272 if (it == end) {
273 return PointT::Zero();
274 }
275
276 const auto third = *it;
277 ++it;
278 if (it == end) {
279 PointT normal = (second - first).cross(third - first);
280 normal.normalize();
281 return normal;
282 }
283
284 const auto fourth = *it;
285 ++it;
286 if (it == end) {
287 PointT normal = (third - first).cross(fourth - second);
288 normal.normalize();
289 return normal;
290 }
291
292 PointT sum;
293 sum.setZero();
294
295 const auto addNewellEdge = [&sum](const PointT& a, const PointT& b) {
296 sum[0] += (a[1] - b[1]) * (a[2] + b[2]);
297 sum[1] += (a[2] - b[2]) * (a[0] + b[0]);
298 sum[2] += (a[0] - b[0]) * (a[1] + b[1]);
299 };
300
301 addNewellEdge(first, second);
302 addNewellEdge(second, third);
303 addNewellEdge(third, fourth);
304
305 prev = fourth;
306 for (; it != end; ++it) {
307 const auto curr = *it;
308 addNewellEdge(prev, curr);
309 prev = curr;
310 }
311
312 addNewellEdge(prev, first);
313
314 sum.normalize();
315 return sum;
316 }

343 template<typename Iterator>
344 static PointT barycenter(Iterator begin, Iterator end)
345 requires (std::is_same_v<typename Iterator::value_type, PointT>)
346 {
347 PointT bar;
348 bar.setZero();
349
350 uint cnt = 0;
351 for (; begin != end; ++begin) {
352 bar += *begin;
353 ++cnt;
354 }
355
356 assert(cnt);
357
358 return bar / cnt;
359 }

391 template<typename Iterator, typename WIterator>
392 static PointT weightedBarycenter(
393 Iterator begin,
394 Iterator end,
395 WIterator wBegin)
396 requires (std::is_same_v<typename Iterator::value_type, PointT>)
397 {
398 using ScalarType = WIterator::value_type;
399
400 PointT bar;
401 bar.setZero();
402 ScalarType wSum = 0;
403
404 for (; begin != end; ++begin, ++wBegin) {
405 bar += *begin * *wBegin;
406 wSum += *wBegin;
407 }
408
409 return bar / wSum;
410 }

449 template<typename Iterator>
450 static ScalarType perimeter(Iterator begin, Iterator end)
451 requires (std::is_same_v<typename Iterator::value_type, PointT>)
452 {
453 using Scalar = PointType::ScalarType;
454
455 Scalar per = 0;
456 for (auto i = begin; i != end; ++i) {
457 const PointT& p0 = *i;
458 auto i1 = i;
459 ++i1;
460 if (i1 == end)
461 i1 = begin;
462 const PointT& p1 = *i1;
463 per += p0.dist(p1);
464 }
465 return per;
466 }

497 template<typename Iterator>
498 static ScalarType area(Iterator begin, Iterator end)
499 requires (std::is_same_v<typename Iterator::value_type, PointT>)
500 {
501 using Scalar = PointType::ScalarType;
502
503 PointType bar = barycenter(begin, end);
504 Scalar area = 0;
505 for (auto i = begin; i != end; ++i) {
506 const PointT& p0 = *i;
507 auto i1 = i;
508 ++i1;
509 if (i1 == end)
510 i1 = begin;
511 const PointT& p1 = *i1;
512
513 area += Triangle<PointT>::area(p0, p1, bar);
514 }
515 return area;
516 }
