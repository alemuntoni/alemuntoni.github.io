24// given a buffer, returns a reference to the element at (i,j) considering the
25// storage type (row-major or column-major)
26inline auto& at(
27 auto* buffer,
28 auto i,
29 auto j,
30 auto numRows,
31 auto numCols,
32 MatrixStorageType storage)
33{
34 if (storage == MatrixStorageType::ROW_MAJOR) {
35 return buffer[i * numCols + j];
36 }
37 else {
38 return buffer[j * numRows + i];
39 }
40}

62template<typename BufferValueType>
63constexpr ushort toMaterialIndex(BufferValueType val)
64{
65 using T = std::remove_cvref_t<BufferValueType>;
66 if constexpr (std::is_floating_point_v<T>) {
67 if (std::isnan(val) || val < 0 || val == static_cast<T>(USHORT_NULL)) {
68 return USHORT_NULL;
69 }
70 return static_cast<ushort>(val);
71 }
72 else if constexpr (std::is_signed_v<T>) {
73 if (val < 0 || val == static_cast<T>(USHORT_NULL)) {
74 return USHORT_NULL;
75 }
76 return static_cast<ushort>(val);
77 }
78 else {
79 if (val == USHORT_NULL || val == bufferSentinel<T>()) {
80 return USHORT_NULL;
81 }
82 return static_cast<ushort>(val);
83 }
84}

92template<typename BufferValueType, typename SizeType>
93void checkBufferIndexCapacity(SizeType elementCount)
94{
95 using ValueType = std::remove_cvref_t<BufferValueType>;
96 if constexpr (std::integral<ValueType>) {
97 if (elementCount > 0) {
98 auto lastIndex = elementCount - 1;
99 if constexpr (std::is_signed_v<ValueType>) {
100 if (std::cmp_greater(
101 lastIndex, std::numeric_limits<ValueType>::max())) {
102 throw WrongSizeException(
103 "Buffer element type is too small to represent the "
104 "indices of the mesh.");
105 }
106 }
107 else {
108 if (std::cmp_greater_equal(
109 lastIndex, std::numeric_limits<ValueType>::max())) {
110 throw WrongSizeException(
111 "Buffer element type is too small to represent the "
112 "indices of the mesh.");
113 }
114 }
115 }
116 }
