77template<MeshConcept MeshType>
78void vertexPositionsFromBuffer(
79 MeshType& mesh,
80 const auto* buffer,
81 uint vertexCount,
82 bool clearBeforeSet = true,
83 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
84 uint numRows = UINT_NULL)
85{
86 using namespace detail;
87
88 const uint NUM_ROWS = numRows == UINT_NULL ? vertexCount : numRows;
89
90 if (clearBeforeSet) {
91 mesh.clearVertices();
92 mesh.resizeVertices(vertexCount);
93 }
94 else {
95 if (vertexCount != mesh.vertexCount()) {
96 throw WrongSizeException(
97 "The input vertex number does not match the number of vertices "
98 "of the mesh\n"
99 "Number of vertices in the mesh: " +
100 std::to_string(mesh.vertexCount()) +
101 "\nNumber of input vertex number: " +
102 std::to_string(vertexCount));
103 }
104 }
105
106 for (uint i = 0; auto& p : mesh.vertices() | views::positions) {
107 p.x() = at(buffer, i, 0, NUM_ROWS, 3, storage);
108 p.y() = at(buffer, i, 1, NUM_ROWS, 3, storage);
109 p.z() = at(buffer, i, 2, NUM_ROWS, 3, storage);
110
111 ++i;
112 }
113}

174template<FaceMeshConcept MeshType>
175void faceIndicesFromBuffer(
176 MeshType& mesh,
177 const auto* buffer,
178 uint faceCount,
179 uint faceSize = 3,
180 bool clearBeforeSet = true,
181 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
182 uint numRows = UINT_NULL)
183{
184 using namespace detail;
185
186 const uint NUM_ROWS = numRows == UINT_NULL ? faceCount : numRows;
187
188 if (clearBeforeSet) {
189 mesh.clearFaces();
190 mesh.resizeFaces(faceCount);
191 }
192 else {
193 if (faceCount != mesh.faceCount()) {
194 throw WrongSizeException(
195 "The input face count does not match the number of faces of "
196 "the mesh\n"
197 "Number of faces in the mesh: " +
198 std::to_string(mesh.faceCount()) +
199 "\nNumber of input face count: " + std::to_string(faceCount));
200 }
201 }
202
203 if constexpr (HasPolygons<MeshType>) {
204 uint i = 0;
205 for (auto& f : mesh.faces()) {
206 uint vertexCount = 0;
207
208 // count the number of vertices of the face
209 while (vertexCount < faceSize &&
210 at(buffer, i, vertexCount, NUM_ROWS, faceSize, storage) !=
211 -1 &&
212 at(buffer, i, vertexCount, NUM_ROWS, faceSize, storage) !=
213 UINT_NULL)
214 vertexCount++;
215
216 f.resizeVertices(vertexCount);
217
218 for (uint j = 0; j < vertexCount; ++j)
219 f.setVertex(j, at(buffer, i, j, NUM_ROWS, faceSize, storage));
220 ++i;
221 }
222 }
223 else { // the vertex number of mesh faces is fixed
224 using FaceType = MeshType::FaceType;
225
226 constexpr int VC = FaceType::VERTEX_COUNT;
227 if (faceSize == VC) { // faces of matrix and mesh have same size
228 uint i = 0;
229 for (auto& f : mesh.faces()) {
230 for (uint j = 0; j < VC; ++j)
231 f.setVertex(
232 j, at(buffer, i, j, NUM_ROWS, faceSize, storage));
233 ++i;
234 }
235 }
236 else {
237 // cannot import the faces if the number of face indices in the
238 // buffer is different w.r.t. face size of the mesh
239 throw WrongSizeException(
240 "The input face buffer has a different face size "
241 "than the vertex count of the faces of the mesh.\n"
242 "Faces vertex count in the mesh: " +
243 std::to_string(VC) +
244 "\nNumber of columns in the input face buffer: " +
245 std::to_string(faceSize));
246 }
247 }
248}

293template<EdgeMeshConcept MeshType>
294void edgeIndicesFromBuffer(
295 MeshType& mesh,
296 const auto* buffer,
297 uint edgeCount,
298 bool clearBeforeSet = true,
299 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
300 uint numRows = UINT_NULL)
301{
302 using namespace detail;
303
304 if (clearBeforeSet) {
305 mesh.clearEdges();
306 mesh.resizeEdges(edgeCount);
307 }
308 else {
309 if (edgeCount != mesh.edgeCount()) {
310 throw WrongSizeException(
311 "The input edge count does not match the number of edges "
312 "of the mesh\n"
313 "Number of edges in the mesh: " +
314 std::to_string(mesh.edgeCount()) +
315 "\nNumber of input edge count: " + std::to_string(edgeCount));
316 }
317 }
318
319 uint i = 0;
320 for (auto& e : mesh.edges()) {
321 e.setVertex(0, at(buffer, i, 0, numRows, 2, storage));
322 e.setVertex(1, at(buffer, i, 1, numRows, 2, storage));
323 i++;
324 }
325}

619template<uint ELEM_ID, MeshConcept MeshType>
620void elementColorsFromBuffer(
621 MeshType& mesh,
622 const auto* buffer,
623 uint channelCount = 4,
624 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
625 Color::Representation representation = Color::Representation::INT_0_255,
626 uint numRows = UINT_NULL)
627{
628 using namespace detail;
629
630 const uint NUM_ROWS =
631 numRows == UINT_NULL ? mesh.template count<ELEM_ID>() : numRows;
632
633 if (channelCount != 3 && channelCount != 4)
634 throw WrongSizeException(
635 "The input " + elementEnumString<ELEM_ID>() +
636 " colors must have 3 or 4 channels.");
637
638 enableIfPerElementComponentOptional<ELEM_ID, CompId::COLOR>(mesh);
639 requirePerElementComponent<ELEM_ID, CompId::COLOR>(mesh);
640
641 for (uint i = 0;
642 auto& c : mesh.template elements<ELEM_ID>() | views::colors) {
643 if (representation == Color::Representation::INT_0_255) {
644 c.x() = at(buffer, i, 0, NUM_ROWS, channelCount, storage);
645 c.y() = at(buffer, i, 1, NUM_ROWS, channelCount, storage);
646 c.z() = at(buffer, i, 2, NUM_ROWS, channelCount, storage);
647
648 if (channelCount == 4)
649 c.w() = at(buffer, i, 3, NUM_ROWS, channelCount, storage);
650 else
651 c.w() = 255;
652 }
653 else {
654 c.x() = at(buffer, i, 0, NUM_ROWS, channelCount, storage) * 255;
655 c.y() = at(buffer, i, 1, NUM_ROWS, channelCount, storage) * 255;
656 c.z() = at(buffer, i, 2, NUM_ROWS, channelCount, storage) * 255;
657
658 if (channelCount == 4)
659 c.w() = at(buffer, i, 3, NUM_ROWS, channelCount, storage) * 255;
660 else
661 c.w() = 255;
662 }
663 ++i;
664 }
665}
