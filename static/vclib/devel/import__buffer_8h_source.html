76template<MeshConcept MeshType>
77void vertexPositionsFromBuffer(
78 MeshType& mesh,
79 const auto* buffer,
80 typename MeshType::IndexType vertexCount,
81 bool clearBeforeSet = true,
82 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
83 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
84{
85 using namespace detail;
86
87 using IndexType = typename MeshType::IndexType;
88
89 const IndexType NUM_ROWS =
90 numRows == MeshType::NULL_INDEX ? vertexCount : numRows;
91
92 if (clearBeforeSet) {
93 mesh.clearVertices();
94 mesh.resizeVertices(vertexCount);
95 }
96 else {
97 if (vertexCount != mesh.vertexCount()) {
98 throw WrongSizeException(
99 "The input vertex number does not match the number of vertices "
100 "of the mesh\n"
101 "Number of vertices in the mesh: " +
102 std::to_string(mesh.vertexCount()) +
103 "\nNumber of input vertex number: " +
104 std::to_string(vertexCount));
105 }
106 }
107
108 for (IndexType i = 0; auto& p : mesh.vertices() | views::positions) {
109 p.x() = at(buffer, i, 0, NUM_ROWS, 3, storage);
110 p.y() = at(buffer, i, 1, NUM_ROWS, 3, storage);
111 p.z() = at(buffer, i, 2, NUM_ROWS, 3, storage);
112
113 ++i;
114 }
115}

178template<FaceMeshConcept MeshType>
179void faceIndicesFromBuffer(
180 MeshType& mesh,
181 const auto* buffer,
182 typename MeshType::IndexType faceCount,
183 uint faceSize = 3,
184 bool clearBeforeSet = true,
185 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
186 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
187{
188 using namespace detail;
189
190 using IndexType = typename MeshType::IndexType;
191
192 const IndexType NUM_ROWS =
193 numRows == MeshType::NULL_INDEX ? faceCount : numRows;
194
195 if (clearBeforeSet) {
196 mesh.clearFaces();
197 mesh.resizeFaces(faceCount);
198 }
199 else {
200 if (faceCount != mesh.faceCount()) {
201 throw WrongSizeException(
202 "The input face count does not match the number of faces of "
203 "the mesh\n"
204 "Number of faces in the mesh: " +
205 std::to_string(mesh.faceCount()) +
206 "\nNumber of input face count: " + std::to_string(faceCount));
207 }
208 }
209
210 if constexpr (HasPolygons<MeshType>) {
211 using BufferValueType = std::remove_cvref_t<decltype(*buffer)>;
212 const auto SENTINEL = bufferSentinel<BufferValueType>();
213
214 IndexType i = 0;
215 for (auto f : mesh.faces()) {
216 IndexType vertexCount = 0;
217
218 // count the number of vertices of the face
219 while (vertexCount < faceSize &&
220 at(buffer, i, vertexCount, NUM_ROWS, faceSize, storage) !=
221 SENTINEL)
222 vertexCount++;
223
224 f.resizeVertices(vertexCount);
225
226 for (IndexType j = 0; j < vertexCount; ++j)
227 f.setVertex(j, at(buffer, i, j, NUM_ROWS, faceSize, storage));
228 ++i;
229 }
230 }
231 else { // the vertex number of mesh faces is fixed
232 using FaceType = FaceType<MeshType>;
233
234 constexpr int VC = FaceType::VERTEX_COUNT;
235 if (faceSize == VC) { // faces of matrix and mesh have same size
236 IndexType i = 0;
237 for (auto f : mesh.faces()) {
238 for (IndexType j = 0; j < VC; ++j)
239 f.setVertex(
240 j, at(buffer, i, j, NUM_ROWS, faceSize, storage));
241 ++i;
242 }
243 }
244 else {
245 // cannot import the faces if the number of face indices in the
246 // buffer is different w.r.t. face size of the mesh
247 throw WrongSizeException(
248 "The input face buffer has a different face size "
249 "than the vertex count of the faces of the mesh.\n"
250 "Faces vertex count in the mesh: " +
251 std::to_string(VC) +
252 "\nNumber of columns in the input face buffer: " +
253 std::to_string(faceSize));
254 }
255 }
256}

301template<EdgeMeshConcept MeshType>
302void edgeIndicesFromBuffer(
303 MeshType& mesh,
304 const auto* buffer,
305 typename MeshType::IndexType edgeCount,
306 bool clearBeforeSet = true,
307 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
308 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
309{
310 using namespace detail;
311
312 using IndexType = typename MeshType::IndexType;
313
314 const IndexType NUM_ROWS =
315 numRows == MeshType::NULL_INDEX ? edgeCount : numRows;
316
317 if (clearBeforeSet) {
318 mesh.clearEdges();
319 mesh.resizeEdges(edgeCount);
320 }
321 else {
322 if (edgeCount != mesh.edgeCount()) {
323 throw WrongSizeException(
324 "The input edge count does not match the number of edges "
325 "of the mesh\n"
326 "Number of edges in the mesh: " +
327 std::to_string(mesh.edgeCount()) +
328 "\nNumber of input edge count: " + std::to_string(edgeCount));
329 }
330 }
331
332 IndexType i = 0;
333 for (auto e : mesh.edges()) {
334 e.setVertex(0, at(buffer, i, 0, NUM_ROWS, 2, storage));
335 e.setVertex(1, at(buffer, i, 1, NUM_ROWS, 2, storage));
336 i++;
337 }

637template<ElemId ELEM_ID, MeshConcept MeshType>
638void elementColorsFromBuffer(
639 MeshType& mesh,
640 const auto* buffer,
641 uint channelCount = 4,
642 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
643 Color::Representation representation = Color::Representation::INT_0_255,
644 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
645{
646 using namespace detail;
647
648 using IndexType = typename MeshType::IndexType;
649
650 const IndexType NUM_ROWS = numRows == MeshType::NULL_INDEX ?
651 mesh.template count<ELEM_ID>() :
652 numRows;
653
654 if (channelCount != 3 && channelCount != 4)
655 throw WrongSizeException(
656 "The input " + elementEnumString<ELEM_ID>() +
657 " colors must have 3 or 4 channels.");
658
659 enableIfElementComponentOptional<ELEM_ID, CompId::COLOR>(mesh);
660 requireElementComponent<ELEM_ID, CompId::COLOR>(mesh);
661
662 for (IndexType i = 0;
663 auto& c : mesh.template elements<ELEM_ID>() | views::colors) {
664 if (representation == Color::Representation::INT_0_255) {
665 c.x() = at(buffer, i, 0, NUM_ROWS, channelCount, storage);
666 c.y() = at(buffer, i, 1, NUM_ROWS, channelCount, storage);
667 c.z() = at(buffer, i, 2, NUM_ROWS, channelCount, storage);
668
669 if (channelCount == 4)
670 c.w() = at(buffer, i, 3, NUM_ROWS, channelCount, storage);
671 else
672 c.w() = 255;
673 }
674 else {
675 c.x() = at(buffer, i, 0, NUM_ROWS, channelCount, storage) * 255;
676 c.y() = at(buffer, i, 1, NUM_ROWS, channelCount, storage) * 255;
677 c.z() = at(buffer, i, 2, NUM_ROWS, channelCount, storage) * 255;
678
679 if (channelCount == 4)
680 c.w() = at(buffer, i, 3, NUM_ROWS, channelCount, storage) * 255;
681 else
682 c.w() = 255;
683 }
684 ++i;
685 }
686}
