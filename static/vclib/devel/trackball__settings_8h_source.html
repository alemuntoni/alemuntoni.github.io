21struct TrackballSettings
22{
23 using DragMotionMap = InputActionMap<MouseInput, TrackballMotionType>;
24 using ScrollAtomicMap = InputActionMap<
25 std::pair<ScrollAxis::Enum, KeyModifiers>,
26 TrackballMotionType>;
27 using KeyAtomicMap =
28 InputActionMap<std::pair<Key::Enum, KeyModifiers>, std::string>;
29 using MouseAtomicMap = InputActionMap<MouseInput, TrackballMotionType>;
30
31 DragMotionMap dragMotionMap = defaultDragMotionMap();
32 ScrollAtomicMap scrollAtomicMap = defaultScrollMotionMap();
33 KeyAtomicMap keyAtomicMap = defaultKeyAtomicMap();
34 MouseAtomicMap mouseAtomicMap = defaultMouseAtomicMap();
35
39 void resetDefaults()
40 {
41 dragMotionMap.resetToDefaults();
42 scrollAtomicMap.resetToDefaults();
43 keyAtomicMap.resetToDefaults();
44 mouseAtomicMap.resetToDefaults();
45 }
46
50 std::vector<std::reference_wrapper<AbstractInputActionMap>> actionMaps()
51 {
52 return {dragMotionMap, scrollAtomicMap, keyAtomicMap, mouseAtomicMap};
53 }
54
58 std::vector<std::reference_wrapper<const AbstractInputActionMap>>
59 actionMaps() const
60 {
61 return {dragMotionMap, scrollAtomicMap, keyAtomicMap, mouseAtomicMap};
62 }
63
68 void loadSettings(const nlohmann::json& j)
69 {
70 for (auto& map : actionMaps()) {
71 map.get().loadSettings(j);
72 }
73 }
74
79 void saveSettings(nlohmann::json& j) const
80 {
81 for (const auto& map : actionMaps()) {
82 map.get().saveSettings(j);
83 }
84 }

87 static DragMotionMap defaultDragMotionMap()
88 {
89 using enum MouseButton::Enum;
90 using enum KeyModifier::Enum;
91
92 using enum TrackballMotionType;
93
94 DragMotionMap map("Trackball Drag Motions");
95 map.registerActions({
96 {ARC,
97 "Arcball Rotation", {MouseInput {LEFT, {NO_MODIFIER}, false}}},
98 {
99 PAN, "Pan",
100 {MouseInput {LEFT, {CONTROL}, false},
101 MouseInput {MIDDLE, {NO_MODIFIER}, false}},
102 },
103 {ZMOVE, "Zoom (Translation)", {MouseInput {LEFT, {ALT}, false}}},
104 {SCALE, "Scale", {MouseInput {LEFT, {SHIFT}, false}}},
105 {ROLL, "Roll", {MouseInput {MIDDLE, {CONTROL}, false}}},
106 {DIR_LIGHT_ARC,
107 "Light Rotation", {MouseInput {LEFT, {SHIFT, CONTROL}, false}}}
108 });
109 return map;
110 }
111
112 static ScrollAtomicMap defaultScrollMotionMap()
113 {
114 using enum KeyModifier::Enum;
115 using enum TrackballMotionType;

117
118 ScrollAtomicMap map("Trackball Scroll Motions");
119 map.registerActions({
120 {SCALE, "Scale", {Input {ScrollAxis::VERTICAL, {NO_MODIFIER}}} },
121 {ROLL, "Roll", {Input {ScrollAxis::VERTICAL, {CONTROL}}} },
122 {FOV,
123 "Field of View", {Input {ScrollAxis::VERTICAL, {SHIFT}},
124 Input {ScrollAxis::HORIZONTAL, {SHIFT}}}}
125 });
126 return map;
127 }
128
129 static MouseAtomicMap defaultMouseAtomicMap()
130 {
131 using enum MouseButton::Enum;
132 using enum KeyModifier::Enum;
133 using enum TrackballMotionType;
134
135 MouseAtomicMap map("Trackball Mouse Atomic Motions");
136 map.registerActions({
137 {FOCUS,
138 "Focus on Object", {MouseInput {LEFT, {NO_MODIFIER}, true}}}
139 });
140 return map;
141 }

154 {"Reset Directional Light",
155 "Reset Directional Light", {Input {R, {CONTROL, SHIFT}}} },
156 {"Rotate X+", "Rotate X+", {Input {NP_2, {NO_MODIFIER}}} },

158 {"Rotate Y+", "Rotate Y+", {Input {NP_6, {NO_MODIFIER}}} },
159 {"Rotate X-", "Rotate X-", {Input {NP_8, {NO_MODIFIER}}} },
160 {"Translate Y+", "Translate Y+", {Input {UP, {NO_MODIFIER}}} },
161 {"Translate Y-", "Translate Y-", {Input {DOWN, {NO_MODIFIER}}} },
162 {"Translate X-", "Translate X-", {Input {LEFT, {NO_MODIFIER}}} },
163 {"Translate X+", "Translate X+", {Input {RIGHT, {NO_MODIFIER}}} },
164 {"View Front", "View Front", {Input {NP_1, {NO_MODIFIER}}} },
165 {"View Top", "View Top", {Input {NP_7, {NO_MODIFIER}}} },
166 {"View Right", "View Right", {Input {NP_3, {NO_MODIFIER}}} },
