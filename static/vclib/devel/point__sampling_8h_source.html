57template<MeshConcept MeshType>
58auto allVerticesPointSampling(
59 const MeshType& m,
60 std::vector<IndexType<MeshType>>& birthVertices,
61 bool onlySelected = false)
62{
63 using PointType = VertexType<MeshType>::PositionType;
64
65 PointSampler<PointType> sampler;
66
67 // Determine the number of vertices to sample
68 std::size_t n = onlySelected ? vertexSelectionCount(m) : m.vertexCount();
69
70 // Reserve space in the sampler and birthVertices vectors
71 sampler.reserve(n);
72 birthVertices.reserve(n);
73 birthVertices.clear();
74
75 // Loop through all the vertices in the mesh and add them to the sampler if
76 // they are selected
77 for (auto&& v : m.vertices()) {
78 if (!onlySelected || v.selected()) {
79 sampler.add(v);
80 birthVertices.push_back(v.index());
81 }
82 }
83 return sampler;
84}

207template<MeshConcept MeshType>
208auto vertexUniformPointSampling(
209 const MeshType& m,
210 uint nSamples,
211 std::vector<IndexType<MeshType>>& birthVertices,
212 bool onlySelected = false,
213 RandomConfig config = std::monostate())
214{
215 using PointType = VertexType<MeshType>::PositionType;
216 using IndexType = IndexType<MeshType>;
217
218 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
219 std::size_t vn =
220 onlySelected ? vertexSelectionCount(m) : m.vertexCount();
221
222 if (nSamples >= vn) {
223 return allVerticesPointSampling(m, birthVertices, onlySelected);
224 }
225
226 PointSampler<PointType> ps;
227
228 // Reserve space in the sampler and birthVertices vectors
229 ps.reserve(vn);
230 birthVertices.reserve(vn);
231 birthVertices.clear();
232
233 std::uniform_int_distribution<IndexType> dist(
234 0, m.vertexContainerSize() - 1);
235
236 std::vector<bool> visited(m.vertexContainerSize(), false);
237 uint nVisited = 0;
238
239 while (nVisited < nSamples) {
240 IndexType vi = dist(gen);
241 if (!m.vertex(vi).deleted() && !visited[vi] &&
242 (!onlySelected || m.vertex(vi).selected())) {
243 visited[vi] = true;
244 nVisited++;
245 ps.add(m.vertex(vi));
246 birthVertices.push_back(vi);
247 }
248 }
249
250 return ps;
251 });
252}

658template<FaceMeshConcept MeshType>
659auto vertexAreaWeightedPointSampling(
660 const MeshType& m,
661 uint nSamples,
662 RandomConfig config = std::monostate())
663{
664 using ScalarType = ScalarType<MeshType>;
665
666 std::vector<ScalarType> weights(m.vertexContainerSize(), 0);
667 std::vector<uint> cnt(m.vertexContainerSize(), 0);
668
669 // for each vertex, store in weights the adjacent faces area and their
670 // number
671 for (auto&& f : m.faces()) {
672 ScalarType area = faceArea(f);
673 for (auto&& v : f.vertices()) {
674 weights[v.index()] += area;
675 cnt[v.index()]++;
676 }
677 }
678
679 // divide each area sum by the number of adjacent faces
680 for (uint i = 0; i < weights.size(); i++) {
681 if (cnt[i] > 0)
682 weights[i] /= cnt[i];
683 }
684
685 // use these weights to create a sapler
686 return vertexWeightedPointSampling(m, weights, nSamples, config);
687}

735template<FaceMeshConcept MeshType>
736auto montecarloPointSampling(
737 const MeshType& m,
738 uint nSamples,
739 std::vector<typename MeshType::IndexType>& birthFaces,
740 RandomConfig config = std::monostate())
741{
742 using PointType = VertexType<MeshType>::PositionType;
743 using ScalarType = PointType::ScalarType;
744 using FaceType = ConstFaceType<MeshType>;
745 using Interval = std::pair<ScalarType, FaceType>;
746
747 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
748 PointSampler<PointType> sampler;
749
750 // Reserve space in the sampler and birthFaces vectors
751 sampler.reserve(nSamples);
752 birthFaces.reserve(nSamples);
753 birthFaces.clear();
754
755 std::uniform_real_distribution<ScalarType> dist(0, 1);
756
757 std::vector<Interval> intervals(m.faceCount());
758 uint i = 0;
759 ScalarType area = 0;
760 for (auto&& f : m.faces()) {
761 area += faceArea(f);
762 intervals[i] = std::make_pair(area, f);
763 i++;
764 }
765
766 ScalarType meshArea = intervals.back().first;
767 for (uint i = 0; i < nSamples; i++) {
768 ScalarType val = meshArea * dist(gen);
769 // lower_bound returns the furthermost iterator i in [first, last)
770 // such that, for every iterator j in [first, i), *j < value. E.g.
771 // An iterator pointing to the first element "not less than" val, or
772 // end() if every element is less than val.
773 auto it =
774 std::ranges::lower_bound(intervals, val, {}, &Interval::first);
775
776 sampler.add(
777 it->second,
778 randomPolygonBarycentricCoordinate<ScalarType>(
779 it->second.vertexCount(), gen));
780 birthFaces.push_back(it->second.index());
781 }
782
783 return sampler;
784 });
785}

813template<FaceMeshConcept MeshType>
814auto stratifiedMontecarloPointSampling(
815 const MeshType& m,
816 uint nSamples,
817 RandomConfig config = std::monostate())
818{
819 using PointType = VertexType<MeshType>::PositionType;
820 using ScalarType = PointType::ScalarType;
821
822 PointSampler<PointType> ps;
823
824 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
825 double area = surfaceArea(m);
826 double samplePerAreaUnit = nSamples / area;
827 // Montecarlo sampling.
828 double floatSampleNum = 0.0;
829
830 for (auto&& f : m.faces()) {
831 // compute # samples in the current face (taking into account of the
832 // remainders)
833 floatSampleNum += faceArea(f) * samplePerAreaUnit;
834 int faceSampleNum = (int) floatSampleNum;
835 // for every sample p_i in T...
836 for (int i = 0; i < faceSampleNum; i++)
837 ps.add(
838 f,
839 randomPolygonBarycentricCoordinate<ScalarType>(
840 f.vertexCount(), gen));
841 floatSampleNum -= (double) faceSampleNum;
842 }
843
844 return ps;
845 });
846}

921template<FaceMeshConcept MeshType>
922auto vertexWeightedMontecarloPointSampling(
923 const MeshType& m,
924 Range auto&& weights,
925 uint nSamples,
926 double variance,
927 RandomConfig config = std::monostate())
928{
929 using PointType = VertexType<MeshType>::PositionType;
930 using MeshScalar = PointType::ScalarType;
931
932 assert(std::ranges::size(weights) == m.vertexContainerSize());
933
934 // lambda function to compute radius weighted area of a face
935 auto weightedArea = [](auto&& f,
936 const MeshType& m,
937 const std::vector<MeshScalar>& r) -> MeshScalar {
938 MeshScalar averageQ = 0;
939 for (uint i = 0; i < f.vertexCount(); ++i)
940 averageQ += r[f.vertexIndex(i)];
941
942 averageQ /= f.vertexCount();
943 return averageQ * averageQ * faceArea(f);
944 };
945
946 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
947 PointSampler<PointType> ps;
948
949 auto radius = vertexRadiusFromWeights(m, weights, 1.0, variance, true);
950
951 MeshScalar wArea = 0;
952 for (auto&& f : m.faces())
953 wArea += weightedArea(f, m, radius);
954
955 MeshScalar samplePerAreaUnit = nSamples / wArea;
956 // Montecarlo sampling.
957 double floatSampleNum = 0.0;
958 for (auto&& f : m.faces()) {
959 // compute # samples in the current face (taking into account of the
960 // remainders)
961 floatSampleNum += weightedArea(f, m, radius) * samplePerAreaUnit;
962 uint faceSampleNum = (uint) floatSampleNum;
963
964 // for every sample p_i in T...
965 for (uint i = 0; i < faceSampleNum; i++)
966 ps.add(
967 f,
968 randomPolygonBarycentricCoordinate<MeshScalar>(
969 f.vertexCount(), gen));
970
971 floatSampleNum -= (double) faceSampleNum;
972 }
973
974 return ps;
975 });
976}
