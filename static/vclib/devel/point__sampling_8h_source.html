57template<MeshConcept MeshType>
58auto allVerticesPointSampling(
59 const MeshType& m,
60 std::vector<uint>& birthVertices,
61 bool onlySelected = false)
62{
63 using VertexType = MeshType::VertexType;
64 using PointType = VertexType::PositionType;
65
66 PointSampler<PointType> sampler;
67
68 // Determine the number of vertices to sample
69 uint n = onlySelected ? vertexSelectionCount(m) : m.vertexCount();
70
71 // Reserve space in the sampler and birthVertices vectors
72 sampler.reserve(n);
73 birthVertices.reserve(n);
74 birthVertices.clear();
75
76 // Loop through all the vertices in the mesh and add them to the sampler if
77 // they are selected
78 for (const VertexType& v : m.vertices()) {
79 if (!onlySelected || v.selected()) {
80 sampler.add(v);
81 birthVertices.push_back(m.index(v));
82 }
83 }
84 return sampler;
85}

209template<MeshConcept MeshType>
210auto vertexUniformPointSampling(
211 const MeshType& m,
212 uint nSamples,
213 std::vector<uint>& birthVertices,
214 bool onlySelected = false,
215 RandomConfig config = std::monostate())
216{
217 using PointType = MeshType::VertexType::PositionType;
218
219 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
220 uint vn = onlySelected ? vertexSelectionCount(m) : m.vertexCount();
221
222 if (nSamples >= vn) {
223 return allVerticesPointSampling(m, birthVertices, onlySelected);
224 }
225
226 PointSampler<PointType> ps;
227
228 // Reserve space in the sampler and birthVertices vectors
229 ps.reserve(vn);
230 birthVertices.reserve(vn);
231 birthVertices.clear();
232
233 std::uniform_int_distribution<uint> dist(
234 0, m.vertexContainerSize() - 1);
235
236 std::vector<bool> visited(m.vertexContainerSize(), false);
237 uint nVisited = 0;
238
239 while (nVisited < nSamples) {
240 uint vi = dist(gen);
241 if (!m.vertex(vi).deleted() && !visited[vi] &&
242 (!onlySelected || m.vertex(vi).selected())) {
243 visited[vi] = true;
244 nVisited++;
245 ps.add(m.vertex(vi));
246 birthVertices.push_back(vi);
247 }
248 }
249
250 return ps;
251 });
252}

656template<FaceMeshConcept MeshType>
657auto vertexAreaWeightedPointSampling(
658 const MeshType& m,
659 uint nSamples,
660 RandomConfig config = std::monostate())
661{
662 using VertexType = MeshType::VertexType;
663 using ScalarType = VertexType::ScalarType;
664 using FaceType = MeshType::FaceType;
665
666 std::vector<ScalarType> weights(m.vertexContainerSize(), 0);
667 std::vector<uint> cnt(m.vertexContainerSize(), 0);
668
669 // for each vertex, store in weights the adjacent faces area and their
670 // number
671 for (const FaceType& f : m.faces()) {
672 ScalarType area = faceArea(f);
673 for (const VertexType* v : f.vertices()) {
674 weights[m.index(v)] += area;
675 cnt[m.index(v)]++;
676 }
677 }
678
679 // divide each area sum by the number of adjacent faces
680 for (uint i = 0; i < weights.size(); i++) {
681 if (cnt[i] > 0)
682 weights[i] /= cnt[i];
683 }
684
685 // use these weights to create a sapler
686 return vertexWeightedPointSampling(m, weights, nSamples, config);
687}

737template<FaceMeshConcept MeshType>
738auto montecarloPointSampling(
739 const MeshType& m,
740 uint nSamples,
741 std::vector<uint>& birthFaces,
742 RandomConfig config = std::monostate())
743{
744 using VertexType = MeshType::VertexType;
745 using PointType = VertexType::PositionType;
746 using ScalarType = PointType::ScalarType;
747 using FaceType = MeshType::FaceType;
748 using Interval = std::pair<ScalarType, const FaceType*>;
749
750 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
751 FirstElementPairComparator<Interval> comparator;
752 PointSampler<PointType> sampler;
753
754 // Reserve space in the sampler and birthFaces vectors
755 sampler.reserve(nSamples);
756 birthFaces.reserve(nSamples);
757 birthFaces.clear();
758
759 std::uniform_real_distribution<ScalarType> dist(0, 1);
760
761 std::vector<Interval> intervals(m.faceCount());
762 uint i = 0;
763 ScalarType area = 0;
764 for (const FaceType& f : m.faces()) {
765 area += faceArea(f);
766 intervals[i] = std::make_pair(area, &f);
767 i++;
768 }
769
770 ScalarType meshArea = intervals.back().first;
771 for (uint i = 0; i < nSamples; i++) {
772 ScalarType val = meshArea * dist(gen);
773 // lower_bound returns the furthermost iterator i in [first, last)
774 // such that, for every iterator j in [first, i), *j < value. E.g.
775 // An iterator pointing to the first element "not less than" val, or
776 // end() if every element is less than val.
777 typename std::vector<Interval>::iterator it = std::lower_bound(
778 intervals.begin(),
779 intervals.end(),
780 std::make_pair(val, nullptr),
781 comparator);
782
783 sampler.add(
784 *it->second,
785 randomPolygonBarycentricCoordinate<ScalarType>(
786 it->second->vertexCount(), gen));
787 birthFaces.push_back(it->second->index());
788 }
789
790 return sampler;
791 });
792}

820template<FaceMeshConcept MeshType>
821auto stratifiedMontecarloPointSampling(
822 const MeshType& m,
823 uint nSamples,
824 RandomConfig config = std::monostate())
825{
826 using FaceType = MeshType::FaceType;
827 using PointType = MeshType::VertexType::PositionType;
828 using ScalarType = PointType::ScalarType;
829
830 PointSampler<PointType> ps;
831
832 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
833 double area = surfaceArea(m);
834 double samplePerAreaUnit = nSamples / area;
835 // Montecarlo sampling.
836 double floatSampleNum = 0.0;
837
838 for (const FaceType& f : m.faces()) {
839 // compute # samples in the current face (taking into account of the
840 // remainders)
841 floatSampleNum += faceArea(f) * samplePerAreaUnit;
842 int faceSampleNum = (int) floatSampleNum;
843 // for every sample p_i in T...
844 for (int i = 0; i < faceSampleNum; i++)
845 ps.add(
846 f,
847 randomPolygonBarycentricCoordinate<ScalarType>(
848 f.vertexCount(), gen));
849 floatSampleNum -= (double) faceSampleNum;
850 }
851
852 return ps;
853 });
854}

912template<FaceMeshConcept MeshType, typename ScalarType>
913auto vertexWeightedMontecarloPointSampling(
914 const MeshType& m,
915 const std::vector<ScalarType>& weights,
916 uint nSamples,
917 double variance,
918 RandomConfig config = std::monostate())
919{
920 using PointType = MeshType::VertexType::PositionType;
921 using FaceType = MeshType::FaceType;
922
923 // lambda function to compute radius weighted area of a face
924 auto weightedArea = [](const FaceType& f,
925 const MeshType& m,
926 const std::vector<ScalarType>& r) -> ScalarType {
927 ScalarType averageQ = 0;
928 for (uint i = 0; i < f.vertexCount(); ++i)
929 averageQ += r[m.index(f.vertex(i))];
930
931 averageQ /= f.vertexCount();
932 return averageQ * averageQ * faceArea(f);
933 };
934
935 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
936 PointSampler<PointType> ps;
937
938 std::vector<ScalarType> radius =
939 vertexRadiusFromWeights(m, weights, 1.0, variance, true);
940
941 ScalarType wArea = 0;
942 for (const FaceType& f : m.faces())
943 wArea += weightedArea(f, m, radius);
944
945 ScalarType samplePerAreaUnit = nSamples / wArea;
946 // Montecarlo sampling.
947 double floatSampleNum = 0.0;
948 for (const FaceType& f : m.faces()) {
949 // compute # samples in the current face (taking into account of the
950 // remainders)
951 floatSampleNum += weightedArea(f, m, radius) * samplePerAreaUnit;
952 uint faceSampleNum = (uint) floatSampleNum;
953
954 // for every sample p_i in T...
955 for (uint i = 0; i < faceSampleNum; i++)
956 ps.add(
957 f,
958 randomPolygonBarycentricCoordinate<ScalarType>(
959 f.vertexCount(), gen));
960
961 floatSampleNum -= (double) faceSampleNum;
962 }
963
964 return ps;
965 });
966}
