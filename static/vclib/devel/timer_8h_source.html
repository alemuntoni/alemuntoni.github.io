119 std::string toString() const
120 {
121 double secs = delay();
122 int mins = static_cast<int>(secs) / 60;
123 if (mins == 0) {
124 return std::format("[{:.3f} secs]\t{}", secs, mCaption);
125 }
126 secs -= mins * 60;
127 int hours = mins / 60;
128 if (hours == 0) {
129 return std::format(
130 "[{} mins; {:.3f} secs]\t{}", mins, secs, mCaption);
131 }
132 mins -= hours * 60;
133 return std::format(
134 "[{} hours; {} mins; {:.3f} secs]\t{}",
135 hours,
136 mins,
137 secs,
138 mCaption);
139 }
