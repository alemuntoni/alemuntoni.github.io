460 void fillTriangleIndices(const FaceMeshConcept auto& mesh, auto* buffer)
461 {
462 using MeshType = std::decay_t<decltype(mesh)>;
463 using FaceType = ConstFaceType<MeshType>;
464
465 triangulatedFaceVertexIndicesToBuffer(
466 mesh, buffer, mIndexMap, MatrixStorageType::ROW_MAJOR, mNumTris);
467 // Update mNumTris to the actual triangle count produced by earCut.
468 // For degenerate faces, earCut may produce fewer than vertexCount()-2
469 // triangles, making the pre-triangulation estimate from
470 // triangulatedFaceCount() too large.
471 mNumTris = mIndexMap.triangleCount();
472 replaceTriangulatedFaceVertexIndicesByVertexDuplicationToBuffer(
473 mesh, mVertsToDuplicate, mFacesToReassign, mIndexMap, buffer);
474
475 // comparator of faces
476 // ordering first by per-vertex material index (if available),
477 // then by per-face material index (if available)
478 auto faceComp = [&](FaceType f1, FaceType f2) {
479 if constexpr (HasVertexMaterialIndex<MeshType>) {
480 if (mesh.isVertexMaterialIndexAvailable()) {
481 uint id1 = f1.vertex(0).materialIndex();
482 uint id2 = f2.vertex(0).materialIndex();
483 if (id1 != id2) { // do not return true if equal
484 return id1 < id2;
485 }
486 }
487 }
488 if constexpr (HasFaceMaterialIndex<MeshType>) {
489 if (mesh.isFaceMaterialIndexAvailable()) {
490 uint id1 = f1.materialIndex();
491 uint id2 = f2.materialIndex();
492 if (id1 != id2) { // do not return true if equal
493 return id1 < id2;
494 }
495 }
496 }
497
498 // if both per-vertex and per-face material indices are equal, sort
499 // by face index to have a stable sorting
500 return f1.index() < f2.index();
501 };
502
503 // permute the triangulated face vertex indices according to the face
504 // sorting by material ID (the function also edits the index map from
505 // polygonal faces (which still refers to the mesh ones) to the
506 // triangulated faces (which refers to the sorted triangles))
507 permuteFaceVertexIndicesByFunctionToBuffer(
508 mesh, buffer, faceComp, mIndexMap);
509
510 fillChunks(mesh);
511 }

1021 template<MeshConcept MeshType>
1022 void updateAuxiliaryData(
1023 const MeshType& mesh,
1024 MeshRenderInfo::BuffersBitSet btu)
1025 {
1026 using enum MRI::Buffers;
1027
1028 if (btu[toUnderlying(VERTICES)] || btu[toUnderlying(WEDGE_TEXCOORDS)] ||
1029 btu[toUnderlying(TRIANGLES)]) {
1030 mVertWedgeMap.clear();
1031 mVertsToDuplicate.clear();
1032 mFacesToReassign.clear();
1033
1034 if constexpr (HasFaceWedgeTexCoords<MeshType>) {
1035 if (mesh.isFaceWedgeTexCoordsAvailable()) {
1036 verticesToDuplicateByWedgeTexCoordsCount(
1037 mesh,
1038 mVertWedgeMap,
1039 mVertsToDuplicate,
1040 mFacesToReassign);
1041 }
1042 }
1043
1044 mNumVerts = mesh.vertexCount() + mVertsToDuplicate.size();
1045 }
1046
1047 if constexpr (HasFaces<MeshType>) {
1048 if (btu[toUnderlying(TRIANGLES)])
1049 mNumTris = triangulatedFaceCount(mesh);
1050 if (btu[toUnderlying(WIREFRAME)])
1051 nWireframeLines = faceVertexReferencesCount(mesh);
1052 }
1053
1054 if constexpr (HasEdges<MeshType>) {
1055 if (btu[toUnderlying(EDGES)])
1056 mNumEdges = mesh.edgeCount();
1057 }
1058 }

1253 void fillChunks(const FaceMeshConcept auto& mesh)
1254 {
1255 using MeshType = std::decay_t<decltype(mesh)>;
1256
1257 mMaterialChunks.clear();
1258 if (mIndexMap.triangleCount() == 0)
1259 return;
1260
1261 uint first = 0;
1262 uint n = 0;
1263
1264 ushort currentVertMatID = USHORT_NULL;
1265 ushort currentFaceMatID = USHORT_NULL;
1266
1267 for (uint i = 0; i < mIndexMap.triangleCount(); ++i) {
1268 uint fIndex = mIndexMap.polygon(i);
1269
1270 ushort vMatId = USHORT_NULL;
1271 if constexpr (HasVertexMaterialIndex<MeshType>) {
1272 if (mesh.isVertexMaterialIndexAvailable()) {
1273 vMatId = mesh.face(fIndex).vertex(0).materialIndex();
1274 }
1275 }
1276
1277 ushort fMatId = USHORT_NULL;
1278 if constexpr (HasFaceMaterialIndex<MeshType>) {
1279 if (mesh.isFaceMaterialIndexAvailable()) {
1280 fMatId = mesh.face(fIndex).materialIndex();
1281 }
1282 }
1283
1284 if (i == 0) {
1285 currentVertMatID = vMatId;
1286 currentFaceMatID = fMatId;
1287 }
1288 else if (vMatId != currentVertMatID || fMatId != currentFaceMatID) {
1289 mMaterialChunks.push_back(
1290 {first, n, currentVertMatID, currentFaceMatID});
1291 first += n;
1292 n = 0;
1293 currentVertMatID = vMatId;
1294 currentFaceMatID = fMatId;
1295 }
1296
1297 n++;
1298 }
1299
1300 if (n > 0) {
1301 mMaterialChunks.push_back(
1302 {first, n, currentVertMatID, currentFaceMatID});
1303 }
1304 }
1305};
