457 void fillTriangleIndices(const FaceMeshConcept auto& mesh, auto* buffer)
458 {
459 using MeshType = std::decay_t<decltype(mesh)>;
460 using FaceType = MeshType::FaceType;
461
462 triangulatedFaceVertexIndicesToBuffer(
463 mesh, buffer, mIndexMap, MatrixStorageType::ROW_MAJOR, mNumTris);
464 // Update mNumTris to the actual triangle count produced by earCut.
465 // For degenerate faces, earCut may produce fewer than vertexCount()-2
466 // triangles, making the pre-triangulation estimate from
467 // triangulatedFaceCount() too large.
468 mNumTris = mIndexMap.triangleCount();
469 replaceTriangulatedFaceVertexIndicesByVertexDuplicationToBuffer(
470 mesh, mVertsToDuplicate, mFacesToReassign, mIndexMap, buffer);
471
472 // comparator of faces
473 // ordering first by per-vertex material index (if available),
474 // then by per-face material index (if available)
475 auto faceComp = [&](const FaceType& f1, const FaceType& f2) {
476 if constexpr (HasPerVertexMaterialIndex<MeshType>) {
477 if (isPerVertexMaterialIndexAvailable(mesh)) {
478 uint id1 = f1.vertex(0)->materialIndex();
479 uint id2 = f2.vertex(0)->materialIndex();
480 if (id1 != id2) { // do not return true if equal
481 return id1 < id2;
482 }
483 }
484 }
485 if constexpr (HasPerFaceMaterialIndex<MeshType>) {
486 if (isPerFaceMaterialIndexAvailable(mesh)) {
487 uint id1 = f1.materialIndex();
488 uint id2 = f2.materialIndex();
489 if (id1 != id2) { // do not return true if equal
490 return id1 < id2;
491 }
492 }
493 }
494
495 // if both per-vertex and per-face material indices are equal, sort
496 // by face index to have a stable sorting
497 return f1.index() < f2.index();
498 };
499
500 // permute the triangulated face vertex indices according to the face
501 // sorting by material ID (the function also edits the index map from
502 // polygonal faces (which still refers to the mesh ones) to the
503 // triangulated faces (which refers to the sorted triangles))
504 permuteFaceVertexIndicesByFunctionToBuffer(
505 mesh, buffer, faceComp, mIndexMap);
506
507 fillChuncks(mesh);
508 }

989 void updateAuxiliaryData(
990 const MeshConcept auto& mesh,
991 MeshRenderInfo::BuffersBitSet btu)
992 {
993 using MeshType = MeshRenderDerived::MeshType;
994 using enum MRI::Buffers;
995
996 if (btu[toUnderlying(VERTICES)] || btu[toUnderlying(WEDGE_TEXCOORDS)] ||
997 btu[toUnderlying(TRIANGLES)]) {
998 mVertWedgeMap.clear();
999 mVertsToDuplicate.clear();
1000 mFacesToReassign.clear();
1001
1002 if constexpr (HasPerFaceWedgeTexCoords<MeshType>) {
1003 if (mesh.isPerFaceWedgeTexCoordsEnabled()) {
1004 verticesToDuplicateByWedgeTexCoordsCount(
1005 mesh,
1006 mVertWedgeMap,
1007 mVertsToDuplicate,
1008 mFacesToReassign);
1009 }
1010 }
1011
1012 mNumVerts = mesh.vertexCount() + mVertsToDuplicate.size();
1013 }
1014
1015 if constexpr (HasFaces<MeshType>) {
1016 if (btu[toUnderlying(TRIANGLES)])
1017 mNumTris = triangulatedFaceCount(mesh);
1018 if (btu[toUnderlying(WIREFRAME)])
1019 nWireframeLines = faceVertexReferencesCount(mesh);
1020 }
1021
1022 if constexpr (HasEdges<MeshType>) {
1023 if (btu[toUnderlying(EDGES)])
1024 mNumEdges = mesh.edgeCount();
1025 }
1026 }

1216 void fillChuncks(const FaceMeshConcept auto& mesh)
1217 {
1218 using MeshType = std::decay_t<decltype(mesh)>;
1219
1220 mMaterialChunks.clear();
1221
1222 uint first = 0;
1223 uint n = 0;
1224
1225 uint currentVertMatID = UINT_NULL;
1226 uint currentFaceMatID = UINT_NULL;
1227
1228 for (uint i = 0; i < mIndexMap.triangleCount(); ++i) {
1229 uint fIndex = mIndexMap.polygon(i);
1230
1231 if constexpr (HasPerVertexMaterialIndex<MeshType>) {
1232 if (isPerVertexMaterialIndexAvailable(mesh)) {
1233 uint mId = mesh.face(fIndex).vertex(0)->materialIndex();
1234 if (mId != currentVertMatID && n != 0) {
1235 if (currentVertMatID != UINT_NULL) {
1236 mMaterialChunks.push_back(
1237 {first, n, currentVertMatID, currentFaceMatID});
1238 first += n;
1239 n = 0;
1240 }
1241 currentVertMatID = mId;
1242 }
1243 }
1244 }
1245 if constexpr (HasPerFaceMaterialIndex<MeshType>) {
1246 if (isPerFaceMaterialIndexAvailable(mesh)) {
1247 uint mId = mesh.face(fIndex).materialIndex();
1248 if (mId != currentFaceMatID && n != 0) {
1249 if (currentFaceMatID != UINT_NULL) {
1250 mMaterialChunks.push_back(
1251 {first, n, currentVertMatID, currentFaceMatID});
1252 first += n;
1253 n = 0;
1254 }
1255 currentFaceMatID = mId;
1256 }
1257 }
1258 }
1259
1260 n++;
1261 }
1262
1263 mMaterialChunks.push_back(
1264 {first, n, currentVertMatID, currentFaceMatID});
1265 }
1266};
