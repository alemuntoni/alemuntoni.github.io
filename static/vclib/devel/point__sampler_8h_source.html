203 template<FaceConcept FaceType>
204 void add(const FaceType& f, const std::vector<ScalarType>& barCoords)
205 {
206 assert(f.vertexCount() <= barCoords.size());
207
208 PointT p;
209 for (uint i = 0; i < f.vertexCount(); i++)
210 p += f.vertex(i).position() * barCoords[i];
211
212 mSamples.push_back(p);
213 }

236
237 PointT p;
238 for (uint i = 0; i < f.vertexCount(); i++)
239 p += f.vertex(i).position() * barCoords[i];
240
241 mSamples[i] = p;
242 }

258 template<FaceConcept FaceType>
259 void add(const FaceType& f, const PointT& barCoords)
260 {
261 static_assert(
262 FaceType::VERTEX_COUNT == 3 || FaceType::VERTEX_COUNT == -1);
263 if constexpr (FaceType::VERTEX_COUNT == -1) {
264 assert(f.vertexCount() == 3);
265 }
266
267 PointT p = triangleBarycentricCoordinatePoint(f, barCoords);
268
269 mSamples.push_back(p);
270 }

293 assert(f.vertexCount() == 3);
294 }
295
296 PointT p = triangleBarycentricCoordinatePoint(f, barCoords);
297
298 mSamples[i] = p;

311 template<MeshConcept MeshType>
312 MeshType toMesh() const
313 {
314 MeshType mesh;
315 mesh.reserveVertices(this->size());
316 for (const auto& p : mSamples) {
317 mesh.addVertex(p);
318 }
319 return mesh;
320 }
