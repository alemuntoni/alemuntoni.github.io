36template<FaceConcept FaceType, Point3Concept PointType>
37auto halfSpaceDeterminant(const FaceType& face, const PointType& point)
38{
39 // - if it is a triangle, compute the determinant using the triangle
40 // - if it is a polygon, use its normal to compute the determinant
41 if constexpr (TriangleFaceConcept<FaceType>) {
42 return halfSpaceDeterminant(
43 face.vertex(0).position(),
44 face.vertex(1).position(),
45 face.vertex(2).position(),
46 point);
47 }
48 else {
49 if (face.vertexCount() == 3) {
50 return halfSpaceDeterminant(
51 face.vertex(0).position(),
52 face.vertex(1).position(),
53 face.vertex(2).position(),
54 point);
55 }
56 else {
57 PointType n = faceNormal(face);
58 return n.dot(point - face.vertex(0).position());
59 }
60 }
61}
