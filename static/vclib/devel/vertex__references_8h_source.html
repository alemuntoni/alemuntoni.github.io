275 template<std::ranges::input_range Rng>
276 void setVertices(this auto&& self, Rng&& rng)
277 {
278 if constexpr (N >= 0) {
279 Base::setRefs(self, std::forward<Rng>(rng));
280 }
281 else {
282 if constexpr (std::ranges::sized_range<Rng>) {
283 self.resizeVertices(std::ranges::size(rng));
284 uint i = 0;
285 for (auto&& v : rng) {
286 self.setVertex(i++, v);
287 }
288 }
289 else if constexpr (std::ranges::forward_range<Rng>) {
290 self.resizeVertices(std::ranges::distance(rng));
291 uint i = 0;
292 for (auto&& v : rng) {
293 self.setVertex(i++, v);
294 }
295 }
296 else {
297 self.resizeVertices(0);
298 for (auto&& v : rng) {
299 self.pushVertex(v);
300 }
301 }
302 }
303 }

403 uint indexOfEdge(this auto&& self, IndexType vi1, IndexType vi2)
404 {
405 uint vid = self.indexOfVertex(vi1);
406 if (vid == UINT_NULL) {
407 return UINT_NULL;
408 }
409 else if (self.vertexIndexMod(vid + 1) == vi2) {
410 return vid;
411 }
412 else if (self.vertexIndexMod(static_cast<int>(vid) - 1) == vi2) {
413 int n = self.vertexCount();
414 return ((static_cast<int>(vid) - 1) % n + n) % n;
415 }
416 else {
417 return UINT_NULL;
418 }
419 }
420
429 template<typename V1, typename V2>
430 uint indexOfEdge(this auto&& self, const V1& v1, const V2& v2)
431 requires (!std::unsigned_integral<V1> || !std::unsigned_integral<V2>)
432 {
433 auto toIdx = [](const auto& v) -> IndexType {
434 if constexpr (requires { v.index(); }) {
435 return v.index();
436 }
437 else {
438 return static_cast<IndexType>(v);
439 }
440 };
441 return self.indexOfEdge(toIdx(v1), toIdx(v2));
442 }

588template<int N, std::unsigned_integral Index>
589template<typename OtherElement>
590void VertexReferences<N, Index>::importFrom(
591 this auto&& self,
592 const OtherElement& other,
593 bool importRefs)
594{
595 if (importRefs) {
596 if constexpr (HasVertexReferences<OtherElement>) {
597 using OtherComp =
598 ComponentTypeT<CompId::VERTEX_REFERENCES, OtherElement>;
599 constexpr int OTHER_N = OtherComp::VERTEX_COUNT;
600
601 if constexpr (N > 0) {
602 if constexpr (N == OTHER_N) {
603 for (uint i = 0; i < N; ++i) {
604 self.setVertex(
605 i, Base::castIndex(other.vertexIndex(i)));
606 }
607 }
608 else if constexpr (OTHER_N < 0) {
609 if (other.vertexCount() == N) {
610 for (uint i = 0; i < N; ++i) {
611 self.setVertex(
612 i, Base::castIndex(other.vertexIndex(i)));
613 }
614 }
615 }
616 }
617 else {
618 self.resizeVertices(other.vertexCount());
619 for (uint i = 0; i < other.vertexCount(); ++i) {
620 self.setVertex(i, Base::castIndex(other.vertexIndex(i)));
621 }
622 }
623 }
624 }
