34template<FaceMeshConcept MeshType, Point3Concept PointType>
35MeshType createCone(
36 const PointType& p1,
37 const PointType& p2,
38 auto radiusBottom,
39 auto radiusTop,
40 const uint subdivisions = 36)
41{
42 using PositionType = PositionType<MeshType>;
43 using ScalarType = PositionType::ScalarType;
44
45 uint vn, fn;
46 if (radiusBottom == 0 || radiusTop == 0) {
47 vn = subdivisions + 2;
48 fn = subdivisions * 2;
49 }
50 else {
51 vn = subdivisions * 2 + 2;
52 fn = subdivisions * 4;
53 }
54
55 MeshType mesh;
56
57 mesh.reserveVertices(vn);
58 mesh.reserveFaces(fn);
59
60 PositionType p1Pos = p1.template cast<ScalarType>();
61 PositionType p2Pos = p2.template cast<ScalarType>();
62
63 mesh.addVertex(p1Pos);
64 mesh.addVertex(p2Pos);
65
66 PositionType d = p2Pos - p1Pos;
67
68 PositionType u(1, 0, 0), v(0, 0, -1);
69 ScalarType length = d.norm();
70 if (length > 0) {
71 PositionType n = d / length;
72 n.orthoBase(u, v);
73 }
74
75 uint b1 = 2;
76 uint b2 = 2;
77
78 if (radiusBottom != 0) {
79 for (uint i = 0; i < subdivisions; ++i) {
80 ScalarType a = toRad(i * 360.0 / subdivisions);
81 mesh.addVertex(
82 p1Pos + u * (radiusBottom * std::cos(a)) +
83 v * (radiusBottom * std::sin(a)));
84 }
85 b2 += subdivisions;
86 }
87
88 if (radiusTop != 0) {
89 for (uint i = 0; i < subdivisions; ++i) {
90 ScalarType a = toRad(i * 360.0 / subdivisions);
91 mesh.addVertex(
92 p2Pos + u * (radiusTop * std::cos(a)) +
93 v * (radiusTop * std::sin(a)));
94 }
95 }
96
97 if (radiusBottom != 0) {
98 for (uint i = 0; i < subdivisions; ++i) {
99 mesh.addFace(0, b1 + (i + 1) % subdivisions, b1 + i);
100 }
101 }
102 else {
103 for (uint i = 0; i < subdivisions; ++i) {
104 mesh.addFace(0, b2 + (i + 1) % subdivisions, b2 + i);
105 }
106 }
107
108 if (radiusTop != 0) {
109 for (uint i = 0; i < subdivisions; ++i) {
110 mesh.addFace(1, b2 + i, b2 + (i + 1) % subdivisions);
111 }
112 }
113 else {
114 for (uint i = 0; i < subdivisions; ++i) {
115 mesh.addFace(1, b1 + i, b1 + (i + 1) % subdivisions);
116 }
117 }
118
119 // side triangles
120 if (radiusBottom != 0 && radiusTop != 0) {
121 for (uint i = 0; i < subdivisions; ++i) {
122 mesh.addFace(b1 + i, b2 + (i + 1) % subdivisions, b2 + i);
123 mesh.addFace(
124 b1 + i,
125 b1 + (i + 1) % subdivisions,
126 b2 + (i + 1) % subdivisions);
127 }
128 }
129
130 return mesh;
131}

150template<FaceMeshConcept MeshType>
151MeshType createCone(
152 auto radiusBottom,
153 auto radiusTop,
154 auto height,
155 const uint subdivisions = 36)
156{
157 using PositionType = PositionType<MeshType>;
158 return createCone<MeshType>(
159 PositionType(0, -height / 2.0, 0),
160 PositionType(0, height / 2.0, 0),
161 radiusBottom,
162 radiusTop,
163 subdivisions);
164}
