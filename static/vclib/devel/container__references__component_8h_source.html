145 template<ElemId TARGET_ELEM_ID>
146 void updateReferences(
147 this auto&& self,
148 const std::vector<IndexType>& newIndices)
149 {
150 if constexpr (TARGET_ELEM_ID == REF_ELEM_ID) {
151 for (auto& ref : Base::container(self)) {
152 if (ref != NULL_INDEX) {
153 assert(ref < newIndices.size());
154 ref = newIndices[ref];
155 }
156 }
157 }
158 }

167 template<ElemId TARGET_ELEM_ID>
168 void shiftReferences(this auto&& self, IndexType offset)
169 {
170 if constexpr (TARGET_ELEM_ID == REF_ELEM_ID) {
171 if (offset != 0) {
172 for (auto& ref : Base::container(self)) {
173 if (ref != NULL_INDEX) {
174 ref += offset;
175 }
176 }
177 }
178 }

308 template<std::ranges::input_range Rng>
309 static void setRefs(auto&& self, Rng&& rng)
310 {
311 if constexpr (N >= 0) {
312 if constexpr (std::ranges::sized_range<Rng>) {
313 assert(std::ranges::size(rng) == static_cast<std::size_t>(N));
314 }
315 uint i = 0;
316 for (auto&& item : rng) {
317 assert(i < static_cast<uint>(N));
318 setRef(self, i++, item);
319 }
320 assert(i == static_cast<uint>(N));
321 }
322 else if constexpr (!TTVC) {
323 if constexpr (std::ranges::sized_range<Rng>) {
324 resizeRefs(self, std::ranges::size(rng));
325 uint i = 0;
326 for (auto&& item : rng) {
327 setRef(self, i++, item);
328 }
329 }
330 else if constexpr (std::ranges::forward_range<Rng>) {
331 resizeRefs(self, std::ranges::distance(rng));
332 uint i = 0;
333 for (auto&& item : rng) {
334 setRef(self, i++, item);
335 }
336 }
337 else {
338 clearRefs(self);
339 for (auto&& item : rng) {
340 pushRef(self, item);
341 }
342 }
343 }
344 else {
345 uint i = 0;
346 for (auto&& item : rng) {
347 assert(i < refCount(self));
348 setRef(self, i++, item);
349 }
350 assert(i == refCount(self));
351 }
352 }

360 template<typename... Args>

362 requires (sizeof...(Args) > 1 && (sizeof...(Args) == N || N < 0))
363 {
364 if constexpr (N < 0 && !TTVC) {
365 resizeRefs(self, sizeof...(Args));
366 }
367 uint i = 0;
368 (setRef(self, i++, args), ...);
