38auto collinearityTest(
39 const PointType& p0,

41 const PointType& p2)
42{
43 return (
44 (p1.x() - p0.x()) * (p2.y() - p0.y()) -

80template<Point2IteratorConcept Iterator>
81bool isCounterClockWise(Iterator begin, Iterator end)
82{
83 using PointType = std::decay_t<decltype(*begin)>;
84 using ScalarType = PointType::ScalarType;
85
86 if (begin == end)
87 return false;
88
89 auto latest = begin;
90 auto it = begin;
91 ScalarType sum = 0;
92 for (++it; it != end; ++it) {
93 sum += (it->x() - latest->x()) * (it->y() + latest->y());
94 latest = it;
95 }
96
97 sum += (latest->x() - begin->x()) * (latest->y() + begin->y());
98
99 return sum < 0;
100}

119template<Point2IteratorConcept Iterator>
120void sortConvexPolygonVertices(Iterator begin, Iterator end)
121{
122 using PointType = std::decay_t<decltype(*begin)>;
123
124 // Find the point with the lowest x-coordinate
125 PointType minPoint = *std::min_element(
126 begin, end, [](const PointType& a, const PointType& b) {
127 return a.y() < b.y();
128 });
129
130 // Sort the points based on the angle with respect to minPoint
131 std::sort(begin, end, [minPoint](const PointType& a, const PointType& b) {
132 double angleA = std::atan2(a.y() - minPoint.y(), a.x() - minPoint.x());
133 double angleB = std::atan2(b.y() - minPoint.y(), b.x() - minPoint.x());
134 return angleA < angleB;
135 });
136}
