81 template<VertFragProgram PROGRAM>
82 bgfx::ProgramHandle getProgram()
83 {
84 uint p = toUnderlying(PROGRAM);
85 if (!bgfx::isValid(mVFPrograms[p])) {
86 mVFPrograms[p] = vcl::createProgram(
87 vcl::loadShader(
88 VertFragLoader<PROGRAM>::vertexShader(mRenderType)),
89 vcl::loadShader(
90 VertFragLoader<PROGRAM>::fragmentShader(mRenderType)));
91 assert(bgfx::isValid(mVFPrograms[p]));
92 }
93 return mVFPrograms[p];
94 }

160 template<size_t... Is>
161 bgfx::ProgramHandle getProgramImpl(
162 VertFragProgram p,
163 std::index_sequence<Is...>)
164 {
165 using GetProgramFn = bgfx::ProgramHandle (ProgramManager::*)();
166 static constexpr std::array<GetProgramFn, sizeof...(Is)> funcs = {
167 &ProgramManager::getProgram<static_cast<VertFragProgram>(Is)>...};
168 return (this->*(funcs[toUnderlying(p)]))();
169 }
170
182 template<size_t... Is>
183 bgfx::ProgramHandle getComputeProgramImpl(
184 ComputeProgram p,

186 {
187 using GetProgramFn = bgfx::ProgramHandle (ProgramManager::*)();
188 static constexpr std::array<GetProgramFn, sizeof...(Is)> funcs = {
189 &ProgramManager::getComputeProgram<static_cast<ComputeProgram>(
190 Is)>...};
191 return (this->*(funcs[toUnderlying(p)]))();
192 }
