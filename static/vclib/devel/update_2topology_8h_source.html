33template<uint ELEM_ID, FaceMeshConcept MeshType>
34void clearPerElementAdjacentFaces(MeshType& mesh)
35{
36 requirePerElementComponent<ELEM_ID, CompId::ADJACENT_FACES>(mesh);
37
38 using AdjacentFacesType = comp::ComponentTypeFromID<
39 CompId::ADJACENT_FACES,
40 typename MeshType::template ElementType<ELEM_ID>::Components>;
41
42 for (auto& e : mesh.template elements<ELEM_ID>()) {
43 if constexpr (comp::IsTiedToVertexCount<AdjacentFacesType>) {
44 for (uint i = 0; i < e.adjFaceCount(); ++i) {
45 e.setAdjFace(i, nullptr);
46 }
47 }
48 else {
49 e.clearAdjFaces();
50 }
51 }
52}

185template<MeshConcept MeshType>
186void updatePerVertexAdjacentVertices(
187 MeshType& mesh,
188 bool includeOnlyFaces = false)
189{
190 clearPerVertexAdjacentVertices(mesh);
191
192 using VertexType = MeshType::VertexType;
193
194 if constexpr (FaceMeshConcept<MeshType>) {
195 // vector that contains edges sorted trough unordered vertex pointers
196 // it contains clusters of "same" edges, but each one of them has its
197 // face pointer note that in case on non-manifold mesh, clusters may be
198 // of size
199 // >= 2
200 std::vector<MeshEdgeUtil<MeshType>> vec =
201 fillAndSortMeshEdgeUtilVector(mesh);
202
203 // store the last pair of vertices
204 VertexType* v1 = nullptr;
205 VertexType* v2 = nullptr;
206 for (uint i = 0; i < vec.size(); ++i) {
207 // if this pair is different from the last pair
208 if (vec[i].v[0] != v1 || vec[i].v[1] != v2) {
209 // update last pair
210 v1 = vec[i].v[0];
211 v2 = vec[i].v[1];
212 v1->pushAdjVertex(v2); // set the pair as adjacent
213 v2->pushAdjVertex(v1);
214 }
215 }
216 }
217
218 if (!includeOnlyFaces) {
219 if constexpr (EdgeMeshConcept<MeshType>) {
220 using EdgeType = MeshType::EdgeType;
221
222 for (EdgeType& e : mesh.edges()) {
223 // set the pair as adjacent only if not already present
224 if (!e.vertex(0)->containsAdjVertex(e.vertex(1)))
225 e.vertex(0)->pushAdjVertex(e.vertex(1));
226 if (!e.vertex(1)->containsAdjVertex(e.vertex(0)))
227 e.vertex(1)->pushAdjVertex(e.vertex(0));
228 }
229 }
230 }
231}

330template<FaceMeshConcept MeshType>
331void updatePerFaceAdjacentFaces(MeshType& mesh)
332{
333 requirePerFaceAdjacentFaces(mesh);
334
335 // vector that contains edges sorted trough unordered vertex pointers
336 // it contains clusters of "same" edges, but each one of them has its face
337 // pointer. note that in case on non-manifold mesh, clusters may be of size
338 // >= 2
339 std::vector<MeshEdgeUtil<MeshType>> vec =
340 fillAndSortMeshEdgeUtilVector(mesh);
341
342 if (vec.size() > 0) {
343 // in this loop, base will point to the first element of a cluster of
344 // edges
345 // increment of clusters into loop
346 for (auto base = vec.begin(); base != vec.end();) {
347 auto first = base; // remember the first to set adj to the last
348
349 // i and j will increment together, and if i == j, i will be adj to
350 // j, but j will not be adj to i (to manage non manifold edges and
351 // make cyclic adj on the same edge)
352 auto i = base;
353 auto j = i + 1;
354 if (j != vec.end()) {
355 // case of cluster composed of one element. adj of i is nullptr
356 if (*i != *j) {
357 i->f->setAdjFace(i->e, nullptr);
358 }
359 else { // at least two edges in the cluster
360 while (j != vec.end() && *i == *j) {
361 i->f->setAdjFace(i->e, j->f);
362 ++i;
363 ++j;
364 }
365 // i now is the last element that was equal to first
366 i->f->setAdjFace(i->e, first->f);
367 }
368 }
369
370 // j is the first different edge from first (or it is vec.end()!)
371 base = j;
372 }
373 }
374}

420template<FaceMeshConcept MeshType>
421void updatePerFaceAdjacentEdges(MeshType& mesh)
422 requires EdgeMeshConcept<MeshType>
423{
424 using FaceType = MeshType::FaceType;
425
426 using AdjacentEdgesType = comp::ComponentTypeFromID<
427 CompId::ADJACENT_EDGES,
428 typename FaceType::Components>;
429
430 clearPerFaceAdjacentEdges(mesh);
431
432 // vector that contains edges sorted trough unordered vertex pointers
433 std::vector<MeshEdgeUtil<MeshType>> vec =
434 fillAndSortMeshEdgeUtilVector(mesh);
435
436 for (auto& e : mesh.edges()) {
437 MeshEdgeUtil<MeshType> meu(e.vertex(0), e.vertex(1));
438
439 // binary search for the edge in the sorted vector
440 auto it = std::lower_bound(vec.begin(), vec.end(), meu);
441
442 while (it != vec.end() && *it == meu) {
443 auto* f = it->f; // the face adjacent to the edge e
444 if constexpr (comp::IsTiedToVertexCount<AdjacentEdgesType>) {
445 f->setAdjEdges(it->e, &e);
446 }
447 else {
448 f->pushAdjEdge(&e);
449 }
450 ++it;
451 }
452 }
453}

532template<EdgeMeshConcept MeshType>
533void updatePerEdgeAdjacentEdges(MeshType& mesh)
534{
535 using EdgeType = MeshType::EdgeType;
536
537 // for each vertex, save the edges incident in it
538 std::vector<std::list<EdgeType*>> edgeMap(mesh.vertexContainerSize());
539
540 clearPerEdgeAdjacentEdges(mesh);
541
542 for (EdgeType& e : mesh.edges()) {
543 edgeMap[e.vertex(0)->index()].push_back(&e);
544 edgeMap[e.vertex(1)->index()].push_back(&e);
545 }
546
547 for (const std::list<EdgeType*>& edges : edgeMap) {
548 // all the edges in edges are adjacent to each other
549 for (auto it1 = edges.begin(); it1 != edges.end(); ++it1) {
550 for (auto it2 = std::next(it1); it2 != edges.end(); ++it2) {
551 (*it1)->pushAdjEdge(*it2);
552 (*it2)->pushAdjEdge(*it1);
553 }
554 }
555 }
