38template<FaceConcept FaceType, bool CNST = std::is_const_v<FaceType>>
39 requires comp::HasAdjacentFaces<FaceType>
40class EdgeAdjFaceIterator
41{
42public:
43 static constexpr bool IS_CONST =
44 CNST ||
45 std::is_const_v<FaceType> ||
46 std::is_const_v<std::remove_reference_t<
47 decltype(std::declval<FaceType>().parentContainer())>>;
48
49 using FT = std::conditional_t<
50 IS_CONST,
51 vcl::ConstFaceType<FaceType>,
52 vcl::FaceType<FaceType>>;
53
54 using VT = std::conditional_t<
55 IS_CONST,
56 ConstVertexType<FaceType>,
57 VertexType<FaceType>>;
58
59 using difference_type = ptrdiff_t;
60 using iterator_category = std::forward_iterator_tag;
61 using iterator_concept = std::forward_iterator_tag;
62 using value_type = FT;
63 using reference = FT;
64 using pointer = FakePointerWithValue<FT>;
65
66private:
67 FT mCurrent;
68 FT mEnd;
69 VT mV0;
70 VT mV1;
71
72public:
76 EdgeAdjFaceIterator() = default;
77
84 EdgeAdjFaceIterator(FT f, uint edge) :
85 mCurrent(f), mEnd(f), mV0(f.vertex(edge)),
86 mV1(f.vertexMod(edge + 1))
87 {
88 assert(f);
89 assert(edge < f.vertexCount());
90 }
91
95 template<FaceConcept OtherFaceType, bool OtherConst>
96 requires (IS_CONST && !OtherConst)
97 EdgeAdjFaceIterator(
98 const EdgeAdjFaceIterator<OtherFaceType, OtherConst>& oi) :
99 mCurrent(oi.current()), mEnd(oi.end()), mV0(oi.v0()), mV1(oi.v1())
100 {
101 }
102
106 bool operator==(const EdgeAdjFaceIterator& oi) const
107 {
108 return mCurrent == oi.mCurrent && mV0 == oi.mV0 && mV1 == oi.mV1;
109 }
110
114 template<FaceConcept OtherFaceType, bool OtherConst>
115 bool operator==(
116 const EdgeAdjFaceIterator<OtherFaceType, OtherConst>& oi) const
117 {
118 return mCurrent == oi.current() && mV0 == oi.v0() && mV1 == oi.v1();
119 }
120
124 EdgeAdjFaceIterator& operator++()
125 {
126 assert(mCurrent);
127 uint edge = mCurrent.indexOfEdge(mV0, mV1);
128 assert(edge != UINT_NULL);
129 mCurrent = mCurrent.adjFace(edge);
130 if (mCurrent == mEnd || !mCurrent) {
131 mCurrent = FT();
132 mV0 = VT();
133 mV1 = VT();
134 }
135 return *this;
136 }
137
138 EdgeAdjFaceIterator operator++(int)
139 {
140 auto it = *this;
141 ++(*this);
142 return it;
143 }
144
145 reference operator*() const { return mCurrent; }
146
147 pointer operator->() const { return pointer(mCurrent); }
148
149 const FT& current() const { return mCurrent; }
150 const FT& end() const { return mEnd; }
151 const VT& v0() const { return mV0; }
152 const VT& v1() const { return mV1; }
153};
