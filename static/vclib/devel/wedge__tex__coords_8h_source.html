405/* importFrom function */
406
407template<typename Scalar, int N, typename ParentElemType, bool OPT>
408template<typename Element>
409void WedgeTexCoords<Scalar, N, ParentElemType, OPT>::importFrom(
410 const Element& e,
411 bool)
412{
413 if constexpr (HasWedgeTexCoords<Element>) {
414 if (isWedgeTexCoordsAvailableOn(e)) {
415 if constexpr (N > 0) {
416 // same static size
417 if constexpr (N == Element::WEDGE_TEX_COORD_COUNT) {
418 importWedgeTexCoordsFrom(e);
419 }
420 // from dynamic to static, but dynamic size == static size
421 else if constexpr (Element::WEDGE_TEX_COORD_COUNT < 0) {
422 if (e.vertexCount() == N) {
423 importWedgeTexCoordsFrom(e);
424 }
425 }
426 else {
427 // do not import in this case: cannot import from
428 // dynamic size != static size
429 }
430 }
431 else {
432 // from static/dynamic to dynamic size: need to resize
433 // first, then import
434 resize(e.vertexCount());
435 importWedgeTexCoordsFrom(e);
436 }
437 }
438 }
439}
