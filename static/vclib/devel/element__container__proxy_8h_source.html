36template<typename Container>
37class ElementContainerProxy :
38 public std::ranges::view_interface<ElementContainerProxy<Container>>
39{
40 Container* mContainer = nullptr;
41 bool mJumpDeleted = true;
42
43public:
44 using ContainerType = std::remove_cv_t<Container>;
45 using IndexType = typename ContainerType::IndexType;
46 using Components = typename ContainerType::Components;
47 using ElementType = std::conditional_t<
48 std::is_const_v<Container>,
49 typename ContainerType::ConstElementType,
50 typename ContainerType::ElementType>;
51 using Iterator = std::conditional_t<
52 std::is_const_v<Container>,
53 typename ContainerType::ConstIterator,
54 typename ContainerType::Iterator>;
55
56 ElementContainerProxy() = default;
57
58 ElementContainerProxy(Container& container, bool jumpDeleted = true) :
59 mContainer(&container), mJumpDeleted(jumpDeleted)
60 {
61 }
62
63 template<typename>
64 friend class ElementContainerProxy;
65
66 // Implicit conversion from mutable proxy to const proxy
67 template<typename OtherContainerType>
68 requires std::is_same_v<Container, const OtherContainerType>
69 ElementContainerProxy(
70 const ElementContainerProxy<OtherContainerType>& other) :
71 mContainer(other.mContainer), mJumpDeleted(other.mJumpDeleted)
72 {
73 }
74
75 /* Element-level iteration and access */
76
77 Iterator begin() const
78 {
79 return mContainer ? mContainer->begin(mJumpDeleted) : Iterator();
80 }
81
82 Iterator end() const
83 {
84 return mContainer ? mContainer->end(mJumpDeleted) : Iterator();
85 }
86
87 ElementType operator[](IndexType i) const
88 {
89 assert(mContainer != nullptr);
90 return mContainer->element(i);
91 }
92
93 std::size_t size() const
94 {
95 if (!mContainer) {
96 return 0;
97 }
98 return mJumpDeleted ? mContainer->elementCount() :
99 mContainer->elementContainerSize();
100 }
101
102 bool empty() const { return size() == 0; }
103
104 bool isCompact() const
105 {
106 return mContainer && mContainer->isCompact();
107 }
108
109 template<CompId COMP_ID>
110 bool isComponentAvailable() const
111 {
112 return mContainer && mContainer->template isComponentAvailable<COMP_ID>();
113 }
114
115 template<CompId COMP_ID>
116 void enableOptionalComponent() const requires (!std::is_const_v<Container>)
117 {
118 assert(mContainer != nullptr);
119 mContainer->template enableOptionalComponent<COMP_ID>();
120 }
121
122 template<CompId COMP_ID>
123 void disableOptionalComponent() const requires (!std::is_const_v<Container>)
124 {
125 assert(mContainer != nullptr);
126 mContainer->template disableOptionalComponent<COMP_ID>();
127 }
128
129 void enableAllOptionalComponents() const requires (!std::is_const_v<Container>)
130 {
131 assert(mContainer != nullptr);
132 mContainer->enableAllOptionalComponents();
133 }
134
135 void disableAllOptionalComponents() const requires (!std::is_const_v<Container>)
136 {
137 assert(mContainer != nullptr);
138 mContainer->disableAllOptionalComponents();
139 }
140
141 /* Component Views */
142
146 auto positions() const requires comp::HasPosition<Container>
147 {
148 return component<CompId::POSITION>();
149 }
150
154 auto normals() const requires comp::HasNormal<Container>
155 {
156 return component<CompId::NORMAL>();
157 }
158
162 auto colors() const requires comp::HasColor<Container>
163 {
164 return component<CompId::COLOR>();
165 }
166
172 auto vertexIndices() const
173 requires comp::HasStaticVertexReferences<Container>
174 {
175 return component<CompId::VERTEX_REFERENCES>();
176 }
177
181 auto principalCurvatures() const
182 requires comp::HasPrincipalCurvature<Container>
183 {
184 return component<CompId::PRINCIPAL_CURVATURE>();
185 }
186
190 auto qualities() const requires comp::HasQuality<Container>
191 {
192 return component<CompId::QUALITY>();
193 }
194
199 auto texCoords() const requires comp::HasTexCoord<Container>
200 {
201 return component<CompId::TEX_COORD>();
202 }
203
207 auto materialIndices() const requires comp::HasMaterialIndex<Container>
208 {
209 return component<CompId::MATERIAL_INDEX>();
210 }
211
215 auto tangents() const requires comp::HasTangent<Container>
216 {
217 return component<CompId::TANGENT>();
218 }
219
226 template<CompId COMP_ID>
227 requires comp::HasComponent<COMP_ID, Container> &&
228 (comp::IsAttributeComponent<COMP_ID> ||
229 (COMP_ID == CompId::VERTEX_REFERENCES &&
230 comp::HasStaticVertexReferences<Container>))
231 auto component() const
232 {
233 assert(mContainer != nullptr);
234 return ComponentView<COMP_ID, Container>(*mContainer, mJumpDeleted);
235 }
236};
