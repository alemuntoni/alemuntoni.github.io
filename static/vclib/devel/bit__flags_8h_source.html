164 void importFlagsFromVCGFormat(this auto&& self, int f)
165 {
166 self.resetBitFlags();
167 if (f & 0x0010)
168 self.visited() = true;
169 if (f & 0x0020)
170 self.selected() = true;
171 if (f & 0x0100)
172 self.onBorder() = true;
173 }

182 int exportFlagsToVCGFormat(this auto&& self)
183 {
184 int f = 0;
185 if (self.visited())
186 f |= 0x0010;
187 if (self.selected())
188 f |= 0x0020;
189 if (self.onBorder())
190 f |= 0x0100;
191 return f;

209template<typename OtherElement>
210void BitFlags::importFrom(this auto&& self, const OtherElement& other, bool)
211{
212 if constexpr (HasBitFlags<OtherElement>) {
213 const bool wasDeleted = self.deleted();
214 const bool isDeleted = other.deleted();
215 self.resetBitFlags();
216 if constexpr (HasFaceBitFlags<OtherElement>) {
217 self.setDeleted(isDeleted);
218 self.selected() = other.selected();
219 self.visited() = other.visited();
220 self.onBorder() = other.onBorder();
221
222 const uint um =
223 std::min(USER_BIT_COUNT, (uint) OtherElement::USER_BIT_COUNT);
224 for (uint i = 0; i < um; ++i) {
225 self.userBit(i) = other.userBit(i);
226 }
227 }
228 else {
229 self.setUnderlyingBitFlags(other.underlyingBitFlags());
230 self.setDeleted(isDeleted);
231 }
232 if constexpr (CanAdjustContainerElemCount<decltype(self)>) {
233 if (self.isValid()) {
234 self.parentContainer().adjustElemCountOnDeletedChange(
235 wasDeleted, isDeleted);
236 }
237 }
238 }
