151 template<MeshConcept Mesh>
152 MeshInfo(const Mesh& m)
153 {
154 setVertices();
155 setVertexPosition(
156 true,
157 getType<typename VertexType<Mesh>::PositionType::ScalarType>());
158 setVertexBitFlags();
159 if constexpr (HasVertexNormal<Mesh>) {
160 if (m.isVertexNormalAvailable()) {
161 setVertexNormal(
162 true,
163 getType<
164 typename VertexType<Mesh>::NormalType::ScalarType>());
165 }
166 }
167 if constexpr (HasVertexTangent<Mesh>) {
168 if (m.isVertexTangentAvailable()) {
169 setVertexTangent(
170 true,
171 getType<
172 typename VertexType<Mesh>::TangentType::ScalarType>());
173 }
174 }
175 if constexpr (HasVertexColor<Mesh>) {
176 if (m.isVertexColorAvailable()) {
177 setVertexColor(true, PrimitiveType::UCHAR);
178 }
179 }
180 if constexpr (HasVertexQuality<Mesh>) {
181 if (m.isVertexQualityAvailable()) {
182 setVertexQuality(
183 true, getType<typename VertexType<Mesh>::QualityType>());
184 }
185 }
186 if constexpr (HasVertexTexCoord<Mesh>) {
187 if (m.isVertexTexCoordAvailable()) {
188 setVertexTexCoord(
189 true,
190 getType<
191 typename VertexType<Mesh>::TexCoordType::ScalarType>());
192 }
193 }
194 if constexpr (HasVertexMaterialIndex<Mesh>) {
195 if (m.isVertexMaterialIndexAvailable()) {
196 setVertexMaterialIndex(true, PrimitiveType::USHORT);
197 }
198 }
199 if constexpr (HasVertexCustomComponents<Mesh>) {
200 auto names = m.vertexCustomComponentNames();
201 for (auto& name : names) {
202 DataType dt = getType(m.vertexCustomComponentType(name));
203 if (dt != PrimitiveType::NONE) {
204 addVertexCustomComponent(name, dt);
205 }
206 }
207 }
208
209 if constexpr (HasFaces<Mesh>) {
210 setFaces();
211 setFaceVertexReferences();
212 setFaceBitFlags();
213 if (HasTriangles<Mesh>) {
214 setTriangleMesh();
215 }
216 else if (HasQuads<Mesh>) {
217 setQuadMesh();
218 }
219 else {
220 setPolygonMesh();
221 }
222 if constexpr (HasFaceNormal<Mesh>) {
223 if (m.isFaceNormalAvailable()) {
224 setFaceNormal(
225 true,
226 getType<
227 typename FaceType<Mesh>::NormalType::ScalarType>());
228 }
229 }
230 if constexpr (HasFaceColor<Mesh>) {
231 if (m.isFaceColorAvailable()) {
232 setFaceColor(true, PrimitiveType::UCHAR);
233 }
234 }
235 if constexpr (HasFaceQuality<Mesh>) {
236 if (m.isFaceQualityAvailable()) {
237 setFaceQuality(
238 true, getType<typename FaceType<Mesh>::QualityType>());
239 }
240 }
241 if constexpr (HasFaceWedgeTexCoords<Mesh>) {
242 if (m.isFaceWedgeTexCoordsAvailable()) {
243 setFaceWedgeTexCoords(
244 true,
245 getType<typename TexCoordType<
246 WedgeType<FaceType<Mesh>>>::ScalarType>());
247 }
248 }
249 if constexpr (HasFaceMaterialIndex<Mesh>) {
250 if (m.isFaceMaterialIndexAvailable()) {
251 setFaceMaterialIndex(true, PrimitiveType::USHORT);
252 }
253 }
254 if constexpr (HasFaceCustomComponents<Mesh>) {
255 auto names = m.faceCustomComponentNames();
256 for (auto& name : names) {
257 DataType dt = getType(m.faceCustomComponentType(name));
258 if (dt != PrimitiveType::NONE) {
259 addFaceCustomComponent(name, dt);
260 }
261 }
262 }
263 }
264
265 if constexpr (HasEdges<Mesh>) {
266 setEdges();
267 setEdgeVertexReferences();
268 setEdgeBitFlags();
269 if constexpr (HasEdgeColor<Mesh>) {
270 if (m.isEdgeColorAvailable()) {
271 setEdgeColor(true, PrimitiveType::UCHAR);
272 }
273 }
274 }
275
276 if constexpr (HasMaterials<Mesh>) {
277 if (m.materialCount() > 0) {
278 setMaterials(true);
279 }
280 }
281 }

844 [[nodiscard]] MeshInfo intersect(const MeshInfo& info) const
845 {
846 MeshInfo res;
847 for (uint i = 0; i < NUM_ELEMENTS; ++i) {
848 res.mElements[i] = mElements[i] && info.mElements[i];
849 for (uint j = 0; j < NUM_COMPONENTS; ++j) {
850 res.mElemComponents[i][j] =
851 mElemComponents[i][j] && info.mElemComponents[i][j];
852
853 if (res.mElemComponents[i][j]) {
854 res.mElemComponentsType(i, j) = mElemComponentsType(i, j);
855 }
856 }
857 }
858
859 if (mType == info.mType) {
860 res.mType = mType;
861 }
862 res.mElemCustomComponents = mElemCustomComponents;
863
864 return res;
865 }

876 template<typename T>
877 static DataType getType()
878 {
879 if constexpr (std::is_same_v<T, char>)
880 return PrimitiveType::CHAR;
881 if constexpr (std::is_same_v<T, unsigned char>)
882 return PrimitiveType::UCHAR;
883 if constexpr (std::is_same_v<T, short>)
884 return PrimitiveType::SHORT;
885 if constexpr (std::is_same_v<T, unsigned short>)
886 return PrimitiveType::USHORT;
887 if constexpr (std::is_same_v<T, int>)
888 return PrimitiveType::INT;
889 if constexpr (std::is_same_v<T, uint>)
890 return PrimitiveType::UINT;
891 if constexpr (std::is_integral_v<T>)
892 return PrimitiveType::INT; // fallback to int
893 if constexpr (std::is_same_v<T, float>)
894 return PrimitiveType::FLOAT;
895 if constexpr (std::is_same_v<T, double>)
896 return PrimitiveType::DOUBLE;
897 if constexpr (std::is_floating_point_v<T>)
898 return PrimitiveType::FLOAT; // fallback to float
899 return PrimitiveType::NONE;
900 }
901
902 static DataType getType(std::type_index ti)
903 {
904 if (ti == typeid(char))
905 return PrimitiveType::CHAR;
906 if (ti == typeid(unsigned char))
907 return PrimitiveType::UCHAR;
908 if (ti == typeid(short))
909 return PrimitiveType::SHORT;
910 if (ti == typeid(unsigned short))
911 return PrimitiveType::USHORT;
912 if (ti == typeid(int))
913 return PrimitiveType::INT;
914 if (ti == typeid(uint))
915 return PrimitiveType::UINT;
916 if (ti == typeid(float))
917 return PrimitiveType::FLOAT;
918 if (ti == typeid(double))
919 return PrimitiveType::DOUBLE;
920 return PrimitiveType::NONE;
921 }

924template<ElemId ELEM_ID, MeshConcept MeshType>
925void addElementCustomComponent(MeshType& m, const MeshInfo::CustomComponent& cc)
926{
927 switch (cc.type) {
928 case PrimitiveType::CHAR:
929 m.template addElementCustomComponent<ELEM_ID, char>(cc.name);
930 break;
931 case PrimitiveType::UCHAR:
932 m.template addElementCustomComponent<ELEM_ID, unsigned char>(
933 cc.name);
934 break;
935 case PrimitiveType::SHORT:
936 m.template addElementCustomComponent<ELEM_ID, short>(cc.name);
937 break;
938 case PrimitiveType::USHORT:
939 m.template addElementCustomComponent<ELEM_ID, unsigned short>(
940 cc.name);
941 break;
942 case PrimitiveType::INT:
943 m.template addElementCustomComponent<ELEM_ID, int>(cc.name);
944 break;
945 case PrimitiveType::UINT:
946 m.template addElementCustomComponent<ELEM_ID, uint>(cc.name);
947 break;
948 case PrimitiveType::FLOAT:
949 m.template addElementCustomComponent<ELEM_ID, float>(cc.name);
950 break;
951 case PrimitiveType::DOUBLE:
952 m.template addElementCustomComponent<ELEM_ID, double>(cc.name);
953 break;
954 default: assert(0);
955 }
956}

989template<MeshConcept MeshType>
990void enableOptionalComponentsFromInfo(MeshInfo& info, MeshType& m)
991{
992 if (info.hasVertices()) {
993 if (info.hasVertexColor()) {
994 if (!enableIfVertexColorOptional(m)) {
995 info.setVertexColor(false);
996 }
997 }
998 if (info.hasVertexNormal()) {
999 if (!enableIfVertexNormalOptional(m)) {
1000 info.setVertexNormal(false);
1001 }
1002 }
1003 if (info.hasVertexQuality()) {
1004 if (!enableIfVertexQualityOptional(m)) {
1005 info.setVertexQuality(false);
1006 }
1007 }
1008 if (info.hasVertexTangent()) {
1009 if (!enableIfVertexTangentOptional(m)) {
1010 info.setVertexTangent(false);
1011 }
1012 }
1013 if (info.hasVertexTexCoord()) {
1014 if (!enableIfVertexTexCoordOptional(m)) {
1015 info.setVertexTexCoord(false);
1016 }
1017 }
1018 if (info.hasVertexMaterialIndex()) {
1019 if (!enableIfVertexMaterialIndexOptional(m)) {
1020 info.setVertexMaterialIndex(false);
1021 }
1022 }
1023 if (info.hasVertexCustomComponents()) {
1024 if constexpr (HasVertexCustomComponents<MeshType>) {
1025 for (const auto& cc : info.vertexCustomComponents()) {
1026 addVertexCustomComponent(m, cc);
1027 }
1028 }
1029 else {
1030 info.clearVertexCustomComponents();
