52template<MeshConcept MeshType>
53void vertexPositionsToBuffer(
54 const MeshType& mesh,
55 auto* buffer,
56 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
57 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
58{
59 using namespace detail;
60 using IndexType = typename MeshType::IndexType;
61
62 const auto NUM_ROWS =
63 numRows == MeshType::NULL_INDEX ? mesh.vertexCount() : numRows;
64 for (IndexType i = 0; const auto& p : mesh.vertices() | views::positions) {
65 at(buffer, i, 0, NUM_ROWS, 3, storage) = p.x();
66 at(buffer, i, 1, NUM_ROWS, 3, storage) = p.y();
67 at(buffer, i, 2, NUM_ROWS, 3, storage) = p.z();
68 ++i;
69 }
70}

90template<MeshConcept MeshType>
91void vertexQuadIndicesToBuffer(const MeshType& mesh, auto* buffer)
92{
93 using IndexType = typename MeshType::IndexType;
94
95 // creates indices for two triangles (quad) for each vertex
96 //
97 // 2-------3
98 // | \ |
99 // | \ |
100 // | \ |
101 // | \ |
102 // | \ |
103 // 0-------1
104 //
105 // - Triangle 1: [0, 1, 2]
106 // - Triangle 2: [1, 3, 2]
107 //
108 for (IndexType i = 0; const auto& v : mesh.vertices()) {
109 const auto baseIdx = i * 6;
110 const auto quadIdx = i * 4;
111 buffer[baseIdx + 0] = quadIdx + 0;
112 buffer[baseIdx + 1] = quadIdx + 1;
113 buffer[baseIdx + 2] = quadIdx + 2;
114 buffer[baseIdx + 3] = quadIdx + 1;
115 buffer[baseIdx + 4] = quadIdx + 3;
116 buffer[baseIdx + 5] = quadIdx + 2;
117 ++i;
118 }
119}

145template<FaceMeshConcept MeshType>
146auto faceSizesToBuffer(const MeshType& mesh, auto* buffer)
147{
148 using IndexType = typename MeshType::IndexType;
149
150 IndexType sum = 0;
151 for (IndexType i = 0; const auto& f : mesh.faces()) {
152 buffer[i] = f.vertexCount();
153 sum += f.vertexCount();
154 ++i;
155 }
156 return sum;

209template<FaceMeshConcept MeshType>
210void faceVertexIndicesToBuffer(
211 const MeshType& mesh,
212 auto* buffer,
213 bool getIndicesAsIfContainerCompact = true)
214{
215 using BufferValueType = std::remove_cvref_t<decltype(*buffer)>;
216 detail::checkBufferIndexCapacity<BufferValueType>(
217 getIndicesAsIfContainerCompact ? mesh.vertexCount() :
218 mesh.vertexContainerSize());
219
220 using IndexType = typename MeshType::IndexType;
221
222 auto vertCompIndices =
223 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
224
225 // lambda to get the vertex index of a face (considering compact indices)
226 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
227
228 for (IndexType i = 0; const auto& f : mesh.faces()) {
229 for (uint j = 0; j < f.vertexCount(); ++j) {
230 buffer[i] = static_cast<BufferValueType>(vIndex(f, j));
231 ++i;
232 }
233 }
234}

285template<FaceMeshConcept MeshType>
286void faceVertexIndicesToBuffer(
287 const MeshType& mesh,
288 auto* buffer,
289 uint largestFaceSize,
290 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
291 bool getIndicesAsIfContainerCompact = true,
292 typename MeshType::IndexType numRows = MeshType::NULL_INDEX)
293{
294 using namespace detail;
295 using BufferValueType = std::remove_cvref_t<decltype(*buffer)>;
296 checkBufferIndexCapacity<BufferValueType>(
297 getIndicesAsIfContainerCompact ? mesh.vertexCount() :
298 mesh.vertexContainerSize());
299
300 using IndexType = typename MeshType::IndexType;
301
302 auto vertCompIndices =
303 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
304
305 // lambda to get the vertex index of a face (considering compact indices)
306 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
307
308 const auto NUM_ROWS =
309 numRows == MeshType::NULL_INDEX ? mesh.faceCount() : numRows;
310
311 const auto SENTINEL = bufferSentinel<BufferValueType>();
312
313 for (IndexType i = 0; const auto& f : mesh.faces()) {
314 for (uint j = 0; j < largestFaceSize; ++j) {
315 at(buffer, i, j, NUM_ROWS, largestFaceSize, storage) =
316 j < f.vertexCount() ?
317 static_cast<BufferValueType>(vIndex(f, j)) :
318 SENTINEL;
319 }
320 ++i;
321 }
322}

370template<FaceMeshConcept MeshType>
371void triangulatedFaceVertexIndicesToBuffer(
372 const MeshType& mesh,
373 auto* buffer,
374 TriPolyIndexBiMap& indexMap = detail::indexMap,
375 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
376 typename MeshType::IndexType numTriangles = MeshType::NULL_INDEX,
377 bool getIndicesAsIfContainerCompact = true)
378{
379 using namespace detail;
380 using BufferValueType = std::remove_cvref_t<decltype(*buffer)>;
381 checkBufferIndexCapacity<BufferValueType>(
382 getIndicesAsIfContainerCompact ? mesh.vertexCount() :
383 mesh.vertexContainerSize());
384
385 using IndexType = typename MeshType::IndexType;
386
387 auto vertCompIndices =
388 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
389
390 // lambda to get the vertex index of a face (considering compact indices)
391 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
392
393 // there will be at least a triangle for each polygon
394 indexMap.clear();
395 indexMap.reserve(mesh.faceCount(), mesh.faceContainerSize());
396
397 if constexpr (TriangleMeshConcept<MeshType>) {
398 // construct the indexMap, which maps each triangle to the face index
399 for (IndexType t = 0; const auto& f : mesh.faces()) {
400 // map the ith triangle to the f face
401 indexMap.insert(t, f.index());
402 ++t;
403 }
404
405 faceVertexIndicesToBuffer(
406 mesh, buffer, 3, storage, getIndicesAsIfContainerCompact);
407 }
408 else {
409 // if the user did not give the number of triangles, and the buffer
410 // storage is column major, we need to compute the number of resulting
411 // triangles
412 if (numTriangles == MeshType::NULL_INDEX &&
413 storage == MatrixStorageType::COLUMN_MAJOR &&
414 mesh.faceCount() > 0) {
415 numTriangles = triangulatedFaceCount(mesh);
416 }
417 for (IndexType t = 0; const auto& f : mesh.faces()) {
418 std::vector<uint> vind = vcl::earCut(f);
419
420 // for each triangle of the triangulation (t is the triangle index)
421 for (uint vi = 0; vi < vind.size(); vi += 3) {
422 // map the t-th triangle to the f polygonal face
423 indexMap.insert(t, f.index());
424
425 for (uint k = 0; k < 3; ++k)
426 at(buffer, t, k, numTriangles, 3, storage) =
427 static_cast<BufferValueType>(vIndex(f, vind[vi + k]));
428
429 ++t;
430 }
431 }
432 }
433
434 if (&indexMap == &detail::indexMap) {
435 // if the user did not give an indexMap, we clear it at the end of the
436 // function to keep the global one empty for the next call.
437 indexMap.clear();
438 }
439}

857void triangulatedFaceNormalsToBuffer(

860 const TriPolyIndexBiMap& indexMap,

862 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,

878 for (auto t = first; t < last; ++t) {
879 at(buffer, t, 0, NUM_ROWS, 3, storage) = n.x();
880 at(buffer, t, 1, NUM_ROWS, 3, storage) = n.y();

1869template<FaceMeshConcept MeshType>
1870void wedgeTexCoordsAsDuplicatedVertexTexCoordsToBuffer(
1871 const MeshType& mesh,
1872 const std::vector<std::pair<IndexType<MeshType>, uint>>& vertWedgeMap,
1873 const std::list<std::list<std::pair<IndexType<MeshType>, uint>>>& facesToReassign,
1874 auto* buffer,
1875 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
1876{
1877 using namespace detail;
1878 using TexCoordType = TexCoordType<WedgeType<FaceType<MeshType>>>;
1879 using IndexType = MeshType::IndexType;
1880
1881 requireFaceWedgeTexCoords(mesh);
1882
1883 const auto NUM_ROWS = mesh.vertexCount() + facesToReassign.size();
1884
1885 // first export the texcoords of the non-duplicated vertices, using the
1886 // vertWedgeMap to get the texcoord index in the face
1887 IndexType vi = 0; // current vertex (or current row in the matrix)
1888 for (const auto& v : mesh.vertices()) {
1889 TexCoordType w;
1890 auto fInd = vertWedgeMap[v.index()].first;
1891 auto wInd = vertWedgeMap[v.index()].second;
1892
1893 // check if the vertex is referenced
1894 if (fInd != MeshType::NULL_INDEX && wInd != UINT_NULL) {
1895 w = mesh.face(fInd).wedge(wInd).texCoord();
1896 }
1897 at(buffer, vi, 0, NUM_ROWS, 2, storage) = w.u();
1898 at(buffer, vi, 1, NUM_ROWS, 2, storage) = w.v();
1899
1900 ++vi;
1901 }
1902
1903 // then append the texcoords of the duplicated vertices, that can be found
1904 // by looking into the any of the facesToReassign element lists
1905 for (const auto& list : facesToReassign) {
1906 assert(list.begin() != list.end());
1907 const auto& p = list.front();
1908 auto fInd = p.first;
1909 auto wInd = p.second;
1910
1911 const auto& w = mesh.face(fInd).wedge(wInd).texCoord();
1912 at(buffer, vi, 0, NUM_ROWS, 2, storage) = w.u();
1913 at(buffer, vi, 1, NUM_ROWS, 2, storage) = w.v();
1914
1915 ++vi;
1916 }
