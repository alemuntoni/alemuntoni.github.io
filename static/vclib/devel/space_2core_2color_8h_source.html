217 uint8_t hsvHue() const
218 {
219 uint8_t rgbMin, rgbMax;
220 uint8_t h;
221
222 rgbMin = x() < y() ? (x() < z() ? x() : z()) : (y() < z() ? y() : z());
223 rgbMax = x() > y() ? (x() > z() ? x() : z()) : (y() > z() ? y() : z());
224
225 if (rgbMax == 0) {
226 return 0;
227 }
228
229 if (255 * long(rgbMax - rgbMin) / rgbMax == 0) {
230 return 0;
231 }
232
233 if (rgbMax == x())
234 h = 0 + 43 * (y() - z()) / (rgbMax - rgbMin);
235 else if (rgbMax == y())
236 h = 85 + 43 * (z() - x()) / (rgbMax - rgbMin);
237 else
238 h = 171 + 43 * (x() - y()) / (rgbMax - rgbMin);
239
240 return h;
241 }

247 uint8_t hsvSaturation() const
248 {
249 uint8_t rgbMin, rgbMax;
250
251 rgbMin = x() < y() ? (x() < z() ? x() : z()) : (y() < z() ? y() : z());
252 rgbMax = x() > y() ? (x() > z() ? x() : z()) : (y() > z() ? y() : z());
253
254 if (rgbMax == 0) {
255 return 0;

257
258 return 255 * long(rgbMax - rgbMin) / rgbMax;
259 }

460 void setHsv(uint8_t h, uint8_t s, uint8_t v, uint8_t alpha = 255)
461 {
462 w() = alpha;
463 if (s == 0) {
464 x() = v;
465 y() = v;
466 z() = v;
467 }
468 else {
469 h = (h / 360.0) * 255;
470 uint8_t region, remainder, p, q, t;
471 region = h / 43;
472 remainder = (h - (region * 43)) * 6;
473
474 p = (v * (255 - s)) >> 8;
475 q = (v * (255 - ((s * remainder) >> 8))) >> 8;
476 t = (v * (255 - ((s * (255 - remainder)) >> 8))) >> 8;
477
478 switch (region) {
479 case 0:
480 x() = v;
481 y() = t;
482 z() = p;
483 break;
484 case 1:
485 x() = q;
486 y() = v;
487 z() = p;
488 break;
489 case 2:
490 x() = p;
491 y() = v;
492 z() = t;
493 break;
494 case 3:
495 x() = p;
496 y() = q;
497 z() = v;
498 break;
499 case 4:
500 x() = t;
501 y() = p;
502 z() = v;
503 break;
504 default:
505 x() = v;
506 y() = p;
507 z() = q;
508 break;
509 }
510 }
511 }

670inline Color colorLerp(const Color& c0, const Color& c1, float value)
671{
672 Color c;
673 if (value < 0) // out of range - left
674 value = 0;
675 else if (value > 1) // out of range - right
676 value = 1;
677 for (uint i = 0; i < 4; i++)
678 c(i) = c1(i) * value + c0(i) * (1 - value);
679 return c;
680}

746inline Color colorFromIntervalParula(float value)
747{
748 if (value < 0) // out of range - left
749 value = 0;
750 else if (value > 1) // out of range - right
751 value = 1;
752
753 static uint paruVal[9] = {
754 0x271680ff,
755 0x0363e1ff,
756 0x1485d4ff,
757 0x06a7c6ff,
758 0x38b99eff,
759 0x92bf73ff,
760 0xd9ba56ff,
761 0xfcce2eff,
762 0xfffa0aff};
763
764 int ind = int(floor(value * 8.0f));
765 float div = (value * 8.0f - ind);
766
767 if (div < 0)
768 div = 0;
769 else if (div > 1)
770 div = 1;
771
772 return colorLerp(
773 (Color::ColorABGR) paruVal[ind],
774 (Color::ColorABGR) paruVal[ind + 1],
775 div);
776}

917inline std::vector<Color> colorScattering(
918 uint n,
919 float sat = 0.3,
920 float val = 0.9)
921{
922 std::vector<Color> scattering;
923 scattering.reserve(n);
924
925 for (uint v = 0; v < n; ++v) {
926 uint value = v, m = n;
927 uint b = 0;
928
929 for (uint k = 1; k < n; k <<= 1) {
930 if (value << 1 >= m) {
931 b += k;
932 value -= (m + 1) >> 1;
933 m >>= 1;
934 }
935 else {
936 m = (m + 1) >> 1;
937 }
938 }
939
940 Color rc;
941 rc.setHsvF(float(b) / float(n), sat, val);
942 scattering.push_back(rc);
943 }
944 return scattering;
945}
