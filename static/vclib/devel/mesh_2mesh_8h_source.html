429 void append(const Mesh& m)
430 {
431 if (this == &m) {
432 Mesh copy = m;
433 append(copy);
434 return;
435 }
436
437 std::array<IndexType, ELEM_COUNT> initialSizes = {0};
438 (saveInitialContainerSize<Rebound<Args>>(initialSizes), ...);
439
440 (appendContainer<Rebound<Args>>(m), ...);
441
442 (shiftReferencesOfContainer<Rebound<Args>>(initialSizes), ...);
443
444 if constexpr (mesh::HasTransformMatrix<Mesh<Args...>>) {
445 using MatrixType = typename Mesh<Args...>::TransformMatrixType;
446
447 MatrixType matrix = this->transformMatrix();
448 matrix = matrix.inverse().eval();
449 matrix *= m.transformMatrix();
450
451 if (matrix != MatrixType::Identity()) {
452 (updatePosAndNormalsOfContainerAfterAppend<Rebound<Args>>(
453 initialSizes, matrix),
454 ...);
455 }
456 }
457
458 if constexpr (mesh::HasMaterials<Mesh<Args...>>) {
459 ushort nMaterials = this->materialCount();
460
461 // mapping from material indices of m to material indices of this
462 std::vector<ushort> mapping(m.materialCount());
463
464 // for each material of the other mesh, add it to this mesh
465 // if it does not exist yet
466 for (ushort i = 0; i < m.materialCount(); ++i) {
467 auto it = std::find(
468 this->materialBegin(), this->materialEnd(), m.material(i));
469
470 if (it == this->materialEnd()) {
471 this->pushMaterial(m.material(i));
472 mapping[i] = nMaterials++;
473 }
474 else {
475 mapping[i] = static_cast<ushort>(
476 std::distance(this->materialBegin(), it));
477 }
478 }
479
480 for (const auto& [path, img] : m.textureImages()) {
481 this->pushTextureImage(path, img);
482 }
483
484 if (!mapping.empty()) {
485 (updateMaterialIndicesOfContainerAfterAppend<Rebound<Args>>(
486 initialSizes, mapping),
487 ...);
488 }
489 }
490 }

544 void deserialize(std::istream& is)
545 {
546 char magic[8];
547 is.read(magic, 8);
548 if (is.gcount() != 8 || std::memcmp(magic, "vcl_mesh", 8) != 0) {
549 throw std::runtime_error(
550 "Invalid mesh file header: expected 'vcl_mesh'");
551 }
552 uint32_t version = 0;
553 vcl::deserialize(is, version);
554 if (version != 1) {
555 throw std::runtime_error(
556 "Unsupported mesh binary format version: " +
557 std::to_string(version));
558 }
559
560 (deserializeBase<Rebound<Args>>(is), ...);
561
562 setParentMeshPointersToContainers();
563 }

1341 template<typename Cont, typename MatrixType>
1342 void updatePosAndNormalsOfContainerAfterAppend(
1343 const std::array<IndexType, ELEM_COUNT>& initialSizes,
1344 const MatrixType& matrix)
1345 {
1346 if constexpr (mesh::ElementContainerConcept<Cont>) {
1347 constexpr ElemId ELEM_ID = Cont::ELEMENT_ID;
1348 const IndexType startFrom = initialSizes[toUnderlying(ELEM_ID)];
1349 if (startFrom >= this->Cont::elementContainerSize()) {
1350 return;
1351 }
1352
1353 if constexpr (comp::HasComponent<CompId::POSITION, Cont>) {
1354 if (this->Cont::template isComponentAvailable<CompId::POSITION>()) {
1355 auto posview = this->template elements<ELEM_ID>(startFrom) |
1356 views::positions;
1357 multiplyPointsByMatrix(posview, matrix);
1358 }
1359 }
1360
1361 if constexpr (comp::HasComponent<CompId::NORMAL, Cont>) {
1362 if (this->Cont::template isComponentAvailable<CompId::NORMAL>()) {
1363 auto norview = this->template elements<ELEM_ID>(startFrom) |
1364 views::normals;
1365 multiplyNormalsByMatrix(norview, matrix);
1366 }
1367 }
1368
1369 if constexpr (comp::HasComponent<CompId::WEDGE_NORMALS, Cont>) {
1370 if (this->Cont::template isComponentAvailable<CompId::WEDGE_NORMALS>()) {
1371 for (auto f : this->template elements<ELEM_ID>(startFrom)) {
1372 for (auto w : f.wedges()) {
1373 w.normal() = multiplyNormalByMatrix(w.normal(), matrix);
1374 }
1375 }
1376 }
1377 }
1378
1379 if constexpr (comp::HasComponent<CompId::TANGENT, Cont>) {
1380 if (this->Cont::template isComponentAvailable<CompId::TANGENT>()) {
1381 auto tanview = this->template elements<ELEM_ID>(startFrom) |
1382 views::tangents;
1383 multiplyNormalsByMatrix(tanview, matrix);
1384
1385 Matrix33<double> m33 =
1386 matrix.template cast<double>().block(0, 0, 3, 3);
1387 if (m33.determinant() < 0) {
1388 for (auto e : this->template elements<ELEM_ID>(startFrom)) {
1389 e.tangentRightHanded() = !e.tangentRightHanded();
1390 }
1391 }
1392 }
1393 }
1394
1395 if constexpr (comp::HasComponent<CompId::PRINCIPAL_CURVATURE, Cont>) {
1396 if (this->Cont::template isComponentAvailable<CompId::PRINCIPAL_CURVATURE>()) {
1397 for (auto e : this->template elements<ELEM_ID>(startFrom)) {
1398 e.principalCurvature().maxDir() =
1399 multiplyNormalByMatrix(e.principalCurvature().maxDir(), matrix);
1400 e.principalCurvature().minDir() =
1401 multiplyNormalByMatrix(e.principalCurvature().minDir(), matrix);
1402 }
1403 }
1404 }
1405 }
1406 }
