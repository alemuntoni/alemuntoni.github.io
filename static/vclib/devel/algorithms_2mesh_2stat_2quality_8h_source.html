51template<ElemId ELEM_ID, MeshConcept MeshType>

53{
54 requireElementComponent<ELEM_ID, CompId::QUALITY>(m);
55
56 auto eq = m.template elements<ELEM_ID>() | views::quality;
57 return std::accumulate(std::ranges::begin(eq), std::ranges::end(eq), 0.0) /

61template<ElemId ELEM_ID, MeshConcept MeshType>
62auto elementQualityHistogram(
63 const MeshType& m,
64 bool selectionOnly = false,
65 uint histSize = 10000)
66{
67 using QualityType =
68 typename MeshType::template ElementType<ELEM_ID>::QualityType;
69
70 requireElementComponent<ELEM_ID, CompId::QUALITY>(m);
71
72 auto minmax = elementQualityMinMax<ELEM_ID>(m);
73
74 Histogram<QualityType> h(minmax.first, minmax.second, histSize);
75 for (const auto& e : m.template elements<ELEM_ID>()) {
76 if (!selectionOnly || e.selected()) {
77 assert(!isDegenerate(e.quality()));
78 h.addValue(e.quality());
79 }
80 }
81 return h;
82}

169template<MeshConcept MeshType>
170auto vertexRadiusFromQuality(
171 const MeshType& m,
172 double diskRadius,
173 double radiusVariance,
174 bool invert = false)
175{
176 requireVertexQuality(m);
177
178 using QualityType = VertexType<MeshType>::QualityType;
179
180 std::vector<QualityType> radius(m.vertexContainerSize());
181 std::pair<QualityType, QualityType> minmax = vertexQualityMinMax(m);
182
183 QualityType minRad = diskRadius;
184 QualityType maxRad = diskRadius * radiusVariance;
185 QualityType deltaQ = minmax.second - minmax.first;
186 QualityType deltaRad = maxRad - minRad;
187 for (const auto& v : m.vertices()) {
188 radius[v.index()] =
189 minRad + deltaRad * ((invert ? minmax.second - v.quality() :
190 v.quality() - minmax.first) /
191 deltaQ);
192 }
193
194 return radius;
195}
