585 void serialize(std::ostream& os) const
586 {
587 // regardless of the type of the container (indices or pointers), the
588 // serialization is always done using the indices
589
590 vcl::serialize(os, adjVertexCount());
591 for (uint i = 0; i < adjVertexCount(); ++i) {
592 vcl::serialize(os, adjVertexIndex(i));
593 }
594 }
595
596 void deserialize(std::istream& is)
597 {
598 uint n;
599 vcl::deserialize(is, n);
600 resizeAdjVertices(n);
601
602 for (uint i = 0; i < adjVertexCount(); ++i) {
603 uint aei;
604 vcl::deserialize(is, aei);
605 setAdjVertex(i, aei);
606 }
607 }

658/* importFrom function */
659
660template<
661 bool STORE_INDICES,
662 typename Vertex,
663 typename ParentElemType,
664 bool VERT,
665 bool OPT>
666template<typename Element>
667void AdjacentVertices<STORE_INDICES, Vertex, ParentElemType, VERT, OPT>::
668 importFrom(const Element& e, bool importRefs)
669{
670 if (importRefs) {
671 if constexpr (HasAdjacentVertices<Element>) {
672 if (isAdjacentVerticesAvailableOn(e)) {
673 // from static/dynamic to dynamic size: need to resize
674 // first, then import
675 resizeAdjVertices(e.adjVertexCount());
676 importIndicesFrom(e);
677 }
678 }
679 }
680}
