276 Scalar angle(const Point& p1) const
277 {
278 Scalar w = Base::norm() * p1.norm();
279 if (w == 0)
280 return -1;
281 Scalar t = Base::dot(p1) / w;
282 if (t > 1)
283 t = 1;
284 else if (t < -1)
285 t = -1;
286 return (Scalar) acos(t);
287 }

423 // TODO: this function should return a pair
444 void orthoBase(Point& u, Point& v) const requires (N == 3)
445 {
446 const double locEps = double(1e-7);
447
448 Point<Scalar, 3> up(0, 1, 0);
449 u = Base::cross(up);
450
451 double len = u.norm();
452 if (len < locEps) {
453 if (std::abs(at(0)) < std::abs(at(1))) {
454 if (std::abs(at(0)) < std::abs(at(2)))
455 up = Point<Scalar, 3>(1, 0, 0); // x is the min
456 else
457 up = Point<Scalar, 3>(0, 0, 1); // z is the min
458 }
459 else {
460 if (std::abs(at(1)) < std::abs(at(2)))
461 up = Point<Scalar, 3>(0, 1, 0); // y is the min
462 else
463 up = Point<Scalar, 3>(0, 0, 1); // z is the min
464 }
465 u = Base::cross(up);
466 }
467 v = Base::cross(u);
468 }

593 Point operator*(const Eigen::Matrix<Scalar, N + 1, N + 1>& m) const
594 requires (N == 3)
595 {
596 Eigen::Matrix<Scalar, 1, N> s;
597
598 s(0) = m(0, 0) * at(0) + m(0, 1) * at(1) + m(0, 2) * at(2) + m(0, 3);
599 s(1) = m(1, 0) * at(0) + m(1, 1) * at(1) + m(1, 2) * at(2) + m(1, 3);
600 s(2) = m(2, 0) * at(0) + m(2, 1) * at(1) + m(2, 2) * at(2) + m(2, 3);
601
602 Scalar w =
603 at(0) * m(3, 0) + at(1) * m(3, 1) + at(2) * m(3, 2) + m(3, 3);
604 if (w != 0) [[likely]]
605 s = s / w;
606 return Point(s);
607 }
