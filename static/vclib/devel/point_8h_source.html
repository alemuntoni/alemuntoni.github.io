248 Scalar angle(const Point& p1) const
249 {
250 Scalar w = Base::norm() * p1.norm();
251 if (w == 0)
252 return -1;
253 Scalar t = Base::dot(p1) / w;
254 if (t > 1)
255 t = 1;
256 else if (t < -1)
257 t = -1;
258 return (Scalar) acos(t);
259 }

395 // TODO: this function should return a pair
416 void orthoBase(Point& u, Point& v) const requires (N == 3)
417 {
418 const double locEps = double(1e-7);
419
420 Point<Scalar, 3> up(0, 1, 0);
421 u = Base::cross(up);
422
423 double len = u.norm();
424 if (len < locEps) {
425 if (std::abs(at(0)) < std::abs(at(1))) {
426 if (std::abs(at(0)) < std::abs(at(2)))
427 up = Point<Scalar, 3>(1, 0, 0); // x is the min
428 else
429 up = Point<Scalar, 3>(0, 0, 1); // z is the min
430 }
431 else {
432 if (std::abs(at(1)) < std::abs(at(2)))
433 up = Point<Scalar, 3>(0, 1, 0); // y is the min
434 else
435 up = Point<Scalar, 3>(0, 0, 1); // z is the min
436 }
437 u = Base::cross(up);
438 }
439 v = Base::cross(u);
440 }

565 Point operator*(const Eigen::Matrix<Scalar, N + 1, N + 1>& m) const
566 requires (N == 3)
567 {
568 Eigen::Matrix<Scalar, 1, N> s;
569
570 s(0) = m(0, 0) * at(0) + m(0, 1) * at(1) + m(0, 2) * at(2) + m(0, 3);
571 s(1) = m(1, 0) * at(0) + m(1, 1) * at(1) + m(1, 2) * at(2) + m(1, 3);
572 s(2) = m(2, 0) * at(0) + m(2, 1) * at(1) + m(2, 2) * at(2) + m(2, 3);
573
574 Scalar w =
575 at(0) * m(3, 0) + at(1) * m(3, 1) + at(2) * m(3, 2) + m(3, 3);
576 if (w != 0) [[likely]]
577 s = s / w;
578 return Point(s);
579 }

1085template<typename Scalar, unsigned int N, typename CharT>
1086struct formatter<vcl::Point<Scalar, N>, CharT> : formatter<Scalar, CharT>
1087{
1088 template<typename FormatContext>
1089 auto format(const vcl::Point<Scalar, N>& p, FormatContext& ctx) const
1090 {
1091 auto out = ctx.out();
1092 *out++ = static_cast<CharT>('[');
1093 for (unsigned int i = 0; i < N; ++i) {
1094 if (i > 0) {
1095 *out++ = static_cast<CharT>(',');
1096 *out++ = static_cast<CharT>(' ');
1097 }
1098 ctx.advance_to(out);
1099 out = formatter<Scalar, CharT>::format(p(i), ctx);
1100 }
1101 *out++ = static_cast<CharT>(']');
1102 return out;
1103 }
