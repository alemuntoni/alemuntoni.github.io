27inline void saveImageToBmp(
28 const std::string& filename,
29 int w,
30 int h,
31 const unsigned char* data)
32{
33 std::string fn = filename;
34 std::string ext = FileInfo::extension(filename);
35 ext = toLower(ext);
36 if (ext != ".bmp") {
37 fn += ".bmp";
38 }
39
40 std::ofstream file(fn, std::ios::binary);
41 if (!file) {
42 throw std::runtime_error("Failed to open file for writing: " + fn);
43 }
44
45 // BMP file header
46 unsigned char fileHeader[14] = {
47 'B',
48 'M', // Signature
49 0,
50 0,
51 0,
52 0, // File size in bytes
53 0,
54 0, // Reserved
55 0,
56 0, // Reserved
57 54,
58 0,
59 0,
60 0 // Offset to pixel data
61 };
62
63 // BMP info header
64 unsigned char infoHeader[40] = {
65 40, 0, 0, 0, // Header size
66 0, 0, 0, 0, // Image width
67 0, 0, 0, 0, // Image height
68 1, 0, // Planes
69 32, 0, // Bits per pixel
70 0, 0, 0, 0, // Compression (none)
71 0, 0, 0, 0, // Image size (can be 0 for uncompressed)
72 0, 0, 0, 0, // X pixels per meter
73 0, 0, 0, 0, // Y pixels per meter
74 0, 0, 0, 0, // Total colors (0 = default)
75 0, 0, 0, 0 // Important colors (0 = all)
76 };
77
78 // Fill in file size
79 int fileSize = 54 + w * h * 4; // Header size + pixel data size
80 fileHeader[2] = (unsigned char) (fileSize);
81 fileHeader[3] = (unsigned char) (fileSize >> 8);
82 fileHeader[4] = (unsigned char) (fileSize >> 16);
83 fileHeader[5] = (unsigned char) (fileSize >> 24);
84
85 // Fill in image width and height
86 infoHeader[4] = (unsigned char) (w);
87 infoHeader[5] = (unsigned char) (w >> 8);
88 infoHeader[6] = (unsigned char) (w >> 16);
89 infoHeader[7] = (unsigned char) (w >> 24);
90 infoHeader[8] = (unsigned char) (h);
91 infoHeader[9] = (unsigned char) (h >> 8);
92 infoHeader[10] = (unsigned char) (h >> 16);
93 infoHeader[11] = (unsigned char) (h >> 24);
94
95 // Write headers
96 file.write(reinterpret_cast<const char*>(fileHeader), sizeof(fileHeader));
97 file.write(reinterpret_cast<const char*>(infoHeader), sizeof(infoHeader));
98
99 // Write pixel data (BMP stores pixels bottom-up)
100 for (int y = h - 1; y >= 0; --y) {
101 file.write(reinterpret_cast<const char*>(data + y * w * 4), w * 4);
102 }
103
104 file.close();
105}

107} // namespace vcl
