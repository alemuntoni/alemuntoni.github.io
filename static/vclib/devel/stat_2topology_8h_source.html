25// function called for the Container Cont of the Mesh
26// mesh: the mesh
27// refs: vector of bools, one for each vertex of the mesh, to store if the
28// vertex is referenced or not
29// nRefs: number of referenced vertices
30template<typename Cont>
31void setReferencedVertices(const auto& mesh, auto& refs, auto& nRefs)
32{
33 // check if the Cont container of the Mesh has vertex references
34 if constexpr (comp::HasVertexReferences<typename Cont::ElementType>) {
35 // if there are still some vertices non-referenced
36 if (nRefs < mesh.vertexCount()) {
37 constexpr ElemId ELEM_ID = Cont::ElementType::ELEMENT_ID;
38 // for each element of the Cont container
39 for (const auto& el : mesh.template elements<ELEM_ID>()) {
40 // for each vertex of the element
41 for (auto vi : el.vertexIndices()) {
42 if (vi != mesh.NULL_INDEX && !mesh.vertex(vi).deleted()) {
43 if (!refs[vi]) {
44 // set the vertex as referenced
45 refs[vi] = true;
46 nRefs++;
47 }
48 }
49 }
50 }
51 }
52 }
53}

55// function called for each container of the Mesh, using variadic templates
56template<typename... Cont>
57void setReferencedVertices(
58 const auto& mesh,
59 auto& refs,
60 auto& nRefs,
61 TypeWrapper<Cont...>)
62{
63 // call the setReferencedVerticesOnVector function for each container of the

65 (setReferencedVertices<Cont>(mesh, refs, nRefs), ...);

68// struct to store the information of the wedge texcoords
69template<typename WedgeTexCoordType>
70struct WedgeTexCoordsInfo
71{
72 WedgeTexCoordType texCoord;
73 uint texCoordIndex;
74
75 bool operator<(const WedgeTexCoordsInfo& other) const
76 {
77 if (texCoordIndex == other.texCoordIndex) {
78 return texCoord < other.texCoord;
79 }
80 return texCoordIndex < other.texCoordIndex;
81 }
82};

84template<FaceMeshConcept MeshType>
85std::vector<bool> nonManifoldVerticesVectorBool(const MeshType& m)
86 requires HasFaceAdjacentFaces<MeshType>
87{
88 requireFaceAdjacentFaces(m);
89
90 std::vector<bool> nonManifoldVertices(m.vertexContainerSize(), false);
91
92 std::vector<uint> TD(m.vertexContainerSize(), 0);
93 std::vector<bool> nonManifoldInc(m.vertexContainerSize(), false);
94 // First Loop, count how many faces are incident on a vertex and store it in
95 // TD, and flag how many vertices are incident on non manifold edges.
96 for (const auto& f : m.faces()) {
97 for (uint i = 0; i < f.vertexCount(); ++i) {
98 auto vi = f.vertexIndex(i);
99 if (vi != m.NULL_INDEX) {
100 TD[vi]++;
101 if (!isFaceManifoldOnEdge(f, i)) {
102 nonManifoldInc[vi] = true;
103 auto viNext = f.vertexIndexMod(i + 1);
104 if (viNext != m.NULL_INDEX) {
105 nonManifoldInc[viNext] = true;
106 }
107 }
108 }
109 }
110 }
111
112 std::vector<bool> visited(m.vertexContainerSize(), false);
113 for (const auto& f : m.faces()) {
114 for (uint i = 0; i < f.vertexCount(); ++i) {
115 auto vi = f.vertexIndex(i);
116 if (vi != m.NULL_INDEX && !visited[vi]) {
117 visited[vi] = true;
118 MeshPos<MeshType> pos(f, i);
119 uint starSize = pos.numberOfAdjacentFacesToV();
120 if (starSize != TD[vi])
121 nonManifoldVertices[vi] = true;
122 }
123 }
124 }
125
126 return nonManifoldVertices;
127}

129template<FaceMeshConcept MeshType>
130auto edgeCount(
131 const MeshType& m,
132 auto& numBoundaryEdges,
133 auto& numNonManifoldEdges)
134{
135 using IndexType = typename MeshType::IndexType;
136
137 std::vector<MeshEdgeUtil<MeshType>> edgeVec =
138 fillAndSortMeshEdgeUtilVector(m);
139
140 IndexType numEdges = 0;
141 numBoundaryEdges = 0;
142 numNonManifoldEdges = 0;
143
144 std::size_t f_on_cur_edge = 1;
145 for (std::size_t i = 0; i < edgeVec.size(); ++i) {
146 if (((i + 1) == edgeVec.size()) || !(edgeVec[i] == edgeVec[i + 1])) {
147 ++numEdges;
148 if (f_on_cur_edge == 1)
149 ++numBoundaryEdges;
150 if (f_on_cur_edge > 2)
151 ++numNonManifoldEdges;
152 f_on_cur_edge = 1;
153 }
154 else {
155 ++f_on_cur_edge;
156 }
157 }
158 return numEdges;
159}

177{

182

466template<
467 FaceMeshConcept MeshType,
468 typename VertWedgeMap,
469 typename VertsToDuplicate,
470 typename FacesToReassign>
471auto verticesToDuplicateByWedgeTexCoordsCount(
472 const MeshType& mesh,
473 VertWedgeMap& vertWedgeMap,
474 VertsToDuplicate& vertsToDuplicate,
475 FacesToReassign& facesToReassign)
476{
477 using IndexType = typename MeshType::IndexType;
478 vcl::requireFaceWedgeTexCoords(mesh);
479
480 bool hasMaterials = false;
481 if constexpr (HasFaceMaterialIndex<MeshType>) {
482 hasMaterials = mesh.isFaceMaterialIndexAvailable();
483 }
484
485 using WedgeTexCoordType = TexCoordType<WedgeType<FaceType<MeshType>>>;
486 using WedgeTexCoordsInfo = detail::WedgeTexCoordsInfo<WedgeTexCoordType>;
487
488 // list of faces that reference a wedge texcoord, with the index of the
489 // vertex in the face
490 using FaceList = std::list<std::pair<IndexType, uint>>;
491
492 vertWedgeMap.resize(mesh.vertexContainerSize());
493 vertsToDuplicate.clear();
494 facesToReassign.clear();
495
496 IndexType count = 0;
497
498 // for each vertex, I'll store a map of WedgeTexCoordsInfo
499 // each element of the map represent a unique wedge texcoord(the texcoord
500 // itself and the index of the texcoord), and for each element it maps a
501 // list of faces that reference the texcoord
502 std::vector<std::map<WedgeTexCoordsInfo, FaceList>> wedges(
503 mesh.vertexContainerSize());
504
505 for (const auto& f : mesh.faces()) {
506 for (uint i = 0; i < f.vertexCount(); ++i) {
507 auto vi = f.vertexIndex(i);
508 if (vi == mesh.NULL_INDEX) {
509 continue;
510 }
511
512 // check if the i-th wedge texcoord of the face already exists
513 uint matIdx = 0;
514 if constexpr (HasFaceMaterialIndex<MeshType>) {
515 if (hasMaterials) {
516 matIdx = f.materialIndex();
517 }
518 }
519 WedgeTexCoordsInfo wi = {f.wedge(i).texCoord(), matIdx};
520 auto it = wedges[vi].find(wi);
521 if (it == wedges[vi].end()) { // if it doesn't exist, add it
522 // if there was already a texcoord for the vertex, it means that
523 // the vertex will be duplicated
524 if (!wedges[vi].empty()) {
525 count++;
526 }
527 wedges[vi][wi].emplace_back(f.index(), i);
528 }
529 else {
530 // if it exists, add the face to the list of faces that
531 // reference the texcoord
532 it->second.emplace_back(f.index(), i);
533 }
534 }
535 }
536
537 // for each vertex, check if there are multiple texcoords
538 // note: here we will modify the maps for convenience: at the end of this
539 // loop, the info will be contained in the two lists vertsToDuplicate and
540 // facesToReassign (the wedges vector of map will be inconsistent)
541 IndexType vi = 0;
542 for (auto& map : wedges) {
543 if (map.size() > 1) {
544 // there are multiple texcoords for the vertex vi, so it will be
545 // duplicated
546
547 // remove from the map the element having the higher number of
548 // faces referencing the texcoord (we do this to reassign the less
549 // number of faces)
550 auto it = std::max_element(
551 map.begin(), map.end(), [](const auto& a, const auto& b) {
552 return a.second.size() < b.second.size();
553 });
554 // store the reference of the texcoord to keep (pair face/vertex
555 // index in the face)
556 vertWedgeMap[vi] = it->second.front();
557 map.erase(it);
558
559 // store the vertex to duplicate, and the faces to reassign
560 for (auto& [wi, fl] : map) {
561 vertsToDuplicate.push_back(vi);
562 facesToReassign.push_back(std::move(fl));
563 }
564 }
565 else {
566 if (map.size() == 1) {
567 // there is only one texcoord for the vertex vi, so store its
568 // reference (pair face/vertex index in the face)
569 vertWedgeMap[vi] = map.begin()->second.front();
570 }
571 else { // the vertex was unreferenced
572 vertWedgeMap[vi] = {mesh.NULL_INDEX, UINT_NULL};
573 }
574 }
575 ++vi;
576 }
577
578 assert(vertWedgeMap.size() == mesh.vertexContainerSize());
579 assert(vertsToDuplicate.size() == count);
580 assert(facesToReassign.size() == count);
581
582 return count;
583}

623template<
624 typename Container = std::vector<bool>,
625 MeshConcept MeshType,
626 typename Integer = typename MeshType::IndexType>
627 requires (!std::is_same_v<std::remove_cvref_t<Integer>, bool>)
628Container referencedVertices(
629 const MeshType& mesh,
630 Integer& nUnref,
631 bool onlyFaces = false)
632{
633 using IndexType = typename MeshType::IndexType;
634 IndexType nRefs = 0;
635
636 Container refVertices(mesh.vertexContainerSize(), false);
637
638 if (onlyFaces) {
639 if constexpr (FaceMeshConcept<MeshType>) {
640 using FaceContainerType =
641 typename MeshType::template ContainerType<ElemId::FACE>;
642 detail::setReferencedVertices<FaceContainerType>(
643 mesh, refVertices, nRefs);
644 }
645 }
646 else {
647 // Iterate over every element container present in this mesh at
648 // compile-time and mark the vertices they reference.
649 detail::setReferencedVertices(
650 mesh, refVertices, nRefs, typename MeshType::ElementContainers());
651 }
652
653 nUnref = static_cast<Integer>(mesh.vertexCount() - nRefs);
654
655 return refVertices;
656}

768template<FaceMeshConcept MeshType>
769auto holeCount(const MeshType& m) requires HasFaceAdjacentFaces<MeshType>
770{
771 requireFaceAdjacentFaces(m);
772
773 using IndexType = typename MeshType::IndexType;
774 IndexType loopNum = 0;
775
776 // create a vector of bools to keep track of visited faces.
777 std::vector<bool> visitedFaces(m.faceContainerSize(), false);
778
779 // Traverse the mesh using a depth-first search algorithm to find all the
780 // holes.
781 for (const auto& f : m.faces()) {
782 uint e = 0;
783 for (const auto& v : f.vertices()) {
784 if (!visitedFaces[f.index()] && !f.adjFace(e)) {
785 MeshPos<MeshType> startPos(f, e);
786 MeshPos<MeshType> curPos = startPos;
787 do {
788 curPos.nextEdgeOnBorderAdjacentToV();
789 curPos.flipVertex();
790 visitedFaces[curPos.face().index()] = true;
791 } while (curPos != startPos);
792 ++loopNum;
793 }
794 ++e;
795 }
796 }
797 return loopNum;
798}

822template<FaceMeshConcept MeshType>
823auto connectedComponents(const MeshType& m)
824 requires HasFaceAdjacentFaces<MeshType>
825{
826 requireFaceAdjacentFaces(m);
827
828 using IndexType = typename MeshType::IndexType;
829 using FaceType = ConstFaceType<MeshType>;
830
831 std::vector<std::set<IndexType>> cc;
832
833 // create a vector of bools to keep track of visited faces.
834 std::vector<bool> visitedFaces(m.faceContainerSize(), false);
835
836 // create a stack to hold the faces that need to be visited during the
837 // depth-first search.
838 std::stack<FaceType> sf;
839
840 // traverse the mesh using a depth-first search algorithm to find the
841 // connected components.
842 for (auto f : m.faces()) {
843 if (!visitedFaces[f.index()]) { // first time I see this face
844 visitedFaces[f.index()] = true;
845
846 // new connected component
847 cc.emplace_back();
848 std::set<IndexType>& ccf = cc.back();
849 ccf.insert(f.index());
850
851 // while the stack is empty, visit the adjacent faces of the top
852 // face of the stack
853 sf.push(f);
854 while (!sf.empty()) {
855 FaceType fpt = sf.top();
856 // remove the top face and add it to the connected component
857 sf.pop();
858 ccf.insert(fpt.index());
859
860 // add the adjacent faces of the current visited in the stack
861 for (uint j = 0; j < fpt.vertexCount(); ++j) {
862 FaceType adjf = fpt.adjFace(j);
863 // if there is an adj face and it has not been visited
864 if (adjf && !visitedFaces[adjf.index()]) {
865 sf.push(adjf);
866 visitedFaces[adjf.index()] = true;
867 }
868 }
869 }
870 }
871 }
872 return cc;
873}
