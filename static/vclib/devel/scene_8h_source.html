110 template<FaceMeshConcept MeshType>
111 Scene(const MeshType& m)
112 {
113 mDevice = rtcNewDevice(nullptr);
114
115 if (!mDevice)
116 throw std::runtime_error("Error creating Embree device.");
117
118 mScene = rtcNewScene(mDevice);
119
120 if (!mScene) {
121 rtcReleaseDevice(mDevice);
122 throw std::runtime_error("Error creating Embree scene.");
123 }
124
125 RTCGeometry geometry =
126 rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_TRIANGLE);
127
128 if (!geometry) {
129 rtcReleaseScene(mScene);
130 rtcReleaseDevice(mDevice);
131 throw std::runtime_error("Error creating Embree geometry.");
132 }
133
134 float* vb = (float*) rtcSetNewGeometryBuffer(
135 geometry,
136 RTC_BUFFER_TYPE_VERTEX,
137 0,
138 RTC_FORMAT_FLOAT3,
139 3 * sizeof(float),
140 m.vertexCount());
141
142 vertexPositionsToBuffer(m, vb);
143
144 uint numTris = vcl::triangulatedFaceCount(m);
145
146 uint* ib = (uint*) rtcSetNewGeometryBuffer(
147 geometry,
148 RTC_BUFFER_TYPE_INDEX,
149 0,
150 RTC_FORMAT_UINT3,
151 3 * sizeof(unsigned),
152 numTris);
153
154 triangulatedFaceVertexIndicesToBuffer(
155 m, ib, mIndexMap, MatrixStorageType::ROW_MAJOR, numTris);
156
157 rtcCommitGeometry(geometry);
158 rtcAttachGeometry(mScene, geometry);
159
160 rtcReleaseGeometry(geometry);
161
162 rtcCommitScene(mScene);
163 }

210 template<typename ScalarType>
211 HitResult firstFaceIntersectedByRay(
212 const Point3<ScalarType>& origin,
213 const Point3<ScalarType>& direction,
214 float near = 0.f,
215 float far = std::numeric_limits<float>::infinity()) const
216 {
217 if (mScene) {
218 RTCRayHit rayhit = initRayHitValues(origin, direction, near, far);
219
220 rtcIntersect1(mScene, &rayhit);
221
222 if (rayhit.hit.geomID != RTC_INVALID_GEOMETRY_ID) {
223 uint fID = mIndexMap.polygon(rayhit.hit.primID);
224 float w = 1.f - rayhit.hit.u - rayhit.hit.v;
225 Point3f barycentricCoords(w, rayhit.hit.u, rayhit.hit.v);
226 uint triID = rayhit.hit.primID - mIndexMap.triangleBegin(fID);
227 return std::make_tuple(fID, barycentricCoords, triID);
228 }
229 }
230
231 return std::make_tuple(UINT_NULL, Point3f(), uint(0));
232 }

317 template<RandomAccessRange R1, RandomAccessRange R2>
318 std::vector<HitResult> firstFaceIntersectedByRays(
319 R1&& origins,
320 R2&& directions,
321 float near = 0.f,
322 float far = std::numeric_limits<float>::infinity()) const
323 requires Point3Concept<std::ranges::range_value_t<R1>> &&
324 Point3Concept<std::ranges::range_value_t<R2>>
325 {
326 using PointType = std::ranges::range_value_t<R1>;
327 using ScalarType = typename PointType::ScalarType;
328
329 assert(
330 std::ranges::size(origins) == std::ranges::size(directions) &&
331 "Origins and directions ranges must have the same size.");
332
333 std::size_t sz = std::ranges::size(origins);
334
335 std::vector<HitResult> results(sz);
336
337 if (mScene) {
338 // some cpus fail to guarantee thread safety for rtcIntersect16
339#ifndef VCL_EMBREE_FORCE_CHUNK_16
340 auto computeRay = [&](uint i) {
341 results[i] = firstFaceIntersectedByRay(
342 origins[i], directions[i], near, far);
343 };
344
345 std::vector<std::size_t> rayIndices(sz);
346 std::iota(rayIndices.begin(), rayIndices.end(), 0);
347
348 vcl::parallelFor(rayIndices, computeRay);
349#else
350 std::size_t chunks = sz / 16;
351 std::size_t rem = sz % 16;
352
353 if (rem != 0) {
354 ++chunks;
355 }
356
357 auto computeChunk = [&](uint chunk) {
358 std::array<int, 16> validMask;
359 std::ranges::fill(validMask, -1);
360
361 RTCRayHit16 rayHits;
362
363 std::size_t first = chunk * 16;
364 std::size_t last = first + 16;
365 if (last > sz) {
366 for (std::size_t i = sz % 16; i < 16; ++i) {
367 validMask[i] = 0;
368 }
369 last = sz;
370 }
371
372 for (std::size_t i = first; i < last; ++i) {
373 std::size_t idx = i - first;
374 initRayHitsValues(
375 rayHits, idx, origins[i], directions[i], near, far);
376 }
377
378 rtcIntersect16(validMask.data(), mScene, &rayHits);
379
380 for (std::size_t i = first; i < last; ++i) {
381 std::size_t idx = i - first;
382 if (rayHits.hit.geomID[idx] != RTC_INVALID_GEOMETRY_ID) {
383 uint fID = mIndexMap.polygon(rayHits.hit.primID[idx]);
384 float w = 1.f - rayHits.hit.u[idx] - rayHits.hit.v[idx];
385 Point3f barycentricCoords(
386 w, rayHits.hit.u[idx], rayHits.hit.v[idx]);
387 uint triID = rayHits.hit.primID[idx] -
388 mIndexMap.triangleBegin(fID);
389 results[i] =
390 std::make_tuple(fID, barycentricCoords, triID);
391 }
392 else {
393 results[i] =
394 std::make_tuple(UINT_NULL, Point3f(), uint(0));
395 }
396 }
397 };
398
399 std::vector<std::size_t> chunkIndices(chunks);
400 std::iota(chunkIndices.begin(), chunkIndices.end(), 0);
401
402 vcl::parallelFor(chunkIndices, computeChunk);
403#endif
404 }
405 else {
406 std::fill(
407 results.begin(),
408 results.end(),
409 std::make_tuple(UINT_NULL, Point3f(), uint(0)));
410 }
411
412 return results;
413 }

438 template<RandomAccessRange R>
439 std::vector<HitResult> firstFaceIntersectedByRays(
440 R&& rays,
441 float near = 0.f,
442 float far = std::numeric_limits<float>::infinity()) const
443 requires Ray3Concept<std::ranges::range_value_t<R>>
444 {
445 auto getOriginView = [](const auto& r) {
446 return r.origin();
447 };
448
449 auto getDirectionView = [](const auto& r) {
450 return r.direction();
451 };
452
453 auto origins = rays | std::views::transform(getOriginView);
454 auto directions = rays | std::views::transform(getDirectionView);
455
456 return firstFaceIntersectedByRays(origins, directions, near, far);
457 }

460 template<Point3Concept PointType>
461 static inline RTCRayHit initRayHitValues(
462 const PointType& origin,
463 const PointType& direction,
464 float tnear = 0.f,
465 float tfar = std::numeric_limits<float>::infinity())
466 {
467 RTCRayHit rayhit;
468 rayhit.ray.org_x = origin.x();
469 rayhit.ray.org_y = origin.y();
470 rayhit.ray.org_z = origin.z();
471 rayhit.ray.dir_x = direction.x();
472 rayhit.ray.dir_y = direction.y();
473 rayhit.ray.dir_z = direction.z();
474 rayhit.ray.tnear = tnear;
475 rayhit.ray.tfar = tfar;
476 rayhit.ray.mask = -1;
477 rayhit.ray.flags = 0;
478 rayhit.hit.geomID = RTC_INVALID_GEOMETRY_ID;
479 rayhit.hit.instID[0] = RTC_INVALID_GEOMETRY_ID;
480 return rayhit;
481 }

492 rayhits.ray.org_x[i] = origin.x();
493 rayhits.ray.org_y[i] = origin.y();
494 rayhits.ray.org_z[i] = origin.z();

496 rayhits.ray.dir_y[i] = direction.y();
497 rayhits.ray.dir_z[i] = direction.z();
498 rayhits.ray.tnear[i] = tnear;
499 rayhits.ray.tfar[i] = tfar;
500 rayhits.ray.mask[i] = -1;
501 rayhits.ray.flags[i] = 0;
502 rayhits.hit.geomID[i] = RTC_INVALID_GEOMETRY_ID;
503 rayhits.hit.instID[0][i] = RTC_INVALID_GEOMETRY_ID;
