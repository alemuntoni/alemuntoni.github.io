81template<PointConcept PointType>
82PointType random(
83 DistConfig<typename PointType::ScalarType> distConf = std::monostate(),
84 RandomConfig config = std::monostate())
85{
86 return callWithDistribution(distConf, [&](auto&& distFunc) {
87 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
88 PointType point;

119template<BoxConcept BoxType>
120BoxType random(
121 DistConfig<typename BoxType::ScalarType> distConf = std::monostate(),
122 RandomConfig config = std::monostate())
123{
124 using PointType = BoxType::PointType;
125
126 return callWithDistribution(distConf, [&](auto&& distFunc) {
127 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
128 PointType minPoint, maxPoint;
129 for (uint i = 0; i < PointType::DIM; ++i) {
130 minPoint[i] = distFunc(gen);
131 maxPoint[i] = distFunc(gen);
132 if (minPoint[i] > maxPoint[i]) {
133 std::swap(minPoint[i], maxPoint[i]);
134 }
135 }
136 return BoxType(minPoint, maxPoint);
137 });
138 });
139}

229template<Point3Concept PointType>
230PointType randomTriangleBarycentricCoordinate(
231 RandomConfig config = std::monostate())
232{
233 using ScalarType = PointType::ScalarType;
234
235 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
236 PointType interp;
237 std::uniform_real_distribution<ScalarType> unif(0, 1);
238
239 interp[1] = unif(gen);
240 interp[2] = unif(gen);
241 if (interp[1] + interp[2] > 1.0) {
242 interp[1] = 1.0 - interp[1];
243 interp[2] = 1.0 - interp[2];
244 }
245
246 interp[0] = 1.0 - (interp[1] + interp[2]);
247 return interp;
248 });
249}

280template<typename ScalarType>
281std::vector<ScalarType> randomPolygonBarycentricCoordinate(
282 uint polySize,
283 RandomConfig config = std::monostate())
284{
285 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
286 std::vector<ScalarType> barCoord(polySize);
287 ScalarType sum = static_cast<ScalarType>(0);
288
289 std::uniform_real_distribution<ScalarType> unif(0, 1);
290
291 for (uint i = 0; i < polySize; ++i) {
292 barCoord[i] = unif(gen);
293 sum += barCoord[i];
294 }
295
296 // Normalize so all weights sum to 1.
297 assert(sum != static_cast<ScalarType>(0));
298 for (uint i = 0; i < polySize; ++i) {
299 barCoord[i] /= sum;
300 }
301 return barCoord;
302 });
303}
