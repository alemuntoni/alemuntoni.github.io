26struct MData
27{
28 // The base path of the mesh, used to resolve relative texture paths.
29 std::string meshPath;
30 // A vector containing all the materials defined for the mesh.
31 std::vector<Material> materials;
32 // A map where keys are texture paths and values are the loaded Image data.
33 std::map<std::string, Image> textureImages;

39 vcl::serialize(os, textureImages);
40 }

42 void deserialize(std::istream& is)
43 {
44 vcl::deserialize(is, meshPath);
45 vcl::deserialize(is, materials);
46 vcl::deserialize(is, textureImages);

48};

334/* importFrom function */

336template<typename OtherMesh>
337void Materials::importFrom(const OtherMesh& m)
338{
339 if constexpr (HasMaterials<OtherMesh>) {
340 clearMaterials();
341 for (const auto& mat : m.materials()) {
342 mats().push_back(mat);
343 }
344 for (const auto& [path, img] : m.textureImages()) {
345 txtImgs()[path] = img;
346 }
347 meshBasePath() = m.meshBasePath();
348 }
349}
