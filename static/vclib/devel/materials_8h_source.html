385/* imoportFrom function */
386
387template<typename Element>
388void Materials::importFrom(const Element& e, bool)
389{
390 if constexpr (HasMaterials<Element>) {
391 mats().clear();
392 txtImgs().clear();
393 for (const auto& mat : e.materials()) {
394 mats().push_back(mat);
395 }
396 for (const auto& [path, img] : e.textureImages()) {
397 txtImgs()[path] = img;
398 }
399 meshBasePath() = e.meshBasePath();
400 }
401}
