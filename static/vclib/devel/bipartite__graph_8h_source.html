35template<class T1, class T2>
36class BipartiteGraph
37{
38 template<typename Graph, typename Iterator>
39 friend class detail::AdjacentLeftNodeIterator;
40
41 template<typename Graph, typename Iterator>
42 friend class detail::AdjacentRightNodeIterator;
43
44protected:
45 std::map<T1, unsigned int> mMapL;
46 std::map<T2, unsigned int> mMapR;
47
48 std::vector<UndirectedNode<T1>> mNodesL;
49 std::vector<UndirectedNode<T2>> mNodesR;
50
51 std::set<unsigned int> mUnusedLNodes;
52 std::set<unsigned int> mUnusedRNodes;
53
54public:
55 using LeftType = T1;
56 using RightType = T2;
57 using LeftNodeType = UndirectedNode<T1>;
58 using RightNodeType = UndirectedNode<T2>;
59
60 using LeftNodeIterator =
61 NodeIterator<typename std::vector<UndirectedNode<T1>>::const_iterator>;
62 using RightNodeIterator =
63 NodeIterator<typename std::vector<UndirectedNode<T2>>::const_iterator>;
64
65 using LeftNodeView = View<LeftNodeIterator>;
66 using RightNodeView = View<RightNodeIterator>;
67
68 using AdjacentLeftNodeIterator = detail::AdjacentLeftNodeIterator<
69 BipartiteGraph<T1, T2>,
70 std::unordered_set<unsigned int>::const_iterator>;
71 using AdjacentRightNodeIterator = detail::AdjacentRightNodeIterator<
72 BipartiteGraph<T1, T2>,
73 std::unordered_set<unsigned int>::const_iterator>;
74
75 using AdjacentLeftNodeView = View<AdjacentLeftNodeIterator>;
76 using AdjacentRightNodeView = View<AdjacentRightNodeIterator>;

81 BipartiteGraph() {}

88 bool leftNodeExists(const T1& lNode) const
89 {
90 return mMapL.find(lNode) != mMapL.end();
91 }
92
98 bool rightNodeExists(const T2& rNode) const
99 {
100 return mMapR.find(rNode) != mMapR.end();

107 uint leftNodeCount() const
108 {
109 return (unsigned int) (mNodesL.size() - mUnusedLNodes.size());
110 }
111
116 uint rightNodeCount() const
117 {
118 return (unsigned int) (mNodesR.size() - mUnusedRNodes.size());
119 }

126 uint adjacentLeftNodeCount(const T1& lNode) const
127 {
128 int uid = getIdLeftNode(lNode);
129 return mNodesL[uid].sizeAdjacentNodes();

137 uint adjacentRightNodeCount(const T2& rNode) const
138 {
139 int vid = getIdRightNode(rNode);

149 bool addLeftNode(const T1& info)
150 {
151 if (mMapL.find(info) == mMapL.end()) {
152 if (mUnusedLNodes.size() == 0) {
153 mMapL[info] = (unsigned int) mNodesL.size();
154 mNodesL.emplace_back(info);
155 }
156 else {
157 unsigned int id = *(mUnusedLNodes.begin());
158 mUnusedLNodes.erase(mUnusedLNodes.begin());
159 mMapL[info] = id;
160 mNodesL[id] = UndirectedNode<T1>(info);
161 }
162 return true;
163 }
164 else
165 return false;
166 }
167
174 bool addRightNode(const T2& info)
175 {
176 if (mMapR.find(info) == mMapR.end()) {
177 if (mUnusedRNodes.size() == 0) {
178 mMapR[info] = (unsigned int) mNodesR.size();
179 mNodesR.emplace_back(info);
180 }
181 else {
182 unsigned int id = *(mUnusedRNodes.begin());
183 mUnusedRNodes.erase(mUnusedRNodes.begin());
184 mMapR[info] = id;
185 mNodesR[id] = UndirectedNode<T2>(info);
186 }
187 return true;
188 }
189 else
190 return false;
191 }
192
198 bool deleteLeftNode(const T1& lNode)
199 {
200 if (clearAdjacenciesLeftNode(lNode)) {
201 mUnusedLNodes.insert(mMapL[lNode]);
202 mMapL.erase(lNode);
203 return true;
204 }
205 else
206 return false;
207 }
208
214 bool deleteRightNode(const T2& rNode)
215 {
216 if (clearAdjacenciesRightNode(rNode)) {
217 mUnusedRNodes.insert(mMapR[rNode]);
218 mMapR.erase(rNode);
219 return true;
220 }
221 else
222 return false;
223 }
224
232 bool addArc(const T1& lNode, const T2& rNode)
233 {
234 try {
235 int uid = getIdLeftNode(lNode);
236 int vid = getIdRightNode(rNode);
237 assert((unsigned int) uid < mNodesL.size());
238 assert((unsigned int) vid < mNodesR.size());
239 mNodesL[uid].addAdjacent(vid);
240 mNodesR[vid].addAdjacent(uid);
241 return true;
242 }
243 catch (...) {
244 return false;
245 }
246 }
247
255 bool deleteArc(const T1& lNode, const T2& rNode)
256 {
257 try {
258 int uid = getIdLeftNode(lNode);
259 int vid = getIdRightNode(rNode);
260 assert((unsigned int) uid < mNodesL.size());
261 assert((unsigned int) vid < mNodesR.size());
262 mNodesL[uid].deleteAdjacent(vid);
263 mNodesR[vid].deleteAdjacent(uid);
264 return true;
265 }
266 catch (...) {
267 return false;
268 }
269 }
270
277 bool clearAdjacenciesLeftNode(const T1& lNode)
278 {
279 try {
280 int uid = getIdLeftNode(lNode);
281 for (unsigned int adj : mNodesL[uid]) {
282 mNodesR[adj].deleteAdjacent(uid);
283 }
284 mNodesL[uid].clearAdjacentNodes();
285 return true;
286 }
287 catch (...) {
288 return false;
289 }
290 }
291
298 bool clearAdjacenciesRightNode(const T2& rNode)
299 {
300 try {
301 int vid = getIdRightNode(rNode);
302 for (unsigned int adj : mNodesR[vid]) {
303 mNodesL[adj].deleteAdjacent(vid);
304 }
305 mNodesR[vid].clearAdjacentNodes();
306 return true;
307 }
308 catch (...) {
309 return false;
310 }
311 }
312
319 bool setLeftNode(const T1& old, const T1& newInfo)
320 {
321 try {
322 int uid = getIdLeftNode(old);
323 mNodesL[uid] = UndirectedNode<T1>(newInfo);
324 mMapL.erase(old);
325 mMapL[newInfo] = uid;
326 return true;
327 }
328 catch (...) {
329 return false;
330 }
331 }
332
339 bool setRightNode(const T2& old, const T2& newInfo)
340 {
341 try {
342 int vid = getIdRightNode(old);
343 mNodesR[vid] = UndirectedNode<T2>(newInfo);
344 mMapR.erase(old);
345 mMapR[newInfo] = vid;
346 return true;
347 }
348 catch (...) {
349 return false;
350 }
351 }

353 AdjacentLeftNodeIterator adjacentLeftNodeBegin(const T1& lNode) const
354 {
355 int uid = getIdLeftNode(lNode);
356 return AdjacentLeftNodeIterator(*this, mNodesL[uid].begin());
357 }
358
359 AdjacentLeftNodeIterator adjacentLeftNodeEnd(const T1& lNode) const
360 {
361 int uid = getIdLeftNode(lNode);

363 }
364
365 AdjacentRightNodeIterator adjacentRightNodeBegin(const T2& rNode) const
366 {
367 int vid = getIdRightNode(rNode);
368 return AdjacentRightNodeIterator(*this, mNodesR[vid].begin());
369 }
370
371 AdjacentRightNodeIterator adjacentRightNodeEnd(const T2& rNode) const
372 {
373 int vid = getIdRightNode(rNode);
374 return AdjacentRightNodeIterator(*this, mNodesR[vid].end());
375 }

377 LeftNodeIterator leftNodeBegin() const
378 {
379 return LeftNodeIterator(mNodesL.begin());
380 }
381
382 LeftNodeIterator leftNodeEnd() const
383 {
384 return LeftNodeIterator(mNodesL.end());
385 }
386
387 RightNodeIterator rightNodeBegin() const
388 {
389 return RightNodeIterator(mNodesR.begin());
390 }
391
392 RightNodeIterator rightNodeEnd() const
393 {
394 return RightNodeIterator(mNodesR.end());
395 }
396
397 LeftNodeView leftNodes() const
398 {
399 return LeftNodeView(leftNodeBegin(), leftNodeEnd());
400 }
401
402 RightNodeView rightNodes() const
403 {
404 return RightNodeView(rightNodeBegin(), rightNodeEnd());
405 }
406
407 AdjacentLeftNodeView adjacentLeftNodes(const T1& lNode) const
408 {
409 return AdjacentLeftNodeView(
410 adjacentLeftNodeBegin(lNode), adjacentLeftNodeEnd(lNode));
411 }
412
413 AdjacentRightNodeView adjacentRightNodes(const T2& rNode) const
414 {
415 return AdjacentRightNodeView(
416 adjacentRightNodeBegin(rNode), adjacentRightNodeEnd(rNode));
417 }
418
419protected:
420 int getIdLeftNode(const T1& uNode) const { return mMapL.at(uNode); }
421
422 int getIdRightNode(const T2& vNode) const { return mMapR.at(vNode); }
423};
