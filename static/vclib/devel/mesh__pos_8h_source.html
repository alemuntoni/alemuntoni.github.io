31template<face::HasAdjacentFaces FaceType>
32class MeshPos
33{
34 const FaceType* mFace = nullptr;
35
36 const typename FaceType::VertexType* mVertex = nullptr;
37
38 short mEdge = -1;
39
40public:
41 using VertexType = FaceType::VertexType;
42
46 MeshPos() = default;
47
48 MeshPos(const FaceType* f, short e) : mFace(f), mEdge(e)
49 {
50 if (f != nullptr)
51 mVertex = f->vertex(e);
52 assert(isValid(mFace, mVertex, mEdge));
53 }
54
55 MeshPos(const FaceType* f, const VertexType* v) : mFace(f), mVertex(v)
56 {
57 for (uint i = 0; i < f->vertexCount(); i++)
58 if (f->vertex(i) == v)
59 mEdge = i;
60 assert(isValid(mFace, mVertex, mEdge));
61 }
62
73 MeshPos(const FaceType* f, const VertexType* v, short e) :
74 mFace(f), mVertex(v), mEdge(e)
75 {
76 assert(isValid(mFace, mVertex, mEdge));
77 }

92 static bool isValid(const FaceType* f, const VertexType* v, short e)
93 {
94 if (f == nullptr || v == nullptr || e < 0)
95 return false;
96 if (!comp::isAdjacentFacesAvailableOn(*f))
97 return false;
98 return (ushort) e < f->vertexCount() &&
99 (v == f->vertex(e) || v == f->vertexMod(e + 1));
100 }
101
102 const FaceType* face() const { return mFace; }
103
104 const VertexType* vertex() const { return mVertex; }
105
106 short edge() const { return mEdge; }
107
108 const FaceType* adjFace() const { return mFace->adjFace(mEdge); }
109
110 const VertexType* adjVertex() const
111 {
112 MeshPos<FaceType> tmpPos = *this;
113 tmpPos.flipVertex();
114 return tmpPos.vertex();
115 }
116

121 return tmpPos.edge();
122 }
123
128 bool isValid() const { return isValid(mFace, mVertex, mEdge); }
129
135 bool isNull() const

137 return mFace == nullptr || mVertex == nullptr || mEdge < 0;
138 }

147 bool isEdgeOnBorder() const { return mFace->adjFace(mEdge) == nullptr; }
148
157 bool isCCWOriented() const { return mFace->vertex(mEdge) == mVertex; }

170 bool flipFace()
171 {
172 const FaceType* nf = mFace->adjFace(mEdge);
173 if (nf != nullptr) {
174 short oldEdge = mEdge;
175 mEdge = nf->indexOfAdjFace(mFace);
176 if (mEdge < 0) {
177 // non-manifold edge. I need to find the edge that has the
178 // current vertex
179 const auto* v0 = mFace->vertex(oldEdge);
180 const auto* v1 = mFace->vertexMod(oldEdge + 1);
181 mEdge = nf->indexOfEdge(v0, v1);
182 // if this assert fails, it means that the edge was not found.
183 // you probably need to update adjacency information of the mesh
184 assert(mEdge >= 0 && mEdge < nf->vertexCount());
185 if (mEdge < 0) {
186 *this = MeshPos<FaceType>(); // reset to null
187 return false;
188 }
189 }
190 mFace = nf;
191 return true;
192 }
193 else {
194 return false;
195 }
196 }
197
202 void flipVertex()
203 {
204 if (mFace->vertexMod(mEdge) == mVertex) {
205 mVertex = mFace->vertexMod(mEdge + 1);
206 }
207 else {
208 mVertex = mFace->vertexMod(mEdge);
209 }
210 }
211
216 void flipEdge()
217 {
218 if (mFace->vertexMod(mEdge + 1) == mVertex) {
219 mEdge = (mEdge + 1) % (short) mFace->vertexCount();
220 }
221 else {
222 short n = mFace->vertexCount();
223 mEdge = ((mEdge - 1) % n + n) %
224 n; // be sure to get the right index of the previous edge
225 }
226 }

239 void nextEdgeAdjacentToV()
240 {
241 flipEdge();
242 flipFace();
243 }
244
253 bool nextEdgeOnBorderAdjacentToV()
254 {
255 MeshPos<FaceType> startPos = *this;
256 do {
257 nextEdgeAdjacentToV();
258 } while (*this != startPos && !isEdgeOnBorder());
259 return (*this != startPos);
260 }
261
270 uint numberOfAdjacentFacesToV() const
271 {
272 bool onBorder = false;
273 uint count = countAdjacentFacesToV(onBorder);
274 // if we visited a border, it means that we visited the all faces
275 // adjacent to v twice to reach the same starting MeshPos.
276 if (onBorder)
277 count /= 2;
278 return count;
279 }
280
281 bool operator==(const MeshPos& op) const
282 {
283 return mFace == op.mFace && mVertex == op.mVertex && mEdge == op.mEdge;
284 }
285
286 bool operator!=(const MeshPos& op) const { return !(*this == op); }
287
288 bool operator<(const MeshPos& op) const
289 {
290 if (mFace == op.mFace) {
291 if (mEdge == op.mEdge)
292 return mVertex < op.mVertex;
293 else
294 return mEdge < op.mEdge;
295 }
296 else {
297 return mFace < op.mFace;
298 }
299 }
300
301private:
302 uint countAdjacentFacesToV(bool& onBorder) const
303 {
304 uint count = 0;
305 onBorder = false;
306 // start from this MeshPos
307 MeshPos<FaceType> mp = *this;
308 do {
309 // go to the next edge in the star of v (if face is on border, will
310 // change just edge)
311 mp.nextEdgeAdjacentToV();
312 ++count;
313 // if we are visiting a border, it means that we will start to
314 // navigate in the opposite sense as we were before.
315 if (mp.isEdgeOnBorder()) // flag that we visited a border
316 onBorder = true;
317 }
318 // we end when we will reach the same pos (same triplet f-v-e) of start
319 while (mp != *this);
320 return count;
321 }
322};
