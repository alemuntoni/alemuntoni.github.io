64 Image(
65 const void* data,
66 uint w,
67 uint h,
68 bool yFlip = false,
69 Color::Format format = Color::Format::ABGR)
70 {
71 if (data) {
72 mImg.resize(h, w);
73 std::size_t size = w * h;
74
75 auto* cdata = reinterpret_cast<const uint32_t*>(data);
76
77 if (format == Color::Format::ABGR) {
78 std::copy(cdata, cdata + size, mImg.data());
79 }
80 else {
81 for (uint i = 0; i < h; i++) {
82 for (uint j = 0; j < w; j++) {
83 Color c(cdata[i * w + j], format);
84 mImg(i, j) = c.abgr();
85 }
86 }
87 }
88
89 if (yFlip) {
90 mirror();
91 }
92 }
93 }

222 bool isAlmostEqual(
223 const Image& other,
224 uint8_t colorTolerance = 0,
225 float failedPixelTolerancePercentage = 0.0f) const
226 {
227 if (width() != other.width() || height() != other.height()) {
228 return false;
229 }
230
231 const unsigned char* d1 = data();
232 const unsigned char* d2 = other.data();
233 std::size_t numPixels = width() * height();
234
235 if (colorTolerance == 0 && failedPixelTolerancePercentage == 0.0f) {
236 return std::memcmp(d1, d2, sizeInBytes()) == 0;
237 }
238
239 std::size_t failedPixels = 0;
240 for (std::size_t i = 0; i < numPixels; ++i) {
241 bool pixelMatch = true;
242 for (int c = 0; c < 4; ++c) {
243 int diff = static_cast<int>(d1[i * 4 + c]) -
244 static_cast<int>(d2[i * 4 + c]);
245 if (std::abs(diff) > colorTolerance) {
246 pixelMatch = false;
247 break;
248 }
249 }
250 if (!pixelMatch) {
251 failedPixels++;
252 }
253 }
254
255 float failedPercentage = static_cast<float>(failedPixels) / numPixels;
256 return failedPercentage <= failedPixelTolerancePercentage;
257 }
