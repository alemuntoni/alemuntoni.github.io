183 template<typename... VC>
184 Index addVertices(const PositionType& p, const VC&... v)
185 requires (std::is_constructible_v<PositionType, VC> && ...)
186 {
187 Index vid = vertexContainerSize();
188 reserveVertices(vid + sizeof...(VC) + 1);
189 addVertex(p);
190 (addVertex(v), ...);
191 return vid;
192 }

201 template<InputRange<PositionType> R>
202 Index addVertices(R&& range)
203 {
204 Index vid = vertexContainerSize();
205 if constexpr (std::ranges::sized_range<R>) {
206 reserveVertices(vid + std::ranges::size(range));
207 }
208 for (const auto& p : range) {
209 addVertex(p);
210 }
211 return vid;
212 }
