26class MaterialUniforms
27{
28 static inline std::array<float, 4> sBaseColor = {1.0, 1.0, 1.0, 1.0};
29
30 // metallic, roughness and occlusion are stored in the B, G and R channels
31 // respectively for consistency with textures
32 static inline std::array<float, 4> sFactorsPack = {
33 1.0, // occlusion strength
34 1.0, // roughness
35 1.0, // metallic
36 1.0 // normal scale
37 };
38
39 // emissive color factor stored in RGB channels, alpha channel is unused so
40 // it can be used to store the alpha cutoff when needed
41 static inline std::array<float, 4> sEmissiveAlphaCutoffPack =
42 {0.0, 0.0, 0.0, 0.5};
43
44 static inline StaticUniform sBaseColorUniform {
45 "u_baseColorFactor",
46 bgfx::UniformType::Vec4};
47 static inline StaticUniform sFactorsPackUniform {
48 "u_FactorsPack",
49 bgfx::UniformType::Vec4};
50 static inline StaticUniform sEmissiveAlphaCutoffPackUniform {
51 "u_emissiveAlphaCutoffPack",
52 bgfx::UniformType::Vec4};
53
54public:
55 MaterialUniforms() = delete;
56
57 static void set(const Material& m)
58 {
59 if (m.alphaMode() ==
60 Material::AlphaMode::ALPHA_MASK) { // alpha mode is MASK
61 sEmissiveAlphaCutoffPack[3] = m.alphaCutoff();
62 }
63 else {
64 sEmissiveAlphaCutoffPack[3] = -1.0f;
65 }
66
67 sBaseColor[0] = m.baseColor().redF();
68 sBaseColor[1] = m.baseColor().greenF();
69 sBaseColor[2] = m.baseColor().blueF();
70 sBaseColor[3] = m.baseColor().alphaF();
71
72 // metallic, roughness and occlusion are stored in the B, G and R
73 // channels respectively for consistency with textures
74 sFactorsPack[0] = m.occlusionStrength();
75 sFactorsPack[1] = m.roughness();
76 sFactorsPack[2] = m.metallic();
77 sFactorsPack[3] = m.normalScale();
78
79 sEmissiveAlphaCutoffPack[0] = m.emissiveColor().redF();
80 sEmissiveAlphaCutoffPack[1] = m.emissiveColor().greenF();
81 sEmissiveAlphaCutoffPack[2] = m.emissiveColor().blueF();
82 }
83
84 static void bind()
85 {
86 sBaseColorUniform.bind(sBaseColor.data());
87 sFactorsPackUniform.bind(sFactorsPack.data());
88 sEmissiveAlphaCutoffPackUniform.bind(sEmissiveAlphaCutoffPack.data());
89 }
90};
