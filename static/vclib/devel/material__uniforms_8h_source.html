26class MaterialUniforms
27{
28 static inline std::array<float, 4> sBaseColor = {1.0, 1.0, 1.0, 1.0};
29
30 // metallic, roughness and occlusion are stored in the B, G and R channels
31 // respectively for consistency with textures
32 static inline std::array<float, 4> sFactorsPack = {
33 1.0, // occlusion strength
34 1.0, // roughness
35 1.0, // metallic
36 1.0 // normal scale
37 };
38
39 // alpha cutoff and maybe other alpha related settings can be stored here
40 static inline std::array<float, 4> sAlphaPack = {-1.0, 0.0, 0.0, 0.0};
41
42 // emissive color factor stored in RGB channels,
43 // alpha stores the emissive strength
44 static inline std::array<float, 4> sEmissivePack = {0.0, 0.0, 0.0, 1.0};
45
46 static inline std::array<float, 4> sClearcoatPack = {
47 0.0, // clearcoat factor
48 0.0, // clearcoat roughness factor
49 1.0, // clearcoat normal scale
50 0.0};
51
52 static inline std::array<float, 4> sAnisotropyPack = {
53 0.0, // anisotropy strength
54 0.0, // anisotropy rotation
55 0.0,
56 0.0};
57
58 // specular color (RGB) and specular factor (A)
59 static inline std::array<float, 4> sSpecularPack = {1.0, 1.0, 1.0, 1.0};
60
61 static inline StaticUniform sBaseColorUniform {
62 "u_baseColorFactor",
63 bgfx::UniformType::Vec4};
64 static inline StaticUniform sFactorsPackUniform {
65 "u_FactorsPack",
66 bgfx::UniformType::Vec4};
67 static inline StaticUniform sAlphaPackUniform {
68 "u_alphaPack",
69 bgfx::UniformType::Vec4};
70 static inline StaticUniform sEmissivePackUniform {
71 "u_emissivePack",
72 bgfx::UniformType::Vec4};
73 static inline StaticUniform sClearcoatPackUniform {
74 "u_clearcoatPack",
75 bgfx::UniformType::Vec4};
76 static inline StaticUniform sAnisotropyPackUniform {
77 "u_anisotropyPack",
78 bgfx::UniformType::Vec4};
79 static inline StaticUniform sSpecularPackUniform {
80 "u_specularPack",
81 bgfx::UniformType::Vec4};
82
83public:
84 MaterialUniforms() = delete;
85
86 static void set(const Material& m)
87 {
88 if (m.alphaMode() ==
89 Material::AlphaMode::ALPHA_MASK) { // alpha mode is MASK
90 sAlphaPack[0] = m.alphaCutoff();
91 }
92 else {
93 sAlphaPack[0] = -1.0f;
94 }
95
96 sBaseColor[0] = m.baseColor().redF();
97 sBaseColor[1] = m.baseColor().greenF();
98 sBaseColor[2] = m.baseColor().blueF();
99 sBaseColor[3] = m.baseColor().alphaF();
100
101 // metallic, roughness and occlusion are stored in the B, G and R
102 // channels respectively for consistency with textures
103 sFactorsPack[0] = m.occlusionStrength();
104 sFactorsPack[1] = m.roughness();
105 sFactorsPack[2] = m.metallic();
106 sFactorsPack[3] = m.normalScale();
107
108 sAnisotropyPack[0] = m.anisotropyStrength();
109 sAnisotropyPack[1] = m.anisotropyRotation();
110
111 sSpecularPack[0] = m.specularColor()[0];
112 sSpecularPack[1] = m.specularColor()[1];
113 sSpecularPack[2] = m.specularColor()[2];
114 sSpecularPack[3] = m.specularFactor();
115
116 sEmissivePack[0] = m.emissiveColor().redF();
117 sEmissivePack[1] = m.emissiveColor().greenF();
118 sEmissivePack[2] = m.emissiveColor().blueF();
119 sEmissivePack[3] = m.emissiveStrength();
120
121 sClearcoatPack[0] = m.clearcoatFactor();
122 sClearcoatPack[1] = m.clearcoatRoughness();
123 sClearcoatPack[2] = m.clearcoatNormalScale();
124 }
125
126 static void bind()
127 {
128 sBaseColorUniform.bind(sBaseColor.data());
129 sFactorsPackUniform.bind(sFactorsPack.data());
130 sAlphaPackUniform.bind(sAlphaPack.data());
131 sEmissivePackUniform.bind(sEmissivePack.data());
132 sClearcoatPackUniform.bind(sClearcoatPack.data());
133 sAnisotropyPackUniform.bind(sAnisotropyPack.data());
134 sSpecularPackUniform.bind(sSpecularPack.data());
135 }
136};
