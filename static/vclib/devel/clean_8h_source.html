42/* Comparison class for vertex position comparison */
43template<typename MeshType>
44class VertPositionComparator
45{
46 const MeshType* mMesh = nullptr;
47
48public:
49 VertPositionComparator(const MeshType& m) : mMesh(&m) {}
50
51 inline bool operator()(
52 typename MeshType::IndexType a,
53 typename MeshType::IndexType b) const
54 {
55 const auto& pa = mMesh->vertex(a).position();
56 const auto& pb = mMesh->vertex(b).position();
57 return (pa == pb) ? (a < b) : (pa < pb);
58 }
59};

73template<typename IndexType, typename SentinelType, int N>
74class SortedIndexContainer
75{
76public:
77 SortedIndexContainer() = default;
78
79 template<Range RangeType>
80 SortedIndexContainer(SentinelType s, RangeType&& rng) :
81 s(s), v(std::forward<RangeType>(rng))
82 {
83 std::sort(v.begin(), v.end());
84 }
85
86 bool operator<(const SortedIndexContainer& other) const
87 {
88 if constexpr (N >= 0) {
89 for (uint i = 0; i < N; ++i) {
90 if (v[i] != other.v[i])
91 return v[i] < other.v[i];
92 }
93 return false;
94 }
95 else {
96 for (std::size_t i = 0; i < v.size() && i < other.v.size(); ++i) {
97 if (v[i] != other.v[i])
98 return v[i] < other.v[i];
99 }
100 return v.size() < other.v.size();
101 }
102 }
103
104 bool operator==(const SortedIndexContainer& other) const
105 {
106 if constexpr (N >= 0) {
107 for (uint i = 0; i < N; ++i) {
108 if (v[i] != other.v[i])
109 return false;
110 }
111 return true;
112 }
113 else {
114 if (v.size() != other.v.size())
115 return false;
116 for (std::size_t i = 0; i < v.size(); ++i) {
117 if (v[i] != other.v[i])
118 return false;
119 }
120 return true;
121 }
122 }
123
124 SentinelType sentinel() const { return s; }
125
126private:
127 Vector<IndexType, N> v;
128 SentinelType s;
129};

131template<typename Cont, MeshConcept MeshType>
132void deleteElementsWithDeletedVertices(
133 MeshType& m,
134 typename MeshType::IndexType& deletedCount)
135{
136 if constexpr (comp::HasVertexReferences<typename Cont::ElementType>) {
137 constexpr ElemId ELEM_ID = Cont::ElementType::ELEMENT_ID;
138 deletedCount +=
139 deleteElementsIf<MeshType, ELEM_ID>(m, [](auto el) {
140 for (auto v : el.vertices()) {
141 if (v == nullptr || v.deleted()) {
142 return true;
143 }
144 }
145 return false;
146 });
147 }
148}

179template<MeshConcept MeshType>
180auto removeUnreferencedVertices(MeshType& m)
181{
182 using IndexType = typename MeshType::IndexType;
183
184 // Generate a vector of boolean flags indicating whether each vertex is
185 // referenced by any of the mesh's elements.
186 IndexType n = 0;
187 std::vector<bool> refVertices = referencedVertices(m, n);
188
189 if (n > 0) {
190 return deleteVerticesIf(
191 m, [&](auto v) { return !refVertices[v.index()]; });
192 }
193
194 return IndexType(0);
195}

215template<MeshConcept MeshType>
216auto removeDuplicateVertices(MeshType& m)
217{
218 using IndexType = typename MeshType::IndexType;
219
220 if (m.vertexCount() == 0)
221 return IndexType(0);
222
223 // a map that will be used to keep track of deleted vertices and their
224 // corresponding indices.
225 std::vector<IndexType> newVertexIndices(m.vertexContainerSize());
226 // assigning each vertex index to itself.
227 std::iota(newVertexIndices.begin(), newVertexIndices.end(), IndexType(0));
228
229 // mark any already deleted vertices as NULL_INDEX
230 for (IndexType v = 0; v < m.vertexContainerSize(); ++v) {
231 if (m.vertex(v).deleted()) {
232 newVertexIndices[v] = MeshType::NULL_INDEX;
233 }
234 }
235
236 IndexType deleted = 0;
237
238 std::vector<IndexType> perm;
239 perm.reserve(m.vertexCount());
240
241 // put all the active vertices into a vector for sorting.
242 for (auto v : m.vertices()) {
243 perm.push_back(v.index());
244 }
245
246 // sort the vector based on the vertices' spatial positions.
247 std::sort(
248 std::execution::par_unseq,
249 perm.begin(),
250 perm.end(),
251 detail::VertPositionComparator<MeshType>(m));
252
253 std::size_t i = 0;
254
255 // compare the i-th position with the next ones while they are equal to the
256 // i-th.
257 while (i + 1 < perm.size()) {
258 std::size_t j = i + 1;
259 while (j < perm.size() &&
260 m.vertex(perm[i]).position() == m.vertex(perm[j]).position()) {
261 // j will be deleted, so we map its index to the i-th vertex's index.
262 newVertexIndices[perm[j]] = perm[i]; // map j -> i
263 m.deleteVertex(perm[j]);
264 j++;
265 deleted++;
266 }
267 // here perm[i] != perm[j], so we need to check perm[j] with the next
268 // vertex.
269 i = j;
270 }
271
272 // update the vertex references to point to the correct vertices, in every
273 // container of the mesh
274 if (deleted > 0) {
275 m.updateVertexReferences(newVertexIndices);
276 }
277
278 // todo:
279 // - add a flag that removes degenerate elements after
280 return deleted;
281}

305template<FaceMeshConcept MeshType>
306auto removeDuplicateFaces(MeshType& m)
307{
308 using IndexType = typename MeshType::IndexType;
309 using FaceType = typename MeshType::Face;
310
311 // create a vector of sorted tuples of indices, where each tuple represents
312 // a face's vertices and its face index.
313 std::vector<detail::SortedIndexContainer<
314 IndexType,
315 IndexType,
316 FaceType::VERTEX_COUNT>>
317 fvec;
318 fvec.reserve(m.faceCount());
319
320 for (auto f : m.faces()) {
321 fvec.emplace_back(f.index(), f.vertexIndices());
322 }
323
324 // sort the vector based on the face vertex indices.
325 std::sort(std::execution::par_unseq, fvec.begin(), fvec.end());
326
327 std::vector<bool> shouldDelete(m.faceContainerSize(), false);
328
329 // iterate over the sorted vector, and mark any duplicate faces as deleted.
330 for (std::size_t i = 0; i + 1 < fvec.size(); ++i) {
331 if (fvec[i] == fvec[i + 1]) {
332 shouldDelete[fvec[i].sentinel()] = true;
333 }
334 }
335
336 return deleteFacesIf(m, [&](auto f) { return shouldDelete[f.index()]; });
337}

362template<MeshConcept MeshType>
363auto removeDegenerateVertices(MeshType& m, bool deleteAlsoIncidentElements = false)
364{
365 using IndexType = typename MeshType::IndexType;
366
367 auto count_vd = deleteVerticesIf(
368 m, [](auto v) { return v.position().isDegenerate(); });
369
370 if (deleteAlsoIncidentElements && count_vd > 0) {
371 IndexType dummy = 0;
372 detail::deleteElementsWithDeletedVertices(
373 m, dummy, typename MeshType::ElementContainers());
374 }
375
376 return count_vd;
377}

410template<FaceMeshConcept MeshType>
411auto removeDegenerateFaces(MeshType& m)
412{
413 using FaceType = typename MeshType::Face;
414
415 if constexpr (PolygonFaceConcept<FaceType>) {
416 for (auto f : m.faces()) {
417 for (int i = int(f.vertexCount()) - 1; i >= 0; --i) {
418 if (f.vertexCount() < 3)
419 break;
420 if (f.vertexIndex(i) == f.vertexIndexMod(i + 1)) {
421 f.eraseVertex(i);
422 }
423 }
424 }
425 }
426
427 return deleteFacesIf(m, [&](auto f) {
428 if constexpr (PolygonFaceConcept<FaceType>) {
429 if (f.vertexCount() < 3) {
430 return true;
431 }
432 }
433 for (uint i = 0; i < f.vertexCount(); ++i) {
434 if (f.vertexIndex(i) == f.vertexIndexMod(i + 1)) {
435 return true;
436 }
437 }
438 return false;
439 });
440}
