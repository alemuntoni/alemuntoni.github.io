42class VertPositionComparator
43{
44public:
45 inline bool operator()(const VertexPointer& a, const VertexPointer& b)
46 {

48 (a < b) :
49 (a->position() < b->position());

65template<typename IndexType, typename SentinelType, int N>
66class SortedIndexContainer
67{
68public:
69 SortedIndexContainer() {}
70
71 template<Range RangeType>
72 SortedIndexContainer(SentinelType s, RangeType rng) : s(s), v(rng)
73 {
74 std::sort(v.begin(), v.end());
75 }
76
77 bool operator<(const SortedIndexContainer& s) const
78 {
79 if constexpr (N >= 0) {
80 for (uint i = 0; i < N; ++i) {
81 if (v[i] != s.v[i])
82 return v[i] < s.v[i];
83 }
84 return false;
85 }
86 else {
87 for (uint i = 0; i < v.size() && i < s.v.size(); ++i) {
88 if (v[i] != s.v[i])
89 return v[i] < s.v[i];
90 }
91 return v.size() < s.v.size();
92 }
93 }
94
95 bool operator==(const SortedIndexContainer& s) const
96 {
97 if constexpr (N >= 0) {
98 for (uint i = 0; i < N; ++i) {
99 if (v[i] != s.v[i])
100 return false;
101 }
102 return true;
103 }
104 else {
105 if (v.size() != s.v.size())
106 return false;
107 for (uint i = 0; i < v.size(); ++i) {
108 if (v[i] != s.v[i])
109 return false;
110 }
111 return true;
112 }
113 }
114
115 SentinelType sentinel() const { return s; }
116
117private:
118 Vector<IndexType, N> v;
119 SentinelType s;
120};

168template<MeshConcept MeshType>
169uint removeUnreferencedVertices(MeshType& m)
170{
171 using VertexType = MeshType::VertexType;
172
173 // Generate a vector of boolean flags indicating whether each vertex is
174 // referenced by any of the mesh's elements.
175
176 uint n = 0;
177 std::vector<bool> refVertices = referencedVertices<std::vector<bool>>(m, n);
178
179 // need to mark as deleted vertices only if the number of unreferenced is
180 // less than vn
181 if (n < m.vertexCount()) {
182 deleteVerticesIf(m, [&](const VertexType& v) {
183 return !refVertices[m.index(v)];
184 });
185 }
186
187 return n;
188}

208template<MeshConcept MeshType>
209uint removeDuplicateVertices(MeshType& m)
210{
211 using VertexType = MeshType::VertexType;
212 using VertexPointer = MeshType::VertexType*;
213
214 if (m.vertexCount() == 0)
215 return 0;
216
217 // a map that will be used to keep track of deleted vertices and their
218 // corresponding pointers.
219 std::vector<uint> newVertexIndices(m.vertexContainerSize());
220 // assigning each vertex index to itself.
221 std::iota(newVertexIndices.begin(), newVertexIndices.end(), 0);
222
223 uint deleted = 0;
224
225 std::vector<VertexPointer> perm(m.vertexCount());
226
227 // put all the vertices into a vector for sorting.
228 uint k = 0;
229 for (VertexType& v : m.vertices())
230 perm[k++] = &v;
231
232 // sort the vector based on the vertices' spatial positions.
233 std::sort(
234 std::execution::par_unseq,
235 perm.begin(),
236 perm.end(),
237 detail::VertPositionComparator<VertexPointer>());
238
239 uint i = 0;
240
241 // compare the i-th position with the next ones while they are equal to the
242 // i-th.
243 while (i < perm.size() - 1) {
244 uint j = i + 1;
245 while (j < perm.size() && perm[i]->position() == perm[j]->position()) {
246 // j will be deleted, so we map its pointer to the i-th vertex's
247 // pointer.
248 newVertexIndices[m.index(perm[j])] = m.index(perm[i]); // map j -> i
249 m.deleteVertex(m.index(perm[j]));
250 j++;
251 deleted++;
252 }
253 // here perm[i] != perm[j], so we need to check perm[j] with the next
254 // vertex.
255 i = j;
256 }
257
258 // update the vertex pointers to point to the correct vertices, in every
259 // container of the mesh
260 m.updateVertexReferences(newVertexIndices);
261
262 // todo:
263 // - add a flag that removes degenerate elements after
264 return deleted;
265}

293template<FaceMeshConcept MeshType>
294uint removeDuplicateFaces(MeshType& m)
295{
296 using VertexType = MeshType::VertexType;
297 using FaceType = MeshType::FaceType;
298
299 // create a vector of sorted tuples of indices, where each tuple represents
300 // a face's vertices and a pointer to the face.
301 std::vector<detail::SortedIndexContainer<
302 VertexType*,
303 FaceType*,
304 FaceType::VERTEX_COUNT>>
305 fvec;
306
307 for (FaceType& f : m.faces()) {
308 fvec.emplace_back(&f, f.vertices());
309 }
310
311 // sort the vector based on the face vertex indices.
312 std::sort(std::execution::par_unseq, fvec.begin(), fvec.end());
313
314 std::vector<bool> shouldDelete(m.faceContainerSize(), false);
315
316 // iterate over the sorted vector, and mark any duplicate faces as deleted.
317 for (uint i = 0; i < fvec.size() - 1; ++i) {
318 if (fvec[i] == fvec[i + 1]) {
319 shouldDelete[m.index(fvec[i].sentinel())] = true;
320 }
321 }
322
323 return deleteFacesIf(m, shouldDelete);
324}

349template<MeshConcept MeshType>
350uint removeDegenerateVertices(MeshType& m, bool deleteAlsoIncidentElements)
351{
352 using VertexType = MeshType::VertexType;
353
354 uint count_vd = deleteVerticesIf(m, [](const VertexType& v) {
355 return v.position().isDegenerate();
356 });
357
358 if (deleteAlsoIncidentElements) {
359 uint dummy = 0;
360 detail::deleteElementsWithDeletedVertices(
361 m, dummy, typename MeshType::Containers());
362 }
363
364 return count_vd;
365}

399template<FaceMeshConcept MeshType>
400uint removeDegenerateFaces(MeshType& m)
401{
402 using FaceType = typename MeshType::FaceType;
403
404 if constexpr (PolygonFaceConcept<FaceType>) {
405 for (FaceType& f : m.faces()) {
406 for (int i = int(f.vertexCount()) - 1; i >= 0; --i) {
407 if (f.vertexCount() < 3)
408 break;
409 if (f.vertex(i) == f.vertexMod(i + 1)) {
410 f.eraseVertex(i);
411 }
412 }
413 }
414 }
415
416 return deleteFacesIf(m, [&](const FaceType& f) {
417 if constexpr (PolygonFaceConcept<FaceType>) {
418 if (f.vertexCount() < 3) {
419 return true;
420 }
421 }
422 for (uint i = 0; i < f.vertexCount(); ++i) {
423 if (f.vertex(i) == f.vertexMod(i + 1)) {
424 return true;
425 }
426 }
427 return false;
428 });
429}
