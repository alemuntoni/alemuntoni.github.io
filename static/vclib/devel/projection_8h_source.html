25template<Point3IteratorConcept Iterator>
26auto project(Iterator begin, Iterator end)
27{
28 using PointType = Iterator::value_type;
29 using ScalarType = PointType::ScalarType;
30
31 Polygon2<ScalarType> projectedPolygon;
32
33 // Calculate the normal vector of the polygon and an orthonormal basis
34 // for the plane containing the polygon.
35 PointType normal = Polygon3<ScalarType>::normal(begin, end);
36 PointType u, v;
37 normal.orthoBase(u, v);
38
39 projectedPolygon.reserve(std::distance(begin, end));
40
41 // Project each vertex onto the plane defined by the orthonormal basis.
42 for (auto it = begin; it != end; ++it) {
43 ScalarType x = (*it).dot(u);
44 ScalarType y = (*it).dot(v);
45 projectedPolygon.pushBack(Point2<ScalarType>(x, y));
46 }
47
48 return projectedPolygon;
49}
