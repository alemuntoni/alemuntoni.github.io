286template<typename T>
287concept FaceConcept =
288 IsDerivedFromSpecializationOfV<T, Face> &&
289 RemoveRef<T>::ELEMENT_ID == ElemId::FACE && face::HasBitFlags<T> &&
290 face::HasVertexReferences<T> &&
291 (RemoveRef<T>::VERTEX_COUNT < 0 || RemoveRef<T>::VERTEX_COUNT >= 3) &&
292 (!face::HasTriangleBitFlags<T> || RemoveRef<T>::VERTEX_COUNT == 3) &&
293 comp::SanityCheckAdjacentEdges<T> && comp::SanityCheckAdjacentFaces<T> &&
294 comp::SanityCheckWedgeColors<T> && comp::SanityCheckWedgeTexCoords<T>;
295
296template<typename T>
297concept TriangleFaceConcept = RemoveRef<T>::VERTEX_COUNT == 3 && FaceConcept<T>;

308template<typename T>
