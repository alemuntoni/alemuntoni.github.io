101template<
102 typename DerivedComponent, // CRTP pattern, derived class
103 uint COMP_ID, // component id
104 typename DataType, // data stored by the component
105 typename ParentElemType, // parent element type
106 bool VERT, // true if component vertical
107 bool OPT, // true if component vertical and optional
108 typename... ReferencedTypes> // types of the refs stored by the component
109class Component : public ReferencesComponentTriggerer<ReferencedTypes>...
110{
111public:
116 using DataValueType = DataType;
117
122 static const bool IS_VERTICAL =
123 !std::is_same_v<ParentElemType, void> && VERT;
124
128 static const uint COMPONENT_ID = COMP_ID;
129
134 static const bool IS_OPTIONAL = VERT && OPT;
135
136private:
137 detail::ComponentData<DataValueType, IS_VERTICAL> mData;
138
139public:
151 bool isAvailable() const
152 {
153 return mData.template isComponentAvailable<ParentElemType>(
154 static_cast<const DerivedComponent*>(this));
155 }
156
157protected:
158 DataValueType& data()
159 {
160 return mData.template get<ParentElemType>(
161 static_cast<DerivedComponent*>(this));
162 }
163
164 const DataValueType& data() const
165 {
166 return mData.template get<ParentElemType>(
167 static_cast<const DerivedComponent*>(this));
168 }
169
170 ParentElemType* parentElement()
171 {
172 static_assert(
173 !std::is_same_v<ParentElemType, void>,
174 "The component should know its parent element type to get access "
175 "to its pointer. You should define the component by passing the "
176 "element type as template parameter. E.G., for a Face element:\n"
177 "vcl::face::TriangleVertexPtrs<Vertex<Scalar>, Face<Scalar>>\n"
178 " ^^^^^^^^^^^^ \n");
179
180 return static_cast<ParentElemType*>(this);
181 }
182
183 const ParentElemType* parentElement() const
184 {
185 static_assert(
186 !std::is_same_v<ParentElemType, void>,
187 "The component should know its parent element type to get access "
188 "to its pointer. You should define the component by passing the "
189 "element type as template parameter. E.G., for a Face element:\n"
190 "vcl::face::TriangleVertexPtrs<Vertex<Scalar>, Face<Scalar>>\n"
191 " ^^^^^^^^^^^^ \n");
192
193 return static_cast<const ParentElemType*>(this);
194 }
195};

209template<uint COMPONENT_ID, typename T>
210bool isComponentAvailableOn(const T& obj)
211{
212 if constexpr (HasOptionalComponentOfType<T, COMPONENT_ID>) {
213 using ComponentType =
214 ComponentOfType<COMPONENT_ID, typename T::Components>;
215 const ComponentType& c = static_cast<const ComponentType&>(obj);
216 return c.isAvailable();
217 }
218 else
219 return HasComponentOfType<T, COMPONENT_ID>;
220}

242// Components can be individuated with their ID, which is an unsigned int.
243// This struct sets its bool `value` to true if this Element has a Component
244// with the given unsigned integer COMP_ID. Sets also `type` with a
245// TypeWrapper containing the Component that satisfied the condition. The
246// TypeWrapper will be empty if no Components were found.
247template<uint COMP_ID, typename TypeWrapperComponents>
248struct ComponentIDPred
249{
250private:
251 template<typename Cont>
252 struct SameCmpPred
253 {
254 static constexpr bool value = Cont::COMPONENT_ID == COMP_ID;
255 };
256
257public:
258 // TypeWrapper of the found component, if any
259 using type =
260 FilterTypesByCondition<SameCmpPred, TypeWrapperComponents>::type;
261 static constexpr bool value = NumberOfTypes<type>::value == 1;
262};

264template<uint COMP_ID, typename TypeWrapperComponents>
265struct GetComponentFromID
266{
267private:
268 template<typename>
269 struct TypeUnwrapper
270 {
271 };
272
273 template<typename C>
274 struct TypeUnwrapper<TypeWrapper<C>>
275 {
276 using type = C;
277 };
278
279public:
280 using type = TypeUnwrapper<
281 typename ComponentIDPred<COMP_ID, TypeWrapperComponents>::type>::type;
282};
