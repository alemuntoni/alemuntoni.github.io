18template<MeshConcept MeshType, typename MatrixType>
19void updateNormalsAndDirections(MeshType& mesh, const MatrixType& matrix)
20{
21 if constexpr (HasVertexNormal<MeshType>) {
22 if (mesh.isVertexNormalAvailable()) {
23 multiplyNormalsByMatrix(
24 mesh.vertices() | vcl::views::normals, matrix);
25 }
26 }
27
28 if constexpr (HasFaceNormal<MeshType>) {
29 if (mesh.isFaceNormalAvailable()) {
30 multiplyNormalsByMatrix(
31 mesh.faces() | vcl::views::normals, matrix);
32 }
33 }
34
35 if constexpr (HasEdgeNormal<MeshType>) {
36 if (mesh.isEdgeNormalAvailable()) {
37 multiplyNormalsByMatrix(
38 mesh.edges() | vcl::views::normals, matrix);
39 }
40 }
41
42 if constexpr (HasFaceWedgeNormals<MeshType>) {
43 if (mesh.isFaceWedgeNormalsAvailable()) {
44 for (auto f : mesh.faces()) {
45 for (auto w : f.wedges()) {
46 w.normal() = multiplyNormalByMatrix(w.normal(), matrix);
47 }
48 }
49 }
50 }
51
52 if constexpr (HasVertexTangent<MeshType>) {
53 if (mesh.isVertexTangentAvailable()) {
54 multiplyNormalsByMatrix(
55 mesh.vertices() | vcl::views::tangents, matrix);
56
57 Matrix33<double> m33;
58 if constexpr (Matrix44Concept<MatrixType>) {
59 m33 = matrix.template cast<double>().block(0, 0, 3, 3);
60 }
61 else {
62 m33 = matrix.template cast<double>();
63 }
64
65 if (m33.determinant() < 0) {
66 for (auto v : mesh.vertices()) {
67 v.tangentRightHanded() = !v.tangentRightHanded();
68 }
69 }
70 }
71 }
72
73 if constexpr (HasVertexPrincipalCurvature<MeshType>) {
74 if (mesh.isVertexPrincipalCurvatureAvailable()) {
75 for (auto v : mesh.vertices()) {
76 v.principalCurvature().maxDir() =
77 multiplyNormalByMatrix(v.principalCurvature().maxDir(), matrix);
78 v.principalCurvature().minDir() =
79 multiplyNormalByMatrix(v.principalCurvature().minDir(), matrix);
80 }
81 }
82 }
83}

115template<MeshConcept MeshType, typename ScalarM>
116void applyTransformMatrix(
117 MeshType& mesh,
118 const Matrix44<ScalarM>& matrix,
119 bool updateNormals = true)
120{
121 multiplyPointsByMatrix(mesh.vertices() | vcl::views::positions, matrix);

139template<MeshConcept MeshType, PointConcept PointType>

141{

159template<MeshConcept MeshType, PointConcept PointType>

216template<MeshConcept MeshType, typename Scalar>
217void rotate(
218 MeshType& mesh,
219 const Matrix33<Scalar>& m,
220 bool updateNormals = true)
221{
222 using ScalarType = ScalarType<MeshType>;
223 Matrix33<ScalarType> rotMat = m.template cast<ScalarType>();
224
225 for (auto&& v : mesh.vertices()) {
226 v.position() = rotMat * v.position();
227 }
228
229 if (updateNormals) {
230 detail::updateNormalsAndDirections(mesh, rotMat);
231 }
232}
