21template<typename PositionType>
22struct LaplacianInfo
23{
24 using ScalarType = PositionType::ScalarType;
25 PositionType sum;
26 ScalarType cnt;
27};
28
29template<typename MeshType>
30void accumulateLaplacianInfo(
31 const MeshType& m,
32 std::vector<LaplacianInfo<PositionType<MeshType>>>& data,
33 bool cotangentFlag = false)
34{
35 using PositionType = PositionType<MeshType>;
36 using ScalarType = PositionType::ScalarType;
37
38 ScalarType weight = 1.0f;
39
40 for (auto&& f : m.faces()) {
41 for (uint j = 0; j < f.vertexCount(); ++j) {
42 if (!f.edgeOnBorder(j)) {
43 auto v0 = f.vertex(j);
44 auto v1 = f.vertexMod(j + 1);
45 auto v2 = f.vertexMod(j + 2);
46 const PositionType& p0 = v0.position();
47 const PositionType& p1 = v1.position();
48 const PositionType& p2 = v2.position();
49 if (cotangentFlag) {
50 ScalarType angle = PositionType(p1 - p2).angle(p0 - p2);
51 weight = std::tan((M_PI * 0.5) - angle);
52 }
53
54 data[v0.index()].sum += p1 * weight;
55 data[v1.index()].sum += p0 * weight;
56 data[v0.index()].cnt += weight;
57 data[v1.index()].cnt += weight;
58 }
59 }
60 }
61 // si azzaera i dati per i vertici di bordo
62 for (auto&& f : m.faces()) {
63 for (uint j = 0; j < f.vertexCount(); ++j) {
64 if (f.edgeOnBorder(j)) {
65 auto v0 = f.vertex(j);
66 auto v1 = f.vertexMod(j + 1);
67 data[v0.index()].sum = v0.position();
68 data[v1.index()].sum = v1.position();
69 data[v0.index()].cnt = 1;
70 data[v1.index()].cnt = 1;
71 }
72 }
73 }
74
75 // se l'edge j e' di bordo si deve mediare solo con gli adiacenti
76 for (auto&& f : m.faces()) {
77 for (uint j = 0; j < f.vertexCount(); ++j) {
78 if (f.edgeOnBorder(j)) {
79 auto v0 = f.vertex(j);
80 auto v1 = f.vertexMod(j + 1);
81 data[v0.index()].sum += v1.position();
82 data[v1.index()].sum += v0.position();
83 ++data[v0.index()].cnt;
84 ++data[v1.index()].cnt;
85 }
86 }
87 }
88}

106template<FaceMeshConcept MeshType>
107void laplacianSmoothing(
108 MeshType& m,
109 uint step,
110 bool smoothSelected = false,
111 bool cotangentWeight = false /*, CallBackPos *cb*/)
112{
113 using PositionType = PositionType<MeshType>;
114
115 const detail::LaplacianInfo<PositionType> lpz = {PositionType(0, 0, 0), 0};
116
117 for (uint i = 0; i < step; ++i) {
118 std::vector<detail::LaplacianInfo<PositionType>> laplData(
119 m.vertexContainerSize(), lpz);
120 detail::accumulateLaplacianInfo(m, laplData, cotangentWeight);
121 for (auto v : m.vertices()) {
122 if (laplData[v.index()].cnt > 0) {
123 if (!smoothSelected || v.selected()) {
124 v.position() = (v.position() + laplData[v.index()].sum) /
125 (laplData[v.index()].cnt + 1);
126 }
127 }
128 }
129 }
130}

132template<FaceMeshConcept MeshType>
133void taubinSmoothing(
134 MeshType& m,
135 uint step,
136 float lambda,
137 float mu,
138 bool smoothSelected = false /*, CallBackPos *cb*/)
139{
140 using PositionType = PositionType<MeshType>;
141
142 const detail::LaplacianInfo<PositionType> lpz = {PositionType(0, 0, 0), 0};
143
144 for (uint i = 0; i < step; ++i) {
145 std::vector<detail::LaplacianInfo<PositionType>> laplData(
146 m.vertexContainerSize(), lpz);
147 detail::accumulateLaplacianInfo(m, laplData);
148 for (auto v : m.vertices()) {
149 if (laplData[v.index()].cnt > 0) {
150 if (!smoothSelected || v.selected()) {
151 PositionType delta =
152 laplData[v.index()].sum / laplData[v.index()].cnt -
153 v.position();
154 v.position() = v.position() + delta * lambda;
155 }
156 }
157 }
158 std::fill(laplData.begin(), laplData.end(), lpz);
159 detail::accumulateLaplacianInfo(m, laplData);
160 for (auto v : m.vertices()) {
161 if (laplData[v.index()].cnt > 0) {
162 if (!smoothSelected || v.selected()) {
163 PositionType delta =
164 laplData[v.index()].sum / laplData[v.index()].cnt -
165 v.position();
166 v.position() = v.position() + delta * mu;
167 }
168 }
169 }
170 }
171}

186template<MeshConcept MeshType, PointConcept PointType>
187void smoothVertexNormalsPointCloud(
188 MeshType& m,
189 const KDTree<PointType>& tree,
190 uint neighborCount,
191 uint iterCount)
192{
193 requireVertexNormal(m);
194
195 using Scalar = PointType::ScalarType;
196 using NormalType = VertexType<MeshType>::NormalType;
197
198 std::vector<NormalType> TD(m.vertexContainerSize(), NormalType(0, 0, 0));
199
200 for (uint ii = 0; ii < iterCount; ++ii) {
201 for (const auto& v : m.vertices()) {
202 std::vector<Scalar> distances;
203
204 std::vector<std::size_t> neighbors = tree.kNearestNeighborsIndices(
205 v.position(), neighborCount, distances);
206
207 for (auto nid : neighbors) {
208 if (m.vertex(nid).normal() * v.normal() > 0) {
209 TD[v.index()] += m.vertex(nid).normal();
210 }
211 else {
212 TD[v.index()] -= m.vertex(nid).normal();
213 }
214 }
215 }
216 for (auto v : m.vertices()) {
217 v.normal() = TD[v.index()];
218 }
219 }
220}
