64template<
65 EdgeMeshConcept EdgeMesh,
66 FaceMeshConcept MeshType,
67 PlaneConcept PlaneType>
68EdgeMesh intersection(const MeshType& m, const PlaneType& pl)
69{
70 using PositionType = PositionType<MeshType>;
71 using IndexType = EdgeMesh::IndexType;
72
73 EdgeMesh em;
74
75 std::vector<double> qH(m.vertexContainerSize());
76
77 for (auto&& v : m.vertices())
78 qH[v.index()] = pl.dist(v.position());
79
80 for (auto&& f : m.faces()) {
81 std::vector<PositionType> ptVec;
82 std::vector<PositionType> nmVec;
83 for (uint j = 0; j < f.vertexCount(); ++j) {
84 if (qH[f.vertexIndex(j)] == 0) {
85 ptVec.push_back(f.vertex(j).position());
86 if constexpr (HasVertexNormal<MeshType>) {
87 if (m.isVertexNormalAvailable()) {
88 nmVec.push_back(f.vertex(j).normal());
89 }
90 }
91 }
92 else if ((qH[f.vertexIndex(j)] * qH[f.vertexIndexMod(j + 1)]) < 0) {
93 const PositionType& p0 = f.vertex(j).position();
94 const PositionType& p1 = f.vertexMod(j + 1).position();
95
96 double q0 = qH[f.vertexIndex(j)];
97 double q1 = qH[f.vertexIndexMod(j + 1)];
98
99 std::pair<PositionType, PositionType> seg(p0, p1);
100 PositionType pp = pl.segmentIntersection(seg);
101 ptVec.push_back(pp);
102 if constexpr (HasVertexNormal<MeshType>) {
103 if (m.isVertexNormalAvailable()) {
104 const auto& n0 = f.vertex(j).normal();
105 const auto& n1 = f.vertexMod(j + 1).normal();
106 PositionType nn =
107 (n0 * std::abs(q1) + n1 * std::abs(q0)) /
108 std::abs(q0 - q1);
109 nmVec.push_back(nn);
110 }
111 }
112 }
113 }
114 if (ptVec.size() >= 2) {
115 IndexType v0 = em.addVertices(2);
116 IndexType v1 = v0 + 1;
117 IndexType eid = em.addEdge(v0, v1);
118 em.vertex(v0).position() = ptVec[0];
119 em.vertex(v1).position() = ptVec[1];
120 if constexpr (
121 HasVertexNormal<MeshType> && HasVertexNormal<EdgeMesh>) {
122 if (m.isVertexNormalAvailable() &&
123 em.isVertexNormalAvailable()) {
124 em.vertex(v0).normal() = nmVec[0];
125 em.vertex(v1).normal() = nmVec[1];
126 }
127 }
128 }
129 }
130

132 // tri::Clean<EdgeMeshType>::RemoveDuplicateVertex(em);
133
134 // // Clean-up: Sort edges ensuring orientation coherence
135 // for (size_t j = 1; j < em.edge.size(); j++) {
136 // auto& n = em.edge[j - 1].V(1);
137 // for (size_t i = j; i < em.edge.size(); i++) {
138 // auto& ei = em.edge[i];
139 // if (ei.V(1) == n)
140 // std::swap(ei.V(0), ei.V(1));
141 // if (ei.V(0) == n) {

148 return em;
149}

169template<FaceMeshConcept MeshType, typename SScalar>
170MeshType intersection(
171 const MeshType& m,
172 const Sphere<SScalar>& sphere,
173 double tol)
174{
175 using IndexType = MeshType::IndexType;
176 using PositionType = PositionType<MeshType>;
177 using ScalarType = ScalarType<MeshType>;
178 using FaceType = ConstFaceType<MeshType>;
179
180 auto faceSphereIntersectionFilter = [&sphere](FaceType f) -> bool {
181 return intersect(f, sphere);
182 };
183
184 MeshType res = faceMeshFilter(m, faceSphereIntersectionFilter);
185
186 IndexType i = 0;
187 while (i < res.faceContainerSize()) {
188 auto f = res.face(i);
189 if (f.deleted()) {
190 ++i;
191 continue;
192 }
193
194 PositionType witness;
195 std::pair<ScalarType, ScalarType> ires(0, 0);
196
197 bool allIn = true;
198 for (auto&& v : f.vertices()) {
199 allIn = allIn && sphere.isInside(v.position());
200 }
201 if (!allIn && intersect(f, sphere, witness, ires)) {
202 if (faceArea(f) > tol) {
203 IndexType v0 = res.addVertices(3);
204 IndexType v1 = v0 + 1;
205 IndexType v2 = v0 + 2;
206 IndexType fi = res.addFaces(4);
207
208 auto currF = res.face(i);
209
210 res.vertex(v0).importFrom(currF.vertex(0), false);
211 res.vertex(v0).position() =
212 (currF.vertex(0).position() + currF.vertex(1).position()) /
213 ScalarType(2);
214 res.vertex(v1).importFrom(currF.vertex(1), false);
215 res.vertex(v1).position() =
216 (currF.vertex(1).position() + currF.vertex(2).position()) /
217 ScalarType(2);
218 res.vertex(v2).importFrom(currF.vertex(2), false);
219 res.vertex(v2).position() =
220 (currF.vertex(2).position() + currF.vertex(0).position()) /
221 ScalarType(2);
222
223 res.face(fi).importFrom(currF, false);
224 res.face(fi).setVertices(currF.vertexIndex(0), v0, v2);
225
226 ++fi;
227 res.face(fi).importFrom(currF, false);
228 res.face(fi).setVertices(currF.vertexIndex(1), v1, v0);
229
230 ++fi;
231 res.face(fi).importFrom(currF, false);
232 res.face(fi).setVertices(v0, v1, v2);
233
234 ++fi;
235 res.face(fi).importFrom(currF, false);
236 res.face(fi).setVertices(v2, v1, currF.vertexIndex(2));
237
238 res.deleteFace(i);
239 }
240 }
241 if (ires.first > 0.0) { // closest point - radius. If >0 is outside
242 res.deleteFace(i);
243 }
244
245 ++i;
246 }
247
248 return res;
249}

265template<FaceMeshConcept MeshType, typename SScalar>
266MeshType intersection(const MeshType& m, const Sphere<SScalar>& sphere)
267{
268 double tol = M_PI * sphere.radius() * sphere.radius() / 100000;
269 return intersection(m, sphere, tol);
