29template<typename ScalarType>
30class Histogram
31{
32 std::vector<ScalarType> mHist; // Counters for bins
33 std::vector<ScalarType> mRanges; // Range for bins
34
35 ScalarType mMinRange = 0; // Minimum range value
36 ScalarType mMaxRange = 1; // Maximum range value
37
38 ScalarType mMin = std::numeric_limits<ScalarType>::max();
39 ScalarType mMax = std::numeric_limits<ScalarType>::lowest();
40
41 // Number of valid bins stored between mMinRange and mMaxRange
42 uint mBinNumber = 0;
43
44 ScalarType mCnt = 0; // Number of accumulated values
45 ScalarType mSum = 0; // Sum of values
46 ScalarType mRMS = 0; // Root mean square
47
48public:
49 Histogram() = default;
50
61 Histogram(
62 ScalarType minRangeValue,
63 ScalarType maxRangeValue,
64 uint nBins,
65 ScalarType gamma = 1.0) :
66 mMinRange(minRangeValue), mMaxRange(maxRangeValue),
67 mBinNumber(nBins)
68 {
69 mHist.resize(nBins + 2, 0);
70
71 mRanges.resize(nBins + 3);
72
73 mRanges[0] = -std::numeric_limits<ScalarType>::max();
74 mRanges[nBins + 2] = std::numeric_limits<ScalarType>::max();
75
76 double delta = (mMaxRange - mMinRange);
77 if (gamma == 1) {
78 for (uint i = 0; i <= nBins; ++i)
79 mRanges[i + 1] = mMinRange + delta * ScalarType(i) / nBins;
80 }
81 else {
82 for (uint i = 0; i <= nBins; ++i)
83 mRanges[i + 1] =
84 mMinRange + delta * std::pow(ScalarType(i) / nBins, gamma);
85 }
86 }
87
91 void clear()
92 {
93 mHist.clear();
94 mRanges.clear();
95 mCnt = 0;
96 mSum = 0;
97 mRMS = 0;
98 mBinNumber = 0;
99 mMinRange = 0;
100 mMaxRange = 1;
101 mMin = std::numeric_limits<ScalarType>::max();
102 mMax = std::numeric_limits<ScalarType>::lowest();
103 }
104
111 void addValue(ScalarType value, ScalarType increment = 1.0)
112 {
113 uint pos = binIndex(value);
114 if (value < mMin)
115 mMin = value;
116 if (value > mMax)
117 mMax = value;
118 assert((pos >= 0) && (pos <= mBinNumber + 1));
119 mHist[pos] += increment;
120 mCnt += increment;
121 mSum += value * increment;
122 mRMS += (value * value) * increment;
123 }

209 ScalarType rangeCount(ScalarType rangeMin, ScalarType rangeMax) const
210 {
211 uint firstBin = binIndex(rangeMin);
212 uint lastBin = binIndex(rangeMax);
213
214 ScalarType sum = 0;
215 for (uint i = firstBin; i <= lastBin; ++i)
216 sum += mHist[i];
217 return sum;
218 }
219
227 ScalarType valueAtPercentile(ScalarType frac) const
228 {
229 if (mHist.size() == 0)
230 return 0;
231
232 // check percentile range
233 assert(frac >= 0 && frac <= 1);
234
235 ScalarType sum = mCnt * frac;
236 ScalarType partsum = 0;
237
238 uint i = 0;
239 while (partsum < sum) {
240 partsum += mHist[i];
241 ++i;
242 }
243
244 return mRanges[i + 1];
245 }
246
251 ScalarType average() const { return mSum / mCnt; }
252
257 ScalarType rootMeanSquare() const { return std::sqrt(mRMS / mCnt); }
258
263 ScalarType variance() const
264 {
265 return std::abs(mRMS / mCnt - average() * average());
266 }
267
272 ScalarType standardDeviation() const { return std::sqrt(variance()); }
273
274private:
278 uint binIndex(ScalarType elem) const
279 {
280 // lower_bound returns the furthermost iterator i in [first, last) such
281 // that, for every iterator j in [first, i), *j < value. E.g. An
282 // iterator pointing to the first element "not less than" val, or end()
283 // if every element is less than val.
284 auto it = std::lower_bound(mRanges.begin(), mRanges.end(), elem);
285
286 assert(it != mRanges.begin());
287 assert(it != mRanges.end());
288
289 uint pos = it - mRanges.begin();
290 pos -= 1;
291
292 assert(mRanges[pos] < elem);
293 assert(elem <= mRanges[pos + 1]);
294 return pos;
295 }
296};
