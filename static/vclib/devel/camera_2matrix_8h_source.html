20// creates a look at matrix and stores it in the given array, that represents a
21// 4x4 column major matrix.
22template<Point3Concept PointType>
23void lookAtMatrix(
24 auto* res,
25 const PointType& eye,
26 const PointType& center,
27 const PointType& up,
28 Handedness handedness = RIGHT_HAND)
29{
30 if (center != eye) {
31 PointType zaxis = handedness == RIGHT_HAND ?
32 (eye - center).normalized() :
33 (center - eye).normalized();
34
35 PointType xaxis = up.cross(zaxis);
36
37 if (xaxis.dot(xaxis) == 0) {
38 xaxis = handedness == RIGHT_HAND ? PointType(1, 0, 0) :
39 PointType(-1, 0, 0);
40 }
41 else {
42 xaxis = xaxis.normalized();
43 }
44
45 PointType yaxis = zaxis.cross(xaxis);
46
47 res[0] = xaxis.x();
48 res[1] = yaxis.x();
49 res[2] = zaxis.x();
50 res[3] = 0.0f;
51
52 res[4] = xaxis.y();
53 res[5] = yaxis.y();
54 res[6] = zaxis.y();
55 res[7] = 0.0f;
56
57 res[8] = xaxis.z();
58 res[9] = yaxis.z();
59 res[10] = zaxis.z();
60 res[11] = 0.0f;
61
62 res[12] = -xaxis.dot(eye);
63 res[13] = -yaxis.dot(eye);
64 res[14] = -zaxis.dot(eye);
65 res[15] = 1.0f;
66 }
67}

69template<typename Scalar>
70void projectionMatrixXYWH(
71 auto* res,
72 Scalar x,
73 Scalar y,
74 Scalar width,
75 Scalar height,
76 Scalar nearPlane,
77 Scalar farPlane,
78 bool homogeneousNDC,
79 Handedness handedness = RIGHT_HAND)
80{
81 // note: don't use 'near' and 'far' variable names, as they are already
82 // defined in windows.h headers
83 Scalar diff = farPlane - nearPlane;
84 Scalar a = homogeneousNDC ? (farPlane + nearPlane) / diff : farPlane / diff;
85 Scalar b =
86 homogeneousNDC ? (2.0 * farPlane * nearPlane) / diff : nearPlane * a;
87
88 std::fill(res, res + 16, 0);
89
90 res[0] = width;
91 res[5] = height;
92 res[8] = handedness == RIGHT_HAND ? x : -x;
93 res[9] = handedness == RIGHT_HAND ? y : -y;
94 res[10] = handedness == RIGHT_HAND ? -a : a;
95 res[11] = handedness == RIGHT_HAND ? -1.0 : 1.0;
96 res[14] = -b;
97}

99template<typename Scalar>
100void projectionMatrix(
101 auto* res,
102 Scalar fov,
103 Scalar aspect,
104 Scalar nearPlane,
105 Scalar farPlane,
106 bool homogeneousNDC,
107 Handedness handedness = RIGHT_HAND)
108{
109 Scalar h = 1.0 / std::tan(vcl::toRad(fov) * 0.5);
110 Scalar w = h * 1.0 / aspect;
111 detail::projectionMatrixXYWH(
112 res,
113 (Scalar) 0,
114 (Scalar) 0,
115 w,
116 h,
117 nearPlane,
118 farPlane,
119 homogeneousNDC,
120 handedness);
121}

123template<typename Scalar>
124void orthoProjectionMatrix(
125 auto* res,
126 Scalar left,
127 Scalar right,
128 Scalar top,
129 Scalar bottom,
130 Scalar nearPlane,
131 Scalar farPlane,
132 bool homogeneousNDC,
133 Handedness handedness = RIGHT_HAND)
134{
135 // note: don't use 'near' and 'far' variable names, as they are already
136 // defined in windows.h headers
137 Scalar c = homogeneousNDC ? 2.0 / (farPlane - nearPlane) :
138 1.0 / (farPlane - nearPlane);
139 Scalar f = homogeneousNDC ?
140 (farPlane + nearPlane) / (nearPlane - farPlane) :
141 nearPlane / (nearPlane - farPlane);
142
143 std::fill(res, res + 16, 0);
144 res[0] = 2.0 / (right - left);
145 res[5] = 2.0 / (top - bottom);
146 res[10] = RIGHT_HAND ? -c : c;
147 res[12] = (right + left) / (left - right);
148 res[13] = (bottom + top) / (bottom - top);
149 res[14] = f;
150 res[15] = 1.0;
151}
