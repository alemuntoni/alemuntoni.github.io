69 void init(const TriPolyIndexBiMap& indexMap, uint numTris)
70 {
71 mIsMappingTrivial = indexMap.isMappingTrivial();
72
73 if (numTris == 0)
74 return;
75
76 const std::size_t numPolys = indexMap.polygonCount();
77
78 // tri_to_poly[triIdx] = polygon index
79 {
80 auto [buf, rel] =
81 Context::getAllocatedBufferAndReleaseFn<uint>(numTris);
82 for (uint i = 0; i < numTris; i++) {
83 buf[i] = indexMap.polygon(i);
84 }
85 mTriToPolyBuffer.create(
86 buf, numTris, PrimitiveType::UINT, bgfx::Access::Read, rel);
87 }
88
89 // poly_to_tri_begin[polyIdx] and poly_to_tri_count[polyIdx]
90 {
91 auto [bufBegin, relBegin] =
92 Context::getAllocatedBufferAndReleaseFn<uint>(numPolys);
93 auto [bufCount, relCount] =
94 Context::getAllocatedBufferAndReleaseFn<uint>(numPolys);
95
96 for (std::size_t i = 0; i < numPolys; i++) {
97 std::size_t begin = indexMap.triangleBegin(i);
98 if (begin != NULL_INDEX<std::size_t>) {
99 bufBegin[i] = begin;
100 bufCount[i] = indexMap.triangleCount(i);
101 }
102 else {
103 // deleted polygon — will never be accessed by a valid
104 // triangle thread
105 bufBegin[i] = 0;
106 bufCount[i] = 0;
107 }
108 }
109
110 mPolyToTriBeginBuffer.create(
111 bufBegin,
112 numPolys,
113 PrimitiveType::UINT,
114 bgfx::Access::Read,
115 relBegin);
116 mPolyToTriCountBuffer.create(
117 bufCount,
118 numPolys,
119 PrimitiveType::UINT,
120 bgfx::Access::Read,
121 relCount);
122 }
123 }
124
125 bool isMappingTrivial() const { return mIsMappingTrivial; }
