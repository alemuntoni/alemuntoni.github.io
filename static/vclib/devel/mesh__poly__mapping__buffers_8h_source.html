68 void init(const TriPolyIndexBiMap& indexMap, uint numTris)
69 {
70 mIsMappingTrivial = indexMap.isMappingTrivial();
71
72 if (numTris == 0)
73 return;
74
75 const uint numPolys = indexMap.polygonCount();
76
77 // tri_to_poly[triIdx] = polygon index
78 {
79 auto [buf, rel] =
80 Context::getAllocatedBufferAndReleaseFn<uint>(numTris);
81 for (uint i = 0; i < numTris; i++) {
82 buf[i] = indexMap.polygon(i);
83 }
84 mTriToPolyBuffer.create(
85 buf, numTris, PrimitiveType::UINT, bgfx::Access::Read, rel);
86 }
87
88 // poly_to_tri_begin[polyIdx] and poly_to_tri_count[polyIdx]
89 {
90 auto [bufBegin, relBegin] =
91 Context::getAllocatedBufferAndReleaseFn<uint>(numPolys);
92 auto [bufCount, relCount] =
93 Context::getAllocatedBufferAndReleaseFn<uint>(numPolys);
94
95 for (uint i = 0; i < numPolys; i++) {
96 uint begin = indexMap.triangleBegin(i);
97 if (begin != UINT_NULL) {
98 bufBegin[i] = begin;
99 bufCount[i] = indexMap.triangleCount(i);
100 }
101 else {
102 // deleted polygon — will never be accessed by a valid
103 // triangle thread
104 bufBegin[i] = 0;
105 bufCount[i] = 0;
106 }
107 }
108
109 mPolyToTriBeginBuffer.create(
110 bufBegin,
111 numPolys,
112 PrimitiveType::UINT,
113 bgfx::Access::Read,
114 relBegin);
115 mPolyToTriCountBuffer.create(
116 bufCount,
117 numPolys,
118 PrimitiveType::UINT,
119 bgfx::Access::Read,
120 relCount);
121 }
122 }
123
124 bool isMappingTrivial() const { return mIsMappingTrivial; }

140 mPolyToTriCountBuffer.bind(stage, bgfx::Access::Read);
141 }
