20inline bool isBinStlMalformed(
21 const std::string& filename,
22 bool& isBinary,
23 std::size_t& fsize)
24{
25 fsize = FileInfo::fileSize(filename);
26 isBinary = FileInfo::isFileBinary(filename);
27
28 if (isBinary) {
29 // we can check if the size of the file is the expected one
30 std::ifstream fp = openInputFileStream(filename);
31 fp.seekg(80); // size of the header
32 uint fnum = io::readUInt<uint>(fp, std::endian::little);
33 std::size_t expectedFileSize =
34 80 + 4 + // header and number of faces
35 fnum * // for each face
36 (3 * sizeof(float) + // 3 floats for the face normal
37 3 * 3 * sizeof(float) + // 3 floats for each vertex of the face
38 sizeof(unsigned short)); // a short containing attributes
39 if (expectedFileSize != fsize) {
40 // sometimes the size is a bit wrong
41 std::size_t diff =
42 std::abs((long int) expectedFileSize - (long int) fsize);
43 if (diff > fsize / 20)
44 return true;
45 }
46 }
47
48 return false;
49}

51inline bool isStlColored(std::istream& fp, bool& magicsMode)
52{
53 bool colored = false;
54
55 char buf[80];
56 fp.read(buf, 80);
57 std::string s(buf);
58 size_t cInd = s.rfind("COLOR=");
59 size_t mInd = s.rfind("MATERIAL=");
60 if (cInd != std::string::npos && mInd != std::string::npos)
61 magicsMode = true;
62 else
63 magicsMode = false;
64 uint fnum = io::readUInt<uint>(fp, std::endian::little);
65 static const uint fmax = 1000;
66 // 3 floats for normal and 9 for v positions
67 static const uint fdataSize = 12 * sizeof(float);
68
69 for (uint i = 0; i < std::min(fnum, fmax); ++i) {
70 fp.seekg(fdataSize, std::ios::cur);
71 unsigned short attr =
72 io::readShort<unsigned short>(fp, std::endian::little);
73 Color c;
74 c.setBgr5(attr);
75 if (c != Color::White)
76 colored = true;
77 }
78 return colored;
79}

81template<MeshConcept MeshType, LoggerConcept LogType>
82void readStlBin(
83 MeshType& m,
84 std::istream& fp,
85 MeshInfo& loadedInfo,
86 const LoadSettings& settings,
87 LogType& log)
88{
89 bool magicsMode, colored;
90 colored = isStlColored(fp, magicsMode);
91
92 if (settings.enableOptionalComponents) {
93 if (colored)
94 loadedInfo.setFaceColor();
95 enableOptionalComponentsFromInfo(loadedInfo, m);
96 }
97 else if (colored) {
98 if constexpr (HasFaceColor<MeshType>) {
99 if (m.isFaceColorAvailable())
100 loadedInfo.setFaceColor();
101 }
102 }
103
104 fp.seekg(80); // size of the header
105 uint fnum = io::readUInt<uint>(fp, std::endian::little);
106 if (fnum > 0)
107 loadedInfo.setTriangleMesh();
108
109 log.startProgress("Loading STL file", fnum);
110
111 m.addVertices(fnum * 3);
112 if constexpr (HasFaces<MeshType>) {
113 m.reserveFaces(fnum);
114 }
115
116 uint vi = 0;
117 for (uint i = 0; i < fnum; ++i) {
118 Point3f norm;
119 norm.x() = io::readFloat<float>(fp, std::endian::little);
120 norm.y() = io::readFloat<float>(fp, std::endian::little);
121 norm.z() = io::readFloat<float>(fp, std::endian::little);
122
123 for (uint j = 0; j < 3; ++j) {
124 m.vertex(vi + j).position().x() =
125 io::readFloat<float>(fp, std::endian::little);
126 m.vertex(vi + j).position().y() =
127 io::readFloat<float>(fp, std::endian::little);
128 m.vertex(vi + j).position().z() =
129 io::readFloat<float>(fp, std::endian::little);
130 }
131
132 unsigned short attr =
133 io::readShort<unsigned short>(fp, std::endian::little);
134
135 if constexpr (HasFaces<MeshType>) {
136 using FaceType = FaceType<MeshType>;
137
138 uint fi = m.addFace();
139 FaceType f = m.face(fi);
140 // we have a polygonal mesh
141 if constexpr (FaceType::VERTEX_COUNT < 0) {
142 // need to resize the face to the right number of verts
143 f.resizeVertices(3);
144 }
145 for (uint j = 0; j < 3; ++j)
146 f.setVertex(j, vi + j);
147 if constexpr (HasFaceNormal<MeshType>) {
148 using ST = FaceType::NormalType::ScalarType;
149 if (m.isFaceNormalAvailable()) {
150 f.normal() = norm.cast<ST>();
151 }
152 }
153 if constexpr (HasFaceColor<MeshType>) {
154 if (m.isFaceColorAvailable() && colored) {
155 Color c;
156 if (magicsMode)
157 c.setBgr5(attr);
158 else
159 c.setRgb5(attr);
160 f.color() = c;
161 }
162 }
163 }
164
165 vi += 3;
166
167 log.progress(i);
168 }
169 log.endProgress();
170}

172template<MeshConcept MeshType, LoggerConcept LogType>
173void readStlAscii(
174 MeshType& m,
175 std::istream& fp,
176 MeshInfo& loadedInfo,
177 const LoadSettings& settings,
178 LogType& log)
179{
180 if (settings.enableOptionalComponents) {
181 enableOptionalComponentsFromInfo(loadedInfo, m);
182 }
183
184 fp.seekg(0, fp.end);
185 std::size_t fsize = fp.tellg();
186 fp.seekg(0, fp.beg);
187 log.startProgress("Loading STL file", fsize);
188
189 Tokenizer tokens = readAndTokenizeNextNonEmptyLineNoThrow(fp);
190 if (fp) {
191 // cycle that reads a face starting from the actual tokenized line
192 do {
193 Tokenizer::iterator token = tokens.begin();
194 if (token != tokens.end() && *token == "facet") {
195 ++token; // skip the "facet" word
196 ++token; // skip the "normal" word
197
198 // add 3 vertices for the face
199 uint vi = m.addVertices(3);
200
201 // read the normal of the face
202 Point3f normal;
203
204 normal.x() = io::readFloat<float>(token, std::endian::little);
205 normal.y() = io::readFloat<float>(token, std::endian::little);
206 normal.z() = io::readFloat<float>(token, std::endian::little);
207
208 readAndTokenizeNextNonEmptyLine(fp); // outer loop
209 // vertex x y z
210 tokens = readAndTokenizeNextNonEmptyLine(fp);
211
212 for (uint i = 0; i < 3; i++) { // read the three vertices
213 token = tokens.begin();
214 ++token; // skip the "vertex" word
215
216 m.vertex(vi + i).position().x() =
217 io::readFloat<float>(token, std::endian::little);
218 m.vertex(vi + i).position().y() =
219 io::readFloat<float>(token, std::endian::little);
220 m.vertex(vi + i).position().z() =
221 io::readFloat<float>(token, std::endian::little);
222
223 // next vertex
224 tokens = readAndTokenizeNextNonEmptyLine(fp);
225 }
226 readAndTokenizeNextNonEmptyLine(fp); // endfacet
227
228 if constexpr (HasFaces<MeshType>) {
229 using FaceType = FaceType<MeshType>;
230 uint fi = m.addFace();
231
232 FaceType f = m.face(fi);
233 // we have a polygonal mesh
234 if constexpr (FaceType::VERTEX_COUNT < 0) {
235 // need to resize the face to the right number of verts
236 f.resizeVertices(3);
237 }
238 for (uint j = 0; j < 3; ++j)
239 f.setVertex(j, vi + j);
240 if constexpr (HasFaceNormal<MeshType>) {
241 using ST = FaceType::NormalType::ScalarType;
242 if (m.isFaceNormalAvailable()) {
243 f.normal() = normal.cast<ST>();
244 }
245 }
246 }
247 }
248 tokens = readAndTokenizeNextNonEmptyLineNoThrow(fp);
249
250 log.progress(fp.tellg());
251 } while (fp);
252
253 if constexpr (HasFaces<MeshType>) {
254 if (m.faceCount() > 0)
255 loadedInfo.setTriangleMesh();
256 }
257 }
258
259 log.endProgress();
260}

294template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
295void loadStl(
296 MeshType& m,
297 std::istream& inputStlStream,
298 MeshInfo& loadedInfo,
299 bool isBinary = false,
300 const LoadSettings& settings = LoadSettings(),
301 LogType& log = nullLogger)
302{
303 loadedInfo.clear();
304 loadedInfo.setVertices();
305 loadedInfo.setVertexPosition();
306
307 if constexpr (HasFaces<MeshType>) {
308 loadedInfo.setFaces();
309 loadedInfo.setFaceVertexReferences();
310 loadedInfo.setFaceNormal();
311 }
312
313 log.log(0, "Loading STL file");
314
315 if (isBinary)
316 detail::readStlBin(m, inputStlStream, loadedInfo, settings, log);
317 else
318 detail::readStlAscii(m, inputStlStream, loadedInfo, settings, log);
319
320 log.log(100, "STL file loaded");
321}

347template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
348void loadStl(
349 MeshType& m,
350 const std::string& filename,
351 MeshInfo& loadedInfo,
352 const LoadSettings& settings = LoadSettings(),
353 LogType& log = nullLogger)
354{
355 log.log(0, "Checking STL file");
356
357 bool isBinary;
358 std::size_t filesize;
359 if (detail::isBinStlMalformed(filename, isBinary, filesize))
360 throw MalformedFileException(filename + " is malformed.");
361
362 log.log(0, "Opening STL file");
363
364 std::ifstream fp = openInputFileStream(filename);
365
366 if constexpr (HasName<MeshType>) {
367 m.name() = FileInfo::fileNameWithoutExtension(filename);
368 }
369
370 loadStl(m, fp, loadedInfo, isBinary, settings, log);
371}
