255template<MatrixConcept Matrix44, Point3Concept PointType>
256PointType unprojectScreenPosition(
257 const PointType& screenPos,
258 const Matrix44& modelViewProjection,
259 const Point4<typename Matrix44::Scalar>& viewport,
260 bool homogeneousNDC)
261{
262 using Scalar = Matrix44::Scalar;
263 const Matrix44 inv = modelViewProjection.inverse();
264 Point4<Scalar> p(
265 (screenPos.x() - viewport[0]) / viewport[2] * 2.0 - 1.0,
266 (screenPos.y() - viewport[1]) / viewport[3] * 2.0 - 1.0,
267 homogeneousNDC ? 2.0 * screenPos.z() - 1.0 : screenPos.z(),
268 1.0);
269 p = inv * p;
270 if (p.w() == 0.0) {
271 throw std::runtime_error("unproject: division by zero");
272 }
273 return p.template head<3>() / p.w();
274}
