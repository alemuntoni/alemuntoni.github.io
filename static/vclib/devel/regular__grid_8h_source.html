117 RegularGrid(
118 const BBoxType& bbox,
119 const Point<Scalar, N>& cellLens,
120 bool extendBBox = true) : mBBox(bbox)
121 {
122 for (uint i = 0; i < N; ++i) {
123 Scalar ratio = mBBox.dim(i) / cellLens(i);
124 mSize(i) = std::max(
125 extendBBox ? uint(std::ceil(ratio)) : uint(std::floor(ratio)),
126 1u);
127 mBBox.max()(i) = mBBox.min()(i) + mSize(i) * cellLens(i);

207 uint indexOfCell(const CellPos& c) const

291 uint cell(uint d, const Scalar& s) const
292 {

350 BBoxType cellBox(const CellPos& c) const
351 {
352 BBoxType b;
353
354 Point<Scalar, N> p;
355 for (size_t i = 0; i < DIM; ++i)
356 p(i) = (Scalar) c(i) * cellLength(i);
357 p += mBBox.min();
358 b.min() = p;
359 b.max() = (p + cellLengths());
360
361 return b;
362 }

475template<PointConcept PointType>
476Point<uint, PointType::DIM> bestGridSize(
477 const PointType& lengths,
478 uint nElements)
479{
480 using Scalar = PointType::ScalarType;
481
482 static const int DIM = PointType::DIM;
483
484 const uint mincells = 1; // Minimum number of cells along each dimension
485 const Scalar GFactor = 1; // GridEntry = NumElem*GFactor
486 const Scalar diag = lengths.norm(); // Diagonal of the bounding box
487 const Scalar eps = diag * 1e-4; // Threshold for degenerate dimensions
488 const uint ncell = nElements * GFactor; // Target number of cells
489
490 Point<uint, PointType::DIM> sizes;
491 sizes.setConstant(mincells);
492
493 bool sanityCheckLengths = true;
494 uint lessEpsDimNumber = 0;
495 std::array<bool, DIM> isLessEps;
496
497 for (uint i = 0; i < DIM; i++) {
498 sanityCheckLengths = sanityCheckLengths && lengths(i) > 0.0;
499 if (lengths(i) < eps) {
500 lessEpsDimNumber++;
501 isLessEps[i] = true;
502 }
503 }
504 uint greaterEpsDimNumber = DIM - lessEpsDimNumber;
505
506 if (nElements > 0 && sanityCheckLengths) {
507 // no lenghts less than epsilon - standard computation for sizes
508 if (greaterEpsDimNumber == DIM) {
509 Scalar product = 1;
510 for (uint i = 0; i < DIM; i++)
511 product *= lengths(i);
512
513 Scalar k = std::pow((ncell / product), (1.0 / DIM));
514
515 for (uint i = 0; i < DIM; i++)
516 sizes(i) = int(lengths(i) * k);
517 }
518 // at least one lenght is less than epsilon
519 else {
520 for (uint i = 0; i < DIM; i++) {
521 if (isLessEps[i]) { // the ith dimension is less than epsilon
522 sizes(i) = 1;
523 }
524 else {
525 // compute the product between all the dimensions that are
526 // not i and not less than epsilon
527 Scalar product = 1;
528 for (uint j = 0; j < DIM; j++)
529 if (j != i && !isLessEps[j])
530 product *= lengths(j);
531
532 // ith dimension size
533 sizes(i) = std::pow(
534 (ncell * lengths(i) / product),
535 (1.0 / greaterEpsDimNumber));
536 }
537 }
538 }
539
540 for (uint i = 0; i < DIM; i++)
541 sizes(i) = std::max(sizes(i), (uint) 1);
542 }
543
544 return sizes;
545}
