145template<IteratorConcept Iterator>
146auto boundingBox(Iterator begin, Iterator end)
147{
148 using BB = decltype(boundingBox(std::declval<typename Iterator::value_type>()));
149
150 BB b;
151
152 for (; begin != end; ++begin)
153 b.add(boundingBox(*begin));
154
155 return b;
156}
