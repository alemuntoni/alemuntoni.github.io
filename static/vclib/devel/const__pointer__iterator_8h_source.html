39template<typename It>
40requires (IsAnyPointer<typename std::iterator_traits<It>::value_type>)
41class ConstPointerIterator
42{
43 using Base = It;
44
45 It mIt;
46
47public:
48 using difference_type = std::iterator_traits<It>::difference_type;
49 using value_type =
50 MakeConstPointerT<typename std::iterator_traits<It>::value_type>;
51 using reference = value_type&;
52 using pointer = value_type*;
53 using iterator_category = std::iterator_traits<It>::iterator_category;
54
55 ConstPointerIterator() = default;
56
57 ConstPointerIterator(It it) : mIt(it) {}
58
59 value_type operator*() const { return *mIt; }
60
61 pointer operator->() const { return &(*mIt); }
62
63 bool operator==(const ConstPointerIterator& oi) const
64 {
65 return mIt == oi.mIt;
66 }
67
68 bool operator!=(const ConstPointerIterator& oi) const
69 {
70 return mIt != oi.mIt;
71 }
72
73 ConstPointerIterator& operator++()
74 {
75 ++mIt;
76 return *this;
77 }
78
79 ConstPointerIterator operator++(int)
80 {
81 ConstPointerIterator tmp = *this;
82 ++mIt;
83 return tmp;
84 }
85
86 ConstPointerIterator& operator--()
87 requires (std::bidirectional_iterator<It>)
88 {
89 --mIt;
90 return *this;
91 }
92
93 ConstPointerIterator operator--(int)
94 requires (std::bidirectional_iterator<It>)
95 {
96 ConstPointerIterator tmp = *this;
97 --mIt;
98 return tmp;
99 }
100
101 ConstPointerIterator& operator+=(difference_type n)
102 requires (std::random_access_iterator<It>)
103 {
104 mIt += n;
105 return *this;
106 }
107
108 ConstPointerIterator operator+(difference_type n) const
109 requires (std::random_access_iterator<It>)
110 {
111 return ConstPointerIterator(mIt + n);
112 }
113
114 friend ConstPointerIterator operator+(
115 difference_type n,
116 const ConstPointerIterator& it)
117 requires (std::random_access_iterator<It>)
118 {
119 return ConstPointerIterator(n + it.mIt);
120 }
121
122 ConstPointerIterator& operator-=(difference_type n)
123 requires (std::random_access_iterator<It>)
124 {
125 mIt -= n;
126 return *this;
127 }
128
129 ConstPointerIterator operator-(difference_type n) const
130 requires (std::random_access_iterator<It>)
131 {
132 return ConstPointerIterator(mIt - n);
133 }
134
135 difference_type operator-(const ConstPointerIterator& oi) const
136 requires (std::random_access_iterator<It>)
137 {
138 return mIt - oi.mIt;
139 }
140
141 value_type operator[](difference_type n) const
142 requires (std::random_access_iterator<It>)
143 {
144 return mIt[n];
145 }
146};
