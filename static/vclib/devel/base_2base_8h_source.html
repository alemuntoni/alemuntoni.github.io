99constexpr int sizeOf(PrimitiveType type) noexcept
100{
101 switch (type) {
102 case PrimitiveType::CHAR: return sizeof(char);
103 case PrimitiveType::UCHAR: return sizeof(unsigned char);
104 case PrimitiveType::SHORT: return sizeof(int16_t);
105 case PrimitiveType::USHORT: return sizeof(uint16_t);
106 case PrimitiveType::INT: return sizeof(int32_t);
107 case PrimitiveType::UINT: return sizeof(uint32_t);
108 case PrimitiveType::FLOAT: return sizeof(float);
109 case PrimitiveType::DOUBLE: return sizeof(double);
110 default: return 0;
111 }
112}
