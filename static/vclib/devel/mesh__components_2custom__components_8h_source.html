109 std::vector<std::string> customComponentNames(this const auto& self)
110 {
111 std::vector<std::string> names;

127 template<typename CompType>
128 std::vector<std::string> customComponentNamesOfType(this const auto& self)
129 {
130 std::vector<std::string> names;
131 std::type_index t(typeid(CompType));
132 for (const auto& [name, type] : self.mCompType) {
133 if (type == t) {
134 names.push_back(name);
135 }
136 }
137 return names;
138 }

190 template<typename CompType>
191 decltype(auto) customComponent(
192 this auto&& self,
193 const std::string& compName)
194 {
195 self.template checkComponentType<CompType>(compName);
196 using ReturnType = std::conditional_t<
197 std::is_const_v<std::remove_reference_t<decltype(self)>>,
198 const CompType&,
199 CompType&>;
200 return std::any_cast<ReturnType>(self.mMap.at(compName));
201 }

210 template<typename CompType>
211 void serializeCustomComponentsOfType(
212 this const auto& self,
213 std::ostream& os)
214 {
215 constexpr uint32_t CCMP_TAG = 0x504D4343; // 'C', 'C', 'M', 'P'
216 vcl::serialize(os, CCMP_TAG);
217 uint32_t typeSize = static_cast<uint32_t>(sizeof(CompType));
218 vcl::serialize(os, typeSize);
219
220 std::vector<std::string> compNames =
221 self.template customComponentNamesOfType<CompType>();
222 vcl::serialize(os, compNames);
223 for (const auto& name : compNames) {
224 const CompType& c = self.template customComponent<CompType>(name);
225 vcl::serialize(os, c);
226 }
227 }
228
236 template<typename CompType>
237 void deserializeCustomComponentsOfType(this auto& self, std::istream& is)
238 {
239 constexpr uint32_t CCMP_TAG = 0x504D4343;
240 uint32_t tag = 0;
241 vcl::deserialize(is, tag);
242 if (tag != CCMP_TAG) {
243 throw std::runtime_error(
244 "Invalid custom component block magic header");
245 }
246 uint32_t typeSize = 0;
247 vcl::deserialize(is, typeSize);
248 if (typeSize != sizeof(CompType)) {
249 throw BadCustomComponentTypeException(
250 "Custom component type size mismatch during deserialization.");
251 }
252
253 std::vector<std::string> compNames;
254 vcl::deserialize(is, compNames);
255 for (const auto& name : compNames) {
256 CompType c;
257 vcl::deserialize(is, c);
258 self.template addCustomComponent<CompType>(name, c);
259 }
260 }
261
262protected:
263 template<typename OtherMesh>
264 void importFrom(const OtherMesh& m)
265 {
266 if constexpr (comp::HasCustomComponents<OtherMesh>) {
267 if constexpr (std::is_base_of_v<CustomComponents, OtherMesh>) {
268 const auto& other = static_cast<const CustomComponents&>(m);
269 mMap = other.mMap;
270 mCompType = other.mCompType;
271 }
272 }
273 }

275private:
276 template<typename CompType>
277 void checkComponentType(this const auto& self, const std::string& compName)
278 {
279 std::type_index t(typeid(CompType));
280 if (t != self.mCompType.at(compName)) {
281 throw BadCustomComponentTypeException(
282 "Expected type " +
283 std::string(self.mCompType.at(compName).name()) + " for " +
284 compName + ", but was " + std::string(t.name()) + ".");
285 }
286 }
