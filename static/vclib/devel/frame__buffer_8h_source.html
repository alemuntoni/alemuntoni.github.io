149 void create(
150 bgfx::TextureHandle* textureHandles,
151 uint8_t count,
152 bool destroyTextures = false)
153 {
154 destroy();
155
156 if (count > 0) {
157 mHandle =
158 bgfx::createFrameBuffer(count, textureHandles, destroyTextures);
159
160 // If we're not destroying textures via bgfx
161 // (destroyTextures=false), but the caller wants us to manage them,
162 // store them Note: bgfx's destroyTextures parameter handles
163 // destruction automatically when set to true, so we only track if
164 // bgfx will destroy them
165 if (destroyTextures) {
166 // bgfx will destroy the textures, so we don't need to track
167 // them
168 mOwnedTextures.clear();
169 }
