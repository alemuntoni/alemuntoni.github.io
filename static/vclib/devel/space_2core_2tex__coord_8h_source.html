33// TODO: TexCoord should inherit from Point2
34template<typename Scalar>
35class TexCoord
36{
37 template<typename S>
38 friend class TexCoord;
39
40 Point2<Scalar> mCoord;
41
42public:
43 using ScalarType = Scalar;
44
45 TexCoord() = default;
46
47 TexCoord(const Scalar& s1, const Scalar& s2) : mCoord(s1, s2) {}
48
49 TexCoord(const Point2<Scalar>& p) : mCoord(p) {}
50
51 template<typename S>
52 auto cast() const
53 {
54 if constexpr (std::is_same<Scalar, S>::value) {
55 return *this;
56 }
57 else {
58 TexCoord<S> tmp;
59 tmp.mCoord = mCoord.template cast<S>();
60 return tmp;
61 }
62 }
63
64 decltype(auto) u(this auto&& self) { return self.mCoord.x(); }
65
66 decltype(auto) v(this auto&& self) { return self.mCoord.y(); }
67
68 void setU(Scalar s)
69 {
70 assert(s >= 0 && s <= 1);
71 mCoord.x() = s;
72 }
73
74 void setV(Scalar s)
75 {
76 assert(s >= 0 && s <= 1);
77 mCoord.y() = s;
78 }
79
80 void set(Scalar u, Scalar v)
81 {
82 setU(u);
83 setV(v);
84 }
85
86 void serialize(std::ostream& os) const { mCoord.serialize(os); }
87
88 void deserialize(std::istream& is) { mCoord.deserialize(is); }
89
90 // operators
91 decltype(auto) operator()(this auto&& self, uint i)
92 {
93 return self.mCoord[i];
94 }
95
96 decltype(auto) operator[](this auto&& self, uint i)
97 {
98 return self.mCoord[i];
99 }
100
101 auto operator<=>(const TexCoord& t1) const = default;
102};
