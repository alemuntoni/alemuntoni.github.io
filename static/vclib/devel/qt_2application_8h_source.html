44inline void initXcb()
45{

47#ifndef VCLIB_RENDER_WITH_WAYLAND

78inline QApplication qAppl(int& argc, char** argv)
79{
80 initXcb();
81#if defined(Q_OS_MACOS) && defined(VCLIB_RENDER_BACKEND_BGFX)
82 // Qt propagates this color space to Metal layers when attaching them and
83 // when the window's backing properties change. VCLib outputs encoded sRGB.
84 QSurfaceFormat format = QSurfaceFormat::defaultFormat();
85 if (!format.colorSpace().isValid()) {
86 format.setColorSpace(QColorSpace::SRgb);
87 QSurfaceFormat::setDefaultFormat(format);
88 }
89#endif
90 return QApplication(argc, argv);
91}

111inline void* getDisplayId()
112{
113 void* displayId = nullptr;
114#ifdef Q_OS_LINUX
115 if (qApp) { // if the QApplication instance is created
116 // Get the display ID depending on whether the system is using X11 or
117 // Wayland
118 if (auto x11App =
119 qApp->nativeInterface<QNativeInterface::QX11Application>()) {
120 displayId = x11App->display();
121 }
122 else if (
123 auto wayApp = qApp->nativeInterface<
124 QNativeInterface::QWaylandApplication>()) {
125 displayId = wayApp->display();
126 }
127 }
128#endif
129 return displayId;
130}
