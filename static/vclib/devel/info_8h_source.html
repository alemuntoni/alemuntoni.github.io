29inline uint gltfNodeMeshCount(const tinygltf::Model& model, uint node)
30{
31 uint nMeshes = 0;
32 if (model.nodes[node].mesh >= 0) {
33 nMeshes = 1;
34 }
35 for (int c : model.nodes[node].children) {
36 if (c >= 0) {
37 nMeshes += gltfNodeMeshCount(model, c);
38 }
39 }
40 return nMeshes;
41}

56inline uint gltfMeshCount(const tinygltf::Model& model)
57{
58 uint nMeshes = 0;
59 for (uint s = 0; s < model.scenes.size(); ++s) {
60 const tinygltf::Scene& scene = model.scenes[s];
61 for (uint n = 0; n < scene.nodes.size(); ++n) {
62 nMeshes += gltfNodeMeshCount(model, scene.nodes[n]);
63 }
64 }
65 return nMeshes;
66}

76template<Matrix44Concept MatrixType>
77MatrixType gltfCurrentNodeMatrix(const tinygltf::Model& model, uint currentNode)
78{
79 using QuatType = Quaternion<typename MatrixType::Scalar>;
80
81 MatrixType currentMatrix = MatrixType::Identity();
82 // if the current node contains a 4x4 matrix
83 if (model.nodes[currentNode].matrix.size() == 16) {
84 for (uint i = 0; i < 4; ++i) {
85 for (uint j = 0; j < 4; ++j) {
86 // set the current matrix element
87 currentMatrix(i, j) =
88 model.nodes[currentNode].matrix[j * 4 + i];
89 }
90 }
91 }
92 // if the current node contains rotation quaternion, scale vector or
93 // translation vector
94 else {
95 // note: if one or more of these are missing, identity is used.
96 // note: final matrix is computed as M = T * R * S, as specified by
97 // gltf docs:
98 // https://github.com/KhronosGroup/glTF-Tutorials/blob/master/gltfTutorial/gltfTutorial_004_ScenesNodes.md
99 // 4x4 matrices associated to rotation, translation and scale
100 MatrixType rot = MatrixType::Identity();
101 MatrixType scale = MatrixType::Identity();
102 MatrixType trans = MatrixType::Identity();
103
104 // if node contains rotation quaternion
105 if (model.nodes[currentNode].rotation.size() == 4) {
106 QuatType qr(
107 model.nodes[currentNode].rotation[3],
108 model.nodes[currentNode].rotation[0],
109 model.nodes[currentNode].rotation[1],
110 model.nodes[currentNode].rotation[2]);
111 // Convert quaternion to 3x3 rotation matrix and insert into
112 // top-left of 4x4 matrix
113 rot.template block<3, 3>(0, 0) = qr.normalized().toRotationMatrix();
114 }
115 // if node contains scale
116 if (model.nodes[currentNode].scale.size() == 3) {
117 // set 4x4 matrix scale
118 scale(0, 0) = model.nodes[currentNode].scale[0];
119 scale(1, 1) = model.nodes[currentNode].scale[1];
120 scale(2, 2) = model.nodes[currentNode].scale[2];
121 }
122 // if node contains translation
123 if (model.nodes[currentNode].translation.size() == 3) {
124 // set 4x4 matrix translation
125 trans(0, 3) = model.nodes[currentNode].translation[0];
126 trans(1, 3) = model.nodes[currentNode].translation[1];
127 trans(2, 3) = model.nodes[currentNode].translation[2];
128 }
129
130 // M = T * R * S
131 currentMatrix = trans * rot * scale;
132 }
133 return currentMatrix;
134}
