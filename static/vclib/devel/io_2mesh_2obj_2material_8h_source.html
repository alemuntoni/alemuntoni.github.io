22class ObjMaterial
23{
24 inline static const Point3f Ka_DEFAULT = {0.2f, 0.2f, 0.2f}; // ambient;
25 inline static const Point3f Kd_DEFAULT = {1.0f, 1.0f, 1.0f}; // diffuse;
26 inline static const Point3f Ks_DEFAULT = {1.0f, 1.0f, 1.0f}; // specular;
27 inline static const Point3f Ke_DEFAULT = {0.0f, 0.0f, 0.0f}; // emissive;
28
29 inline static const float d_DEFAULT = 1.0f; // alpha
30 inline static const int illum_DEFAULT = 2; // specular illumination
31 inline static const float Ns_DEFAULT = 0.f;
32
33public:
34 // id of the material in the mesh, used when loading materials
35 ushort matId = USHORT_NULL;
36 std::string matName;
37
38 Point3f Ka = Ka_DEFAULT; // ambient
39 Point3f Kd = Kd_DEFAULT; // diffuse
40 Point3f Ks = Ks_DEFAULT; // specular
41 Point3f Ke = Ke_DEFAULT; // emissive
42
43 float d = d_DEFAULT; // alpha
44
45 int illum = illum_DEFAULT; // specular illumination
46 float Ns = Ns_DEFAULT;
47
48 std::string map_Kd; // filename texture
49 std::string map_Ke; // filename emissive map
50 std::string map_bump; // filename bump map
51
52 ObjMaterial() = default;
53
54 ObjMaterial(const Material& mat, ushort id)
55 {
56 using enum Material::TextureType;
57
58 matId = id;
59 matName = mat.name();
60
61 Kd.x() = mat.baseColor().redF();
62 Kd.y() = mat.baseColor().greenF();
63 Kd.z() = mat.baseColor().blueF();
64 Ns = std::pow(1.0f - mat.roughness(), 2) * 1000.f;
65 if (mat.alphaMode() == Material::AlphaMode::ALPHA_BLEND) {
66 d = mat.baseColor().alphaF();
67 }
68
69 Ke.x() = mat.emissiveColor().redF();
70 Ke.y() = mat.emissiveColor().greenF();
71 Ke.z() = mat.emissiveColor().blueF();
72
73 map_Kd = mat.baseColorTextureDescriptor().path();
74 map_Ke = mat.textureDescriptor(EMISSIVE).path();
75 // map_bump = mat.textureDescriptor(NORMAL).path();
76 }
77
87 Material toMaterial()
88 {
89 using enum Material::TextureType;
90
91 Material m;
92 m.name() = matName;
93 m.baseColor() =
94 vcl::Color(Kd.x() * 255, Kd.y() * 255, Kd.z() * 255, 255);
95
96 float ns = std::clamp(Ns, 0.f, 1000.f);
97 m.roughness() = 1.0f - std::sqrt(ns) / std::sqrt(1000.f);
98
99 if (d < 1.0) {
100 m.baseColor().alpha() = d * 255;
101 m.alphaMode() = Material::AlphaMode::ALPHA_BLEND;
102 }
103
104 m.metallic() = 0.0f; // obj materials are non-metallic;
105
106 m.emissiveColor() =
107 vcl::Color(Ke.x() * 255, Ke.y() * 255, Ke.z() * 255, 255);
108
109 if (!map_Kd.empty()) {
110 m.baseColorTextureDescriptor().path() = map_Kd;
111 }
112 if (!map_Ke.empty()) {
113 m.textureDescriptor(EMISSIVE).path() = map_Ke;
114 }
115 // if (!map_bump.empty()) {
116 // m.textureDescriptor(NORMAL).path() = map_bump;
117 // }
118
119 return m;
120 }
121
122 bool isValid() const { return matId != USHORT_NULL; }
123
124 Color color() const
125 {
126 return Color(Kd.x() * 255, Kd.y() * 255, Kd.z() * 255, d * 255);
127 }
128
129 const std::string& texture() const { return map_Kd; }
130
131 ushort textureId() const { return matId; }
132
133 bool operator==(const ObjMaterial& m) const = default;
134
135 friend inline std::ostream& operator<<(
136 std::ostream& out,
137 const ObjMaterial& m);
138};

140inline std::ostream& operator<<(std::ostream& out, const ObjMaterial& m)
141{
142 if (m.Ka != m.Ka_DEFAULT)
143 out << "Ka " << m.Ka.x() << " " << m.Ka.y() << " " << m.Ka.z()
144 << std::endl;
145
146 if (m.Kd != m.Kd_DEFAULT)
147 out << "Kd " << m.Kd.x() << " " << m.Kd.y() << " " << m.Kd.z()
148 << std::endl;
149
150 if (m.Ks != m.Ks_DEFAULT)
151 out << "Ks " << m.Ks.x() << " " << m.Ks.y() << " " << m.Ks.z()
152 << std::endl;
153
154 if (m.Ke != m.Ke_DEFAULT)
155 out << "Ke " << m.Ke.x() << " " << m.Ke.y() << " " << m.Ke.z()
156 << std::endl;
157
158 if (m.d != m.d_DEFAULT)
159 out << "d " << m.d << std::endl;
160
161 if (m.illum != m.illum_DEFAULT)
162 out << "illum " << m.illum << std::endl;
163
164 if (m.Ns != m.Ns_DEFAULT)
165 out << "Ns " << m.Ns << std::endl;
166
167 if (!m.map_Kd.empty()) {
168 out << "map_Kd " << m.map_Kd << std::endl;
169 }
170
171 if (!m.map_Ke.empty()) {
172 out << "map_Ke " << m.map_Ke << std::endl;
173 }
174
175 // if (m.map_bump.empty()) {
176 // out << "map_bump " << m.map_bump << std::endl;
177 // }
178
179 return out;
180}
