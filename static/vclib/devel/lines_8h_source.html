220 template<Range R>
221 requires Point3Concept<std::ranges::range_value_t<R>>
222 void setVertices(R&& verts)
223 {
224 mVerPosCount = std::ranges::size(verts);
225
226 uint padding = (4 - (mVerPosCount % 4)) % 4;
227 uint nv = mVerPosCount + padding;
228
229 VertexBuffer vertBuff;
230 auto [buffer, releaseFn] =
231 Context::getAllocatedBufferAndReleaseFn<float>(nv * 3);
232
233 size_t i = 0;
234 for (const auto& v : verts) {
235 buffer[i * 3 + 0] = v.x();
236 buffer[i * 3 + 1] = v.y();
237 buffer[i * 3 + 2] = v.z();
238 ++i;
239 }
240
241 // Use create for compute binding (vertex pulling)
242 vertBuff.create(
243 buffer,
244 nv,
245 bgfx::Attrib::Position,
246 3,
247 PrimitiveType::FLOAT,
248 releaseFn);
249 mVertexPositions.setOwned(std::move(vertBuff));
250 mIsUpdateProgramNeeded = true;
251 }
252
256 template<Range R>

261
262 IndexBuffer indexBuff;
263 auto [buffer, releaseFn] =
264 Context::getAllocatedBufferAndReleaseFn<uint>(mIndexCount);
265
266 size_t i = 0;
267 for (const auto& idx : indices) {
268 buffer[i] = idx;
269 ++i;
270 }
271
272 indexBuff.create(buffer, mIndexCount, releaseFn);
273
274 mIndices.setOwned(std::move(indexBuff));
275 mIsUpdateProgramNeeded = true;

412 template<Range R>
413 requires std::convertible_to<std::ranges::range_value_t<R>, bool>
414 void setLineSelections(R&& selections)
415 {
416 mLineSelCount = std::ranges::size(selections);
417 BooleanBuffer buf;
418 buf.init(mLineSelCount);
419
420 vcl::BitVector<true> backup(mLineSelCount, false);
421
422 uint vidx = 0;
423 for (bool sel : selections) {
424 backup[vidx++] = sel;
425 }
426
427 buf.setFromCPUBuffer(backup);
428 mLineSelections.setOwned(std::move(buf));
429 mIsUpdateProgramNeeded = true;
430 }
