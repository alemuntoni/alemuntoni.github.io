222 uint linesCount() const
223 {
224 uint nv = mIndices.isValid() ? mIndexCount : mVerPosCount;
225 if (mTopology == Topology::LINES || mTopology == Topology::VECTORS) {
226 return nv / 2;
227 }
228 else if (mTopology == Topology::LINE_STRIP) {
229 return nv > 1 ? nv - 1 : 0;
230 }
231 return 0;
232 }

251 template<Range R>
252 requires Point3Concept<std::ranges::range_value_t<R>>
253 void setVertices(R&& verts)
254 {
255 mVerPosCount = std::ranges::size(verts);
256
257 uint padding = (4 - (mVerPosCount % 4)) % 4;
258 uint nv = mVerPosCount + padding;
259
260 VertexBuffer vertBuff;
261 auto [buffer, releaseFn] =
262 Context::getAllocatedBufferAndReleaseFn<float>(nv * 3);
263
264 size_t i = 0;
265 for (const auto& v : verts) {
266 buffer[i * 3 + 0] = v.x();
267 buffer[i * 3 + 1] = v.y();
268 buffer[i * 3 + 2] = v.z();
269 ++i;
270 }
271
272 // Use create for compute binding (vertex pulling)
273 vertBuff.create(
274 buffer,
275 nv,
276 bgfx::Attrib::Position,
277 3,
278 PrimitiveType::FLOAT,
279 releaseFn);
280 mVertexPositions.setOwned(std::move(vertBuff));
281 mIsUpdateProgramNeeded = true;
282 }

287 template<Range R>
288 requires std::integral<std::ranges::range_value_t<R>>
289 void setIndices(R&& indices)
290 {
291 mIndexCount = std::ranges::size(indices);
292
293 IndexBuffer indexBuff;
294 auto [buffer, releaseFn] =
295 Context::getAllocatedBufferAndReleaseFn<uint>(mIndexCount);
296
297 size_t i = 0;
298 for (const auto& idx : indices) {
299 buffer[i] = idx;
300 ++i;
301 }
302
303 indexBuff.create(buffer, mIndexCount, releaseFn);
304
305 mIndices.setOwned(std::move(indexBuff));
306 mIsUpdateProgramNeeded = true;
307 }

443 template<Range R>
444 requires std::convertible_to<std::ranges::range_value_t<R>, bool>
445 void setLineSelections(R&& selections)
446 {
447 mLineSelCount = std::ranges::size(selections);
448 BooleanBuffer buf;
449 buf.init(mLineSelCount);
450
451 vcl::BitVector<true> backup(mLineSelCount, false);
452
453 uint vidx = 0;
454 for (bool sel : selections) {
455 backup[vidx++] = sel;
456 }
457
458 buf.setFromCPUBuffer(backup);
459 mLineSelections.setOwned(std::move(buf));
460 mIsUpdateProgramNeeded = true;
461 }
