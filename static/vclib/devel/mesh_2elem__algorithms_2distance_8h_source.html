91template<Point3Concept PointType, FaceConcept FaceType, typename ScalarType>
92auto boundedDistance(
93 const PointType& p,
94 const FaceType& f,
95 ScalarType maxDist,
96 PointType& closest,
97 bool signedDist = false)
98{
99 if constexpr (TriangleFaceConcept<FaceType>) {
100 TriangleWrapper tw(
101 f.vertex(0)->position(),
102 f.vertex(1)->position(),
103 f.vertex(2)->position());
104
105 return boundedDistance(p, tw, maxDist, closest, signedDist);
106 }
107 else {
108 if (f.vertexCount() == 3) {
109 TriangleWrapper tw(
110 f.vertex(0)->position(),
111 f.vertex(1)->position(),
112 f.vertex(2)->position());
113
114 return boundedDistance(p, tw, maxDist, closest, signedDist);
115 }
116
117 ScalarType minDist = maxDist;
118
119 std::vector<uint> tris = earCut(f);
120 for (uint i = 0; i < tris.size(); i += 3) {
121 PointType w;
122 ScalarType d = boundedDistance(
123 p,
124 TriangleWrapper(
125 f.vertex(tris[i])->position(),
126 f.vertex(tris[i + 1])->position(),
127 f.vertex(tris[i + 2])->position()),
128 minDist,
129 w,
130 signedDist);
131
132 if (std::abs(d) < minDist) {
133 minDist = std::abs(d);
134 closest = w;
135 }
136 }
137
138 return minDist;
139 }
140}
