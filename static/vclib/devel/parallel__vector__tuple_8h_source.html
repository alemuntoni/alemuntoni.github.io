135 void enableAllVectors()
136 {
137 mVecEnabled.fill(true);
138 if (mSize > 0) {
139 auto func = [this](auto&& vec) {
140 vec.resize(mSize);
141 };
142 applyToAllEnabledVectors<TUPLE_SIZE - 1>(func);
143 }
144 }
145
152 void disableAllVectors()
153 {
154 mVecEnabled.fill(false);
155 auto func = [](auto& vec) {
156 vec.clear();
157 };
158 std::apply(func, mVecTuple);
159 // don't change the size!
160 }

171 template<uint I>
172 auto span() const
173 {
174 if (!isVectorEnabled<I>()) {
175 throw std::runtime_error(
176 "Accessing disabled vector at index " + std::to_string(I));
177 }
178 using RT = TypeAtT<I, Types...>;
179 return std::span<const RT>(std::get<I>(mVecTuple));

192 auto span()

194 if (!isVectorEnabled<I>()) {

206 void resize(std::size_t size)
207 {
208 if constexpr (TUPLE_SIZE > 0) {
209 auto func = [this, size](auto&& vec) {
210 vec.resize(size);
211 };
212 applyToAllEnabledVectors<TUPLE_SIZE - 1>(func);
213 }
214 mSize = size;
215 }
216
223 void reserve(std::size_t size)
224 {
225 if constexpr (TUPLE_SIZE > 0) {
226 auto func = [this, size](auto&& vec) {
227 vec.reserve(size);
228 };
229 applyToAllEnabledVectors<TUPLE_SIZE - 1>(func);
230 }
231 }
232
237 void clear()
238 {
239 if constexpr (TUPLE_SIZE > 0) {
240 auto func = [](auto& vec) {
241 vec.clear();
242 };
243 applyToAllEnabledVectors<TUPLE_SIZE - 1>(func);
244 }
245 mSize = 0;
246 }
247
248private:
249 template<std::size_t N>
250 void applyToAllEnabledVectors(auto&& func)
251 {
252 if (mVecEnabled[N]) {
253 func(std::get<N>(mVecTuple));
254 }
255 if constexpr (N != 0)
256 applyToAllEnabledVectors<N - 1>(func);
257 }
258};
