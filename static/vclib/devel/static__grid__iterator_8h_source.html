15template<typename KeyType, typename ValueType, typename GridType>
16class StaticGridIterator
17{
18 using VecIt = std::vector<std::pair<uint, ValueType>>::iterator;
19
20 VecIt mVecIt;
21 const GridType* mGrid = nullptr;
22
23public:
24 using T = SecondRefPair<KeyType, ValueType>;
25 using value_type = T;
26
27 class ArrowHelper
28 {
29 T mValue;
30
31 public:
32 ArrowHelper(T value) : mValue(value) {}
33
34 const T* operator->() const { return &mValue; }
35 };
36
37 StaticGridIterator() = default;
38
39 StaticGridIterator(VecIt it, const GridType& g) : mVecIt(it), mGrid(&g) {}
40
41 value_type operator*() const
42 {
43 KeyType cell = mGrid->cellOfIndex(mVecIt->first);
44 return value_type(cell, mVecIt->second);
45 }
46
47 ArrowHelper operator->() const { return **this; }
48
49 bool operator==(const StaticGridIterator& oi) const
50 {
51 return mVecIt == oi.mVecIt;
52 }
53
54 bool operator!=(const StaticGridIterator& oi) const
55 {
56 return mVecIt != oi.mVecIt;
57 }
58
59 StaticGridIterator operator++()
60 {
61 ++mVecIt;
62 return *this;
63 }
64
65 StaticGridIterator operator++(int)
66 {
67 StaticGridIterator old = *this;
68 ++mVecIt;
69 return old;
70 }
71};
