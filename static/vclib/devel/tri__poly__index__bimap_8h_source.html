147 void insert(std::size_t triangleIndex, std::size_t polygonIndex)
148 {
149 // resize the vectors if needed
150 if (triangleIndex >= mTriToPoly.size()) {
151 mTriToPoly.resize(triangleIndex + 1, NULL_INDEX<std::size_t>);
152 }
153 if (polygonIndex >= mPolyToTri.size()) {
154 mPolyToTri.resize(polygonIndex + 1, NULL_INDEX<std::size_t>);
155 mPolyToTriCount.resize(polygonIndex + 1, 0);
156 }
157
158 // add the index of the polygon associated to the triangle
159 mTriToPoly[triangleIndex] = polygonIndex;
160
161 // update the number of triangles associated to the polygon
162 mPolyToTriCount[polygonIndex]++;
163 // add the index of the triangle associated to the polygon,
164 // but only if it is the first triangle index of the polygon!
165 if (mPolyToTri[polygonIndex] == NULL_INDEX<std::size_t> ||
166 triangleIndex < mPolyToTri[polygonIndex])
167 mPolyToTri[polygonIndex] = triangleIndex;
168 }

174 std::size_t triangleCount() const { return mTriToPoly.size(); }
175
180 std::size_t polygonCount() const { return mPolyToTri.size(); }

187 bool isMappingTrivial() const
188 {
189 // We cannot simply compare the total number of triangles to the total
190 // number of polygons because deleted polygons have a triangle count of
191 // 0. A mesh with deleted faces and quadrangles could coincidentally
192 // have the same number of total triangles and total polygons.
193 // Furthermore, if faces are sorted or permuted (e.g. sorted by material),
194 // each polygon may map to 1 triangle but not with the identity index.
195 // Therefore, we must explicitly verify that every polygon maps to
196 // exactly one triangle and that polygon i maps to triangle i.
197 if (mPolyToTri.size() != mTriToPoly.size())
198 return false;
199
200 for (std::size_t i = 0; i < mPolyToTri.size(); ++i) {
201 if (mPolyToTriCount[i] != 1 || mPolyToTri[i] != i)
202 return false;
203 }
204 return true;
205 }
206};
