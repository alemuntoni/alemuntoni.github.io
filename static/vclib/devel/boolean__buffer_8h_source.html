61 void init(uint elementCount)
62 {
63 mNumElements = elementCount;
64
65 if (elementCount == 0) {
66 mGpuBuffer.destroy();
67 mNumUints = 0;
68 return;
69 }
70
71 mNumUints = uint(std::ceil(double(elementCount) / 32.0));
72
73 auto [buffer, releaseFn] =
74 Context::getAllocatedBufferAndReleaseFn<uint>(mNumUints);
75
76 // clear the buffer with zeroes
77 std::fill(buffer, buffer + mNumUints, 0);
78
79 mGpuBuffer.create(
80 buffer,
81 mNumUints,
82 vcl::PrimitiveType::UINT,
83 bgfx::Access::ReadWrite,
84 releaseFn);
85
86 // Resize readback handler. We need bytes, so ceil(elementCount / 8.0)
87 uint bytes = uint(std::ceil(double(elementCount) / 8.0));
88 mReadbackHandler = std::move(detail::ReadFromGPUBuffer(bytes));
89 }

96 void setFromCPUBuffer(const vcl::BitVector<true>& backup)
97 {
98 mBackup = backup;
99
100 if (backup.empty() || !mGpuBuffer.isValid())
101 return;
102
103 const auto& bytes = backup.bytes();
104 const uint elementCount =
105 (uint(bytes.size()) + uint(sizeof(uint32_t)) - 1u) /
106 uint(sizeof(uint32_t));
107
108 if (elementCount == 0)
109 return;
110
111 auto [buffer, releaseFn] =
112 Context::getAllocatedBufferAndReleaseFn<uint32_t>(elementCount);
113
114 // The backup bytes are big-endian (uintRGBAToVec4Color stores MSB
115 // in the R channel). Reconstruct each uint32 with proper byte order.
116 // Clear the buffer with zeroes first
117 std::memset(buffer, 0, elementCount * sizeof(uint32_t));
118 // Copy the raw bytes directly
119 std::memcpy(buffer, bytes.data(), bytes.size());
120
121 // The backup bytes are big-endian (uintRGBAToVec4Color stores MSB
122 // in the R channel). Reconstruct each uint32 with proper byte order.
123 if constexpr (std::endian::native == std::endian::little) {
124 for (uint i = 0; i < elementCount; i++) {
125 uint32_t v = buffer[i];
126 buffer[i] = ((v & 0xFF000000) >> 24) | ((v & 0x00FF0000) >> 8) |
127 ((v & 0x0000FF00) << 8) | ((v & 0x000000FF) << 24);
128 }
129 }
130
131 mGpuBuffer.destroy();
132 mGpuBuffer.create(
133 buffer,
134 elementCount,
135 vcl::PrimitiveType::UINT,
136 bgfx::Access::ReadWrite,
137 releaseFn);
138 }
