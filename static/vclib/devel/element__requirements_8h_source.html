65template<uint ELEM_ID, uint COMP_ID, MeshConcept MeshType>
66bool isPerElementComponentAvailable(const MeshType& m)
67{
68 if constexpr (mesh::HasElementContainer<MeshType, ELEM_ID>) {
69 using Container = mesh::ContainerOfElementType<ELEM_ID, MeshType>;
70 using Element = Container::ElementType;
71 if constexpr (comp::HasOptionalComponentOfType<Element, COMP_ID>) {
72 return m.template isPerElementComponentEnabled<ELEM_ID, COMP_ID>();
73 }
74 else {
75 return comp::HasComponentOfType<Element, COMP_ID>;
76 }
77 }
78 else {
79 return false;
80 }
81}

177template<uint ELEM_ID, uint COMP_ID, MeshConcept MeshType>
178void requirePerElementComponent(const MeshType& m)
179{
180 // TODO C++26: use elementEnumString<ELEM_ID>() and
181 // componentEnumString<COMP_ID>() in static_assert messages
182 static_assert(
183 mesh::HasElementContainer<MeshType, ELEM_ID>,
184 "The input Mesh does not have a Container of the given Element ID.");
185
186 using ElementType = MeshType::template ElementType<ELEM_ID>;
187
188 static_assert(
189 comp::HasComponentOfType<ElementType, COMP_ID>,
190 "The Element of the input Mesh does not have a Component of the given "
191 "Component ID.");
192
193 if (!isPerElementComponentAvailable<ELEM_ID, COMP_ID>(m)) {
194 throw MissingComponentException(
195 "Per " + std::string(elementEnumString<ELEM_ID>()) + " " +
196 std::string(componentEnumString<COMP_ID>()) +
197 " Component is not enabled.");
198 }
199}
