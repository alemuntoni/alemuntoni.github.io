761 template<typename OtherMesh>
762 void manageImportTriFromPoly(const OtherMesh& m)
763 {
764 if constexpr (OtherMesh::template hasContainerOf<ElemId::FACE>()) {
765 using OtherFace = OtherMesh::ConstFace;
766
767 // if this is not a triangle mesh nor a polygon mesh (meaning that
768 // we can't control the number of vertex references in this mesh), and
769 // this mesh does not have the same number of vertex references of the
770 // other, it means that we don't know how to convert these type of
771 // meshes (e.g. we don't know how to convert a polygon mesh into a
772 // quad mesh, or convert a quad mesh into a pentagonal mesh...)
773 static_assert(
774 !(Face::VERTEX_COUNT != 3 && Face::VERTEX_COUNT > 0 &&
775 Face::VERTEX_COUNT != OtherFace::VERTEX_COUNT),
776 "Cannot import from that type of Mesh. Don't know how to "
777 "convert faces.");
778
779 // we need to manage conversion from poly or faces with cardinality
780 // > 3 (e.g. quads) to triangle meshes. In this case, we triangulate
781 // the polygon using the earcut algorithm.
782 if constexpr (
783 Face::VERTEX_COUNT == 3 &&
784 (OtherFace::VERTEX_COUNT > 3 || OtherFace::VERTEX_COUNT < 0)) {
785 for (const auto& mf : m.faces()) {
786 // if the current face has the same number of vertices of
787 // this mf face (3), then the vertex indices have been
788 // correctly imported from the component importFrom
789 // function. The component importFrom function does
790 // nothing when importing from a face with at least 4
791 // vertices
792 if (mf.vertexCount() != 3) {
793 // triangulate mf; the first triangle of the
794 // triangulation will be this->face(mf.index()); the
795 // other triangles will be added at the end of the
796 // container
797
798 std::vector<uint> tris =
799 vcl::earCut(mf.vertices() | vcl::views::positions);
800
801 if (tris.size() >= 3) {
802 Face f = face(mf.index());
803 importTriCorners(f, mf, tris, 0);
804
805 // number of other faces to add
806 uint nf = tris.size() / 3 - 1;
807 Index baseFid = addFaces(nf);
808
809 uint t = 3; // index that cycles into tris
810 for (uint k = 0; k < nf; ++k, t += 3) {
811 Face fk = face(baseFid + k);
812 fk.importFrom(mf, false);
813 importTriCorners(fk, mf, tris, t);
814 }
815 }
816 else {
817 this->deleteFace(mf.index());
818 }
819 }
820 }
821 }
822 }
823 }

839 template<typename OtherFace>
840 static void importTriCorners(
841 Face& f,
842 const OtherFace& mf,
843 const std::vector<uint>& tris,
844 uint basetri)
845 {
846 for (uint j = 0; j < 3; ++j) {
847 f.setVertex(j, castIndex(mf.vertexIndex(tris[basetri + j])));
848 }
849
850 // Wedge Colors
851 if constexpr (
852 comp::HasWedgeColors<Face> && comp::HasWedgeColors<OtherFace>) {
853 if (f.template isComponentAvailable<CompId::WEDGE_COLORS>() &&
854 mf.template isComponentAvailable<CompId::WEDGE_COLORS>()) {
855 for (uint j = 0; j < 3; ++j) {
856 f.wedge(j).color() = mf.wedge(tris[basetri + j]).color();
857 }
858 }
859 }
860
861 // Wedge TexCoords
862 if constexpr (
863 comp::HasWedgeTexCoords<Face> &&
864 comp::HasWedgeTexCoords<OtherFace>) {
865 if (f.template isComponentAvailable<CompId::WEDGE_TEX_COORDS>() &&
866 mf.template isComponentAvailable<CompId::WEDGE_TEX_COORDS>()) {
867 using S = typename std::remove_cvref_t<
868 decltype(f.wedge(0).texCoord())>::ScalarType;
869 for (uint j = 0; j < 3; ++j) {
870 f.wedge(j).texCoord() =
871 mf.wedge(tris[basetri + j]).texCoord().template cast<S>();
872 }
873 }
874 }
875
876 // Wedge Normals
877 if constexpr (
878 comp::HasWedgeNormals<Face> && comp::HasWedgeNormals<OtherFace>) {
879 if (f.template isComponentAvailable<CompId::WEDGE_NORMALS>() &&
880 mf.template isComponentAvailable<CompId::WEDGE_NORMALS>()) {
881 using S = typename std::remove_cvref_t<
882 decltype(f.wedge(0).normal())>::ScalarType;
883 for (uint j = 0; j < 3; ++j) {
884 f.wedge(j).normal() =
885 mf.wedge(tris[basetri + j]).normal().template cast<S>();
886 }
887 }
888 }
889 }
