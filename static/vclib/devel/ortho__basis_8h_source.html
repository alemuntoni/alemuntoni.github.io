48template<typename Scalar, uint N>

66 OrthoBasis()

108 static OrthoBasis fromNormal(const PointType& normal) requires (N == 3)
109 {
110 PointType n = normal.normalized();
111 PointType u, v;
112 n.orthoBase(u, v);
113 u.normalize();
114 v.normalize();
115 return OrthoBasis(u, v, n);
116 }

123 const PointType& axis(uint i) const { return mAxes[i]; }

143 PointType toWorld(const PointType& local) const
144 {
145 PointType p;
146 p.setZero();
147 for (uint i = 0; i < N; ++i)
148 p += mAxes[i] * local(i);
149 return p;
150 }
151
163 PointType toLocal(const PointType& world) const
164 {
165 PointType local;
166 for (uint i = 0; i < N; ++i)
167 local(i) = world.dot(mAxes[i]);
168 return local;
169 }

181 Matrix<Scalar, N, N> rotationMatrix() const
182 {
183 Matrix<Scalar, N, N> r;
184 for (uint i = 0; i < N; ++i)
185 r.col(i) = mAxes[i];
186 return r;
187 }
188
194 template<typename S>
195 auto cast() const
196 {
197 if constexpr (std::is_same_v<Scalar, S>) {
198 return *this;
199 }
200 else {
201 std::array<Point<S, N>, N> newAxes;
202 for (uint i = 0; i < N; ++i)
203 newAxes[i] = mAxes[i].template cast<S>();
204 return OrthoBasis<S, N>(newAxes);
205 }
206 }
207
212 void serialize(std::ostream& os) const
213 {
214 for (uint i = 0; i < N; ++i)

216 }
217
222 void deserialize(std::istream& is)

225 mAxes[i].deserialize(is);
226 }
227
228 bool operator==(const OrthoBasis& b) const { return mAxes == b.mAxes; }
229
230 bool operator!=(const OrthoBasis& b) const { return !(*this == b); }
231};
