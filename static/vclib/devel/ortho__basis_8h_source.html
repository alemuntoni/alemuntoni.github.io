108 static OrthoBasis fromNormal(const PointType& normal) requires (N == 3)
109 {
110 PointType n = normal.normalized();
111 PointType u, v;
112 n.orthoBase(u, v);
113 u.normalize();
114 v.normalize();
115 return OrthoBasis(u, v, n);

136 PointType toWorld(const PointType& local) const
137 {
138 PointType p;
139 p.setZero();
140 for (uint i = 0; i < N; ++i)
141 p += mAxes[i] * local(i);
142 return p;
143 }

156 PointType toLocal(const PointType& world) const
157 {
158 PointType local;
159 for (uint i = 0; i < N; ++i)
160 local(i) = world.dot(mAxes[i]);
161 return local;
162 }

174 Matrix<Scalar, N, N> rotationMatrix() const
175 {
176 Matrix<Scalar, N, N> r;
177 for (uint i = 0; i < N; ++i)
178 r.col(i) = mAxes[i];
179 return r;
180 }
181
187 template<typename S>
188 auto cast() const
189 {
190 if constexpr (std::is_same_v<Scalar, S>) {
191 return *this;
192 }
193 else {
194 std::array<Point<S, N>, N> newAxes;
195 for (uint i = 0; i < N; ++i)
196 newAxes[i] = mAxes[i].template cast<S>();
197 return OrthoBasis<S, N>(newAxes);
198 }
199 }
200
205 void serialize(std::ostream& os) const

221 bool operator==(const OrthoBasis& b) const { return mAxes == b.mAxes; }
222
223 bool operator!=(const OrthoBasis& b) const { return !(*this == b); }
