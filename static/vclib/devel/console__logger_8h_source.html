86 void flush() const override
87 {
88 if (mErrFile)
89 std::fflush(mErrFile);
90 if (mWarnFile && mWarnFile != mErrFile)
91 std::fflush(mWarnFile);
92 if (mMsgFile && mMsgFile != mWarnFile && mMsgFile != mErrFile)
93 std::fflush(mMsgFile);
94 if (mProgFile && mProgFile != mMsgFile && mProgFile != mWarnFile &&
95 mProgFile != mErrFile)
96 std::fflush(mProgFile);
97 if (mDebugFile && mDebugFile != mProgFile && mDebugFile != mMsgFile &&
98 mDebugFile != mWarnFile && mDebugFile != mErrFile)
99 std::fflush(mDebugFile);
100 }
101
102private:
103 std::FILE* targetFile(LogLevel lvl) const
104 {
105 switch (lvl) {
106 case ERROR_LOG: return mErrFile;
107 case WARNING_LOG: return mWarnFile;
108 case MESSAGE_LOG: return mMsgFile;
109 case PROGRESS_LOG: return mProgFile;
110 case DEBUG_LOG: return mDebugFile;
111 }
112 return nullptr;
113 }
