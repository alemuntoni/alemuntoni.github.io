48 static const uint TIMER_MAX_CHAR_COUNT = 12;

50 std::recursive_mutex mMutex;

52 uint mPercPrecision = 0;

54 // on top of the stack, we save the interval percentages of the current task
55 // values are between 0 and 100
56 std::stack<std::pair<double, double>> mIntervals;
57
58 // actual percentage (0 - 100), that is in the interval in top of the stack
59 double mGlobalPercProgress = 0;

61 double mStep = 1; // the value that corresponds to 1% on the current task

89 void startNewTask(double fromPerc, double toPerc, const std::string& action)
90 override final
91 {
92 std::lock_guard<std::recursive_mutex> lock(mMutex);
93 printLine(action, START);
94
95 assert(fromPerc >= 0);
96 assert(toPerc <= 100);
97 std::pair<double, double> actualP = mIntervals.top();
98 std::pair<double, double> newP;
99 newP.first =
100 actualP.first + (actualP.second - actualP.first) * (fromPerc / 100);
101 newP.second =
102 actualP.first + (actualP.second - actualP.first) * (toPerc / 100);
103 mGlobalPercProgress = newP.first;
104 mIntervals.push(newP);
105 updateStep();
106 }
107
108 void endTask(const std::string& action) override final
109 {
110 std::lock_guard<std::recursive_mutex> lock(mMutex);
111 mGlobalPercProgress = mIntervals.top().second;
112 if (mIntervals.size() > 1) {
113 mIntervals.pop();
114 updateStep();
115
116 printLine(action, END);
117 }
118 }
119
120 double percentage() const override final
121 {
122 std::lock_guard<std::recursive_mutex> lock(
123 const_cast<std::recursive_mutex&>(mMutex));
124 double k = std::pow(10, mPercPrecision);
125 uint c = mGlobalPercProgress * k;
126 return c / k;
127 }

129 virtual void setPercentage(uint newPerc) override

131 std::lock_guard<std::recursive_mutex> lock(mMutex);

133 mGlobalPercProgress = (mIntervals.top().first) + mStep * newPerc;

166 void startProgress(
167 const std::string& msg,
168 uint progressSize,
169 ProgressMode mode = ProgressMode::TIME,
170 double printInterval = 1.0,
171 uint startPerc = 0,
172 uint endPerc = 100) override final
173 {
174 std::lock_guard<std::recursive_mutex> lock(mMutex);
175 assert(printInterval > 0);
176 assert((endPerc - startPerc) > 0);
177
178 ProgressStatus prog;
179 prog.isActive = true;
180 prog.message = msg;
181 prog.size = progressSize;
182 prog.startPerc = startPerc;
183 prog.endPerc = endPerc;
184 prog.perc = startPerc;
185 prog.mode = mode;
186 prog.currentN = 0;
187
188 if (!mProgresses.empty() && mProgresses.top().isActive) {
189 auto& parent = mProgresses.top();
190 uint p_start =
191 parent.startPerc +
192 (parent.currentN * (parent.endPerc - parent.startPerc)) /
193 parent.size;
194 uint p_end =
195 parent.startPerc +
196 ((parent.currentN + 1) * (parent.endPerc - parent.startPerc)) /
197 parent.size;
198
199 prog.startPerc = p_start + (startPerc * (p_end - p_start)) / 100;
200 prog.endPerc = p_start + (endPerc * (p_end - p_start)) / 100;
201 prog.perc = prog.startPerc;
202 }
203
204 if (mode == ProgressMode::PERCENTAGE) {
205 uint unscaledDiff = endPerc - startPerc;
206 uint unscaledPercStep = static_cast<uint>(printInterval);
207 if (unscaledPercStep == 0)
208 unscaledPercStep = 1;
209
210 if (unscaledDiff == 0) {
211 prog.step = progressSize;
212 }
213 else {
214 uint diff_step = unscaledDiff / unscaledPercStep;
215 if (diff_step == 0)
216 diff_step = 1;
217 prog.step = (progressSize + 1) / diff_step;
218 if (prog.step == 0)
219 prog.step = progressSize;
220 }
221 }
222 else {
223 prog.secondsStep = printInterval;
224 prog.progressTimer.start();
225 prog.throttleCounter = 0;
226 }
227
228 prog.lastProgress = 0;
229 mProgresses.push(prog);
230 }
231
232 void endProgress() override final
233 {
234 std::lock_guard<std::recursive_mutex> lock(mMutex);
235 if (!mProgresses.empty()) {
236 ProgressStatus& prog = mProgresses.top();
237 if (prog.isActive) {
238 if (prog.perc < prog.endPerc) {
239 prog.perc = prog.endPerc;
240 if (mPrintMsgDuringProgress)
241 log(prog.perc, prog.message, PROGRESS_LOG);
242 else
243 setPercentage(prog.perc);
244 }
245 prog.isActive = false;
246 }
247 mProgresses.pop();
248 }
249 }
250
251 void progress(uint n) override final
252 {
253 std::lock_guard<std::recursive_mutex> lock(mMutex);
254 if (mProgresses.empty())
255 return;
256 ProgressStatus& prog = mProgresses.top();
257 if (!prog.isActive)
258 return;
259 prog.currentN = n;
260 if (n > prog.size)
261 n = prog.size;
262
263 if (prog.mode == ProgressMode::TIME) {
264 if (++prog.throttleCounter >= 64) {
265 prog.throttleCounter = 0;
266 double elapsed = prog.progressTimer.delay();
267 if (elapsed >= prog.secondsStep) {
268 prog.perc =
269 prog.startPerc +
270 (n * (prog.endPerc - prog.startPerc)) / prog.size;
271 if (prog.perc > prog.endPerc)
272 prog.perc = prog.endPerc;
273
274 if (mPrintMsgDuringProgress)
275 log(prog.perc, prog.message, PROGRESS_LOG);
276 else
277 setPercentage(prog.perc);
278
279 prog.progressTimer.start();
280 }
281 }
282 }
283 else {
284 uint progress = n / prog.step;
285 if (prog.lastProgress < progress) {
286 prog.perc = prog.startPerc +
287 (n * (prog.endPerc - prog.startPerc)) / prog.size;
288 if (prog.perc > prog.endPerc)
289 prog.perc = prog.endPerc;
290 if (mPrintMsgDuringProgress)
291 log(prog.perc, prog.message, PROGRESS_LOG);
292 else
293 setPercentage(prog.perc);
294 prog.lastProgress = progress;
295 }
296 }

318 void printLine(const std::string& msg, uint lvl) const
319 {
320 if (!mPrintPerc && msg.empty())
321 return;
322
323 // lvl could be also InternalLogLevel BEGIN or END
324 LogLevel l = PROGRESS_LOG;
325 if (lvl <= DEBUG_LOG) {
326 l = (LogLevel) lvl;
327 }
328 if (l > mPrintLevel)
329 return;
330 Stream* stream = levelStream(l);
331
332 if (stream) {
333 std::ostringstream ss;
334 uint s = 0;
335
336 if (mPrintPerc) {
337 s = printPercentage(ss);
338 }
339 s += printIndentation(ss);
340 printMessage(ss, msg, lvl, s);
341 printElapsedTime(ss);
342 ss << "\n";
343 *stream << ss.str().c_str();
344 flush(*stream);
345 }
346 }

348 uint printPercentage(std::ostream& o) const
349 {
350 uint size = 3;
351 if (mPercPrecision > 0)
352 size += 1 + mPercPrecision;
353
354 o << "[";
355 o << std::right;
356 o << std::setw(size);
357 o << percentage() << "%]";
358 return size + 3;
359 }
360
361 uint printIndentation(std::ostream& o) const
362 {
363 uint s = 0;
364 if (mIndent) {
365 uint n = mIntervals.size() - 1;
366 for (uint i = 0; i < n; i++) {
367 o << " ";
368 s += 2;
369 }
370 }
371 return s;
372 }
373
374 void printMessage(std::ostream& o, const std::string& msg, uint lvl, uint n)
375 const
376 {
377 uint maxMsgSize = mLineWidth - n;
378 if (mPrintTimer)
379 maxMsgSize -= TIMER_MAX_CHAR_COUNT;
380 switch (lvl) {
381 case LogLevel::ERROR_LOG:
382 maxMsgSize -= 8;
383 o << " ERROR: ";
384 break;
385 case LogLevel::WARNING_LOG:
386 maxMsgSize -= 10;
387 o << " WARNING: ";
388 break;
389 case LogLevel::MESSAGE_LOG:
390 case LogLevel::PROGRESS_LOG:
391 maxMsgSize -= 1;
392 o << " ";
393 break;
394 case LogLevel::DEBUG_LOG:
395 maxMsgSize -= 9;
396 o << " (debug) ";
397 break;
398 case InternalLogLevel::START:
399 maxMsgSize -= 7;
400 o << " Start ";
401 break;
402 case InternalLogLevel::END:
403 maxMsgSize -= 5;
404 o << " End ";
405 break;
406 }
407 o << std::setw(maxMsgSize);
408 o << std::left;
409 o << msg.c_str();
410 }
411
412 void printElapsedTime(std::ostream& o) const
413 {
414 if (mPrintTimer) {
415 o << "[";
416 o << std::setw(TIMER_MAX_CHAR_COUNT - 3);
417 o << std::right;
418 o << mTimer.delay() << "s]";
419 }
420 }
