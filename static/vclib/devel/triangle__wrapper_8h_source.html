265template<typename T>
266concept TriangleConcept =
267 std::derived_from< // same type or derived type
268 std::remove_cvref_t<T>,
269 Triangle<typename RemoveRef<T>::PointType>> ||
270 std::derived_from< // same type or derived type
271 std::remove_cvref_t<T>,
272 TriangleWrapper<typename RemoveRef<T>::PointType>>;
