50 SelectionUndoRedoAction(std::vector<MeshState> states) :

53 }
54
56 void undo() override { swapState(); }

60
62 std::string name() const override { return "Selection Changed"; }
63
64private:
65 void swapState()
66 {
67 for (auto& state : mStates) {
68 if (auto lock = state.obj.lock()) {
69 if (auto* mesh =
70 dynamic_cast<AbstractDrawableMesh*>(lock.get())) {
71 if (!state.vertexSelection.empty() ||
72 !mesh->vertexSelectionBitVector().empty()) {
73 auto current = mesh->vertexSelectionBitVector();
74 mesh->setVertexSelectionBitVector(
75 state.vertexSelection);
76 state.vertexSelection = std::move(current);
77 }
78 if (!state.faceSelection.empty() ||
79 !mesh->faceSelectionBitVector().empty()) {
80 auto current = mesh->faceSelectionBitVector();
81 mesh->setFaceSelectionBitVector(state.faceSelection);
82 state.faceSelection = std::move(current);
83 }
84 }
85 }
86 }
87 }
88};
