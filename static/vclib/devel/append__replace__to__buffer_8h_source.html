88template<MeshConcept MeshType>
89void appendDuplicateVertexPositionsToBuffer(
90 const MeshType& mesh,
91 const std::list<uint>& vertsToDuplicate,
92 auto* buffer,
93 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
94{
95 using namespace detail;
96
97 // no vertices to duplicate, do nothing
98 if (vertsToDuplicate.empty())
99 return;
100
101 const uint NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
102
103 for (uint i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
104 const auto& pos = mesh.vertex(v).position();
105
106 at(buffer, i, 0, NUM_ROWS, 3, storage) = pos.x();
107 at(buffer, i, 1, NUM_ROWS, 3, storage) = pos.y();
108 at(buffer, i, 2, NUM_ROWS, 3, storage) = pos.z();
109
110 ++i;
111 }
112}

159template<FaceMeshConcept MeshType>
160void replaceFaceVertexIndicesByVertexDuplicationToBuffer(
161 const MeshType& mesh,
162 const std::list<uint>& vertsToDuplicate,
163 const std::list<std::list<std::pair<uint, uint>>>& facesToReassign,
164 auto* buffer,
165 uint largestFaceSize = 3,
166 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
167{
168 using namespace detail;
169
170 // no vertices have been duplicated, do nothing
171 if (vertsToDuplicate.empty())
172 return;
173
174 assert(vertsToDuplicate.size() == facesToReassign.size());
175
176 const uint NUM_ROWS = mesh.faceCount();
177
178 uint vFirst = mesh.vertexCount();
179 uint vLast = mesh.vertexCount() + vertsToDuplicate.size();
180 for (uint vi = vFirst; const auto& faces : facesToReassign) {
181 for (const auto& f : faces) {
182 at(buffer, f.first, f.second, NUM_ROWS, largestFaceSize, storage) =
183 vi;
184 }
185 ++vi;
186 }
187}

236template<FaceMeshConcept MeshType>
237void replaceTriangulatedFaceVertexIndicesByVertexDuplicationToBuffer(
238 const MeshType& mesh,
239 const std::list<uint>& vertsToDuplicate,
240 const std::list<std::list<std::pair<uint, uint>>>& facesToReassign,
241 const TriPolyIndexBiMap& indexMap,
242 auto* buffer,
243 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
244{
245 using namespace detail;
246
247 // no vertices have been duplicated, do nothing
248 if (vertsToDuplicate.empty())
249 return;
250
251 assert(vertsToDuplicate.size() == facesToReassign.size());
252
253 uint vFirst = mesh.vertexCount();
254 uint vLast = mesh.vertexCount() + vertsToDuplicate.size();
255
256 const uint NUM_ROWS = indexMap.triangleCount();
257
258 // the facesToReassign lists for each vertex contain pairs that in the
259 // second element store the index of the vertex in the face. However, the
260 // face has been triangulated, and this info is not useful anymore. We need
261 // to look into the triangles generated for the face (first elem of the
262 // pair) and look for the vertex index to set (that is the vertex index
263 // stored in the vertsToDuplicate list).
264 for (uint vi = vFirst; const auto& [vert, faces] :
265 std::views::zip(vertsToDuplicate, facesToReassign)) {
266 // vert is the vertex index in the mesh that we need to reassign with
267 // the index vi in the buffer
268
269 // for each face f that has at least a vertex to reassign
270 for (const auto& f : faces) {
271 // get the triangle indices of the face using the index map
272 uint tBegin = indexMap.triangleBegin(f.first);
273 uint tEnd = tBegin + indexMap.triangleCount(f.first);
274 for (uint t = tBegin; t < tEnd; ++t) { // look into the triangles
275 for (uint j = 0; j < 3; ++j) { // for each vertex of triangle
276 auto& triVert = at(buffer, t, j, NUM_ROWS, 3, storage);
277 if (triVert == vert) {
278 triVert = vi;
279 }
280 }
281 }
282 }
283 ++vi;
284 }
285}

311template<FaceMeshConcept MeshType>
312void permuteFaceVertexIndicesByFunctionToBuffer(
313 const MeshType& mesh,
314 auto* buffer,
315 const std::function<bool(
316 const typename MeshType::FaceType&,
317 const typename MeshType::FaceType&)>& func,
318 TriPolyIndexBiMap& indexMap = detail::indexMap)
319{
320 // get the list of face indices sorted by material ID and using the given
321 // face comparator function. The result contains original container indices
322 // of non-deleted faces in sorted order:
323 std::vector<uint> sortedFaceContainerIndices =
324 sortFaceIndicesByFunction(mesh, func, false);
325
326 // sortedFaceContainerIndices[newPos] = originalContainerIndex
327 // (only non-deleted faces, in the desired new order)
328
329 bool isIndexMapEmpty = indexMap.triangleCount() == 0;
330
331 // temporary copy of the buffer
332 uint buffSize =
333 isIndexMapEmpty ? mesh.faceCount() : indexMap.triangleCount();
334 std::vector<uint> bufferCopy(buffSize * 3);
335
336 // temporary indexmap (not used if input map is empty)
337 TriPolyIndexBiMap indexMapCopy;
338
339 if (!isIndexMapEmpty)
340 indexMapCopy.reserve(indexMap.triangleCount(), indexMap.polygonCount());
341
342 uint copiedTriangles = 0;
343
344 for (uint i = 0; i < sortedFaceContainerIndices.size(); ++i) {
345 // place the triangles associated to the face at new position i
346 uint polyIndex = sortedFaceContainerIndices[i];
347 uint firstTri =
348 isIndexMapEmpty ? polyIndex : indexMap.triangleBegin(polyIndex);
349 uint nTris = isIndexMapEmpty ? 1 : indexMap.triangleCount(polyIndex);
350
351 std::copy(
352 buffer + firstTri * 3,
353 buffer + (firstTri + nTris) * 3,
354 bufferCopy.data() + copiedTriangles * 3);
355
356 if (!isIndexMapEmpty) {
357 for (uint t = copiedTriangles; t < copiedTriangles + nTris; ++t) {
358 indexMapCopy.insert(t, polyIndex);
359 }
360 }
361
362 copiedTriangles += nTris;
363 }
364
365 // copy back
366 std::copy(
367 bufferCopy.begin(), bufferCopy.begin() + copiedTriangles * 3, buffer);
368
369 if (!isIndexMapEmpty)
370 indexMap = std::move(indexMapCopy);
371}

466void appendDuplicateVertexNormalsToBuffer(
467 const MeshType& mesh,
468 const std::list<uint>& vertsToDuplicate,
469 auto* buffer,
470 bool normalize = false,
471 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
472{
473 using namespace detail;
474
475 // no vertices to duplicate, do nothing

479 requirePerVertexNormal(mesh);
480

482
483 for (uint i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
484 auto normal = mesh.vertex(v).normal();
485 if (normalize)

487 at(buffer, i, 0, NUM_ROWS, 3, storage) = normal.x();
488 at(buffer, i, 1, NUM_ROWS, 3, storage) = normal.y();
489 at(buffer, i, 2, NUM_ROWS, 3, storage) = normal.z();

491 ++i;

533template<MeshConcept MeshType>
534void appendDuplicateVertexColorsToBuffer(
535 const MeshType& mesh,
536 const std::list<uint>& vertsToDuplicate,
537 auto* buffer,
538 Color::Representation representation = Color::Representation::INT_0_255,
539 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
540{
541 using namespace detail;
542
543 // no vertices to duplicate, do nothing
544 if (vertsToDuplicate.empty())
545 return;
546
547 requirePerVertexColor(mesh);
548
549 const bool R_INT = representation == Color::Representation::INT_0_255;
550 const uint NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
551
552 for (uint i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
553 const auto& c = mesh.vertex(v).color();
554 at(buffer, i, 0, NUM_ROWS, 4, storage) = R_INT ? c.red() : c.redF();
555 at(buffer, i, 1, NUM_ROWS, 4, storage) = R_INT ? c.green() : c.greenF();
556 at(buffer, i, 2, NUM_ROWS, 4, storage) = R_INT ? c.blue() : c.blueF();
557 at(buffer, i, 3, NUM_ROWS, 4, storage) = R_INT ? c.alpha() : c.alphaF();
558
559 ++i;
560 }

796template<MeshConcept MeshType>
797void appendDuplicateVertexTangentsToBuffer(
798 const MeshType& mesh,
799 const std::list<uint>& vertsToDuplicate,
800 auto* buffer,
801 bool storeHandednessAsW = true,
802 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
803{
804 using namespace detail;
805
806 // no vertices to duplicate, do nothing
807 if (vertsToDuplicate.empty())
808 return;
809
810 requirePerVertexTangent(mesh);
811
812 const uint NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
813 const uint NUM_COLS = storeHandednessAsW ? 4 : 3;
814
815 for (uint i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
816 const auto& t = mesh.vertex(v).tangent();
817
818 at(buffer, i, 0, NUM_ROWS, NUM_COLS, storage) = t.x();
819 at(buffer, i, 1, NUM_ROWS, NUM_COLS, storage) = t.y();
820 at(buffer, i, 2, NUM_ROWS, NUM_COLS, storage) = t.z();
821 if (storeHandednessAsW) {
822 at(buffer, i, 3, NUM_ROWS, NUM_COLS, storage) =
823 mesh.vertex(v).tangentRightHanded() ? 1.0 : -1.0;
824 }
825
826 ++i;
827 }
