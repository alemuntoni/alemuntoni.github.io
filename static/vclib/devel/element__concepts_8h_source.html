23/* =========================================================================
24 * Base Element Identity Concepts
25 *
26 * ElementConcept, VertexConcept, FaceConcept, EdgeConcept, and WedgeConcept
27 * are the primary identity concepts defined alongside their respective
28 * element and proxy definitions:
29 * - ElementConcept<T>: defined in <vclib/mesh/elements/base/element_proxy.h>
30 * - VertexConcept<T>: defined in <vclib/mesh/elements/vertex.h>
31 * - FaceConcept<T>: defined in <vclib/mesh/elements/face.h>
32 * - EdgeConcept<T>: defined in <vclib/mesh/elements/edge.h>
33 * - WedgeConcept<T>: defined in <vclib/mesh/components/wedges/wedge_proxy.h>
34 * ========================================================================= */
35

45template<typename T>
46concept TriangleFaceConcept = FaceConcept<T> && requires {

55template<typename T>

65template<typename T>
66concept PolygonFaceConcept = FaceConcept<T> && requires {
