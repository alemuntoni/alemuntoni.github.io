134 /*
135 * This member function is called when we need to update the indices in
136 * this container after a reallocation (the pointer of the first element of
137 * the container of Elem is changed).
138 *
139 * With respect to the updateReferences function of the
140 * PointerContainerComponent, we need to do something only when offset is
141 * different from 0, because in this case the indices of the elements have
142 * changed (append case).
143 *
144 * When we append elements, this function is called only on the elements
145 * that have been appended, and the offset is the number of referenced
146 * elements that were in the container before the append operation.
147 */
148 void updateReferences(const Elem*, std::size_t offset = 0)
149 {
150 if (offset != 0) {
151 auto& baseContainer = Base::container();
152
153 for (uint j = 0; j < baseContainer.size(); ++j) {
154 if (baseContainer.at(j) != UINT_NULL) {
155 baseContainer.at(j) += offset;
156 }
157 }
158 }
159 }
160
161 /*
162 * This member function is called when we need to update the pointers or
163 * indices in this containers, usually after a compaction of the container
164 * (but not always).
165 *
166 * In this case, indices must be updated, because the indices of the
167 * elements have changed.
168 */
169 void updateReferences(const std::vector<uint>& newIndices)
170 {
171 auto& baseContainer = Base::container();
172
173 for (uint j = 0; j < baseContainer.size(); ++j) {
174 if (baseContainer.at(j) != UINT_NULL) {
175 baseContainer.at(j) = newIndices.at(baseContainer.at(j));
176 }
177 }
178 }
179};
