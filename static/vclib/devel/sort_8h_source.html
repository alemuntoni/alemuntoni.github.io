19template<FaceMeshConcept MeshType>
20std::vector<MeshEdgeUtil<MeshType>> fillAndSortMeshEdgeUtilVector(
21 MeshType& m,
22 bool includeFauxEdges = true)
23{
24 using FaceType = MeshType::FaceType;
25
26 std::vector<MeshEdgeUtil<MeshType>> vec;
27
28 int n_edges = 0;
29 for (const FaceType& f : m.faces())
30 n_edges += f.vertexCount();
31
32 vec.reserve(n_edges);
33
34 for (FaceType& f : m.faces()) { // fill it with face data
35 for (uint j = 0; j < f.vertexCount(); ++j) {
36 if (includeFauxEdges || !f.edgeFaux(j)) {
37 vec.emplace_back(f, j);
38 }
39 }
40 }
41
42 // Sort it by vertices
43 std::sort(std::execution::par_unseq, vec.begin(), vec.end());
44
45 return vec;
46}

100template<uint ELEM_ID, MeshConcept MeshType>
101std::vector<uint> sortElemIndicesByFunction(
102 const MeshType& mesh,
103 const std::function<bool(
104 const typename MeshType::template ElementType<ELEM_ID>&,
105 const typename MeshType::template ElementType<ELEM_ID>&)>& func,
106 bool getIndicesAsIfContainerCompact = false)
107{
108 using ElemType = typename MeshType::template ElementType<ELEM_ID>;
109
110 auto compactIndices = detail::elemCompactIndices<ELEM_ID>(
111 mesh, getIndicesAsIfContainerCompact);
112
113 std::vector<uint> indices;
114 indices.reserve(mesh.template count<ELEM_ID>());
115
116 // Initialize indices with the actual non-deleted element container indices
117 for (const auto& el : mesh.template elements<ELEM_ID>()) {
118 indices.push_back(el.index());
119 }
120
121 std::sort(indices.begin(), indices.end(), [&](uint a, uint b) {
122 return func(
123 mesh.template element<ELEM_ID>(a),
124 mesh.template element<ELEM_ID>(b));
125 });
126
127 if (!compactIndices.empty()) {
128 for (auto& idx : indices) {
129 idx = compactIndices[idx];
130 }
131 }
132
133 return indices;
134}
