229 template<Range R>
230 requires Point3Concept<std::ranges::range_value_t<R>>
231 void setVertices(R&& verts)
232 {
233 mVerPosCount = std::ranges::size(verts);
234
235 // Compute padding to ensure the buffer size is a multiple of 4 floats
236 // (16 bytes). This is required because the vertex shader reads the
237 // buffer as vec4 elements.
238 uint padding = (4 - (mVerPosCount % 4)) % 4;
239 uint nv = mVerPosCount + padding;
240
241 VertexBuffer vertBuff;
242 auto [buffer, releaseFn] =
243 Context::getAllocatedBufferAndReleaseFn<float>(nv * 3);
244
245 for (size_t i = 0; const auto& v : verts) {
246 buffer[i * 3 + 0] = v.x();
247 buffer[i * 3 + 1] = v.y();
248 buffer[i * 3 + 2] = v.z();
249 ++i;
250 }
251
252 vertBuff.create(
253 buffer,
254 nv,
255 bgfx::Attrib::Position,
256 3,
257 PrimitiveType::FLOAT,
258 releaseFn);
259 mVertexPositions.setOwned(std::move(vertBuff));
260 mIsUpdateProgramNeeded = true;
261 }

319 void setVertexColors(R&& vertColors)
320 {
321 mVerColCount = std::ranges::size(vertColors);
322
323 // Compute padding to ensure the buffer size is a multiple of 16 bytes.
324 uint padding = (4 - (mVerColCount % 4)) % 4;
325 uint nc = mVerColCount + padding;

327 VertexBuffer vColsBuff;

329 auto [buffer, releaseFn] =

332 for (uint i = 0; const auto& c : vertColors) {

334 ++i;

340 bgfx::Attrib::Color0,

343 true,
344 releaseFn);

355 template<Range R>
356 requires std::convertible_to<std::ranges::range_value_t<R>, bool>
357 void setVertexSelection(R&& selections)
358 {
359 mVerSelCount = std::ranges::size(selections);
360 BooleanBuffer buf;
361 buf.init(mVerSelCount);
362
363 vcl::BitVector<true> backup(mVerSelCount, false);
364
365 uint vidx = 0;
366 for (bool sel : selections) {
367 backup[vidx++] = sel;
368 }
369
370 buf.setFromCPUBuffer(backup);
371 mSelectionBuffer.setOwned(std::move(buf));
372 mIsUpdateProgramNeeded = true;
373 }
