383template<typename T>

400template<typename T>

415/* Implementation of member functions */
416
417template<int N, std::unsigned_integral Index, bool TTVC, bool OPT>
418template<typename OtherElement>
419void AdjacentFaces<N, Index, TTVC, OPT>::importFrom(
420 this auto&& self,
421 const OtherElement& other,
422 bool importRefs)
423{
424 if (importRefs) {
425 if constexpr (HasAdjacentFaces<OtherElement>) {
426 if (other.template isComponentAvailable<CompId::ADJACENT_FACES>()) {
427 using OtherComp =
428 ComponentTypeT<CompId::ADJACENT_FACES, OtherElement>;
429 constexpr int OTHER_N = OtherComp::ADJ_FACE_COUNT;
430
431 if constexpr (N > 0) {
432 if constexpr (N == OTHER_N) {
433 for (uint i = 0; i < N; ++i) {
434 self.setAdjFace(
435 i, Base::castIndex(other.adjFaceIndex(i)));
436 }
437 }
438 else if constexpr (OTHER_N < 0) {
439 if (other.adjFaceCount() == N) {
440 for (uint i = 0; i < N; ++i) {
441 self.setAdjFace(
442 i, Base::castIndex(other.adjFaceIndex(i)));
443 }
444 }
445 }
446 }
447 else if constexpr (!TTVC) {
448 self.resizeAdjFaces(other.adjFaceCount());
449 for (uint i = 0; i < other.adjFaceCount(); ++i) {
450 self.setAdjFace(
451 i, Base::castIndex(other.adjFaceIndex(i)));
452 }
453 }
454 else {
455 if (self.adjFaceCount() == other.adjFaceCount()) {
456 for (uint i = 0; i < self.adjFaceCount(); ++i) {
457 self.setAdjFace(
458 i, Base::castIndex(other.adjFaceIndex(i)));
459 }
460 }
461 }
462 }
463 }
464 }
