322 template<typename... I>
323 ConstPointer data(I... indices) const requires (sizeof...(indices) < N)
324 {
325 constexpr std::size_t n = sizeof...(indices);
326 if constexpr (n == 0) {
327 return mVec.data();
328 }
329 else {
330 std::size_t args[] = {static_cast<std::size_t>(indices)...};
331 std::size_t ind = args[0];
332 assert(args[0] < mSizes[0]);
333 uint i;
334 for (i = 1; i < n; i++) {
335 assert(args[i] < mSizes[i]);
336 ind *= mSizes[i];
337 ind += args[i];
338 }
339 for (; i < N; i++) {
340 ind *= mSizes[i];
341 }
342 return &mVec[ind];
343 }
344 }

440 template<typename... Sizes>
441 void conservativeResize(Sizes... s) requires (sizeof...(s) == N)
442 {
443 std::size_t newSizes[N] = {static_cast<std::size_t>(s)...};
444 std::size_t newTotalSize = 1;
445 for (uint i = 0; i < N; i++)
446 newTotalSize *= newSizes[i];
447 std::vector<T> newVector(newTotalSize);
448
449 for (std::size_t i = 0; i < mVec.size(); i++) {
450 std::array<std::size_t, N> indices = reverseIndex(i);
451 bool outOfBound = false;
452 for (std::size_t j = 0; j < N; j++)
453 if (indices[j] >= newSizes[j] || indices[j] >= mSizes[j])
454 outOfBound = true;
455 if (!outOfBound) {
456 newVector[getIndex(indices.data(), newSizes)] = mVec[i];
457 }
458 }
459
460 for (uint i = 0; i < mSizes.size(); i++) {
461 mSizes[i] = newSizes[i];
462 }
463
464 mVec = std::move(newVector);
465 }

500 Array<T, N - 1> subArray(uint r) const requires (N > 1)
501 {
502 assert(r < mSizes[0]);
503 Array<T, N - 1> sub;
504 size_t size = 1;
505 for (uint i = 0; i < mSizes.size() - 1; i++) {
506 sub.mSizes[i] = mSizes[i + 1];
507 size *= sub.mSizes[i];
508 }
509 sub.mVec = std::vector<T>(
510 mVec.begin() + r * size, mVec.begin() + (r + 1) * size);
511 return sub;
512 }

559 std::size_t getIndex(const std::size_t indices[]) const
560 {
561#ifndef NDEBUG
562 for (uint i = 0; i < N; i++) {
563 assert(indices[i] < mSizes[i]);
564 }
565#endif
566 return linearizeIndex(mSizes.data(), indices, N);
567 }

569 std::array<std::size_t, N> reverseIndex(uint index)
570 {
571 std::array<std::size_t, N> indices;
572 for (long int i = N - 1; i >= 0; i--) {
573 indices[i] = index % mSizes[i];
574 index /= mSizes[i];
575 }
576 return indices;
577 }

579 static std::size_t getIndex(
580 const std::size_t indices[],
581 const std::size_t sizes[])
582 {
583#ifndef NDEBUG
584 for (uint i = 0; i < N; i++) {
585 assert(indices[i] < sizes[i]);
586 }
587#endif
588 return linearizeIndex(sizes, indices, N);
589 }

591 void initializeNestedLists(NestedInitializerLists<T, N> values)
592 {
593 std::list<std::size_t> szs =
594 NestedInitializerListsProcessor<T, N>::maxDimensionsLevels(values);
595
596 uint i = 0;
597 size_t totalSize = 1;
598 for (std::size_t s : szs) {
599 totalSize *= s;
600 mSizes[i++] = s;
601 }
602 mVec.resize(totalSize);
603
604 typename std::vector<T>::iterator iterator = mVec.begin();
605 NestedInitializerListsProcessor<T, N>::processElements(
606 values,
607 [&iterator](T value) {
608 *(iterator++) = value;
609 },
610 szs);
611 }
