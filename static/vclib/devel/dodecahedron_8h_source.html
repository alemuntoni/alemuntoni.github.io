25template<PolygonMeshConcept MeshType, LoggerConcept LogType = NullLogger>
26MeshType createDodecahedron(LogType& log = nullLogger)
27{
28 using VertexType = MeshType::VertexType;
29 using PositionType = VertexType::PositionType;
30 using ScalarType = PositionType::ScalarType;
31
32 MeshType mesh;
33
34 const ScalarType fi = (1 + std::sqrt(5)) / 2;
35 const ScalarType s0 = 0;
36 const ScalarType s1 = fi;
37 const ScalarType s2 = std::pow(fi, 2);
38 const ScalarType s3 = 1;
39
40 log.log(0, "Adding vertices to PolyMesh...");
41
42 mesh.addVertices(
43 PositionType(-s1, -s1, s1),
44 PositionType(s2, s3, s0),
45 PositionType(s2, -s3, s0),
46 PositionType(-s2, s3, s0),
47 PositionType(-s2, -s3, s0),
48 PositionType(s0, s2, s3),
49 PositionType(s0, s2, -s3),
50 PositionType(s3, s0, -s2),
51 PositionType(-s3, s0, -s2),
52 PositionType(s0, -s2, -s3),
53 PositionType(s0, -s2, s3),
54 PositionType(s3, s0, s2),
55 PositionType(-s3, s0, s2),
56 PositionType(s1, s1, -s1),
57 PositionType(s1, s1, s1),
58 PositionType(-s1, s1, -s1),
59 PositionType(-s1, s1, s1),
60 PositionType(s1, -s1, -s1),
61 PositionType(s1, -s1, s1),
62 PositionType(-s1, -s1, -s1));
63
64 log.log(50, "Vertices added to PolyMesh.");
65 log.log(50, "Adding faces to PolyMesh...");
66
67 mesh.reserveFaces(12);
68 mesh.addFace(14, 11, 18, 2, 1);
69 mesh.addFace(2, 17, 7, 13, 1);
70 mesh.addFace(15, 8, 19, 4, 3);
71 mesh.addFace(4, 0, 12, 16, 3);
72 mesh.addFace(16, 5, 6, 15, 3);
73 mesh.addFace(13, 6, 5, 14, 1);
74 mesh.addFace(18, 10, 9, 17, 2);
75 mesh.addFace(19, 9, 10, 0, 4);
76 mesh.addFace(17, 9, 19, 8, 7);
77 mesh.addFace(13, 7, 8, 15, 6);
78 mesh.addFace(16, 12, 11, 14, 5);
79 mesh.addFace(18, 11, 12, 0, 10);
80
81 log.log(100, "Faces added to PolyMesh.");
82
83 return mesh;
84}

98template<TriangleMeshConcept MeshType, LoggerConcept LogType = NullLogger>
99MeshType createDodecahedron(LogType& log = nullLogger)
100{
101 log.startNewTask(0, 75, "Create Polygonal Dodecahedron.");
102
103 detail::TMPSimplePolyMesh pmesh =
104 createDodecahedron<detail::TMPSimplePolyMesh>(log);
105
106 log.endTask("Create Polygonal Dodecahedron.");
107 log.log(75, "Copying vertices into TriMesh...");
108
109 MeshType mesh;
110 using ST = MeshType::VertexType::PositionType::ScalarType;
111 mesh.reserveVertices(pmesh.vertexCount());
112 for (const auto& v : pmesh.vertices()) {
113 mesh.addVertex(v.position().cast<ST>());
114 }
115
116 log.log(80, "Vertices copied into TriMesh.");
117 log.log(80, "Triangularize and copy Faces into TriMesh...");
118
119 for (const auto& f : pmesh.faces()) {
120 std::vector<uint> ind = earCut(f);
121 for (uint i = 0; i < ind.size(); i += 3) {
122 mesh.addFace(
123 pmesh.index(f.vertex(ind[i])),
124 pmesh.index(f.vertex(ind[i + 1])),
125 pmesh.index(f.vertex(ind[i + 2])));
126 }
127 }
128
129 log.log(100, "Faces triangularized and copied into TriMesh.");
130
131 return mesh;
132}
