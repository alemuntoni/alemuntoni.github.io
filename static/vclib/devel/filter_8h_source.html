19template<MeshConcept OutMeshType, uint ELEM_ID, MeshConcept InMeshType>
20OutMeshType perElementMeshFilter(
21 const InMeshType& m,
22 Range auto&& elemFilterRng,
23 bool saveBirthIndicesInCustomComponent = true)
24{
25 using OutElemType = OutMeshType::template ElementType<ELEM_ID>;
26
27 std::string ccname = "birth" + elementEnumString<ELEM_ID>();
28
29 OutMeshType res;
30 res.enableSameOptionalComponentsOf(m);
31
32 // enable the custom component birth element
33 if constexpr (comp::HasCustomComponents<OutElemType>) {
34 if (saveBirthIndicesInCustomComponent) {
35 res.template addPerElementCustomComponent<ELEM_ID, uint>(ccname);
36 }
37 }
38
39 for (const auto& [birthV, filter] :
40 std::views::zip(m.template elements<ELEM_ID>(), elemFilterRng)) {
41 if (filter) {
42 uint v = res.template add<ELEM_ID>();
43 // import all the components from the input mesh
44 res.template element<ELEM_ID>(v).importFrom(birthV, false);
45 // set the birth element
46 if constexpr (comp::HasCustomComponents<OutElemType>) {
47 if (saveBirthIndicesInCustomComponent) {
48 res.template element<ELEM_ID>(v)
49 .template customComponent<uint>(ccname) =
50 m.index(birthV);
51 }
52 }
53 }
54 }
55
56 return res;
57}

73template<MeshConcept OutMeshType, uint ELEM_ID, MeshConcept InMeshType>
74OutMeshType perElementMeshFilterWithVRefs(
75 const InMeshType& m,
76 Range auto&& elemFilterRng,
77 bool saveBirthIndicesInCustomComponent = true)
78{
79 using InVertexType = InMeshType::VertexType;
80 using OutElemType = OutMeshType::template ElementType<ELEM_ID>;
81
82 std::string ccname = "birth" + elementEnumString<ELEM_ID>();
83
84 OutMeshType res;
85 res.enableSameOptionalComponentsOf(m);
86
87 // enable the custom component birthVertex
88 if constexpr (HasPerVertexCustomComponents<OutMeshType>) {
89 if (saveBirthIndicesInCustomComponent) {
90 res.template addPerVertexCustomComponent<uint>("birthVertex");
91 }
92 }
93
94 // enable the custom component birth element
95 if constexpr (comp::HasCustomComponents<OutElemType>) {
96 if (saveBirthIndicesInCustomComponent) {
97 res.template addPerElementCustomComponent<ELEM_ID, uint>(ccname);
98 }
99 }
100
101 std::vector<uint> vertexMapping(m.vertexContainerSize(), UINT_NULL);
102
103 for (const auto& [birthF, filter] :
104 std::views::zip(m.template elements<ELEM_ID>(), elemFilterRng)) {
105 if (filter) {
106 std::vector<uint> verts(birthF.vertexCount(), UINT_NULL);
107 uint vi = 0; // incremented with vertices of the element
108 // set all the vertex indices in the verts vector
109 // two cases here:
110 // - the ith vertex of the face has been already added, we need just
111 // to take its id in the out mesh from the vertexMapping vector
112 // - the ith vertex of the face has not been added: we need to add
113 // it and import all its components, and update the
114 // vertexMappingVector
115 for (const InVertexType* v : birthF.vertices()) {
116 // the vertex has not already added
117 if (vertexMapping[m.index(v)] == UINT_NULL) {
118 // add the vertex to the out mesh
119 uint ov = res.addVertex();
120 // import all the components from the input mesh
121 res.vertex(ov).importFrom(*v, false);
122 if constexpr (HasPerVertexCustomComponents<OutMeshType>) {
123 // set the birth vertex
124 if (saveBirthIndicesInCustomComponent) {
125 res.vertex(ov).template customComponent<uint>(
126 "birthVertex") = m.index(v);
127 }
128 }
129 vertexMapping[m.index(v)] = ov;
130 verts[vi] = ov;
131 }
132 else {
133 verts[vi] = vertexMapping[m.index(v)];
134 }
135 ++vi;
136 }
137
138 // now all the vertices of the elements are in the out mesh, we can
139 // add the actual element
140 uint f = res.template add<ELEM_ID>();
141 // import all the components from the input mesh
142 res.template element<ELEM_ID>(f).importFrom(birthF, false);
143
144 if constexpr (OutElemType::VERTEX_COUNT < 0) {
145 res.template element<ELEM_ID>(f).resizeVertices(verts.size());
146 }
147
148 res.template element<ELEM_ID>(f).setVertices(verts);
149
150 if constexpr (comp::HasCustomComponents<OutElemType>) {
151 // set the birth elements
152 if (saveBirthIndicesInCustomComponent) {
153 res.template element<ELEM_ID>(f)
154 .template customComponent<uint>(ccname) =
155 m.index(birthF);
156 }
157 }
158 }
159 }
160 return res;
161}

564template<EdgeMeshConcept OutMeshType, FaceMeshConcept InMeshType>
565OutMeshType perFaceEdgeMeshFilter(
566 const InMeshType& m,
567 const std::function<bool(const typename InMeshType::FaceType&, uint)>&
568 faceEdgeFilter,
569 bool dontDuplicateEdges = true,
570 bool saveBirthIndicesInCustomComponent = true)
571{
572 using InVertexType = InMeshType::VertexType;
573 using OutEdgeType = OutMeshType::EdgeType;
574
575 OutMeshType res;
576 // todo: enable only per vertex same optional components of m
577 res.enableSameOptionalComponentsOf(m);
578
579 // enable the custom component birthVertex
580 if constexpr (HasPerVertexCustomComponents<OutMeshType>) {
581 if (saveBirthIndicesInCustomComponent) {
582 res.template addPerVertexCustomComponent<uint>("birthVertex");
583 }
584 }
585
586 std::vector<uint> vertexMapping(m.vertexContainerSize(), UINT_NULL);
587
588 std::set<std::pair<uint, uint>, UnorderedPairComparator<uint>>
589 unorderedEdges;
590
591 for (const auto& f : m.faces()) {
592 for (uint ei = 0; ei < f.vertexCount(); ++ei) {
593 if (faceEdgeFilter(f, ei)) {
594 std::array<uint, 2> verts = {UINT_NULL, UINT_NULL};
595 for (uint i = 0; i < 2; ++i) {
596 const InVertexType* v = f.vertexMod(ei + i);
597 if (vertexMapping[m.index(v)] == UINT_NULL) {
598 uint ov = res.addVertex();
599 res.vertex(ov).importFrom(*v, false);
600 if constexpr (HasPerVertexCustomComponents<
601 OutMeshType>) {
602 if (saveBirthIndicesInCustomComponent) {
603 res.vertex(ov).template customComponent<uint>(
604 "birthVertex") = m.index(v);
605 }
606 }
607 vertexMapping[m.index(v)] = ov;
608 verts[i] = ov;
609 }
610 else {
611 verts[i] = vertexMapping[m.index(v)];
612 }
613 }
614
615 std::pair<uint, uint> ep {verts[0], verts[1]};
616 if (!dontDuplicateEdges || !unorderedEdges.contains(ep)) {
617 uint e = res.addEdge(verts[0], verts[1]);
618 unorderedEdges.insert(ep);
619 }
620 }
621 }
622 }
623
624 return res;
625}
