198 void bindVertexBuffers(const MeshRenderSettings& mrs) const
199 {
200 // TODO: streams cannot be higher than
201 // "bgfx::getCaps()->limits.maxVertexStreams". Buffers must be bound
202 // only if necessary (using MeshRenderSettings). right now, this is
203 // managed only for uvs (per vertex or per wedge, not both).
204 // We MUST be sure that the limit is not exceeded.
205
206 using enum MeshRenderInfo::Surface;
207
208 uint stream = 0;
209
210 // streams MUST be consecutive starting from 0
211 // otherwise on metal it won't work
212 mVertexPositionsBuffer.bind(stream++);
213
214 if (mVertexNormalsBuffer.isValid()) {
215 // bgfx limitation
216 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
217 mVertexNormalsBuffer.bind(stream++);
218 }
219
220 if (mVertexTangentsBuffer.isValid()) {
221 // bgfx limitation
222 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
223 mVertexTangentsBuffer.bind(stream++);
224 }
225
226 if (mVertexColorsBuffer.isValid()) {
227 // bgfx limitation
228 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
229 mVertexColorsBuffer.bind(stream++);
230 }
231
232 if (mVertexUVBuffer.isValid() && mrs.isSurface(COLOR_VERTEX_TEX)) {
233 // bgfx limitation
234 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
235 mVertexUVBuffer.bind(stream++);
236 }
237
238 if (mVertexWedgeUVBuffer.isValid() && mrs.isSurface(COLOR_WEDGE_TEX)) {
239 // bgfx limitation
240 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
241 mVertexWedgeUVBuffer.bind(stream++);
242 }
243 }

289 uint bindTextures(
290 const MeshRenderSettings& mrs,
291 uint chunkNumber,
292 const MeshType& m) const
293 {
294 ushort materialId = Base::materialIndex(mrs, chunkNumber);
295
296 uint boundTextures = 0;
297
298 DrawableMeshUniforms::TextureType tt =
299 DrawableMeshUniforms::TextureType::BASE_COLOR;
300
301 if (materialId != USHORT_NULL) {
302 for (uint j = 0; j < N_TEXTURE_TYPES; ++j) {
303 const auto& td = m.material(materialId).textureDescriptor(j);
304 const std::string& path = td.path();
305 if (!path.empty()) {
306 const Texture& tex = mMaterialTextures.at(path);
307 if (tex.isValid()) {
308 uint flags = Texture::samplerFlagsFromTexture(td);
309 tex.bind(
310 boundTextures,
311 sTextureSamplerUniforms[j].get().handle(),
312 flags);
313
314 tt = static_cast<DrawableMeshUniforms::TextureType>(j);
315 DrawableMeshUniforms::setTextureStage(
316 tt, boundTextures);
317 boundTextures++;
318 }
319 }
320 }
321 }
322 return boundTextures;
323 }

325 void updateEdgeSettings(const MeshRenderSettings& mrs)
326 {
327 using enum MeshRenderInfo::Edges;
328 using enum Lines::ColorSetting;
329
330 mEdgeLines.setWidth(mrs.edgesWidth());
331
332 Lines::Shading sh = Lines::Shading::NONE;
333 if (mrs.isEdges(SHADING_SMOOTH)) {
334 sh = Lines::Shading::PER_VERTEX;
335 }
336 else if (mrs.isEdges(SHADING_FLAT)) {
337 sh = Lines::Shading::PER_LINE;
338 }
339 mEdgeLines.setShading(sh);
340
341 if (mrs.isEdges(COLOR_USER)) {
342 mEdgeLines.setGeneralColor(mrs.edgesUserColor());
343 mEdgeLines.setColorSetting(GENERAL);
344 }
345 else if (mrs.isEdges(COLOR_MESH)) {
346 mEdgeLines.setGeneralColor(mMeshColor);
347 mEdgeLines.setColorSetting(GENERAL);
348 }
349 else if (mrs.isEdges(COLOR_VERTEX)) {
350 mEdgeLines.setColorSetting(PER_VERTEX);
351 }
352 else if (mrs.isEdges(COLOR_EDGE)) {
353 mEdgeLines.setColorSetting(PER_LINE);
354 }
355
356 mEdgeLines.setSelectionVisibility(mrs.isEdges(SELECTION));
357 mEdgeLines.setSelectionColor(mrs.edgesSelectionColor());
358 }

362 using enum MeshRenderInfo::Wireframe;

374 mWireframeLines.setColorSetting(GENERAL);
375 }
376 else if (mrs.isWireframe(COLOR_MESH)) {
377 mWireframeLines.setGeneralColor(mMeshColor);
378 mWireframeLines.setColorSetting(GENERAL);
379 }
380 else if (mrs.isWireframe(COLOR_VERTEX)) {
381 mWireframeLines.setColorSetting(PER_VERTEX);
382 }

391 void updatePointsSettings(const MeshRenderSettings& mrs)
392 {
393 using enum MeshRenderInfo::Points;
394 using enum Points::ColorSetting;
395

414 mPoints.setGeneralColor(mrs.pointUserColor());
415 mPoints.setColorSetting(GENERAL);
416 }
417 else if (mrs.isPoints(COLOR_MESH)) {
418 mPoints.setGeneralColor(mMeshColor);

422 mPoints.setColorSetting(PER_VERTEX);

670 void setTextures(const MeshType& mesh) // override
671 {
672 auto downsampleNormalMap2x2 = [](const uint8_t* src,
673 uint8_t* dst,
674 uint srcWidth,
675 uint srcHeight,
676 uint dstWidth,
677 uint dstHeight) {
678 for (uint y = 0; y < dstHeight; ++y) {
679 for (uint x = 0; x < dstWidth; ++x) {
680 uint sx = x * 2;
681 uint sy = y * 2;
682
683 uint sx1 = std::min(sx + 1, srcWidth - 1);
684 uint sy1 = std::min(sy + 1, srcHeight - 1);
685
686 // Fetch the 4 neighboring pixels from the higher resolution
687 // mip level
688 const uint8_t* p00 = src + (sy * srcWidth + sx) * 4;
689 const uint8_t* p10 = src + (sy * srcWidth + sx1) * 4;
690 const uint8_t* p01 = src + (sy1 * srcWidth + sx) * 4;
691 const uint8_t* p11 = src + (sy1 * srcWidth + sx1) * 4;
692
693 float nx = 0.0f, ny = 0.0f, nz = 0.0f, alpha = 0.0f;
694 const uint8_t* pixels[4] = {p00, p10, p01, p11};
695
696 for (int i = 0; i < 4; ++i) {
697 // Remap from [0, 255] to [-1.0, 1.0] to get the true
698 // normal vector direction
699 nx += (pixels[i][0] / 255.0f) * 2.0f - 1.0f;
700 ny += (pixels[i][1] / 255.0f) * 2.0f - 1.0f;
701 nz += (pixels[i][2] / 255.0f) * 2.0f - 1.0f;
702 // Accumulate alpha for standard bilinear interpolation
703 alpha += pixels[i][3];
704 }
705
706 // Normalize the averaged normal vector to preserve unit
707 // length
708 float len = std::sqrt(nx * nx + ny * ny + nz * nz);
709 if (len > 0.0001f) {
710 nx /= len;
711 ny /= len;
712 nz /= len;
713 }
714 else {
715 // Fallback to pointing upwards if the vector is
716 // degenerate
717 nx = 0.0f;
718 ny = 0.0f;
719 nz = 1.0f;
720 }
721
722 // Remap the normalized vector back to [0, 255] and write to
723 // destination
724 uint8_t* out = dst + (y * dstWidth + x) * 4;
725 out[0] = static_cast<uint8_t>(std::max(
726 0.0f,
727 std::min(255.0f, (nx * 0.5f + 0.5f) * 255.0f + 0.5f)));
728 out[1] = static_cast<uint8_t>(std::max(
729 0.0f,
730 std::min(255.0f, (ny * 0.5f + 0.5f) * 255.0f + 0.5f)));
731 out[2] = static_cast<uint8_t>(std::max(
732 0.0f,
733 std::min(255.0f, (nz * 0.5f + 0.5f) * 255.0f + 0.5f)));
734 out[3] = static_cast<uint8_t>(
735 std::max(0.0f, std::min(255.0f, alpha / 4.0f + 0.5f)));
736 }
737 }

740 // lambda that sets a texture
741 auto setTexture = [&](const Image& img,
742 const std::string& path,
743 bool generateMips,
744 bool isNormalMap) {
745 const uint size = img.width() * img.height();
746 assert(size > 0);
747
748 uint sizeWithMips = bimg::imageGetSize(
749 nullptr,
750 img.width(),
751 img.height(),
752 1,
753 false,
754 generateMips,
755 1,
756 bimg::TextureFormat::RGBA8) /
757 4; // in uints
758
759 uint numMips = 1;
760 if (generateMips)
761 numMips = bimg::imageGetNumMips(
762 bimg::TextureFormat::RGBA8, img.width(), img.height());
763
764 auto [buffer, releaseFn] =
765 Context::getAllocatedBufferAndReleaseFn<uint>(sizeWithMips);
766
767 const uint* tdata = reinterpret_cast<const uint*>(img.data());
768
769 std::copy(tdata, tdata + size, buffer); // mip level 0
770
771 if (numMips > 1) {
772 uint* source = buffer;
773 uint* dest;
774 uint offset = size;
775
776 for (uint mip = 1; mip < numMips; mip++) {
777 dest = source + offset;
778
779 uint srcWidth = std::max(1, img.width() >> (mip - 1));
780 uint srcHeight = std::max(1, img.height() >> (mip - 1));
781 uint dstWidth = std::max(1, img.width() >> mip);
782 uint dstHeight = std::max(1, img.height() >> mip);
783 uint mipSize = dstWidth * dstHeight;
784
785 if (isNormalMap) {
786 const uint8_t* src8 =
787 reinterpret_cast<const uint8_t*>(source);
788 uint8_t* dst8 = reinterpret_cast<uint8_t*>(dest);
789 downsampleNormalMap2x2(
790 src8,
791 dst8,
792 srcWidth,
793 srcHeight,
794 dstWidth,
795 dstHeight);
796 }
797 else {
798 bimg::imageRgba8Downsample2x2(
799 dest, // output location
800 srcWidth, // input width
801 srcHeight, // input height
802 1, // depth, always 1 for 2D textures
803 srcWidth * 4, // input pitch
804 dstWidth * 4, // output pitch
805 source // input location
806 );
807 }
808
809 source = dest;
810 offset = mipSize;
811 }
812 }
813
814 uint64_t flags = BGFX_TEXTURE_NONE | BGFX_SAMPLER_NONE;
815
816 if (img.colorSpace() == Image::ColorSpace::SRGB)
817 flags |= BGFX_TEXTURE_SRGB;
818
819 Texture tex;
820 tex.set(
821 buffer,
822 vcl::Point2i(img.width(), img.height()),
823 generateMips,
824 flags,
825 bgfx::TextureFormat::RGBA8,
826 false,
827 releaseFn);
828
829 // at() does not insert if already present, thus safe in parallel
830 mMaterialTextures.at(path) = std::move(tex);
831 };
832
833 auto loadImageAndSetTexture = [&](const std::pair<std::string, uint>&
834 pathPair) {
835 const std::string& path = pathPair.first;
836
837 uint materialId = pathPair.second / N_TEXTURE_TYPES;
838 uint textureType = pathPair.second % N_TEXTURE_TYPES;
839 // copy the image because it could be not loaded,
840 // and at the end it needs to be mirrored.
841 vcl::Image txtImg = mesh.textureImage(path);
842 if (txtImg.isNull()) { // try to load it just for rendering
843 try {
844 txtImg = vcl::loadImage(mesh.meshBasePath() + path);
845 txtImg.colorSpace() = Material::textureTypeToColorSpace(
846 static_cast<Material::TextureType>(textureType));
847 }
848 catch (...) {
849 // do nothing
850 }
851 if (txtImg.isNull()) {
852 // still null, use a dummy texture
853 txtImg = createCheckBoardImage(512);
854 }
855 }
856
857 // if loading succeeded (or dummy texture has been created)
858 if (!txtImg.isNull()) {
859 const TextureDescriptor& tex =
860 mesh.material(materialId).textureDescriptor(textureType);
861 using enum TextureDescriptor::MinificationFilter;
862
863 TextureDescriptor::MinificationFilter minFilter =
864 tex.minFilter();
865
866 bool hasMips =
867 minFilter >= NEAREST_MIPMAP_NEAREST ||
868 minFilter == NONE; // default LINEAR_MIPMAP_LINEAR
869
870 txtImg.mirror();
871 setTexture(
872 txtImg,
873 path,
874 hasMips,
875 textureType == toUnderlying(Material::TextureType::NORMAL));
876 }
877 };
878
879 mMaterialTextures.clear();
880
881 if constexpr (vcl::HasMaterials<MeshType>) {
882 // textures could be missing from the textureImages of the mesh
883 // setting the texture paths in a map - key is the path and value is
884 // an uint where materialIndex and textureType are encoded
885 // map is used to avoid duplicates, then is moved to a vector for
886 // parallel processing
887 std::map<std::string, uint> texturePaths;
888 for (ushort i = 0; i < mesh.materialCount(); ++i) {
889 for (uint j = 0; j < N_TEXTURE_TYPES; ++j) {
890 const vcl::TextureDescriptor& td =
891 mesh.material(i).textureDescriptor(j);
892 if (!td.path().empty()) {
893 texturePaths[td.path()] = i * N_TEXTURE_TYPES + j;
894
895 // create a null texture in the map
896 // this is crucial to avoid insertions during the
897 // actual creation, that is done in parallel
898 mMaterialTextures[td.path()] = Texture();
899 }
900 }
901 }
902
903 // move to vector for parallel processing
904 std::vector<std::pair<std::string, uint>> texturePathVec;
905 texturePathVec.reserve(texturePaths.size());
906 for (const auto& tp : texturePaths) {
907 texturePathVec.push_back(tp);
908 }
909
910 parallelFor(texturePathVec, loadImageAndSetTexture);
911 }
912 }
