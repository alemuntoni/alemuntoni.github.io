174 bool)
175{
176 if constexpr (HasCustomComponents<OtherElement>) {
177 auto& thisCC = self.parentContainer().mCustomComponents;
178 const auto& otherCC = other.parentContainer().mCustomComponents;
179 for (const auto& name : thisCC.allComponentNames()) {

181 self.index(), other.index(), name, otherCC);
