36template<typename ScalarType>
37RegularGrid2<ScalarType> makeGrid(
38 const Box2<ScalarType>& bbPlane,
39 const Point2<ScalarType>& gridCellSideLengths)
40{
41 if (gridCellSideLengths.x() <= 0.0 || gridCellSideLengths.y() <= 0.0) {
42 Point<uint, 2> oneCell(1u, 1u);
43 return RegularGrid2<ScalarType>(bbPlane, oneCell);

45 return RegularGrid2<ScalarType>(bbPlane, gridCellSideLengths);

48inline double segmentVolume(
49 double cellArea,
50 double startD,
51 double endD,
52 double epsilon,
53 bool first = false)
54{
55 const double segLength = endD - startD;
56 const bool validLength = first || (segLength >= epsilon);
57
58 double segVolume = 0;
59 if (validLength) {
60 segVolume = cellArea * segLength;
61 }
62
63 return segVolume;
64}

66template<FaceMeshConcept MeshType, typename ScalarType>
67ScalarType processCell(
68 uint i,
69 uint j,
70 const RegularGrid2<ScalarType>& grid,
71 const OrthoFrame3<ScalarType>& frame,
72 const Scene& scene,
73 const MeshType& m,
74 auto epsilon,
75 VolumeResultMeshes& outMeshes)
76{
77 using PointType = Point3<ScalarType>;
78
79 const PointType& n = frame.axis(2);
80
81 RegularGrid2<double>::CellPos cellPos(i, j);
82
83 const auto c2d = grid.cellCenter(cellPos);
84 const PointType cellCenter = frame.toWorld(PointType(c2d(0), c2d(1), 0.0));
85
86 ScalarType volumeAcc = 0.0;
87
88 std::vector<Scene::HitResult> hits =
89 scene.facesIntersectedByRay(cellCenter, n, static_cast<float>(epsilon));
90
91 const ScalarType cellArea = grid.cellVolume();
92
93 ScalarType prevT = -epsilon;
94 PointType prevNormal = n;
95 bool first = true;
96
97 int hitsMesh = 0;
98
99 for (const auto& hit : hits) {
100 auto [hitFaceId, _, __, tHit] = hit;
101
102 if (hitFaceId == UINT_NULL) {
103 continue;
104 }
105
106 const auto& face = m.face(hitFaceId);
107
108 const PointType endPoint = cellCenter + n * tHit;
109 PointType endNormal = face.normal();
110 if (endNormal.norm() >= epsilon) {
111 endNormal.normalize();
112 }
113
114 if (!first && tHit <= prevT) {
115 prevT = tHit;
116 prevNormal = endNormal;
117 continue;
118 }
119
120 const ScalarType startDot = prevNormal.dot(n);
121 const ScalarType endDot = endNormal.dot(n);
122
123 if (endDot < 0.0) {
124 if (startDot > 0.0 && hitsMesh == 0) {
125 double volSeg =
126 segmentVolume(cellArea, prevT, tHit, epsilon, first);
127
128 if (outMeshes.computeMeshes && volSeg > 0) {
129 using VMesh = decltype(outMeshes.exteriorVolumeMesh);
130
131 const auto u0 = grid.min()(0) + i * grid.cellLength(0);
132 const auto u1 = u0 + grid.cellLength(0);
133 const auto v0 = grid.min()(1) + j * grid.cellLength(1);
134 const auto v1 = v0 + grid.cellLength(1);
135 VMesh hex = createHexahedron<VMesh>(
136 Point3d(u0, v0, prevT), Point3d(u1, v1, tHit));
137 for (auto& vert : hex.vertices()) {
138 const PointType p =
139 vert.position().template cast<ScalarType>();
140 vert.position() =
141 frame.toWorld(p).template cast<double>();
142 }
143 {
144 std::lock_guard<std::mutex> lock(outMeshes.appendMutex);
145 outMeshes.exteriorVolumeMesh.append(hex);
146 }
147 }
148
149 volumeAcc += volSeg;
150 }
151 hitsMesh += 1;
152 }
153 else {
154 hitsMesh -= 1;
155 }
156
157 prevT = tHit;
158 prevNormal = endNormal;
159 first = false;
160 }
161
162 return volumeAcc;
163}

211template<
212 FaceMeshConcept MeshType,
213 typename ScalarType,
214 Point2Concept Point2Type>
215auto heightfieldExteriorVolume(
216 const MeshType& m,
217 const Scene& scene,
218 const Point3<ScalarType>& direction,
219 const Point2Type& gridCellSideLengths,
220 auto epsilon,
221 VolumeResultMeshes& outMeshes)
222{
223 using namespace vcl;
224
225 using PointType = Point3<ScalarType>;
226 using OrthoFrameType = OrthoFrame<ScalarType, 3>;
227
228 PointType n = direction;
229 if (n.norm() <= epsilon) {
230 throw std::invalid_argument(
231 "Direction vector must have non-zero length greater than epsilon.");
232 }
233 n.normalize();
234
235 ScalarType minProj = std::numeric_limits<ScalarType>::infinity();
236 for (const auto& vv : m.vertices()) {
237 minProj = std::min(minProj, vv.position().dot(n));
238 }
239 const PointType planeOrigin = n * minProj;
240 const Plane<ScalarType> plane(planeOrigin, n);
241
242 const OrthoFrameType frame = OrthoFrameType::fromNormal(planeOrigin, n);
243
244 Box2<ScalarType> bbPlane;
245
246 for (const auto& vert : m.vertices()) {
247 const PointType projected = plane.projectPoint(vert.position());
248 const PointType rel = projected - frame.origin();
249
250 const ScalarType pu = rel.dot(frame.axis(0));
251 const ScalarType pv = rel.dot(frame.axis(1));
252 const Point2Type projUV(pu, pv);
253
254 bbPlane.add(projUV.template cast<ScalarType>());
255 }
256
257 if (bbPlane.dim(0) <= epsilon || bbPlane.dim(1) <= epsilon) {
258 return std::numeric_limits<ScalarType>::infinity();
259 }
260
261 const RegularGrid2<ScalarType> grid = detail::makeGrid(
262 bbPlane, gridCellSideLengths.template cast<ScalarType>());
263
264 if (outMeshes.computeMeshes) {
265 // out grid
266 for (uint ii = 0; ii <= grid.cellCount(0); ++ii) {
267 const ScalarType cu = grid.min()(0) + grid.cellLength(0) * ii;
268 const uint a = outMeshes.grid2dMesh.addVertex(
269 frame.toWorld(PointType(cu, grid.min()(1), 0.0))
270 .template cast<double>());
271 const uint b = outMeshes.grid2dMesh.addVertex(
272 frame.toWorld(PointType(cu, grid.max()(1), 0.0))
273 .template cast<double>());
274 outMeshes.grid2dMesh.addEdge(a, b);
275 outMeshes.grid2dMesh.vertex(a).normal() =
276 direction.template cast<double>();
277 outMeshes.grid2dMesh.vertex(b).normal() =
278 direction.template cast<double>();
279 }
280 for (uint jj = 0; jj <= grid.cellCount(1); ++jj) {
281 const ScalarType cv = grid.min()(1) + grid.cellLength(1) * jj;
282 const uint a = outMeshes.grid2dMesh.addVertex(
283 frame.toWorld(PointType(grid.min()(0), cv, 0.0))
284 .template cast<double>());
285 const uint b = outMeshes.grid2dMesh.addVertex(
286 frame.toWorld(PointType(grid.max()(0), cv, 0.0))
287 .template cast<double>());
288 outMeshes.grid2dMesh.addEdge(a, b);
289 outMeshes.grid2dMesh.vertex(a).normal() =
290 direction.template cast<double>();
291 outMeshes.grid2dMesh.vertex(b).normal() =
292 direction.template cast<double>();
293 }
294 }
295
296 ScalarType totalVolume = 0.0;
297
298 std::vector<uint> allCells(grid.totalCellCount());
299 std::iota(allCells.begin(), allCells.end(), 0);
300
301 std::vector<ScalarType> cellVolumes(grid.totalCellCount(), 0.0);
302
303 vcl::parallelFor(allCells, [&](uint idx) {
304 uint i = idx % grid.cellCount(0);
305 uint j = idx / grid.cellCount(0);
306
307 cellVolumes[idx] = detail::processCell(
308 i, j, grid, frame, scene, m, epsilon, outMeshes);
309 });
310
311 totalVolume += std::accumulate(cellVolumes.begin(), cellVolumes.end(), 0.0);
312
313 return totalVolume;
314}

323auto heightfieldExteriorVolume(
324 const MeshType& m,
325 const Scene& scene,

327 const Point2Type& gridCellSideLengths,

373template<
374 FaceMeshConcept MeshType,
375 typename P2Scalar,
376 LoggerConcept LogType = NullLogger>
377auto findBestOrientationByHeightfieldExteriorVolume(
378 const MeshType& m,
379 const Point2<P2Scalar>& gridCellSideLengths,
380 vcl::uint nDirections,
381 double epsilon = 1e-6,
382 LogType& log = nullLogger)
383{
384 using namespace vcl;
385
386 using PointType = MeshType::VertexType::PositionType;
387 using ScalarType = typename PointType::ScalarType;
388
389 if (nDirections == 0) {
390 throw std::invalid_argument(
391 "Number of directions must be greater than zero.");
392 }
393
394 log.startNewTask(
395 0, 100, "Finding best orientation by heightfield exterior volume...");
396
397 requirePerFaceNormal(m);
398
399 Box3<ScalarType> bbox = boundingBox(m);
400
401 epsilon *= bbox.diagonal();
402
403 // Ray tracing: shoot rays from grid cell centers through the mesh.
404 embree::Scene scene(m);
405
406 std::vector<PointType> fibNormals =
407 sphericalFibonacciPointSet<PointType>(nDirections);
408
409 uint bestPlaneId = 0;
410 ScalarType bestVolume = std::numeric_limits<ScalarType>::infinity();
411 PointType bestNormal = fibNormals.front();
412
413 for (uint i = 0; i < fibNormals.size(); ++i) {
414 ScalarType vol = heightfieldExteriorVolume(
415 m, scene, fibNormals[i], gridCellSideLengths, epsilon);
416
417 if (vol < bestVolume) {
418 bestVolume = vol;
419 bestPlaneId = i;
420 bestNormal = fibNormals[i];
421
422 log.setPercentage(i / static_cast<double>(fibNormals.size()) * 100);
423 log.info() << "New best: Plane id: " << i << "/"
424 << fibNormals.size() - 1 << ", normal: " << fibNormals[i]
425 << ", volume: " << std::to_string(vol);
426 }
427 }
428
429 log.setPercentage(100);
430 log.info() << "Fibonacci planes tested: " << fibNormals.size()
431 << "\nBest plane id: " << bestPlaneId
432 << "\nBest plane normal: " << bestNormal
433 << "\nMinimum volume: " << bestVolume;
434 log.endTask("Best orientation found.");
435
436 return bestNormal;
437}
