352 template<typename R>
353 requires std::ranges::input_range<R>
354 uint pushDrawableObjects(R&& objs)
355 {
356 uint startIdx = mDrawList->size();
357 for (auto&& obj : objs) {
358 using ElementType = std::remove_cvref_t<decltype(obj)>;
359 constexpr bool canMove =
360 std::is_rvalue_reference_v<decltype(obj)> ||
361 std::is_rvalue_reference_v<R&&>;
362
363 if constexpr (std::is_convertible_v<
364 ElementType,
365 std::shared_ptr<DrawableObject>>) {
366 if constexpr (canMove) {
367 mDrawList->pushBack(std::move(obj));
368 }
369 else {
370 mDrawList->pushBack(obj);
371 }
372 }
373 else {
374 static_assert(
375 std::derived_from<ElementType, DrawableObject> &&
376 (!std::is_abstract_v<ElementType>),
377 "Objects must derive from DrawableObject and not be "
378 "abstract.");
379 if constexpr (canMove) {
380 mDrawList->pushBack(std::move(obj));
381 }
382 else {
383 mDrawList->pushBack(obj);
384 }
385 }
386 mDrawList->back()->init();
387 }
388 if (mDrawList->size() > startIdx) {
389 refreshEditors();
390 }
391 return startIdx;
392 }
393
394 bool removeDrawableObject(uint id)
395 {
396 if (id >= mDrawList->size())
397 return false;
398 mDrawList->erase(id);
399 refreshEditors();
400 requestUpdate();
401 return true;
402 }

560 bool onKeyPress(Key::Enum key, const KeyModifiers& modifiers) override
561 {
562 bool block = false;
563
564 // global actions (e.g. Escape to toggle editor events) always fire,
565 // even when editor events are disabled, otherwise they could never
566 // be used to re-enable them
567 auto actionNameOpt =
568 mViewerSettings.globalActionMap.action({key, modifiers});
569 if (actionNameOpt.has_value()) {
570 auto it = mGlobalActionRegistry.find(actionNameOpt.value());
571 if (it != mGlobalActionRegistry.end()) {
572 it->second();
573 return true;
574 }
575 }
576
577 if (mEditorsEventsEnabled) {
578 for (const auto& editor : mEditors) {
579 if (!block && editor->isActive())
580 block = editor->onKeyPress(key, modifiers);
581 }
582 }
583
584 if (!block)
585 block = Base::onKeyPress(key, modifiers);
586
587 return block;
588 }

736 void readDepthRequest(double x, double y, bool homogeneousNDC = true)
737 {
738 using ReadData = ReadBufferTypes::ReadData;
739 using FloatData = ReadBufferTypes::FloatData;
740 using MatrixType = Base::MatrixType;
741
742 if (mReadRequested)
743 return;
744
745 // get point
746 const Point2d p(x, y);
747
748 // create the callback
749 const auto proj = Base::projectionMatrix();
750 const auto view = Base::viewMatrix();
751 // viewport
752 auto size = DRA::DRW::canvasSize(derived());
753
754 const Point4f vp = {.0f, .0f, float(size.x()), float(size.y())};
755
756 auto callback = [=, this](const ReadData& dt) {
757 const auto& data = std::get<FloatData>(dt);
758 assert(data.size() == 1);
759 const float depth = data[0];
760
761 mReadRequested = false;
762
763 // if the depth is 1.0, the point is not in the scene
764 if (depth == 1.0f) {
765 return;
766 }
767
768 // unproject the point
769 const Point3f p2d(p.x(), vp[3] - p.y(), depth);
770 auto unproj = unprojectScreenPosition(
771 p2d, MatrixType(proj * view), vp, homogeneousNDC);
772
773 this->focus(unproj);
774 derived()->update();
775 };
776
777 mReadRequested =
778 DRA::DRW::readDepth(derived(), Point2i(p.x(), p.y()), callback);
779 if (mReadRequested)
780 derived()->update();
781 }
782
783 void readIdRequest(double x, double y, std::function<void(uint)> idCallback)
784 {
785 using ReadData = ReadBufferTypes::ReadData;
786
787 if (mReadRequested)
788 return;
789
790 // get point
791 const Point2d p(x, y);
792
793 // create the callback
794 auto callback = [=, this](const ReadData& dt) {
795 const auto& data = std::get<ReadBufferTypes::ByteData>(dt);
796 assert(data.size() == 4);
797 // TODO: check how to do this properly
798 const uint id = *(uint32_t*) &data[0];
799
800 mReadRequested = false;
801
802 idCallback(id);
803 derived()->update();
804 };
805
806 mReadRequested =
807 DRA::DRW::readId(derived(), Point2i(p.x(), p.y()), callback);
808 if (mReadRequested)
809 derived()->update();
810 }
