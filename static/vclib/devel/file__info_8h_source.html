77 static bool isFileBinary(const std::string& filename)
78 {
79 const std::size_t CHECK_BUFF_SIZE = 1000;
80
81 std::ifstream file(filename, std::ios::binary | std::ios::ate);
82 std::size_t fsize = file.tellg();
83 std::size_t size = std::min(fsize, CHECK_BUFF_SIZE);
84 file.seekg(0, std::ios::beg);
85
86 std::vector<unsigned char> buffer(size);
87 if (file.read((char*) buffer.data(), size)) {
88 for (uint i = 0; i < buffer.size(); ++i) {
89 if (buffer[i] > 127)
90 return true;
91 }
92 }
93 else {
94 throw MalformedFileException("Cannot read data from file.");
95 }
96 return false;
97 }

122 static void separateExtensionFromFileName(
123 const std::string& fullname,
124 std::string& rawName,
125 std::string& extension)
126 {
127 // https://stackoverflow.com/questions/6417817
128
129 size_t lastIndex = fullname.find_last_of(".");
130 if (lastIndex != std::string::npos) {
131 rawName = fullname.substr(0, lastIndex);
132 extension = fullname.substr(lastIndex, fullname.size());
133 }
134 else {
135 rawName = fullname;
136 }
137 }

157 static void separateFileNameFromPath(
158 const std::string& fullpath,
159 std::string& path,
160 std::string& filename)
161 {
162 size_t lastIndex = fullpath.find_last_of("/");
163 if (lastIndex != std::string::npos) {
164 path = fullpath.substr(0, lastIndex + 1);
165 filename = fullpath.substr(lastIndex + 1, fullpath.size());
166 }
167 else {
168 filename = fullpath;
169 }
170 }

277 static std::string addExtensionIfNeeded(
278 const std::string& filename,
279 const std::string& ext)
280 {
281 std::string actualFileName;
282 size_t lastIndex = filename.find_last_of(".");
283 if (lastIndex != std::string::npos) {
284 std::string e = filename.substr(lastIndex + 1, filename.size());
285 if (e == ext)
286 actualFileName = filename;
287 else
288 actualFileName = filename + "." + ext;
289 }
290 else
291 actualFileName = filename + "." + ext;
292 return actualFileName;
293 }
