33template<typename DerivedRenderApp>
34class ViewerDrawerOpenGL2 : public AbstractViewerDrawer<DerivedRenderApp>
35{
36 using ParentViewer = AbstractViewerDrawer<DerivedRenderApp>;
37
38public:
39 ViewerDrawerOpenGL2(uint width = 1024, uint height = 768) :
40 ParentViewer(width, height)
41 {
42 }
43
44 void onInit(uint viewId) override
45 {
46 ParentViewer::onInit(viewId);
47 glEnable(GL_LIGHTING);
48 glEnable(GL_LIGHT0);
49 glEnable(GL_DEPTH_TEST);
50 glEnable(GL_COLOR_MATERIAL);
51 glEnable(GL_NORMALIZE);
52
53 float lightAmbient[] = {.2f, .2f, .2f, 1.0f};
54 float lightDiffuse[] = {.5f, .5f, .5f, 1.0f};
55
56 glLightfv(GL_LIGHT0, GL_AMBIENT, lightAmbient);
57 glLightfv(GL_LIGHT0, GL_DIFFUSE, lightDiffuse);
58
59 if (ParentViewer::mDrawList) {
60 for (auto& obj : *(ParentViewer::mDrawList)) {
61 obj->init();
62 }
63 }
64 }
65
66 void onDrawContent(uint) override
67 {
68 auto tmp = ParentViewer::light().direction();
69 vcl::Point4f lPos(tmp.x(), tmp.y(), tmp.z(), 0.0f);
70
71 glMatrixMode(GL_PROJECTION);
72 glLoadIdentity();
73 glMultMatrixf(ParentViewer::projectionMatrix().data());
74 glMatrixMode(GL_MODELVIEW);
75 glLoadIdentity();
76 glLightfv(GL_LIGHT0, GL_POSITION, lPos.data());
77 glMultMatrixf(ParentViewer::viewMatrix().data());
78
79 for (auto& obj : *(ParentViewer::mDrawList))
80 obj->draw();
81 }
82
83 // events
84 bool onMouseDoubleClick(
85 MouseButton::Enum button,
86 double x,
87 double y,
88 const KeyModifiers& modifiers) override
89 {
90 bool block = ParentViewer::onMouseDoubleClick(button, x, y, modifiers);
91 if (!block && button == MouseButton::LEFT) {
92 ParentViewer::readDepthRequest(x, y);
93 }
94 return block;
95 }
96};
