153 Point3<Scalar> operator*(const Point3<Scalar>& p) const
154 {
155 Eigen::Matrix<Scalar, 3, 1> fc = Base::vec().cross(p);
156
157 Eigen::Matrix<Scalar, 3, 1> fd = p * Base::w();
158
159 Eigen::Matrix<Scalar, 3, 1> s = fc + fd;
160
161 Eigen::Matrix<Scalar, 3, 1> sc = Base::vec().cross(s);
162
163 return Point3<Scalar>(p + sc * 2.0);
164 }
