44template<typename Scalar, uint N>
45class OrthoFrame
46{
47 static_assert(N > 0, "Number of dimensions of OrthoFrame must be > 0.");
48
49 Point<Scalar, N> mOrigin;
50 OrthoBasis<Scalar, N> mBasis;
51
52public:
53 using ScalarType = Scalar;
54 using PointType = Point<Scalar, N>;
55 using BasisType = OrthoBasis<Scalar, N>;
56
57 static const uint DIM = N;
58
65 OrthoFrame() = default;
66
77 template<typename... Axes>
78 OrthoFrame(const PointType& origin, Axes&&... axes)
79 requires (sizeof...(Axes) == N)
80 : mOrigin(origin), mBasis(std::forward<Axes>(axes)...)
81 {
82 }

91 OrthoFrame(const PointType& origin, const std::array<PointType, N>& axes) :
92 mOrigin(origin), mBasis(axes)
93 {
94 }

103 OrthoFrame(const PointType& origin, const BasisType& basis) :
104 mOrigin(origin), mBasis(basis)
105 {
106 }
107
122 static OrthoFrame fromNormal(
123 const PointType& origin,
124 const PointType& normal) requires (N == 3)
125 {
126 return OrthoFrame(origin, BasisType::fromNormal(normal));
127 }
128
133 decltype(auto) origin(this auto&& self) { return (self.mOrigin); }
134
139 decltype(auto) basis(this auto&& self) { return (self.mBasis); }
140
149 decltype(auto) axis(this auto&& self, uint i)
150 {
151 return self.mBasis.axis(i);
152 }
153
166 PointType toWorld(const PointType& local) const
167 {
168 PointType p = mOrigin;
169 p += mBasis.toWorld(local);
170 return p;
171 }
172
184 PointType toLocal(const PointType& world) const
185 {
186 return mBasis.toLocal(world - mOrigin);
187 }
188
199 Eigen::Transform<Scalar, N, Eigen::Affine> transformMatrix() const
200 {
201 Eigen::Transform<Scalar, N, Eigen::Affine> t;
202 t.linear() = mBasis.rotationMatrix();
203 t.translation() = mOrigin;
204 return t;
205 }
206
212 template<typename S>
213 auto cast() const
214 {
215 if constexpr (std::is_same_v<Scalar, S>) {
216 return *this;
217 }
218 else {
219 return OrthoFrame<S, N>(
220 mOrigin.template cast<S>(), mBasis.template cast<S>());
221 }
222 }
223
228 void serialize(std::ostream& os) const
229 {
230 mOrigin.serialize(os);
231 mBasis.serialize(os);
232 }
233
238 void deserialize(std::istream& is)
239 {
240 mOrigin.deserialize(is);
241 mBasis.deserialize(is);
242 }
243
244 bool operator==(const OrthoFrame& f) const
245 {
246 return mOrigin == f.mOrigin && mBasis == f.mBasis;
247 }
248
249 bool operator!=(const OrthoFrame& f) const { return !(*this == f); }
250};
