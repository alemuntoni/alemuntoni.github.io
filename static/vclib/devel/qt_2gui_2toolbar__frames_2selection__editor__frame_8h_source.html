19template<typename ViewerType>
20class SelectionEditorFrame : public GenericEditorFrame
21{
22 using Base = GenericEditorFrame;
23
24 std::shared_ptr<vcl::SelectionEditor<ViewerType>> mSelectionEditor;
25
26public:
27 explicit SelectionEditorFrame(
28 std::shared_ptr<vcl::SelectionEditor<ViewerType>> ptr,
29 QWidget* parent = nullptr) : GenericEditorFrame(parent)
30 {
31 mSelectionEditor = ptr;
32
33 auto& settings = mSelectionEditor->settings();
34
35 QIcon icv(":/icons/select_vertex.png");
36 QPushButton* selectVerticesButton = Base::addButton(icv);
37 selectVerticesButton->setToolTip("Select Vertices");
38
39 QIcon icf(":/icons/select_face.png");
40 QPushButton* selectFacesButton = Base::addButton(icf);
41 selectFacesButton->setToolTip("Select Faces");
42
43 auto onSelectVerticesButtonClicked = [&](bool checked) {
44 settings.selectVertices = checked;
45 bool selFaces = settings.selectFaces;
46 mSelectionEditor->setActive(checked || selFaces);
47 };
48
49 auto onSelectFacesButtonClicked = [&](bool checked) {
50 settings.selectFaces = checked;
51 bool selVertices = settings.selectVertices;
52 mSelectionEditor->setActive(checked || selVertices);
53 };
54
55 connect(
56 selectVerticesButton,
57 &QPushButton::clicked,
58 this,
59 onSelectVerticesButtonClicked);
60
61 connect(
62 selectFacesButton,
63 &QPushButton::clicked,
64 this,
65 onSelectFacesButtonClicked);
66
67 SelectionEditorSettingsFrame* sf =
68 Base::setSettingsFrame<SelectionEditorSettingsFrame>(settings);
69
70 connect(sf, SIGNAL(settingsUpdated()), this, SLOT(refreshSettings()));
71
72 mSelectionEditor->setOnStateUpdatedCallback(
73 [this, selectVerticesButton, selectFacesButton]() {
74 selectVerticesButton->setChecked(
75 mSelectionEditor->settings().selectVertices);
76 selectFacesButton->setChecked(
77 mSelectionEditor->settings().selectFaces);
78 });
79 }
80
81private slots:
82
83 // probably not needed here (no drawing to update when settings change)
84 void refreshSettings() override
85 {
86 if (mSelectionEditor) {
87 mSelectionEditor->refreshSettings();
88 }
89 }
90};
