397template<int N, std::unsigned_integral Index, bool TTVC, bool OPT>
398template<typename OtherElement>
399void AdjacentEdges<N, Index, TTVC, OPT>::importFrom(
400 this auto&& self,
401 const OtherElement& other,
402 bool importRefs)
403{
404 if (importRefs) {
405 if constexpr (HasAdjacentEdges<OtherElement>) {
406 if (other.template isComponentAvailable<CompId::ADJACENT_EDGES>()) {
407 using OtherComp =
408 ComponentTypeT<CompId::ADJACENT_EDGES, OtherElement>;
409 constexpr int OTHER_N = OtherComp::ADJ_EDGE_COUNT;
410
411 if constexpr (N > 0) {
412 if constexpr (N == OTHER_N) {
413 for (uint i = 0; i < N; ++i) {
414 self.setAdjEdge(
415 i, Base::castIndex(other.adjEdgeIndex(i)));
416 }
417 }
418 else if constexpr (OTHER_N < 0) {
419 if (other.adjEdgeCount() == N) {
420 for (uint i = 0; i < N; ++i) {
421 self.setAdjEdge(
422 i, Base::castIndex(other.adjEdgeIndex(i)));
423 }
424 }
425 }
426 }
427 else if constexpr (!TTVC) {
428 self.resizeAdjEdges(other.adjEdgeCount());
429 for (uint i = 0; i < other.adjEdgeCount(); ++i) {
430 self.setAdjEdge(
431 i, Base::castIndex(other.adjEdgeIndex(i)));
432 }
433 }
434 else {
435 if (self.adjEdgeCount() == other.adjEdgeCount()) {
436 for (uint i = 0; i < self.adjEdgeCount(); ++i) {
437 self.setAdjEdge(
438 i, Base::castIndex(other.adjEdgeIndex(i)));
439 }
440 }
441 }
442 }
443 }
444 }
445}
