603 for (uint i = 0; i < adjEdgeCount(); ++i) {

610 if constexpr (N < 0) {

613 Base::resize(n);
614 }
615 for (uint i = 0; i < adjEdgeCount(); ++i) {
616 uint aei;
617 vcl::deserialize(is, aei);
618 setAdjEdge(i, aei);
619 }

702/* importFrom function */
703
704template<
705 bool STORE_INDICES,
706 typename Edge,
707 int N,
708 bool TTVC,
709 typename ParentElemType,
710 bool VERT,
711 bool OPT>
712template<typename Element>
713void AdjacentEdges<STORE_INDICES, Edge, N, TTVC, ParentElemType, VERT, OPT>::
714 importFrom(const Element& e, bool importRefs)
715{
716 if (importRefs) {
717 if constexpr (HasAdjacentEdges<Element>) {
718 if (isAdjacentEdgesAvailableOn(e)) {
719 if constexpr (N > 0) {
720 // same static size
721 if constexpr (N == Element::ADJ_EDGE_COUNT) {
722 importIndicesFrom(e);
723 }
724 // from dynamic to static, but dynamic size == static size
725 else if constexpr (Element::ADJ_EDGE_COUNT < 0) {
726 if (e.adjEdgeCount() == N) {
727 importIndicesFrom(e);
728 }
729 }
730 else {
731 // do not import in this case: cannot import from
732 // dynamic size != static size
733 }
734 }
735 else {
736 // from static/dynamic to dynamic size: need to resize
737 // first, then import
738 Base::resize(e.adjEdgeCount());
739 importIndicesFrom(e);
740 }
741 }
742 }
743 }
