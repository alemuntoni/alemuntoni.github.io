33class ScreenSpaceBox
34{
35 // ---- BGFX rendering resources ----
36 DynamicVertexBuffer mPosBuf;
37 IndexBuffer mTriIndexBuf;
38 bool mInitialized = false;
39
40 // ---- Box state ----
41 Color mColor = {27, 120, 249, 64};
42 bool mVisible = true;
43 Box2d mBox;
44
45public:
46 ScreenSpaceBox() = default;
47
54 void init()
55 {
56 if (mInitialized)
57 return;
58
59 // Create position dynamic vertex buffer
60 bgfx::VertexLayout vertexLayout;
61 vertexLayout.begin()
62 .add(bgfx::Attrib::Position, 2, bgfx::AttribType::Float)
63 .end();
64 mPosBuf.create(4, vertexLayout);
65
66 // Create index buffer for the two box triangles
67 std::array<uint, 6> indices {1, 2, 0, 2, 3, 0};
68 mTriIndexBuf.create(bgfx::copy(indices.data(), 6 * sizeof(uint)));
69
70 mInitialized = true;
71 }
72
76 bool isInitialized() const { return mInitialized; }
77
81 void setBox(const Box2d& box) { mBox = box; }
82
86 const Box2d& box() const { return mBox; }
87
91 void setColor(const Color& color) { mColor = color; }
92
96 const Color& color() const { return mColor; }
97
101 void setVisible(bool visible) { mVisible = visible; }
102
106 bool isVisible() const { return mVisible; }
107
113 void draw(uint viewId)
114 {
115 if (!mInitialized || !mVisible || mBox.isNull())
116 return;
117
118 std::array<Point2f, 4> temp;
119 temp[0] = vcl::boxVertex(mBox, 0).cast<float>();
120 temp[1] = vcl::boxVertex(mBox, 1).cast<float>();
121 temp[2] = vcl::boxVertex(mBox, 2).cast<float>();
122 temp[3] = vcl::boxVertex(mBox, 3).cast<float>();
123
124 bgfx::setState(
125 0 | BGFX_STATE_WRITE_RGB | BGFX_STATE_DEPTH_TEST_ALWAYS |
126 BGFX_STATE_BLEND_ALPHA);
127 mTriIndexBuf.bind();
128
129 mPosBuf.update(0, bgfx::copy(&temp[0], 8 * sizeof(float)));
130 mPosBuf.bind(0);
131
132 ScreenSpaceBoxUniforms::setColor(mColor);
133 ScreenSpaceBoxUniforms::bind();
134
135 bgfx::submit(
136 viewId,
137 Context::instance()
138 .programManager()
139 .getProgram<VertFragProgram::SCREENSPACE_BOX>());
140 }
141};
