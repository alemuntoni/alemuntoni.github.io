138 template<typename D, typename... Others>
139 void callEventFunForDrawers(auto lambda)
140 {
141 bool block = false;
142 if constexpr (EventDrawerConcept<D>) {
143 if constexpr (BlockerEventDrawerConcept<D>) {
144 block = lambda.template operator()<D>(this);
145 }
146 else {
147 lambda.template operator()<D>(this);
148 }
149 }
150 if constexpr (sizeof...(Others) > 0) {
151 if (!block) {
152 callEventFunForDrawers<Others...>(lambda);
153 }
154 }
155 }
156
157 // Base case when there are no drawers
158 void callEventFunForDrawers(auto lambda) {}
