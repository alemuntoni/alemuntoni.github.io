13namespace vcl {

15template<typename ViewerDrawer>
16class MeshSelectorEditor : public Editor<ViewerDrawer>
17{
18public:
19 enum class MeshSelectorAction { SELECT_MESH };
20 using MouseMap = BindingMap<
21 std::pair<MouseButton::Enum, KeyModifiers>,
22 MeshSelectorAction>;
23
24private:
25 using Base = Editor<ViewerDrawer>;
26
27 // a callback function called when an object is selected
28 std::function<void(uint)> mOnObjectSelectedFunction = nullptr;
29
30 EditorSettings mSettings;
31
32 MouseMap mMouseBindings = {
33 {{MouseButton::RIGHT, {KeyModifier::NO_MODIFIER}},
34 MeshSelectorAction::SELECT_MESH}
35 };
36
37public:
38 void setOnObjectSelectedFunction(const std::function<void(uint)>& f)
39 {
40 mOnObjectSelectedFunction = f;
41 }
42
43 MouseMap& mouseBindings() { return mMouseBindings; }
44
45 const MouseMap& mouseBindings() const { return mMouseBindings; }
46
47 // Editor implementation
48
49 void onViewerSet() override
50 {
51 Base::viewerRegisterGlobalAction(
52 "Toggle Mesh Selector",
53 {Key::M, {KeyModifier::NO_MODIFIER}},
54 [this]() {
55 Base::setActive(!Base::isActive());
56 Base::viewerUpdate();
57 });
58 }
59
60 std::string name() const override { return "Mesh Selector"; }
61
62 EditorSettings& settings() override { return mSettings; }
63
64 const EditorSettings& settings() const override { return mSettings; }
65
66 void draw(uint viewId) override {}
67
68 bool onMousePress(
69 vcl::MouseButton::Enum button,
70 double x,
71 double y,
72 const vcl::KeyModifiers& modifiers) override
73 {
74 bool block = Base::onMousePress(button, x, y, modifiers);
75 if (block)
76 return true;
77
78 auto action = mMouseBindings.action({button, modifiers});
79 if (action.has_value() &&
80 action.value() == MeshSelectorAction::SELECT_MESH) {
81 auto callback = [&](uint id) {
82 if (id == vcl::UINT_NULL)
83 return;
84
85 if (mOnObjectSelectedFunction)
86 mOnObjectSelectedFunction(id);
87 else
88 Base::drawList()->setSelectedObjectId(id);
89 };
90
91 Base::viewerReadIdRequest(x, y, callback);
92 return true; // Smart blocking: consumed event
93 }
94
95 return false;
96 }
97};
98
99} // namespace vcl
100
101#endif // VCL_RENDER_EDITORS_MESH_SELECTOR_EDITOR_H
