39template<typename T, typename Index = uint, typename... Args>
40void compactVector(
41 std::vector<T, Args...>& vec,
42 const std::vector<Index>& newIndices)
43{
44 assert(vec.size() == newIndices.size());
45 std::size_t newSize = 0;
46 for (std::size_t i = 0; i < newIndices.size(); ++i) {
47 if (newIndices[i] != NULL_INDEX<Index>) {
48 ++newSize;
49 if (newIndices[i] != i) {
50 // must move the element from position i to position
51 // newIndices[i]
52 vec[newIndices[i]] = std::move(vec[i]);
53 }
54 }
55 }
56 vec.resize(newSize);
57}

95template<typename T, typename Index = uint, typename... Args>
96requires std::unsigned_integral<Index>
97void permuteInPlace(
98 std::vector<T, Args...>& vec,
99 const std::vector<Index>& newIndices)
100{
101 using std::swap;
102
103 assert(vec.size() == newIndices.size());
104
105 std::vector<bool> visited(vec.size(), false);
106
107 for (std::size_t i = 0; i < vec.size(); ++i) {
108 if (visited[i] || newIndices[i] == i) {
109 continue;
110 }
111
112 std::size_t current = i;
113 T temp = std::move(vec[i]);
114
115 while (!visited[current]) {
116 visited[current] = true;
117 std::size_t next = newIndices[current];
118 swap(temp, vec[next]);
119 current = next;
120 }
121 }
122}
