117template<CompId COMP_ID>
118constexpr const char* componentEnumString()
119{
120 static_assert(
121 ComponentString<COMP_ID>().str != nullptr,
122 "Invalid ComponentIDEnum. You should specialize the 'ComponentString' "
123 "struct with your COMP_ID value.");
