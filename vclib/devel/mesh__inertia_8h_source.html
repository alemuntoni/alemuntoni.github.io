45template<FaceMeshConcept MeshType>
46class MeshInertia
47{
48 using ScalarType = vcl::ScalarType<MeshType>;
49
50 enum { X = 0, Y = 1, Z = 2 };
51
52 int mA; // alpha
53 int mB; // beta
54 int mC; // gamma
55
56 /* projection integrals */
57 double mP1, mPa, mPb, mPaa, mPab, mPbb, mPaaa, mPaab, mPabb, mPbbb;
58
59 /* face integrals */
60 double mFa, mFb, mFc, mFaa, mFbb, mFcc;
61 double mFaaa, mFbbb, mFccc, mFaab, mFbbc, mFcca;
62
63 /* volume integrals */
64 double mT0 = 0, mT1[3] = {0, 0, 0}, mT2[3] = {0, 0, 0}, mTP[3] = {0, 0, 0};
65
66public:
67 MeshInertia(const MeshType& m)
68 {
69 double nx, ny, nz;
70
71 for (const auto&& f : m.faces()) {
72 if (faceArea(f) > 0) {
73 Point3d fn = faceNormal(f).template cast<double>();
74 fn.normalize();
75
76 nx = std::abs(fn[0]);
77 ny = std::abs(fn[1]);
78 nz = std::abs(fn[2]);
79 if (nx > ny && nx > nz)
80 mC = X;
81 else
82 mC = (ny > nz) ? Y : Z;
83 mA = (mC + 1) % 3;
84 mB = (mA + 1) % 3;
85
86 faceIntegrals(f, fn);
87
88 mT0 += fn[X] * ((mA == X) ? mFa : ((mB == X) ? mFb : mFc));
89
90 mT1[mA] += fn[mA] * mFaa;
91 mT1[mB] += fn[mB] * mFbb;
92 mT1[mC] += fn[mC] * mFcc;
93 mT2[mA] += fn[mA] * mFaaa;
94 mT2[mB] += fn[mB] * mFbbb;
95 mT2[mC] += fn[mC] * mFccc;
96 mTP[mA] += fn[mA] * mFaab;
97 mTP[mB] += fn[mB] * mFbbc;
98 mTP[mC] += fn[mC] * mFcca;
99 }
100 }
101
102 mT1[X] /= 2;
103 mT1[Y] /= 2;
104 mT1[Z] /= 2;
105 mT2[X] /= 3;
106 mT2[Y] /= 3;
107 mT2[Z] /= 3;
108 mTP[X] /= 2;
109 mTP[Y] /= 2;
110 mTP[Z] /= 2;
111 }
112
118 ScalarType volume() const { return mT0; }
119
124 Point3<ScalarType> centerOfMass() const
125 {
126 Point3<ScalarType> r;
127 r[X] = mT1[X] / mT0;
128 r[Y] = mT1[Y] / mT0;
129 r[Z] = mT1[Z] / mT0;
130 return r;
131 }
132
137 template<typename Matrix33>
138 Matrix33 inertiaTensor() const
139 {
140 Matrix33 J;
141 Point3d r;
142 r[X] = mT1[X] / mT0;
143 r[Y] = mT1[Y] / mT0;
144 r[Z] = mT1[Z] / mT0;
145 /* compute inertia tensor */
146 J(X, X) = (mT2[Y] + mT2[Z]);
147 J(Y, Y) = (mT2[Z] + mT2[X]);
148 J(Z, Z) = (mT2[X] + mT2[Y]);
149 J(X, Y) = J(Y, X) = -mTP[X];
150 J(Y, Z) = J(Z, Y) = -mTP[Y];
151 J(Z, X) = J(X, Z) = -mTP[Z];
152
153 J(X, X) -= mT0 * (r[Y] * r[Y] + r[Z] * r[Z]);
154 J(Y, Y) -= mT0 * (r[Z] * r[Z] + r[X] * r[X]);
155 J(Z, Z) -= mT0 * (r[X] * r[X] + r[Y] * r[Y]);
156 J(X, Y) = J(Y, X) += mT0 * r[X] * r[Y];
157 J(Y, Z) = J(Z, Y) += mT0 * r[Y] * r[Z];
158 J(Z, X) = J(X, Z) += mT0 * r[Z] * r[X];
159 return J;
160 }
161
172 template<typename Matrix33>
173 void inertiaTensorEigen(
174 Matrix33& eigenVectors,
175 Point3<ScalarType>& eigenValues) const
176 {
177 Eigen::Matrix3d it = inertiaTensor<Eigen::Matrix3d>();
178 Eigen::SelfAdjointEigenSolver<Eigen::Matrix3d> eig(it);
179 Eigen::Vector3d c_val = eig.eigenvalues();
180 // eigenvector are stored as columns.
181 Eigen::Matrix3d c_vec = eig.eigenvectors();
182 c_vec.transposeInPlace();
183 for (uint i = 0; i < 3; i++)
184 for (uint j = 0; j < 3; j++)
185 eigenVectors(i, j) = c_vec(i, j);
186 eigenValues = Point3<ScalarType>(c_val.template cast<ScalarType>());
187 }

189private:
190 static double sqr(double x) { return x * x; }
191
192 static double cube(double x) { return x * x * x; }
193
194 void faceIntegrals(FaceConcept auto&& f, const Point3d& n)
195 {
196 double w;
197 double k1, k2, k3, k4;
198
199 projectionIntegrals(f);
200
201 w = -f.vertex(0).position().template cast<double>().dot(n);
202 k1 = 1 / n[mC];
203 k2 = k1 * k1;
204 k3 = k2 * k1;
205 k4 = k3 * k1;
206
207 mFa = k1 * mPa;
208 mFb = k1 * mPb;
209 mFc = -k2 * (n[mA] * mPa + n[mB] * mPb + w * mP1);
210
211 mFaa = k1 * mPaa;
212 mFbb = k1 * mPbb;
213 mFcc = k3 * (sqr(n[mA]) * mPaa + 2 * n[mA] * n[mB] * mPab +
214 sqr(n[mB]) * mPbb +
215 w * (2 * (n[mA] * mPa + n[mB] * mPb) + w * mP1));
216
217 mFaaa = k1 * mPaaa;
218 mFbbb = k1 * mPbbb;
219 mFccc = -k4 * (cube(n[mA]) * mPaaa + 3 * sqr(n[mA]) * n[mB] * mPaab +
220 3 * n[mA] * sqr(n[mB]) * mPabb + cube(n[mB]) * mPbbb +
221 3 * w *
222 (sqr(n[mA]) * mPaa + 2 * n[mA] * n[mB] * mPab +
223 sqr(n[mB]) * mPbb) +
224 w * w * (3 * (n[mA] * mPa + n[mB] * mPb) + w * mP1));
225
226 mFaab = k1 * mPaab;
227 mFbbc = -k2 * (n[mA] * mPabb + n[mB] * mPbbb + w * mPbb);
228 mFcca = k3 * (sqr(n[mA]) * mPaaa + 2 * n[mA] * n[mB] * mPaab +
229 sqr(n[mB]) * mPabb +
230 w * (2 * (n[mA] * mPaa + n[mB] * mPab) + w * mPa));
231 }

237 void projectionIntegrals(FaceConcept auto&& f)
238 {
239 double a0, a1, da;
240 double b0, b1, db;
241 double a0_2, a0_3, a0_4, b0_2, b0_3, b0_4;
242 double a1_2, a1_3, b1_2, b1_3;
243 double C1, Ca, Caa, Caaa, Cb, Cbb, Cbbb;
244 double Cab, Kab, Caab, Kaab, Cabb, Kabb;
245
246 mP1 = mPa = mPb = mPaa = mPab = mPbb = mPaaa = mPaab = mPabb = mPbbb =
247 0.0;
248
249 for (uint i = 0; i < f.vertexCount(); i++) {
250 a0 = f.vertex(i).position()[mA];
251 b0 = f.vertex(i).position()[mB];
252 a1 = f.vertexMod(i + 1).position()[mA];
253 b1 = f.vertexMod(i + 1).position()[mB];
254 da = a1 - a0;
255 db = b1 - b0;
256 a0_2 = a0 * a0;
257 a0_3 = a0_2 * a0;
258 a0_4 = a0_3 * a0;
259 b0_2 = b0 * b0;
260 b0_3 = b0_2 * b0;
261 b0_4 = b0_3 * b0;
262 a1_2 = a1 * a1;
263 a1_3 = a1_2 * a1;
264 b1_2 = b1 * b1;
265 b1_3 = b1_2 * b1;
266
267 C1 = a1 + a0;
268 Ca = a1 * C1 + a0_2;
269 Caa = a1 * Ca + a0_3;
270 Caaa = a1 * Caa + a0_4;
271 Cb = b1 * (b1 + b0) + b0_2;
272 Cbb = b1 * Cb + b0_3;
273 Cbbb = b1 * Cbb + b0_4;
274 Cab = 3 * a1_2 + 2 * a1 * a0 + a0_2;
275 Kab = a1_2 + 2 * a1 * a0 + 3 * a0_2;
276 Caab = a0 * Cab + 4 * a1_3;
277 Kaab = a1 * Kab + 4 * a0_3;
278 Cabb = 4 * b1_3 + 3 * b1_2 * b0 + 2 * b1 * b0_2 + b0_3;
279 Kabb = b1_3 + 2 * b1_2 * b0 + 3 * b1 * b0_2 + 4 * b0_3;
280
281 mP1 += db * C1;
282 mPa += db * Ca;
283 mPaa += db * Caa;
284 mPaaa += db * Caaa;
285 mPb += da * Cb;
286 mPbb += da * Cbb;
287 mPbbb += da * Cbbb;
288 mPab += db * (b1 * Cab + b0 * Kab);
289 mPaab += db * (b1 * Caab + b0 * Kaab);
290 mPabb += da * (a1 * Cabb + a0 * Kabb);
291 }
292
293 mP1 /= 2.0;
294 mPa /= 6.0;
295 mPaa /= 12.0;
296 mPaaa /= 20.0;
297 mPb /= -6.0;
298 mPbb /= -12.0;
299 mPbbb /= -20.0;
300 mPab /= 24.0;
301 mPaab /= 60.0;
302 mPabb /= -60.0;
303 }
304};
