50 // The number of mip levels in the specular cubemap,

53
54 // The HDR environment map,
55 // this is used if the input file is either in the HDR or EXR format
56 // and it is an equirectangular image.
57 Texture mHdrTexture;
58
59 // The cubemap environment map,
60 // this is either used directly if the input file is in the KTX or DDS
61 // format or contains the cubemap generated from an equirectangular image.
62 // It is used for display in the environment and for caluclating the
63 // irradiance and specular maps for the image-based lighting.
64 Texture mCubeMapTexture;
65
66 // The irradiance cubemap, generated from the environment cubemap.
67 // It contains the diffuse part of the image-based lighting and
68 // it is used in the shader to compute the diffuse contribution of the
69 // environment lighting.
70 Texture mIrradianceTexture;
71
72 // The specular cubemap, generated from the environment cubemap.
73 // It contains the specular part of the image-based lighting and
74 // it is used in the shader to compute the specular contribution of the
75 // environment lighting.
76 Texture mSpecularTexture;
77
78 // The BRDF lookup texture, generated independently from the environment
79 // map. It is used in the shader to compute the BRDF part of the image-based
80 // lighting.
81 Texture mBrdfLuTexture;
82
83 // Vertex buffer for rendering a full-screen triangle, used for drawing the
84 // environment.
