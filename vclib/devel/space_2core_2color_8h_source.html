193 uint8_t hsvHue() const
194 {
195 uint8_t rgbMin, rgbMax;
196 uint8_t h;
197
198 rgbMin = x() < y() ? (x() < z() ? x() : z()) : (y() < z() ? y() : z());
199 rgbMax = x() > y() ? (x() > z() ? x() : z()) : (y() > z() ? y() : z());
200
201 if (rgbMax == 0) {
202 return 0;
203 }
204
205 if (255 * long(rgbMax - rgbMin) / rgbMax == 0) {
206 return 0;
207 }
208
209 if (rgbMax == x())
210 h = 0 + 43 * (y() - z()) / (rgbMax - rgbMin);
211 else if (rgbMax == y())
212 h = 85 + 43 * (z() - x()) / (rgbMax - rgbMin);
213 else
214 h = 171 + 43 * (x() - y()) / (rgbMax - rgbMin);
215
216 return h;
217 }

223 uint8_t hsvSaturation() const

229
230 if (rgbMax == 0) {

436 void setHsv(uint8_t h, uint8_t s, uint8_t v, uint8_t alpha = 255)
437 {
438 w() = alpha;
439 if (s == 0) {
440 x() = v;
441 y() = v;
442 z() = v;
443 }
444 else {
445 h = (h / 360.0) * 255;
446 uint8_t region, remainder, p, q, t;
447 region = h / 43;
448 remainder = (h - (region * 43)) * 6;
449
450 p = (v * (255 - s)) >> 8;
451 q = (v * (255 - ((s * remainder) >> 8))) >> 8;
452 t = (v * (255 - ((s * (255 - remainder)) >> 8))) >> 8;
453
454 switch (region) {
455 case 0:
456 x() = v;
457 y() = t;
458 z() = p;
459 break;
460 case 1:
461 x() = q;
462 y() = v;
463 z() = p;
464 break;
465 case 2:
466 x() = p;
467 y() = v;
468 z() = t;
469 break;
470 case 3:
471 x() = p;
472 y() = q;
473 z() = v;
474 break;
475 case 4:
476 x() = t;
477 y() = p;
478 z() = v;
479 break;
480 default:
481 x() = v;
482 y() = p;
483 z() = q;
484 break;
485 }
486 }
487 }

646inline Color colorLerp(const Color& c0, const Color& c1, float value)
647{
648 Color c;
649 if (value < 0) // out of range - left
650 value = 0;
651 else if (value > 1) // out of range - right
652 value = 1;
653 for (uint i = 0; i < 4; i++)
654 c(i) = c1(i) * value + c0(i) * (1 - value);
655 return c;
656}

722inline Color colorFromIntervalParula(float value)
723{
724 if (value < 0) // out of range - left
725 value = 0;
726 else if (value > 1) // out of range - right
727 value = 1;
728
729 static uint paruVal[9] = {
730 0x271680ff,
731 0x0363e1ff,
732 0x1485d4ff,
733 0x06a7c6ff,
734 0x38b99eff,
735 0x92bf73ff,
736 0xd9ba56ff,
737 0xfcce2eff,
738 0xfffa0aff};
739
740 int ind = int(floor(value * 8.0f));
741 float div = (value * 8.0f - ind);
742
743 if (div < 0)
744 div = 0;
745 else if (div > 1)
746 div = 1;
747
748 return colorLerp(
749 (Color::ColorABGR) paruVal[ind],
750 (Color::ColorABGR) paruVal[ind + 1],
751 div);
752}

893inline std::vector<Color> colorScattering(
894 uint n,
895 float sat = 0.3,
896 float val = 0.9)
897{
898 std::vector<Color> scattering;
899 scattering.reserve(n);
900
901 for (uint v = 0; v < n; ++v) {
902 uint value = v, m = n;
903 uint b = 0;
904
905 for (uint k = 1; k < n; k <<= 1) {
906 if (value << 1 >= m) {
907 b += k;
908 value -= (m + 1) >> 1;
909 m >>= 1;
910 }
911 else {
912 m = (m + 1) >> 1;
913 }
914 }
915
916 Color rc;
917 rc.setHsvF(float(b) / float(n), sat, val);
918 scattering.push_back(rc);
919 }
920 return scattering;
921}
