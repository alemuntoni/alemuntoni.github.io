136template<
137 MeshConcept MeshType,
138 MatrixConcept VMatrix,
139 MatrixConcept FMatrix = Eigen::MatrixX3i,
140 MatrixConcept EMatrix = Eigen::MatrixX2i>
141void meshFromMatrices(
142 MeshType& mesh,
143 const VMatrix& vertices,
144 const FMatrix& faces = FMatrix(),
145 const EMatrix& edges = EMatrix())
146{
147 mesh.clear();
148 mesh.disableAllOptionalComponents();
149
150 vertexPositionsFromMatrix(mesh, vertices);
151
152 if constexpr (HasFaces<MeshType>) {
153 if (faces.rows() > 0)
154 faceIndicesFromMatrix(mesh, faces);
155 }
156
157 if constexpr (HasEdges<MeshType>) {
158 if (edges.rows() > 0)
159 edgeIndicesFromMatrix(mesh, edges);
160 }
161}

348template<uint ELEM_ID, MeshConcept MeshType, Range R>
349void elementSelectionFromRange(MeshType& mesh, R&& selection)
350{
351 if (std::ranges::size(selection) != mesh.template count<ELEM_ID>())
352 throw WrongSizeException(
353 "The input selection range must have the same number of elements "
354 "as the number of " +
355 elementEnumString<ELEM_ID>() + " element in the mesh");
356
357 auto s = selection.begin();
358 for (auto& e : mesh.template elements<ELEM_ID>()) {
359 e.selected() = *s;
360 ++s;
361 }
362}

575template<uint ELEM_ID, MeshConcept MeshType, MatrixConcept CMatrix>
576void elementColorsFromMatrix(MeshType& mesh, const CMatrix& colors)
577{
578 using MatrixScalar = CMatrix::Scalar;
579
580 Color::Representation repr = std::integral<MatrixScalar> ?
581 Color::Representation::INT_0_255 :
582 Color::Representation::FLOAT_0_1;
583
584 if (colors.rows() != mesh.template count<ELEM_ID>())
585 throw WrongSizeException(
586 "The input colors matrix must have the same number of rows as the "
587 "number of " +
588 elementEnumString<ELEM_ID>() + " elements in the mesh");
589
590 elementColorsFromBuffer<ELEM_ID>(
591 mesh, colors.data(), colors.cols(), matrixStorageType<CMatrix>(), repr);
592}

619template<uint ELEM_ID, MeshConcept MeshType, Range R>
620void elementColorsFromRange(

636 e.color() = Color(*c, colorFormat);
