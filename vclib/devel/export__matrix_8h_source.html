1160template<
1161 template<typename, typename...> typename Container,
1162 typename T,
1163 MeshConcept MeshType>
1164Container<Container<T>> vertexAdjacentVerticesVectors(const MeshType& mesh)
1165{
1166 requireVertexContainerCompactness(mesh);
1167 requirePerVertexAdjacentVertices(mesh);
1168
1169 Container<Container<T>> vv(mesh.vertexCount());
1170
1171 auto vvIt = vv.begin();
1172 for (const auto& v : mesh.vertices()) {
1173 auto& vec = *vvIt;
1174
1175 vec.resize(v.adjVertexCount());
1176 auto vecIt = vec.begin();
1177 for (const auto* ve : v.adjVertices()) {
1178 uint idx = ve ? ve->index() : UINT_NULL;
1179 *vecIt = T(idx);
1180 ++vecIt;
1181 }
1182
1183 ++vvIt;
1184 }
1185
1186 return vv;
1187}
