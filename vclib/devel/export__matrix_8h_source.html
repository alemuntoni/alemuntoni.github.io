1175template<
1176 template<typename, typename...> typename Container,
1177 typename T,
1178 MeshConcept MeshType>
1179Container<Container<T>> vertexAdjacentVerticesVectors(const MeshType& mesh)
1180{
1181 requireVertexCompactness(mesh);
1182 requireVertexAdjacentVertices(mesh);
1183
1184 Container<Container<T>> vv(mesh.vertexCount());
1185
1186 auto vvIt = vv.begin();
1187 for (const auto& v : mesh.vertices()) {
1188 auto& vec = *vvIt;
1189
1190 vec.resize(v.adjVertexCount());
1191 auto vecIt = vec.begin();
1192 for (auto idx : v.adjVertexIndices()) {
1193 *vecIt = T(idx);
1194 ++vecIt;
1195 }
1196
1197 ++vvIt;
1198 }
1199
1200 return vv;
1201}
