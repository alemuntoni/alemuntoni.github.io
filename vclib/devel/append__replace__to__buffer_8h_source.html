91template<MeshConcept MeshType>
92void appendDuplicateVertexPositionsToBuffer(
93 const MeshType& mesh,
94 const std::list<IndexType<MeshType>>& vertsToDuplicate,
95 auto* buffer,
96 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
97{
98 using namespace detail;
99 using IndexType = IndexType<MeshType>;
100
101 // no vertices to duplicate, do nothing
102 if (vertsToDuplicate.empty())
103 return;
104
105 const IndexType NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
106
107 for (IndexType i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
108 const auto& pos = mesh.vertex(v).position();
109
110 at(buffer, i, 0, NUM_ROWS, 3, storage) = pos.x();
111 at(buffer, i, 1, NUM_ROWS, 3, storage) = pos.y();
112 at(buffer, i, 2, NUM_ROWS, 3, storage) = pos.z();
113
114 ++i;
115 }
116}

164template<FaceMeshConcept MeshType>
165void replaceFaceVertexIndicesByVertexDuplicationToBuffer(
166 const MeshType& mesh,
167 const std::list<IndexType<MeshType>>& vertsToDuplicate,
168 const std::list<std::list<std::pair<IndexType<MeshType>, uint>>>&
169 facesToReassign,
170 auto* buffer,
171 uint largestFaceSize = 3,
172 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
173{
174 using namespace detail;
175 using IndexType = IndexType<MeshType>;
176
177 // no vertices have been duplicated, do nothing
178 if (vertsToDuplicate.empty())
179 return;
180
181 assert(vertsToDuplicate.size() == facesToReassign.size());
182
183 const IndexType NUM_ROWS = mesh.faceCount();
184 auto faceCompIndices = detail::faceCompactIndices(mesh, true);
185
186 IndexType vFirst = mesh.vertexCount();
187 IndexType vLast = mesh.vertexCount() + vertsToDuplicate.size();
188 for (IndexType vi = vFirst; const auto& faces : facesToReassign) {
189 for (const auto& f : faces) {
190 IndexType fRow = faceCompIndices.empty() ? f.first : faceCompIndices[f.first];
191 at(buffer, fRow, f.second, NUM_ROWS, largestFaceSize, storage) =
192 vi;
193 }
194 ++vi;
195 }
196}

246template<FaceMeshConcept MeshType>
247void replaceTriangulatedFaceVertexIndicesByVertexDuplicationToBuffer(
248 const MeshType& mesh,
249 const std::list<IndexType<MeshType>>& vertsToDuplicate,
250 const std::list<std::list<std::pair<IndexType<MeshType>, uint>>>&
251 facesToReassign,
252 const TriPolyIndexBiMap& indexMap,
253 auto* buffer,
254 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
255{
256 using namespace detail;
257 using IndexType = IndexType<MeshType>;
258
259 // no vertices have been duplicated, do nothing
260 if (vertsToDuplicate.empty())
261 return;
262
263 assert(vertsToDuplicate.size() == facesToReassign.size());
264
265 IndexType vFirst = mesh.vertexCount();
266 IndexType vLast = mesh.vertexCount() + vertsToDuplicate.size();
267
268 const IndexType NUM_ROWS = indexMap.triangleCount();
269 auto vertCompIndices = detail::vertCompactIndices(mesh, true);
270
271 // the facesToReassign lists for each vertex contain pairs that in the
272 // second element store the index of the vertex in the face. However, the
273 // face has been triangulated, and this info is not useful anymore. We need
274 // to look into the triangles generated for the face (first elem of the
275 // pair) and look for the vertex index to set (that is the vertex index
276 // stored in the vertsToDuplicate list).
277 for (IndexType vi = vFirst; const auto& [vert, faces] : std::views::zip(
278 vertsToDuplicate, facesToReassign)) {
279 // vert is the vertex index in the mesh that we need to reassign with
280 // the index vi in the buffer
281 IndexType targetVert = vertCompIndices.empty() ? vert : vertCompIndices[vert];
282
283 // for each face f that has at least a vertex to reassign
284 for (const auto& f : faces) {
285 // get the triangle indices of the face using the index map
286 IndexType tBegin = indexMap.triangleBegin(f.first);
287 IndexType tEnd = tBegin + indexMap.triangleCount(f.first);
288 for (IndexType t = tBegin; t < tEnd;
289 ++t) { // look into the triangles
290 for (uint j = 0; j < 3; ++j) { // for each vertex of triangle
291 auto& triVert = at(buffer, t, j, NUM_ROWS, 3, storage);
292 if (triVert == targetVert) {
293 triVert = vi;
294 }
295 }
296 }
297 }
298 ++vi;
299 }
300}

326template<FaceMeshConcept MeshType>
327void permuteFaceVertexIndicesByFunctionToBuffer(
328 const MeshType& mesh,
329 auto* buffer,
330 const std::function<bool(ConstFaceType<MeshType>, ConstFaceType<MeshType>)>&
331 func,
332 TriPolyIndexBiMap& indexMap = detail::indexMap)
333{
334 using IndexType = IndexType<MeshType>;
335 using BufferElementType = std::remove_cvref_t<decltype(*buffer)>;
336
337 // get the list of face indices sorted by material ID and using the given
338 // face comparator function. The result contains original container indices
339 // of non-deleted faces in sorted order:
340 auto sortedFaceContainerIndices =
341 sortFaceIndicesByFunction(mesh, func, false);
342
343 // sortedFaceContainerIndices[newPos] = originalContainerIndex
344 // (only non-deleted faces, in the desired new order)
345
346 bool isIndexMapEmpty = indexMap.triangleCount() == 0;
347
348 // temporary copy of the buffer
349 IndexType buffSize =
350 isIndexMapEmpty ? mesh.faceCount() : indexMap.triangleCount();
351 std::vector<BufferElementType> bufferCopy(buffSize * 3);
352
353 // temporary indexmap (not used if input map is empty)
354 TriPolyIndexBiMap indexMapCopy;
355
356 if (!isIndexMapEmpty)
357 indexMapCopy.reserve(indexMap.triangleCount(), indexMap.polygonCount());
358
359 IndexType copiedTriangles = 0;
360 auto faceCompIndices = detail::faceCompactIndices(mesh, true);
361
362 for (std::size_t i = 0; i < sortedFaceContainerIndices.size(); ++i) {
363 // place the triangles associated to the face at new position i
364 IndexType polyIndex = sortedFaceContainerIndices[i];
365 IndexType firstTri =
366 isIndexMapEmpty ?
367 (faceCompIndices.empty() ? polyIndex : faceCompIndices[polyIndex]) :
368 indexMap.triangleBegin(polyIndex);
369 IndexType nTris =
370 isIndexMapEmpty ? 1 : indexMap.triangleCount(polyIndex);
371
372 std::copy(
373 buffer + firstTri * 3,
374 buffer + (firstTri + nTris) * 3,
375 bufferCopy.data() + copiedTriangles * 3);
376
377 if (!isIndexMapEmpty) {
378 for (IndexType t = copiedTriangles; t < copiedTriangles + nTris;
379 ++t) {
380 indexMapCopy.insert(t, polyIndex);
381 }
382 }
383
384 copiedTriangles += nTris;
385 }
386
387 // copy back
388 std::copy(
389 bufferCopy.begin(), bufferCopy.begin() + copiedTriangles * 3, buffer);
390
391 if (!isIndexMapEmpty)
392 indexMap = std::move(indexMapCopy);
393}

492template<MeshConcept MeshType>
493void appendDuplicateVertexNormalsToBuffer(
494 const MeshType& mesh,
495 const std::list<IndexType<MeshType>>& vertsToDuplicate,
496 auto* buffer,
497 bool normalize = false,
498 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
499{
500 using namespace detail;
501 using IndexType = IndexType<MeshType>;
502
503 // no vertices to duplicate, do nothing
504 if (vertsToDuplicate.empty())
505 return;
506
507 requireVertexNormal(mesh);
508
509 const IndexType NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
510
511 for (IndexType i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
512 auto normal = mesh.vertex(v).normal();
513 if (normalize)
514 normal.normalize();
515 at(buffer, i, 0, NUM_ROWS, 3, storage) = normal.x();
516 at(buffer, i, 1, NUM_ROWS, 3, storage) = normal.y();
517 at(buffer, i, 2, NUM_ROWS, 3, storage) = normal.z();
518
519 ++i;
520 }

835template<MeshConcept MeshType>
836void appendDuplicateVertexTangentsToBuffer(
837 const MeshType& mesh,
838 const std::list<IndexType<MeshType>>& vertsToDuplicate,
839 auto* buffer,
840 bool storeHandednessAsW = true,
841 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
842{
843 using namespace detail;
844 using IndexType = IndexType<MeshType>;
845
846 // no vertices to duplicate, do nothing
847 if (vertsToDuplicate.empty())
848 return;
849
850 requireVertexTangent(mesh);
851
852 const IndexType NUM_ROWS = mesh.vertexCount() + vertsToDuplicate.size();
853 const uint NUM_COLS = storeHandednessAsW ? 4 : 3;
854
855 for (IndexType i = mesh.vertexCount(); const auto& v : vertsToDuplicate) {
856 const auto& t = mesh.vertex(v).tangent();
857
858 at(buffer, i, 0, NUM_ROWS, NUM_COLS, storage) = t.x();
859 at(buffer, i, 1, NUM_ROWS, NUM_COLS, storage) = t.y();
860 at(buffer, i, 2, NUM_ROWS, NUM_COLS, storage) = t.z();
861 if (storeHandednessAsW) {
862 at(buffer, i, 3, NUM_ROWS, NUM_COLS, storage) =
863 mesh.vertex(v).tangentRightHanded() ? 1.0 : -1.0;
864 }
865
866 ++i;
867 }
