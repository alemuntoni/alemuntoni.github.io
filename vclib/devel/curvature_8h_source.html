27template<FaceMeshConcept MeshType, LoggerConcept LogType = NullLogger>
28void updatePrincipalCurvatureTaubin95(MeshType& m, LogType& log = nullLogger)
29{
30 requireVertexPrincipalCurvature(m);
31 requireVertexAdjacentFaces(m);
32 requireFaceAdjacentFaces(m);
33
34 using ConstVertexType = ConstVertexType<MeshType>;
35 using PositionType = PositionType<MeshType>;
36 using ScalarType = PositionType::ScalarType;
37
38 // Auxiliary data structure
39 struct AdjVertex
40 {
41 ConstVertexType vert;
42 double doubleArea;
43 bool isBorder;
44 };
45
46 log.log(0, "Updating per vertex normals...");
47
48 updateVertexNormalsAngleWeighted(m);
49 normalizeVertexNormals(m);
50
51 log.log(5, "Computing per vertex curvature...");
52 // log every 5%, starting from 5% to 100%
53 log.startProgress(
54 "", m.vertexCount(), LogType::ProgressMode::TIME, 1.0, 5, 100);
55
56 for (auto v : m.vertices()) {
57 std::vector<ScalarType> weights;
58 std::vector<AdjVertex> vertices;
59
60 auto pos = MeshPos(v.adjFace(0), v);
61 auto firstVertex = pos.adjVertex();
62 auto tmpVertex = ConstVertexType();
63 float totalDoubleAreaSize = 0;
64
65 // compute the area of each triangle around the central vertex as well
66 // as their total area
67 do {
68 pos.nextEdgeAdjacentToV();
69 tmpVertex = pos.adjVertex();
70 AdjVertex adjV;
71 adjV.isBorder = pos.isEdgeOnBorder();
72 adjV.vert = tmpVertex;
73 adjV.doubleArea = faceArea(pos.face()) * 2;
74 totalDoubleAreaSize += adjV.doubleArea;
75 vertices.push_back(adjV);
76 } while (tmpVertex != firstVertex);
77
78 weights.reserve(vertices.size());
79
80 for (size_t i = 0; i < vertices.size(); ++i) {
81 if (vertices[i].isBorder) {
82 weights.push_back(vertices[i].doubleArea / totalDoubleAreaSize);
83 }
84 else {
85 weights.push_back(
86 0.5f *
87 (vertices[i].doubleArea +
88 vertices[(i + vertices.size() - 1) % vertices.size()]
89 .doubleArea) /
90 totalDoubleAreaSize);
91 }
92 assert(weights.back() < 1.0f);
93 }
94
95 // compute I-NN^t to be used for computing the T_i's
96 Matrix33<ScalarType> Tp;
97
98 auto n = v.normal();
99 for (int i = 0; i < 3; ++i)
100 Tp(i, i) = 1.0f - std::pow(n[i], 2);
101 Tp(0, 1) = Tp(1, 0) = -1.0f * (n[0] * n[1]);
102 Tp(1, 2) = Tp(2, 1) = -1.0f * (n[1] * n[2]);
103 Tp(0, 2) = Tp(2, 0) = -1.0f * (n[0] * n[2]);
104
105 // for all neighbors vi compute the directional curvatures k_i and the
106 // T_i compute M by summing all w_i k_i T_i T_i^t
107 Matrix33<ScalarType> tempMatrix;
108 Matrix33<ScalarType> M = Matrix33<ScalarType>::Zero();
109 for (size_t i = 0; i < vertices.size(); ++i) {
110 PositionType edge = (v.position() - vertices[i].vert.position());
111 float curvature =
112 (2.0f * (v.normal().dot(edge))) / edge.squaredNorm();
113 PositionType t = Tp * edge;
114 t.normalize();
115 tempMatrix = t.outerProduct(t);
116 M += tempMatrix * weights[i] * curvature;
117 }
118
119 // compute vector W for the Householder matrix
120 PositionType w;
121 PositionType e1(1.0f, 0.0f, 0.0f);
122 if ((e1 - v.normal()).squaredNorm() > (e1 + v.normal()).squaredNorm())
123 w = e1 - v.normal();
124 else
125 w = e1 + v.normal();
126 w.normalize();
127
128 // compute the Householder matrix I - 2WW^t
129 Matrix33<ScalarType> Q = Matrix33<ScalarType>::Identity();
130 tempMatrix = w.outerProduct(w);
131 Q -= tempMatrix * 2.0f;
132
133 // compute matrix Q^t M Q
134 Matrix33<ScalarType> QtMQ = (Q.transpose() * M * Q);
135
136 Eigen::Matrix<ScalarType, 1, 3> T1 = Q.col(1);
137 Eigen::Matrix<ScalarType, 1, 3> T2 = Q.col(2);
138
139 // diagonalize the 2x2 restriction of M in the tangent plane basis (T1, T2)
140 Eigen::Matrix<ScalarType, 2, 2> minor2x2;
141 minor2x2(0, 0) = QtMQ(1, 1);
142 minor2x2(0, 1) = QtMQ(1, 2);
143 minor2x2(1, 0) = QtMQ(2, 1);
144 minor2x2(1, 1) = QtMQ(2, 2);
145
146 Eigen::SelfAdjointEigenSolver<Eigen::Matrix<ScalarType, 2, 2>> es(
147 minor2x2);
148 // es.eigenvalues() are sorted in ascending order: lambda0 <= lambda1
149 ScalarType l0 = es.eigenvalues()[0];
150 ScalarType l1 = es.eigenvalues()[1];
151
152 // Taubin95 formula:
153 // k0 = 3*l0 - l1
154 // k1 = 3*l1 - l0
155 // Since l0 <= l1, k0 <= k1 holds: (3*l1 - l0) - (3*l0 - l1) = 4*(l1 - l0) >= 0
156 ScalarType k0 = 3.0f * l0 - l1;
157 ScalarType k1 = 3.0f * l1 - l0;
158
159 auto ev0 = es.eigenvectors().col(0);
160 auto ev1 = es.eigenvectors().col(1);
161
162 Eigen::Matrix<ScalarType, 1, 3> dir0 = T1 * ev0[0] + T2 * ev0[1];
163 Eigen::Matrix<ScalarType, 1, 3> dir1 = T1 * ev1[0] + T2 * ev1[1];
164
165 v.principalCurvature().maxValue() = k1;
166 v.principalCurvature().minValue() = k0;
167 v.principalCurvature().maxDir() = dir1.normalized();
168 v.principalCurvature().minDir() = dir0.normalized();
169
170 log.progress(v.index());
171 }
172
173 log.endProgress();
174 log.log(100, "Per vertex curvature computed.");
175}

189template<FaceMeshConcept MeshType, LoggerConcept LogType = NullLogger>
190void updatePrincipalCurvaturePCA(
191 MeshType& m,
192 auto radius,
193 bool montecarloSampling = true,
194 LogType& log = nullLogger)
195{
196 using VertexType = VertexType<MeshType>;
197 using PositionType = VertexType::PositionType;
198 using ScalarType = PositionType::ScalarType;
199
200 using VGrid = StaticGrid3<VertexType, ScalarType>;
201
202 VGrid pGrid;
203 ScalarType area;
204
205 log.log(0, "Updating per vertex normals...");
206
207 updateVertexNormalsAngleWeighted(m);
208 normalizeVertexNormals(m);
209
210 log.log(0, "Computing per vertex curvature...");
211 log.startProgress("", m.vertexCount());
212
213 if (montecarloSampling) {
214 area = surfaceArea(m);
215 pGrid = VGrid(m.vertices());
216 }
217
218 parallelFor(m.vertices(), [&](VertexType v) {
219 // for (VertexType& v : m.vertices()) {
220 Matrix33<ScalarType> A, eigenvectors;
221 PositionType bp, eigenvalues;
222 if (montecarloSampling) {
223 Sphere s(v.position(), radius);
224 auto vec = pGrid.valuesInSphere(s); // vector of iterators
225 std::vector<PositionType> points;
226 points.reserve(vec.size());
227 for (const auto& it : vec) {
228 points.push_back(it->second.position());
229 }
230 A = covarianceMatrixOfPointCloud(points);
231 A *= area * area / 1000;
232 }
233 else {
234 Sphere<ScalarType> sph(v.position(), radius);
235 MeshType tmpMesh = intersection(m, sph);
236
237 A = covarianceMatrixOfMesh(tmpMesh);
238 }
239
240 Eigen::SelfAdjointEigenSolver<Eigen::Matrix<ScalarType, 3, 3>> eig(A);
241 eigenvalues = PositionType(eig.eigenvalues());
242 eigenvectors = eig.eigenvectors(); // eigenvector are stored as columns.
243 // get the estimate of curvatures from eigenvalues and eigenvectors
244 // find the 2 most tangent eigenvectors (by finding the one closest to
245 // the normal)
246 uint best = 0;
247 ScalarType bestv = std::abs(
248 v.normal().dot(PositionType(eigenvectors.col(0).normalized())));
249 for (uint i = 1; i < 3; ++i) {
250 ScalarType prod = std::abs(
251 v.normal().dot(PositionType(eigenvectors.col(i).normalized())));
252 if (prod > bestv) {
253 bestv = prod;
254 best = i;
255 }
256 }
257 v.principalCurvature().maxDir() =
258 (eigenvectors.col((best + 1) % 3).normalized());
259 v.principalCurvature().minDir() =
260 (eigenvectors.col((best + 2) % 3).normalized());
261
262 Matrix33<ScalarType> rot;
263 ScalarType angle;
264 angle = acos(v.principalCurvature().maxDir().dot(v.normal()));
265
266 rot = rotationMatrix<Matrix33<ScalarType>>(
267 PositionType(v.principalCurvature().maxDir().cross(v.normal())),
268 -(M_PI * 0.5 - angle));
269
270 v.principalCurvature().maxDir() = rot * v.principalCurvature().maxDir();
271
272 angle = acos(v.principalCurvature().minDir().dot(v.normal()));
273
274 rot = rotationMatrix<Matrix33<ScalarType>>(
275 PositionType(v.principalCurvature().minDir().cross(v.normal())),
276 -(M_PI * 0.5 - angle));
277
278 v.principalCurvature().minDir() = rot * v.principalCurvature().minDir();
279
280 // computes the curvature values
281 const ScalarType r5 = std::pow(radius, 5);
282 const ScalarType r6 = r5 * radius;
283 v.principalCurvature().maxValue() =
284 (2.0 / 5.0) *
285 (4.0 * M_PI * r5 + 15 * eigenvalues[(best + 2) % 3] -
286 45.0 * eigenvalues[(best + 1) % 3]) /
287 (M_PI * r6);
288 v.principalCurvature().minValue() =
289 (2.0 / 5.0) *
290 (4.0 * M_PI * r5 + 15 * eigenvalues[(best + 1) % 3] -
291 45.0 * eigenvalues[(best + 2) % 3]) /
292 (M_PI * r6);
293 if (v.principalCurvature().maxValue() <
294 v.principalCurvature().minValue()) {
295 std::swap(
296 v.principalCurvature().minValue(),
297 v.principalCurvature().maxValue());
298 std::swap(
299 v.principalCurvature().minDir(),
300 v.principalCurvature().maxDir());
301 }
302
303 log.progress(v.index());
304 //}
305 });
306
307 log.endProgress();
308 log.log(100, "Per vertex curvature computed.");
309}

311template<FaceMeshConcept MeshType, LoggerConcept LogType = NullLogger>
312void updatePrincipalCurvature(MeshType& m, LogType& log = nullLogger)
313{
314 requireVertexPrincipalCurvature(m);
315
316 updatePrincipalCurvatureTaubin95(m, log);
317}

319template<FaceMeshConcept MeshType, LoggerConcept LogType = NullLogger>
320void updatePrincipalCurvature(
321 MeshType& m,
322 PrincipalCurvatureAlgorithm alg = PrincipalCurvatureAlgorithm::TAUBIN95,
323 LogType& log = nullLogger)
324{
325 using enum PrincipalCurvatureAlgorithm;
326 requireVertexPrincipalCurvature(m);
327
328 double radius;
329 switch (alg) {
330 case TAUBIN95: updatePrincipalCurvatureTaubin95(m, log); break;
331 case PCA:
332 radius = boundingBox(m).diagonal() * 0.1;
333 updatePrincipalCurvaturePCA(m, radius, true, log);
334 }
335}
