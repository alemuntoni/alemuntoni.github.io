54template<>
55class CustomComponentsVectorMap<true>
56{
57 // the actual map containing, for each name of a custom component, the
58 // vector of values (a value for each element(vertex/face...) of the mesh)
59 std::unordered_map<std::string, std::vector<std::any>> mMap;
60 // map that stores if some custom components need to be initialized.
61 // this happens when we resize the container of an element: we need to
62 // resize also the vectors of the custom components, but we cannot
63 // initialize them with the right type (because we don't know it in resize
64 // phase). The initialization will be made when the type is known (get of
65 // the custom component vector).
66 mutable std::unordered_map<std::string, bool> mNeedToInitialize;
67 // types used for each custom component
68 std::unordered_map<std::string, std::type_index> mCompType;
69
70public:
74 void clear()
75 {
76 mMap.clear();
77 mNeedToInitialize.clear();
78 mCompType.clear();
79 }
80
85 void reserve(uint size)
86 {
87 for (auto& p : mMap) {
88 p.second.reserve(size);
89 }
90 }
91
97 void resize(uint size)
98 {
99 for (auto& p : mMap) {
100 // At this call, we don't know statically the types of each custom
101 // component, therefore we cannot initialize them (std::any of each
102 // vector will be in an invalid state). Therefore, we mark all the
103 // resized custom components as 'needToInitialize'. Initiaization
104 // will be made just on the uninitialized values of the vectors at
105 // the first access of each custom component.
106 if (p.second.size() < size)
107 mNeedToInitialize.at(p.first) = true;
108 p.second.resize(size);
109 }
110 }

124 void compact(const std::vector<uint>& newIndices)

131 void swapCustomComponents(uint i, uint j)

135 }

151 template<typename CompType>
152 void addNewComponent(const std::string& name, uint size)
153 {
154 std::vector<std::any>& v = mMap[name];
155 v.resize(size, CompType());
156 mNeedToInitialize[name] = false;
157 mCompType.emplace(name, typeid(CompType));
158 }
159
165 void deleteComponent(const std::string& name)
166 {
167 mMap.erase(name);
168 mNeedToInitialize.erase(name);
169 mCompType.erase(name);
170 }

177 void assertComponentExists(const std::string& compName) const

188 bool componentExists(const std::string& compName) const

198 std::vector<std::string> allComponentNames() const
199 {
200 std::vector<std::string> names;
201 names.reserve(mMap.size());
202 for (const auto& p : mMap)
203 names.push_back(p.first);

246 template<typename CompType>
247 std::vector<std::string> allComponentNamesOfType() const
248 {
249 std::vector<std::string> names;
250 std::type_index t(typeid(CompType));
251 for (const auto& p : mCompType) {
252 if (p.second == t)

270 template<typename CompType>
271 const std::vector<std::any>& componentVector(
272 const std::string& compName) const
273 {
274 checkComponentType<CompType>(compName);
275 std::vector<std::any>& v =
276 const_cast<std::vector<std::any>&>(mMap.at(compName));
277 // if the component needs to be initialized (e.g. we made a resize)
278 if (mNeedToInitialize.at(compName)) {
279 for (std::any& a : v) {
280 if (!a.has_value())
281 a = CompType();
282 }
283 mNeedToInitialize.at(compName) = false;
284 }
285 return v;
286 }

319 void importSameCustomComponentFrom(
320 uint thisPos,
321 uint otherPos,
322 const std::string& compName,
323 const CustomComponentsVectorMap<true>& other)
324 {
325 if (other.componentExists(compName) && componentExists(compName)) {
326 if (other.componentType(compName) == componentType(compName)) {
327 mMap.at(compName)[thisPos] = other.mMap.at(compName)[otherPos];
328 }

332 template<typename CompType>
333 void serializeCustomComponentsOfType(std::ostream& os) const
334 {
335 std::vector<std::string> compNames =
336 allComponentNamesOfType<CompType>();
337 vcl::serialize(os, compNames);
338 for (const auto& name : compNames) {
339 bool b = mNeedToInitialize.at(name);
340 vcl::serialize(os, b);
341 const std::vector<std::any>& v = componentVector<CompType>(name);
342 vcl::serialize<CompType>(os, v);
343 }
344 }
345
346 template<typename CompType>
347 void deserializeCustomComponentsOfType(std::istream& is)
348 {
349 std::vector<std::string> compNames;
350 vcl::deserialize(is, compNames);
351 for (const auto& name : compNames) {
352 bool b;
353 vcl::deserialize(is, b);
354 std::vector<std::any> v;
355 vcl::deserialize<CompType>(is, v);
356 mMap[name] = v;
357 mNeedToInitialize[name] = b;
358 mCompType.emplace(name, typeid(CompType));
359 }
360 }
361
362private:
363 template<typename CompType>
364 void checkComponentType(const std::string& compName) const
365 {
366 std::type_index t(typeid(CompType));
367 if (t != mCompType.at(compName)) {
368 throw BadCustomComponentTypeException(
369 "Expected type " + std::string(mCompType.at(compName).name()) +
370 " for " + compName + ", but was " + std::string(t.name()) +
371 ".");
372 }
373 }
