39inline std::set<FileFormat> loadImageFormats()
40{
41 std::set<FileFormat> ff;
42
43#ifdef VCLIB_WITH_QT
44 auto fqt = qt::loadImageFormats();
45 ff.insert(fqt.begin(), fqt.end());
46#endif
47
48#ifdef VCLIB_WITH_STB
49 auto fstb = stb::loadImageFormats();
50 ff.insert(fstb.begin(), fstb.end());
51#endif
52 return ff;

55inline std::shared_ptr<unsigned char> loadImageData(
56 const std::string& filename,
57 int& w,
58 int& h)
59{
60 FileFormat ff = FileInfo::fileFormat(filename);
61
62#ifdef VCLIB_WITH_QT
63 if (qt::loadImageFormats().contains(ff)) {
64 return qt::loadImageData(filename, w, h);
65 }
66#endif
67
68#ifdef VCLIB_WITH_STB
69 if (stb::loadImageFormats().contains(ff)) {
70 return stb::loadImageData(filename, w, h);
71 }
72#endif
73 throw UnknownFileFormatException(ff.extensions().front());
74}
75
76inline Image loadImage(const std::string& filename)
77{
78 int w, h;
79 auto data = loadImageData(filename, w, h);
80 if (!data) {
81 return Image();
82 }
83 return Image(data.get(), w, h);
84}

101template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
102void loadTextureImages(
103 MeshType& mesh,
104 std::string basePath = "",
105 const BitSet8& textureTypesToLoad = BitSet8::ALL(),
106 LogType& log = nullLogger) requires HasMaterials<MeshType>
107{
108 if (basePath.empty()) {
109 basePath = mesh.meshBasePath();
110 }
111
112 for (const Material& mat : mesh.materials()) {
113 using enum Material::TextureType;
114 const uint N_TEXTURE_TYPES = toUnderlying(COUNT);
115 for (uint i = 0; i < N_TEXTURE_TYPES; ++i) {
116 auto texType = static_cast<Material::TextureType>(i);
117 // supported textures to load
118 if (textureTypesToLoad[i]) {
119 const TextureDescriptor& tex = mat.textureDescriptor(i);
120 if (!tex.isNull()) {
121 // if not already loaded
122 if (mesh.textureImage(tex.path()).isNull()) {
123 Image img = loadImage(basePath + tex.path());
124 if (img.isNull()) {
125 log.log(
126 "Cannot load texture " + tex.path(),
127 LogType::WARNING_LOG);
128 }
129 else {
130 img.colorSpace() =
131 Material::textureTypeToColorSpace(texType);
132 mesh.pushTextureImage(tex.path(), std::move(img));
133 }
134 }
135 }
136 }
137 }
138 }
139}
