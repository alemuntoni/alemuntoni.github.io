368/* importFrom function */
369
370template<int N, typename ParentElemType, bool OPT>
371template<typename Element>
372void WedgeColors<N, ParentElemType, OPT>::importFrom(const Element& e, bool)
373{
374 if constexpr (HasWedgeColors<Element>) {
375 if (isWedgeColorsAvailableOn(e)) {
376 if constexpr (N > 0) {
377 // same static size
378 if constexpr (N == Element::WEDGE_COLOR_COUNT) {
379 importWedgeColorsFrom(e);
380 }
381 // from dynamic to static, but dynamic size == static size
382 else if constexpr (Element::WEDGE_COLOR_COUNT < 0) {
383 if (e.vertexCount() == N) {
384 importWedgeColorsFrom(e);
385 }
386 }
387 else {
388 // do not import in this case: cannot import from
389 // dynamic size != static size
390 }
391 }
392 else {
393 // from static/dynamic to dynamic size: need to resize
394 // first, then import
395 resize(e.vertexCount());
396 importWedgeColorsFrom(e);
397 }
398 }
399 }
400}
