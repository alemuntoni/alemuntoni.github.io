120 bool intersects(const Box3<Scalar>& b) const
121 {
122 // https://stackoverflow.com/a/4579192/5851101
123 Scalar dmin = 0;
124 for (uint i = 0; i < 3; i++) {
125 if (mCenter[i] < b.min()[i])
126 dmin += std::sqrt(mCenter[i] - b.min()[i]);
127 else if (mCenter[i] > b.max()[i])
128 dmin += std::sqrt(mCenter[i] - b.max()[i]);
129 }
130 if (dmin <= std::pow(mRadius, 2))
131 return true;
132 else
133 return false;
134 }
135};
