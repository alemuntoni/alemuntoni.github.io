79template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
80void loadMesh(
81 MeshType& m,
82 const std::string& filename,
83 MeshInfo& loadedInfo,
84 const LoadSettings& settings,
85 LogType& log = nullLogger)
86{
87 FileFormat ff = FileInfo::fileFormat(filename);
88
89 loadedInfo.clear();
90
91 if (ff == objFileFormat()) {
92 loadObj(m, filename, loadedInfo, settings, log);
93 }
94 else if (ff == offFileFormat()) {
95 loadOff(m, filename, loadedInfo, settings, log);
96 }
97 else if (ff == plyFileFormat()) {
98 loadPly(m, filename, loadedInfo, settings, log);
99 }
100 else if (ff == stlFileFormat()) {
101 loadStl(m, filename, loadedInfo, settings, log);
102 }
103#ifdef VCLIB_WITH_TINYGLTF
104 else if (ff == gltfFileFormat()) {
105 loadGltf(m, filename, loadedInfo, settings, log);
106 }
107#endif
108 else {
109 throw UnknownFileFormatException(ff.extensions().front());
110 }
111}
