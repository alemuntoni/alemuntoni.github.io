27template<MeshConcept MeshType>
28auto barycenter(const MeshType& m) -> PositionType<MeshType>
29{
30 using PositionType = PositionType<MeshType>;
31
32 PositionType bar;
33
34 for (auto&& v : m.vertices()) {
35 bar += v.position();
36 }
37
38 return bar / m.vertexCount();
39}

55template<MeshConcept MeshType>
56auto weightedBarycenter(const MeshType& m, Range auto&& weights)
57 -> PositionType<MeshType>
58{
59 using PositionType = PositionType<MeshType>;
60 using RType = std::ranges::range_value_t<decltype(weights)>;
61
62 assert(std::ranges::size(weights) == m.vertexCount());
63
64 PositionType bar;
65 RType weightedSum = 0;
66
67 for (const auto& [v, w] : std::views::zip(m.vertices(), weights)) {
68 bar += v.position() * w;
69 weightedSum += w;
70 }
71
72 return bar / weightedSum;
73}

91auto qualityWeightedBarycenter(const MeshType& m) -> PositionType<MeshType>
92{
93 requireVertexQuality(m);

113template<FaceMeshConcept MeshType>
114auto shellBarycenter(const MeshType& m) -> PositionType<MeshType>
115{
116 using PositionType = PositionType<MeshType>;
117 using ScalarType = PositionType::ScalarType;
118
119 PositionType bar;
120 bar.setZero();
121 ScalarType areaSum = 0;
122
123 for (auto&& f : m.faces()) {
124 ScalarType area = faceArea(f);
125 bar += faceBarycenter(f) * area;
126 areaSum += area;
127 }
128
129 return bar / areaSum;
130}
