26template<MeshConcept MeshType>
27auto barycenter(const MeshType& m) -> MeshType::VertexType::PositionType
28{
29 using VertexType = MeshType::VertexType;
30 using PositionType = VertexType::PositionType;
31
32 PositionType bar;
33
34 for (const VertexType& v : m.vertices()) {
35 bar += v.position();
36 }
37
38 return bar / m.vertexCount();
39}

55template<MeshConcept MeshType>
56auto weightedBarycenter(const MeshType& m, Range auto&& weights)
57 -> MeshType::VertexType::PositionType
58{
59 using VertexType = MeshType::VertexType;
60 using PositionType = VertexType::PositionType;
61 using RType = std::ranges::range_value_t<decltype(weights)>;
62
63 assert(std::ranges::size(weights) == m.vertexCount());
64
65 PositionType bar;
66 RType weightedSum = 0;
67
68 for (const auto& [v, w] : std::views::zip(m.vertices(), weights)) {
69 bar += v.position() * w;
70 weightedSum += w;
71 }
72
73 return bar / weightedSum;
74}

115template<FaceMeshConcept MeshType>
116auto shellBarycenter(const MeshType& m) -> MeshType::VertexType::PositionType
117{
118 using VertexType = MeshType::VertexType;
119 using FaceType = MeshType::FaceType;
120 using PositionType = VertexType::PositionType;
121 using ScalarType = PositionType::ScalarType;
122
123 PositionType bar;
124 bar.setZero();
125 ScalarType areaSum = 0;
126
127 for (const FaceType& f : m.faces()) {
128 ScalarType area = faceArea(f);
129 bar += faceBarycenter(f) * area;
130 areaSum += area;
131 }
132
133 return bar / areaSum;
134}
