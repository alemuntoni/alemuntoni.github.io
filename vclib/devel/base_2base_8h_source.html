84{
85 switch (type) {
86 case PrimitiveType::CHAR: return sizeof(char);
87 case PrimitiveType::UCHAR: return sizeof(unsigned char);
88 case PrimitiveType::SHORT: return sizeof(int16_t);
89 case PrimitiveType::USHORT: return sizeof(uint16_t);

92 case PrimitiveType::FLOAT: return sizeof(float);
93 case PrimitiveType::DOUBLE: return sizeof(double);
94 default: return 0;
