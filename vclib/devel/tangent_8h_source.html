107 P bitangent(this auto&& self)
108 {
109 if constexpr (HasNormal<decltype(self)>) {
110 if (self.template isComponentAvailable<CompId::NORMAL>()) {
111 const auto& n = self.normal();
112 if constexpr (std::is_same_v<
113 typename P::ScalarType,
114 typename std::remove_cvref_t<
115 decltype(n)>::ScalarType>) {
116 return self.tangentRightHanded() ? n.cross(self.tangent()) :
117 self.tangent().cross(n);
118 }
119 else {
120 auto nCast = n.template cast<typename P::ScalarType>();
121 return self.tangentRightHanded() ?
122 nCast.cross(self.tangent()) :
123 self.tangent().cross(nCast);
124 }
125 }
126 }
127 throw MissingComponentException(
128 "Normal Component is not available on this element.");
129 }

165template<PointConcept P, bool OPT>
166template<typename OtherElement>
167void Tangent<P, OPT>::importFrom(
168 this auto&& self,
169 const OtherElement& e,
170 bool)
171{
172 using ScalarType = typename TangentType::ScalarType;
173 if constexpr (HasTangent<OtherElement>) {
174 if (e.template isComponentAvailable<CompId::TANGENT>()) {
175 self.tangent() = e.tangent().template cast<ScalarType>();
176 self.tangentRightHanded() = e.tangentRightHanded();
177 }
178 }
