136 P bitangent() const
137 {
138 const ParentElemType& parent =
139 static_cast<const ParentElemType&>(*this);
140
141 if (isComponentAvailableOn<CompId::NORMAL>(parent)) {
142 // compute bitangent using the tangent and normal
143 // assuming that normal is available
144 const auto& n = static_cast<const ParentElemType&>(*this).normal();
145 return tangentRightHanded() ? n.cross(tangent()) :
146 tangent().cross(n);
147 }
148 else {
149 throw MissingComponentException(
150 "Per " +
151 std::string(elementEnumString<ParentElemType::ELEMENT_ID>()) +
152 " Normal Component is not enabled.");
153 return P();
154 }
155 }

217 using ScalarType = TangentType::ScalarType;
218 if constexpr (HasTangent<Element>) {
219 if (isTangentAvailableOn(e)) {
220 tangent() = e.tangent().template cast<ScalarType>();
