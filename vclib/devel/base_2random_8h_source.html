92inline std::mt19937 randomGenerator(std::optional<uint> seed = std::nullopt)
93{
94 std::mt19937 gen;
95
96 if (seed.has_value()) {
97 gen.seed(seed.value());
98 }
99 else {
100 std::random_device rd;
101 gen.seed(rd()); // random seed
102 }
103
104 return gen;
105}

139template<typename Func>
140decltype(auto) callWithRandomGenerator(RandomConfig config, Func&& func)
141{
142 // Case 1: the user provides a reference to a std::mt19937 generator
143 if (auto* userGen =
144 std::get_if<std::reference_wrapper<std::mt19937>>(&config)) {
145 return func(userGen->get());
146 }
147 // Case 2: the user provides a seed for a temporary generator
148 else if (auto* seed = std::get_if<uint>(&config)) {
149 std::mt19937 tempGen = randomGenerator(*seed);
150 return func(tempGen);
151 }
152 // Case 3: the user provides std::monostate or no config, use a thread-local
153 // generator
154 else {
155 thread_local std::mt19937 internal_gen = randomGenerator();
156 return func(internal_gen);
157 }
158}

193template<Numeric T, typename Func>
194decltype(auto) callWithDistribution(const DistConfig<T>& distConf, Func&& func)
195{
196 // 1. Promote T to a safe type if it's an 8-bit integer to avoid
197 // non-supported types in std::uniform_int_distribution.
198 using SafeT = std::conditional_t<
199 std::is_integral_v<T> && sizeof(T) == 1,
200 std::conditional_t<std::is_signed_v<T>, short, unsigned short>,
201 T>;
202
203 // 2. compiler chooses the correct distribution type based on SafeT
204 using UniformDist = std::conditional_t<
205 std::is_floating_point_v<T>,
206 std::uniform_real_distribution<SafeT>,
207 std::uniform_int_distribution<SafeT>>;
208
209 // Case 1: The user provides a pair of bounds for a uniform distribution
210 if (auto* bounds = std::get_if<std::pair<T, T>>(&distConf)) {
211 UniformDist dist(
212 static_cast<SafeT>(bounds->first),
213 static_cast<SafeT>(bounds->second));
214
215 // 'func' takes a lambda that captures the distribution and calls it
216 // with a random generator
217 return func([&dist](std::mt19937& g) {
218 // generate using SafeT, but then casts to T
219 return static_cast<T>(dist(g));
220 });
221 }
222 // Case 2: The user provides a custom distribution function
223 else if (
224 auto* custom =
225 std::get_if<std::function<T(std::mt19937&)>>(&distConf)) {
226 return func([&custom](std::mt19937& g) { return (*custom)(g); });
227 }
228 // Case 3: The user provides std::monostate or no config, use default
229 // uniform distribution with standard bounds based on the type T
230 else {
231 // Extract the maximum and minimum limits for the ORIGINAL T, to avoid
232 // dist(min, max) generating numbers outside the range of T if we have
233 // promoted it.
234 SafeT min_val = std::is_floating_point_v<T> ?
235 static_cast<SafeT>(0.0) :
236 static_cast<SafeT>(std::numeric_limits<T>::min());
237
238 SafeT max_val = std::is_floating_point_v<T> ?
239 static_cast<SafeT>(1.0) :
240 static_cast<SafeT>(std::numeric_limits<T>::max());
241
242 UniformDist dist(min_val, max_val);
243 return func([&dist](std::mt19937& g) {
244 // generate using SafeT, but then casts to T
245 return static_cast<T>(dist(g));
246 });
247 }
248}

260template<Range R>

262{
263 callWithRandomGenerator(config, [&](std::mt19937& gen) {
