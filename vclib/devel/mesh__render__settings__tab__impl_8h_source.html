37class MeshRenderSettingsTabImpl : public SettingsDialogTab
38{
39 MeshRenderSettings& mGlobalSettings;
40 std::shared_ptr<MeshRenderSettings> mTempSettings;
41 QString mName;
42 std::function<
43 GenericMeshRenderSettingsFrame*(MeshRenderSettings&, QWidget*)>
44 mFactory;
45 std::function<void(MeshRenderSettings&)> mResetFunc;
46
47public:
48 MeshRenderSettingsTabImpl(
49 MeshRenderSettings& globalSettings,
50 std::shared_ptr<MeshRenderSettings> tempSettings,
51 const QString& name,
52 std::function<
53 GenericMeshRenderSettingsFrame*(MeshRenderSettings&, QWidget*)>
54 factory,
55 std::function<void(MeshRenderSettings&)> resetFunc) :
56 mGlobalSettings(globalSettings), mTempSettings(tempSettings),
57 mName(name), mFactory(factory), mResetFunc(resetFunc)
58 {
59 }
60
61 QString category() const override { return "Mesh Render Defaults"; }
62
63 QString name() const override { return mName; }
64
65 QWidget* createWidget(QWidget* parent) override
66 {
67 QWidget* page = new QWidget(parent);
68 QVBoxLayout* layout = new QVBoxLayout(page);
69 layout->setContentsMargins(0, 0, 0, 0);
70
71 GenericMeshRenderSettingsFrame* frame = mFactory(*mTempSettings, page);
72 frame->updateFrameFromSettings();
73 if (auto cb = frame->visibilityCheckBox()) {
74 cb->setText(QString("Enable %1").arg(mName));
75 }
76
77 QPushButton* resetBtn = new QPushButton("Reset to Defaults", page);
78 resetBtn->setObjectName("resetDefaultButton");
79 QObject::connect(resetBtn, &QPushButton::clicked, [this, frame]() {
80 mResetFunc(*mTempSettings);
81 frame->updateFrameFromSettings();
82 });
83
84 layout->addWidget(frame);
85 layout->addWidget(resetBtn);
86 // Note: Stretch is automatically added by SettingsDialog.cpp, but
87 // adding one here doesn't hurt.
88
89 return page;
90 }
91
92 void applySettings() override { mGlobalSettings = *mTempSettings; }
93
94 void saveSettings(nlohmann::json& j) const override
95 {
96 if (mTempSettings) {
97 j["MeshRenderSettings"] = *mTempSettings;
98 }
99 }
100
101 void updateToolbarFrames(QToolBar*) override {}
102};
103
104inline void setupMeshRenderSettingsTabs(
105 SettingsDialogData& data,
106 MeshRenderSettings& defaults)
107{
108 auto tempMeshSettings = std::make_shared<MeshRenderSettings>(defaults);
109
110 data.addTab(
111 std::make_shared<MeshRenderSettingsTabImpl>(
112 defaults,
113 tempMeshSettings,
114 "Points",
115 [](MeshRenderSettings& s, QWidget* p) {
116 return new PointsFrame(s, p);
117 },
118 [](MeshRenderSettings& s) {
119 s.setDefaultPointSettingsFromCapability();
120 }));
121
122 data.addTab(
123 std::make_shared<MeshRenderSettingsTabImpl>(
124 defaults,
125 tempMeshSettings,
126 "Surface",
127 [](MeshRenderSettings& s, QWidget* p) {
128 return new SurfaceFrame(s, p);
129 },
130 [](MeshRenderSettings& s) {
131 s.setDefaultSurfaceSettingsFromCapability();
132 }));
133
134 data.addTab(
135 std::make_shared<MeshRenderSettingsTabImpl>(
136 defaults,
137 tempMeshSettings,
138 "Wireframe",
139 [](MeshRenderSettings& s, QWidget* p) {
140 return new WireframeFrame(s, p);
141 },
142 [](MeshRenderSettings& s) {
143 s.setDefaultWireframeSettingsFromCapability();
144 }));
145
146 data.addTab(
147 std::make_shared<MeshRenderSettingsTabImpl>(
148 defaults,
149 tempMeshSettings,
150 "Edges",
151 [](MeshRenderSettings& s, QWidget* p) {
152 return new EdgesFrame(s, p);
153 },
154 [](MeshRenderSettings& s) {
155 s.setDefaultEdgeSettingsFromCapability();
156 }));
157}
