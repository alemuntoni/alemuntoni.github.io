33template<ElemId ELEM_ID, FaceMeshConcept MeshType>
34void clearElementAdjacentFaces(MeshType& mesh)
35{
36 requireElementComponent<ELEM_ID, CompId::ADJACENT_FACES>(mesh);
37
38 using AdjacentFacesType = comp::ComponentTypeT<
39 CompId::ADJACENT_FACES,
40 typename MeshType::template ElementType<ELEM_ID>::Components>;
41
42 for (auto e : mesh.template elements<ELEM_ID>()) {
43 if constexpr (comp::IsTiedToVertexCount<AdjacentFacesType>) {
44 for (uint i = 0; i < e.adjFaceCount(); ++i) {
45 e.setAdjFace(i, MeshType::NULL_INDEX);
46 }
47 }
48 else {
49 e.clearAdjFaces();
50 }
51 }
52}

182template<MeshConcept MeshType>
183void updateVertexAdjacentVertices(MeshType& mesh, bool includeOnlyFaces = false)
184{
185 clearVertexAdjacentVertices(mesh);
186
187 using IndexType = MeshType::IndexType;
188
189 if constexpr (FaceMeshConcept<MeshType>) {
190 // vector that contains edges sorted trough unordered vertex pointers
191 // it contains clusters of "same" edges, but each one of them has its
192 // face pointer note that in case on non-manifold mesh, clusters may be
193 // of size
194 // >= 2
195 std::vector<MeshEdgeUtil<MeshType>> vec =
196 fillAndSortMeshEdgeUtilVector(mesh);
197
198 // store the last pair of vertices
199 IndexType v1 = MeshType::NULL_INDEX;
200 IndexType v2 = MeshType::NULL_INDEX;
201 for (uint i = 0; i < vec.size(); ++i) {
202 // if this pair is different from the last pair
203 if (vec[i].v[0] != v1 || vec[i].v[1] != v2) {
204 // update last pair
205 v1 = vec[i].v[0];
206 v2 = vec[i].v[1];
207 mesh.vertex(v1).pushAdjVertex(v2); // set the pair as adjacent
208 mesh.vertex(v2).pushAdjVertex(v1);
209 }
210 }
211 }
212
213 if (!includeOnlyFaces) {
214 if constexpr (EdgeMeshConcept<MeshType>) {
215 for (auto e : mesh.edges()) {
216 // set the pair as adjacent only if not already present
217 if (!e.vertex(0).containsAdjVertex(e.vertex(1)))
218 e.vertex(0).pushAdjVertex(e.vertex(1));
219 if (!e.vertex(1).containsAdjVertex(e.vertex(0)))
220 e.vertex(1).pushAdjVertex(e.vertex(0));
221 }
222 }
223 }
224}

321template<FaceMeshConcept MeshType>
322void updateFaceAdjacentFaces(MeshType& mesh)
323{
324 requireFaceAdjacentFaces(mesh);
325
326 // vector that contains edges sorted trough unordered vertex pointers
327 // it contains clusters of "same" edges, but each one of them has its face
328 // pointer. note that in case on non-manifold mesh, clusters may be of size
329 // >= 2
330 std::vector<MeshEdgeUtil<MeshType>> vec =
331 fillAndSortMeshEdgeUtilVector(mesh);
332
333 if (vec.size() > 0) {
334 // in this loop, base will point to the first element of a cluster of
335 // edges
336 // increment of clusters into loop
337 for (auto base = vec.begin(); base != vec.end();) {
338 auto first = base; // remember the first to set adj to the last
339
340 // i and j will increment together, and if i == j, i will be adj to
341 // j, but j will not be adj to i (to manage non manifold edges and
342 // make cyclic adj on the same edge)
343 auto i = base;
344 auto j = i + 1;
345 if (j == vec.end() || *i != *j) {
346 // case of cluster composed of one element. adj of i is NULL_INDEX
347 mesh.face(i->f).setAdjFace(i->e, MeshType::NULL_INDEX);
348 }
349 else { // at least two edges in the cluster
350 while (j != vec.end() && *i == *j) {
351 mesh.face(i->f).setAdjFace(i->e, j->f);
352 ++i;
353 ++j;
354 }
355 // i now is the last element that was equal to first
356 mesh.face(i->f).setAdjFace(i->e, first->f);
357 }
358
359 // j is the first different edge from first (or it is vec.end()!)
360 base = j;
361 }
362 }
363}

408template<FaceMeshConcept MeshType>
409void updateFaceAdjacentEdges(MeshType& mesh) requires EdgeMeshConcept<MeshType>
410{
411 using FaceType = FaceType<MeshType>;
412
413 using AdjacentEdgesType = comp::ComponentTypeT<
414 CompId::ADJACENT_EDGES,
415 typename FaceType::Components>;
416
417 clearFaceAdjacentEdges(mesh);
418
419 // vector that contains edges sorted trough unordered vertex pointers
420 std::vector<MeshEdgeUtil<MeshType>> vec =
421 fillAndSortMeshEdgeUtilVector(mesh);
422
423 for (auto e : mesh.edges()) {
424 MeshEdgeUtil<MeshType> meu(e.vertexIndex(0), e.vertexIndex(1));
425
426 // binary search for the edge in the sorted vector
427 auto it = std::lower_bound(vec.begin(), vec.end(), meu);
428
429 while (it != vec.end() && *it == meu) {
430 auto f = it->f; // the face adjacent to the edge e
431 if constexpr (comp::IsTiedToVertexCount<AdjacentEdgesType>) {
432 mesh.face(f).setAdjEdge(it->e, e);
433 }
434 else {
435 mesh.face(f).pushAdjEdge(e);
436 }
437 ++it;
438 }
439 }
440}

517template<EdgeMeshConcept MeshType>
518void updateEdgeAdjacentEdges(MeshType& mesh)
519{
520 using EdgeType = EdgeType<MeshType>;
521
522 // for each vertex, save the edges incident in it
523 std::vector<std::list<EdgeType>> edgeMap(mesh.vertexContainerSize());
524
525 clearEdgeAdjacentEdges(mesh);
526
527 for (EdgeType e : mesh.edges()) {
528 edgeMap[e.vertexIndex(0)].push_back(e);
529 edgeMap[e.vertexIndex(1)].push_back(e);
530 }
531
532 for (std::list<EdgeType>& edges : edgeMap) {
533 // all the edges in edges are adjacent to each other
534 for (auto it1 = edges.begin(); it1 != edges.end(); ++it1) {
535 for (auto it2 = std::next(it1); it2 != edges.end(); ++it2) {
536 it1->pushAdjEdge(it2->index());
537 it2->pushAdjEdge(it1->index());
538 }
539 }
540 }
