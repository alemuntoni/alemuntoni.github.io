123 bool isNull() const
124 {
125 for (uint i = 0; i < PointT::DIM; ++i) {

225 bool overlap(const Box<PointT>& b) const
226 {
227 // Check if any dimension of the boxes does not overlap.
228 for (uint i = 0; i < PointT::DIM; ++i) {
229 if (b.mMin[i] >= mMax[i] || b.mMax[i] <= mMin[i])
230 return false;
231 }
232 // All dimensions overlap, the boxes collide.
233 return true;
234 }

285 auto volume() const
286 {
287 // Initialize volume as the width of the first dimension.
288 auto vol = mMax[0] - mMin[0];
289 // Multiply by the width of each dimension.
290 for (uint i = 1; i < PointT::DIM; ++i) {
291 vol *= mMax[i] - mMin[i];
292 }
293 return vol;
294 }

303 auto dim(uint i) const

316 auto minDim() const
317 {
318 auto m = mMax[0] - mMin[0];
319 for (uint i = 1; i < PointT::DIM; ++i) {
320 if (mMax[i] - mMin[i] < m)
321 m = mMax[i] - mMin[i];
322 }
323 return m;
324 }

333 auto maxDim() const
334 {
335 auto m = mMax[0] - mMin[0];
336 for (uint i = 1; i < PointT::DIM; ++i) {
337 if (mMax[i] - mMin[i] > m)
338 m = mMax[i] - mMin[i];
339 }
340 return m;

350 Box<PointT> intersection(const Box<PointT>& b) const
351 {
352 Box<PointT> res = *this;
353 for (uint i = 0; i < PointT::DIM; ++i) {
354 if (mMin[i] < b.mMin[i]) {
355 // set the minimum point to the larger value
356 res.mMin[i] = b.mMin[i];
357 }
358 if (mMax[i] > b.mMax[i]) {
359 // set the maximum point to the smaller value
360 res.mMax[i] = b.mMax[i];
361 }
362 if (res.mMin[i] > res.mMax[i]) {
363 // if the minimum point is larger than the maximum point, the
364 // box is null
365 res.setNull();
366 return res;
367 }
368 }
369 return res;
370 }
371
379 void setNull()
380 {
381 mMin = PointT::max();
382 mMax = PointT::min();
383 }
384
394 void add(const PointT& p)
395 {
396 if (isNull()) {
397 // If the box was null, set both the minimum and maximum points to
398 // the given point
399 *this = Box(p);
400 }
401 else {
402 // Expand the current box to include the added point
403 mMin = vcl::min(mMin, p);
404 mMax = vcl::max(mMax, p);
405 }
406 }
407
420 template<typename Scalar>
421 void add(const PointT& p, Scalar radius)
422 {
423 if (isNull()) {
424 // If the box was null, initialize it with the minimum and maximum
425 // points obtained by subtracting and adding the radius to the given
426 // point
427 *this = Box(p - radius, p + radius);
428 }
429 else {
430 // Expand the current box to include the added point and all points
431 // within the radius around it
432 mMin = vcl::min(mMin, p - radius);
433 mMax = vcl::max(mMax, p + radius);
434 }
435 }
436
445 void add(const Box<PointT>& b)
446 {
447 // Adding a null box should do nothing
448 if (!b.isNull()) {
449 if (isNull()) {
450 *this = b; // if the current box is null, set it equal to b
451 }
452 else {
453 // expand this box to contain the minimum and maximum point of b
454 add(b.mMin);
455 add(b.mMax);
456 }
457 }
458 }

465 void translate(const PointT& p)
466 {
467 mMin += p; // translate the minimum point by p
468 mMax += p; // translate the maximum point by p
469 }

497 bool operator==(const Box<PointT>& b) const
498 {
499 return mMin == b.mMin && mMax == b.mMax;
500 }

508 bool operator!=(const Box<PointT>& b) const { return !(*this == b); }
