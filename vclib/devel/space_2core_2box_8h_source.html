109 bool isNull() const
110 {
111 for (uint i = 0; i < PointT::DIM; ++i) {
112 if (mMin[i] > mMax[i])
113 return true;
114 }

128 {
129 if (isNull())

147 bool isEmpty() const { return mMin == mMax; }

160 bool isInside(const PointT& p) const
161 {
162 for (uint i = 0; i < PointT::DIM; ++i) {
163 if (p[i] < mMin[i] || p[i] > mMax[i])
164 return false;
165 }
166 return true;

180 bool isInsideStrict(const PointT& p) const
181 {
182 for (uint i = 0; i < PointT::DIM; ++i) {
183 if (p[i] <= mMin[i] || p[i] >= mMax[i])
184 return false;
185 }
186 return true;

196 bool isInsideOpenBox(const PointT& p) const
197 {
198 for (uint i = 0; i < PointT::DIM; ++i) {
199 if (p[i] < mMin[i] || p[i] >= mMax[i])
200 return false;
201 }
202 return true;
203 }

211 bool overlap(const Box<PointT>& b) const
212 {
213 // Check if any dimension of the boxes does not overlap.
214 for (uint i = 0; i < PointT::DIM; ++i) {
215 if (b.mMin[i] >= mMax[i] || b.mMax[i] <= mMin[i])
216 return false;
217 }
218 // All dimensions overlap, the boxes collide.
219 return true;
220 }
221
228 bool collide(const Box<PointT>& b) const { return overlap(b); }

236 bool intersects(const Box<PointT>& b) const { return overlap(b); }

243 auto diagonal() const { return mMin.dist(mMax); }
244
249 auto squaredDiagonal() const { return mMin.squaredDist(mMax); }
250
256 PointT center() const { return (mMax + mMin) / 2; }
257
264 PointT size() const { return mMax - mMin; }
265
271 auto volume() const
272 {
273 // Initialize volume as the width of the first dimension.
274 auto vol = mMax[0] - mMin[0];
275 // Multiply by the width of each dimension.
276 for (uint i = 1; i < PointT::DIM; ++i) {
277 vol *= mMax[i] - mMin[i];
278 }
279 return vol;
280 }

289 auto dim(uint i) const

292 return mMax[i] - mMin[i];
293 }

302 auto minDim() const
303 {
304 auto m = mMax[0] - mMin[0];
305 for (uint i = 1; i < PointT::DIM; ++i) {
306 if (mMax[i] - mMin[i] < m)
307 m = mMax[i] - mMin[i];
308 }
309 return m;
310 }

319 auto maxDim() const
320 {
321 auto m = mMax[0] - mMin[0];
322 for (uint i = 1; i < PointT::DIM; ++i) {
323 if (mMax[i] - mMin[i] > m)
324 m = mMax[i] - mMin[i];
325 }
326 return m;

336 Box<PointT> intersection(const Box<PointT>& b) const
337 {
338 Box<PointT> res = *this;
339 for (uint i = 0; i < PointT::DIM; ++i) {
340 if (mMin[i] < b.mMin[i]) {
341 // set the minimum point to the larger value
342 res.mMin[i] = b.mMin[i];
343 }
344 if (mMax[i] > b.mMax[i]) {
345 // set the maximum point to the smaller value
346 res.mMax[i] = b.mMax[i];
347 }
348 if (res.mMin[i] > res.mMax[i]) {
349 // if the minimum point is larger than the maximum point, the
350 // box is null
351 res.setNull();
352 return res;
353 }
354 }
355 return res;
356 }
357
365 void setNull()
366 {
367 mMin = PointT::max();
368 mMax = PointT::min();
369 }

380 void add(const PointT& p)
381 {
382 if (isNull()) {
383 // If the box was null, set both the minimum and maximum points to
384 // the given point
385 *this = Box(p);
386 }
387 else {
388 // Expand the current box to include the added point
389 mMin = vcl::min(mMin, p);
390 mMax = vcl::max(mMax, p);
391 }
392 }
393
406 template<typename Scalar>
407 void add(const PointT& p, Scalar radius)
408 {
409 if (isNull()) {
410 // If the box was null, initialize it with the minimum and maximum
411 // points obtained by subtracting and adding the radius to the given
412 // point
413 *this = Box(p - radius, p + radius);
414 }
415 else {
416 // Expand the current box to include the added point and all points
417 // within the radius around it
418 mMin = vcl::min(mMin, p - radius);
419 mMax = vcl::max(mMax, p + radius);
420 }
421 }
422
431 void add(const Box<PointT>& b)
432 {
433 // Adding a null box should do nothing
434 if (!b.isNull()) {
435 if (isNull()) {
436 *this = b; // if the current box is null, set it equal to b
437 }
438 else {
439 // expand this box to contain the minimum and maximum point of b
440 add(b.mMin);
441 add(b.mMax);
442 }
443 }
444 }
445
451 void translate(const PointT& p)
452 {
453 mMin += p; // translate the minimum point by p
454 mMax += p; // translate the maximum point by p
455 }

483 bool operator==(const Box<PointT>& b) const
484 {
485 return mMin == b.mMin && mMax == b.mMax;
486 }
487
494 bool operator!=(const Box<PointT>& b) const { return !(*this == b); }
