23// Internal helper to load the glTF model once (no duplication).
24namespace detail {
25inline tinygltf::Model loadTinyGltfCameraModel(const std::string& filename)
26{
27 tinygltf::TinyGLTF loader;
28 tinygltf::Model model;
29 std::string err, warn;
30
31 std::string ext = toLower(FileInfo::extension(filename));
32 bool ret = false;
33 if (ext == ".gltf")
34 ret = loader.LoadASCIIFromFile(&model, &err, &warn, filename);
35 else if (ext == ".glb")
36 ret = loader.LoadBinaryFromFile(&model, &err, &warn, filename);
37 else
38 throw UnknownFileFormatException(ext);
39
40 if (!ret)
41 throw std::runtime_error(
42 "Failed to load glTF cameras: " + err +
43 (warn.empty() ? "" : " " + warn));
44 return model;
45}
46} // namespace detail

48// Load all cameras from a glTF file.
49template<CameraConcept CameraType = Camera<float>>
50inline std::vector<CameraType> loadCamerasGltf(const std::string& filename)
51{
52 using Scalar = typename CameraType::ScalarType;
53 tinygltf::Model model = detail::loadTinyGltfCameraModel(filename);
54
55 std::vector<CameraType> cams;
56 cams.reserve(model.cameras.size());
57
58 for (size_t cameraIdx = 0; cameraIdx < model.cameras.size(); ++cameraIdx) {
59 const tinygltf::Camera& gltfCamera = model.cameras[cameraIdx];
60 CameraType camera;
61
62 if (gltfCamera.type == "perspective") {
63 camera.projectionMode() = CameraType::ProjectionMode::PERSPECTIVE;
64 camera.fieldOfView() = gltfCamera.perspective.yfov * 180.0 / M_PI;
65 // TODO check what to do with the aspect ratio
66 camera.aspectRatio() = gltfCamera.perspective.aspectRatio;
67 camera.nearPlane() = gltfCamera.perspective.znear;
68 camera.farPlane() = gltfCamera.perspective.zfar;
69 }
70 else if (gltfCamera.type == "orthographic") {
71 camera.projectionMode() = CameraType::ProjectionMode::ORTHO;
72 camera.verticalHeight() = gltfCamera.orthographic.ymag * 2.0;
73 // TODO check what to do with the aspect ratio
74 camera.aspectRatio() =
75 gltfCamera.orthographic.xmag / gltfCamera.orthographic.ymag;
76 camera.nearPlane() = gltfCamera.orthographic.znear;
77 camera.farPlane() = gltfCamera.orthographic.zfar;
78 }
79
80 bool foundNode = false;
81 for (const auto& node : model.nodes) {
82 if (node.camera == static_cast<int>(cameraIdx)) {
83 // create a vcl matrix44 of double, not typededuced
84 vcl::Matrix44d mat = vcl::Matrix44d::Identity();
85 if (node.matrix.size() == 16) {
86 // copy node matrix to vcl matrix
87 mat = vcl::Matrix44d(node.matrix.data());
88 }
89 else {
90 // construct matrix from TRS
91 vcl::Point3d translation(0, 0, 0);
92 vcl::Quaterniond rotation = vcl::Quaterniond::Identity();
93 vcl::Point3d scale(1, 1, 1);
94 if (node.translation.size() == 3) {
95 translation = vcl::Point3d(
96 node.translation[0],
97 node.translation[1],
98 node.translation[2]);
99 }
100 if (node.rotation.size() == 4) {
101 rotation = vcl::Quaterniond(
102 node.rotation[0],
103 node.rotation[1],
104 node.rotation[2],
105 node.rotation[3]);
106 }
107 if (node.scale.size() == 3) {
108 scale = vcl::Point3d(
109 node.scale[0], node.scale[1], node.scale[2]);
110 }
111 // robut contruction from TRS use affine type then convert
112 // to matrix44
113 vcl::Affine3d affine = vcl::Affine3d::Identity();
114 affine.translate(translation);
115 affine.rotate(rotation);
116 affine.scale(scale);
117 mat = affine.matrix();
118 }
119 // Use mat instead of node.matrix
120 typename CameraType::PointType eye(
121 mat(0, 3), mat(1, 3), mat(2, 3));
122 camera.eye() = eye;
123
124 typename CameraType::PointType forward(
125 -mat(0, 2), -mat(1, 2), -mat(2, 2));
126 forward = forward.normalized();
127
128 typename CameraType::PointType up(
129 mat(0, 1), mat(1, 1), mat(2, 1));
130 camera.up() = up.normalized();
131
132 Scalar distance = 1.0;
133 camera.center() = eye + forward * distance;
134 foundNode = true;
135 break;
136 }
137 }
138
139 if (!foundNode) {
140 // Invalid gltf camera: no node references it.
141 throw std::runtime_error(
142 "Invalid glTF camera: no node references camera index " +
143 std::to_string(cameraIdx));
144 }
145
146 cams.push_back(camera);
147 }
148
149 assert(cams.size() == model.cameras.size());
150 return cams;
151}

153// Single camera loader now delegates to the vector version.
154template<CameraConcept CameraType = Camera<float>>
155inline CameraType loadCameraGltf(
156 const std::string& filename,
157 uint cameraIdx = 0)
158{
159 auto cams = loadCamerasGltf<CameraType>(filename);
160 if (cameraIdx >= cams.size())
161 throw std::out_of_range(
162 "Camera index " + std::to_string(cameraIdx) +
163 " is out of range. Total cameras: " + std::to_string(cams.size()));
164 return cams[cameraIdx];
165}
