97template<Point3Concept PointType, FaceConcept FaceType, typename ScalarType>
98auto boundedDistance(
99 const PointType& p,
100 const FaceType& f,
101 ScalarType maxDist,
102 PointType& closest,
103 bool signedDist = false)
104{
105 if constexpr (TriangleFaceConcept<FaceType>) {
106 TriangleWrapper tw(
107 f.vertex(0).position(),
108 f.vertex(1).position(),
109 f.vertex(2).position());
110
111 return boundedDistance(p, tw, maxDist, closest, signedDist);
112 }
113 else {
114 if (f.vertexCount() == 3) {
115 TriangleWrapper tw(
116 f.vertex(0).position(),
117 f.vertex(1).position(),
118 f.vertex(2).position());
119
120 return boundedDistance(p, tw, maxDist, closest, signedDist);
121 }
122
123 ScalarType minDist = maxDist;
124
125 std::vector<uint> tris = earCut(f);
126 for (uint i = 0; i < tris.size(); i += 3) {
127 PointType w;
128 ScalarType d = boundedDistance(
129 p,
130 TriangleWrapper(
131 f.vertex(tris[i]).position(),
132 f.vertex(tris[i + 1]).position(),
133 f.vertex(tris[i + 2]).position()),
134 minDist,
135 w,
136 signedDist);
137
138 if (std::abs(d) < minDist) {
139 minDist = std::abs(d);
140 closest = w;
141 }
142 }
143
144 return minDist;
145 }
146}
