63template<FaceMeshConcept MeshType, Point3Concept PointType>
64MeshType createTetrahedron(
65 const PointType& p0,
66 const PointType& p1,
67 const PointType& p2,
68 const PointType& p3)
69{
70 using ScalarType = ScalarType<MeshType>;
71
72 MeshType m;
73
74 m.addVertices(
75 p0.template cast<ScalarType>(),
76 p1.template cast<ScalarType>(),
77 p2.template cast<ScalarType>(),
78 p3.template cast<ScalarType>());
79
80 m.reserveFaces(4);
81 m.addFace(0, 1, 2);
82 m.addFace(0, 2, 3);
83 m.addFace(0, 3, 1);
84 m.addFace(3, 2, 1);
85
86 return m;
87}
