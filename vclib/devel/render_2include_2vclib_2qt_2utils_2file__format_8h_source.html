17inline QString filterFormatToQString(const FileFormat& format)
18{
19 QString filter = QString::fromStdString(format.description());
20 filter += " (";
21 for (const auto& ext : format.extensions()) {
22 filter += "*." + QString::fromStdString(ext) + " ";
23 }
24 filter.chop(1);
25 filter += ")";
26 return filter;
27}
28
29inline QString filterFormatsToQString(
30 const std::vector<FileFormat>& formats,
31 bool allFormats = false)
32{
33 QString filter;
34 if (allFormats) {
35 filter += "All supported formats (";
36 for (const auto& format : formats) {
37 for (const auto& ext : format.extensions()) {
38 filter += "*." + QString::fromStdString(ext) + " ";
39 }
40 }
41 filter.chop(1);
42 filter += ");;";
43 }
44 for (const auto& format : formats) {
45 filter += filterFormatToQString(format);
46 filter += ";;";
47 }
48 // remove last two chars
49 filter.chop(2);
50 return filter;
51}
52
53inline FileFormat formatFromQStringFilter(const QString& filter)
54{
55 // find the first "(*."
56 int start = filter.indexOf("(*.") + 3;
57 // find the first ")" after start
58 int end = filter.indexOf(")", start);
59 // extract the extension between start and end
60 QString ext = filter.mid(start, end - start);
61 return FileFormat(ext.toStdString());
62}
