31template<Point3Concept PointType>
32auto halfSpaceDeterminant(
33 const PointType& p0,
34 const PointType& p1,
35 const PointType& p2,
36 const PointType& p)
37{
38 return (p1 - p0).cross(p2 - p0).dot(p - p0);

72template<Point3Concept PointType>
73bool arePointsCollinear(
74 const PointType& p0,
75 const PointType& p1,
76 const PointType& p2)
77{
78 PointType c = (p1 - p0).cross(p2 - p0);
