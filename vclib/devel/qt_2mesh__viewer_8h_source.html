199 template<template<typename> typename EditorT>
200 auto pushEditor(bool active = false)
201 {
202 nlohmann::json j;
203 std::string filePath = mSettingsFilePath;
204 std::ifstream in(filePath);
205 if (in.is_open()) {
206 try {
207 in >> j;
208 }
209 catch (...) {
210 // Ignore parse errors and reset to empty object to avoid
211 // crashes
212 j = nlohmann::json::object();
213 }
214 }
215
216 auto editor = viewer().template pushEditor<EditorT>(active, j);
217
218 if constexpr (std::is_same_v<
219 EditorT<ViewerType>,
220 vcl::MeshSelectorEditor<ViewerType>>) {
221 editor->setOnObjectSelectedFunction([this](uint id) {
222 drawableObjectVectorTree().setSelectedItem(id);
223 });
224 }
225
226 // Store editor for settings dialog if it has a settings frame
227 using SettingsFrame =
228 typename EditorFrameTraits<EditorT, ViewerType>::SettingsFrameType;
229 if constexpr (!std::is_same_v<SettingsFrame, void>) {
230 mSettingsData.addTab(
231 std::make_shared<
232 EditorSettingsTabImpl<EditorT<ViewerType>, SettingsFrame>>(
233 editor, QString::fromStdString(editor->name())));
234 }
235
236 using ToolbarFrameType =
237 typename EditorFrameTraits<EditorT, ViewerType>::ToolbarFrameType;
238 if constexpr (!std::is_same_v<ToolbarFrameType, void>) {
239 addEditorFrame(new ToolbarFrameType(editor));
240 }
241
242 return editor;
243 }

320private:
321 void applyDefaultMeshRenderSettings(auto& obj)
322 {
323 using ValType = std::remove_cvref_t<decltype(obj)>;
324 if constexpr (std::is_convertible_v<
325 ValType,
326 std::shared_ptr<vcl::DrawableObject>>) {
327 auto mesh =
328 std::dynamic_pointer_cast<vcl::AbstractDrawableMesh>(obj);
329 if (mesh) {
330 MeshRenderSettings rs = mesh->renderSettings();
331 rs.updateIfCapable(mDefaultMeshRenderSettings);
332 mesh->setRenderSettings(rs);
333 }
334 }
335 else if constexpr (std::is_base_of_v<
336 vcl::AbstractDrawableMesh,
337 ValType>) {
338 auto rs = obj.renderSettings();
339 rs.updateIfCapable(mDefaultMeshRenderSettings);
340 obj.setRenderSettings(rs);
341 }
342 }
