213 template<template<typename> typename EditorT>
214 auto pushEditor(bool active = false)
215 {
216 nlohmann::json j;
217 std::string filePath = mSettingsFilePath;
218 if (filePath.empty()) {
219 std::filesystem::path configDir = vcl::appConfigDirectory("vclib");
220 filePath = (configDir / vcl::RENDER_SETTINGS_FILE_NAME).string();
221 }
222 std::ifstream in(filePath);
223 if (in.is_open()) {
224 try {
225 in >> j;
226 }
227 catch (...) {
228 // Ignore parse errors and reset to empty object to avoid
229 // crashes
230 j = nlohmann::json::object();
231 }
232 }
233
234 auto editor = viewer().template pushEditor<EditorT>(active, j);
235
236 if constexpr (std::is_same_v<
237 EditorT<ViewerType>,
238 vcl::MeshSelectorEditor<ViewerType>>) {
239 editor->setOnObjectSelectedFunction([this](uint id) {
240 drawableObjectVectorTree().setSelectedItem(id);
241 });
242 }
243
244 // Store editor for settings dialog if it has a settings frame
245 using SettingsFrame =
246 typename EditorFrameTraits<EditorT, ViewerType>::SettingsFrameType;
247 if constexpr (!std::is_same_v<SettingsFrame, void>) {
248 mSettingsData.addTab(
249 std::make_shared<
250 EditorSettingsTabImpl<EditorT<ViewerType>, SettingsFrame>>(
251 editor, QString::fromStdString(editor->name())));
252 }
253
254 using ToolbarFrameType =
255 typename EditorFrameTraits<EditorT, ViewerType>::ToolbarFrameType;
256 if constexpr (!std::is_same_v<ToolbarFrameType, void>) {
257 addEditorFrame(new ToolbarFrameType(editor));
258 }
259
260 return editor;
261 }
