22struct ColorAvgInfo
23{
24 Point4<uint> c;
25 uint cnt = 0;
26};

28template<uint ELEM_ID, MeshConcept MeshType>
29void setPerElemColorFromVertexColor(MeshType& m)
30{
31 requirePerVertexColor(m);
32 requirePerElementComponent<ELEM_ID, CompId::VERTEX_REFERENCES>(m);
33 requirePerElementComponent<ELEM_ID, CompId::COLOR>(m);
34
35 for (auto& e : m.template elements<ELEM_ID>()) {
36 Point4<uint> avg(0, 0, 0, 0);
37 for (const auto* v : e.vertices()) {
38 avg += v->color().template cast<uint>();
39 }
40 avg /= e.vertexCount();
41 e.color() = avg.template cast<uint8_t>();
42 }
43}
44
45template<uint ELEM_ID, MeshConcept MeshType>
46void setPerVertexColorFromElemColor(MeshType& m)
47{
48 requirePerVertexColor(m);
49 requirePerElementComponent<ELEM_ID, CompId::VERTEX_REFERENCES>(m);
50 requirePerElementComponent<ELEM_ID, CompId::COLOR>(m);
51
52 std::vector<ColorAvgInfo> avgColors(m.vertexContainerSize());
53
54 for (const auto& e : m.template elements<ELEM_ID>()) {
55 for (const auto* v : e.vertices()) {
56 avgColors[v->index()].c += e.color().template cast<uint>();
57 avgColors[v->index()].cnt++;
58 }
59 }
60
61 for (auto& v : m.vertices()) {
62 if (avgColors[v.index()].cnt > 0) {
63 avgColors[v.index()].c /= avgColors[v.index()].cnt;
64 v.color() = avgColors[v.index()].c.template cast<uint8_t>();
65 }
66 }
67}

69template<uint ELEM_ID, MeshConcept MeshType>
70void setPerElemColorFromMaterial(MeshType& m)
71{
72 static_assert(
73 HasMaterials<MeshType>,
74 "The input Mesh must have the Materials component.");
75 requirePerElementComponent<ELEM_ID, CompId::COLOR>(m);
76 requirePerElementComponent<ELEM_ID, CompId::MATERIAL_INDEX>(m);
77
78 for (auto& e : m.template elements<ELEM_ID>()) {
79 uint matIndex = e.materialIndex();
80 if (matIndex < m.materialCount()) {
81 e.color() = m.materials()[matIndex].baseColor();
82 }
83 }
84}

279template<MeshConcept MeshType>
280void setPerVertexColorFromQuality(
281 MeshType& m,
282 Color::ColorMap colorMap = Color::ColorMap::RedBlue,
283 typename MeshType::VertexType::QualityType minQuality = 0,
284 typename MeshType::VertexType::QualityType maxQuality = 0)
285{
286 requirePerVertexColor(m);
287 requirePerVertexQuality(m);
288
289 using VertexType = MeshType::VertexType;
290 using QualityType = VertexType::QualityType;
291
292 if (minQuality == maxQuality) {
293 std::pair<QualityType, QualityType> pair = vertexQualityMinMax(m);
294 minQuality = pair.first;
295 maxQuality = pair.second;
296 }
297 for (VertexType& v : m.vertices()) {
298 v.color() =
299 colorFromInterval(minQuality, maxQuality, v.quality(), colorMap);
300 }
301}

421template<FaceMeshConcept MeshType>
422void setPerVertexColorFromFaceBorderFlag(
423 MeshType& m,
424 Color borderColor = Color::Blue,
425 Color internalColor = Color::White,
426 Color mixColor = Color::Cyan)
427{
428 requirePerVertexColor(m);
429
430 using FaceType = MeshType::FaceType;
431
432 const Color baseColor = Color::Green;
433
434 setPerVertexColor(m, baseColor);
435
436 for (FaceType& f : m.faces()) {
437 for (uint i = 0; i < f.vertexCount(); ++i) {
438 if (f.edgeOnBorder(i)) {
439 if (f.vertex(i)->color() == baseColor)
440 f.vertex(i)->color() = borderColor;
441 if (f.vertex(i)->color() == internalColor)
442 f.vertex(i)->color() = mixColor;
443 if (f.vertexMod(i + 1)->color() == baseColor)
444 f.vertexMod(i + 1)->color() = borderColor;
445 if (f.vertexMod(i + 1)->color() == internalColor)
446 f.vertexMod(i + 1)->color() = mixColor;
447 }
448 else {
449 if (f.vertex(i)->color() == baseColor)
450 f.vertex(i)->color() = internalColor;
451 if (f.vertex(i)->color() == borderColor)
452 f.vertex(i)->color() = mixColor;
453 if (f.vertexMod(i + 1)->color() == baseColor)
454 f.vertexMod(i + 1)->color() = internalColor;
455 if (f.vertexMod(i + 1)->color() == borderColor)
456 f.vertexMod(i + 1)->color() = mixColor;
457 }
458 }
459 }

547template<FaceMeshConcept MeshType>
548void setPerFaceColorScattering(
549 MeshType& m,
550 uint nColors = 50,
551 bool checkFauxEdges = true)
552{
553 requirePerFaceColor(m);
554
555 using FaceType = MeshType::FaceType;
556
557 Color baseColor = Color::Black;
558 setPerFaceColor(m, baseColor);
559
560 std::vector<Color> vc = colorScattering(nColors);
561
562 for (FaceType& f : m.faces()) {
563 if (f.color() == baseColor) {
564 f.color() = vc[m.index(f) % nColors];
565 }
566 if constexpr (HasPerFaceAdjacentFaces<MeshType>) {
567 if (checkFauxEdges && isPerFaceAdjacentFacesAvailable(m)) {
568 for (uint i = 0; i < f.vertexCount(); ++i) {
569 if (f.edgeFaux(i)) {
570 assert(f.adjFace(i) != nullptr);
571 f.adjFace(i)->color() = f.color();
572 }
573 }
574 }
575 }
576 }

600template<MeshConcept MeshType, Point3Concept PointType>
601void setPerVertexColorPerlinNoise(
602 MeshType& m,
603 const PointType& period,
604 const PointType& offset = PointType(0, 0, 0),
605 bool onSelected = false)
606{
607 requirePerVertexColor(m);
608
609 using VertexType = MeshType::VertexType;
610
611 PointType p[3];
612
613 for (VertexType& v : m.vertices()) {
614 if (!onSelected || v.selected()) {
615 p[0] = (v.position() / period[0]) + offset;
616 p[1] = (v.position() / period[1]) + offset;
617 p[2] = (v.position() / period[2]) + offset;
618 v.color() = Color(
619 127 + 128.0 * perlinNoise(p[0][0], p[0][1], p[0][2]),
620 127 + 128.0 * perlinNoise(p[1][0], p[1][1], p[1][2]),
621 127 + 128.0 * perlinNoise(p[2][0], p[2][1], p[2][2]),
622 255);
623 }
624 }
625}

645template<MeshConcept MeshType, Point3Concept PointType>
646void setPerVertexPerlinColor(
647 MeshType& m,
648 double period,
649 const PointType& offset = PointType(0, 0, 0),
650 const Color& color1 = Color::Black,
651 const Color& color2 = Color::White,
652 bool onSelected = false)
653{
654 requirePerVertexColor(m);
655
656 using VertexType = MeshType::VertexType;
657
658 for (VertexType& v : m.vertices()) {
659 if (!onSelected || v.selected()) {
660 PointType p = v.position() / period + offset;
661
662 double factor = (perlinNoise(p[0], p[1], p[2]) + 1.0) / 2.0;
663
664 int rr = (color1[0] * factor) + (color2[0] * (1.0 - factor));
665 int gg = (color1[1] * factor) + (color2[1] * (1.0 - factor));
666 int bb = (color1[2] * factor) + (color2[2] * (1.0 - factor));
667 int aa = (color1[3] * factor) + (color2[3] * (1.0 - factor));
668
669 v.color() = Color(rr, gg, bb, aa);
670 }
671 }
