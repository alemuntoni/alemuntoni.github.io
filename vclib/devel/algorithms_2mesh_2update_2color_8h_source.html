23struct ColorAvgInfo
24{
25 Point4<uint> c;
26 uint cnt = 0;
27};

29template<ElemId ELEM_ID, MeshConcept MeshType>
30void setElemColorFromVertexColor(MeshType& m)
31{
32 requireVertexColor(m);
33 requireElementComponent<ELEM_ID, CompId::VERTEX_REFERENCES>(m);
34 requireElementComponent<ELEM_ID, CompId::COLOR>(m);
35
36 for (auto e : m.template elements<ELEM_ID>()) {
37 Point4<uint> avg(0, 0, 0, 0);
38 for (auto&& v : e.vertices()) {
39 avg += v.color().template cast<uint>();
40 }
41 avg /= e.vertexCount();
42 e.color() = avg.template cast<uint8_t>();
43 }
44}

46template<ElemId ELEM_ID, MeshConcept MeshType>
47void setVertexColorFromElemColor(MeshType& m)
48{
49 requireVertexColor(m);
50 requireElementComponent<ELEM_ID, CompId::VERTEX_REFERENCES>(m);
51 requireElementComponent<ELEM_ID, CompId::COLOR>(m);
52
53 std::vector<ColorAvgInfo> avgColors(m.vertexContainerSize());
54
55 for (const auto& e : m.template elements<ELEM_ID>()) {
56 for (auto&& v : e.vertices()) {
57 avgColors[v.index()].c += e.color().template cast<uint>();
58 avgColors[v.index()].cnt++;
59 }
60 }
61
62 for (auto v : m.vertices()) {
63 if (avgColors[v.index()].cnt > 0) {
64 avgColors[v.index()].c /= avgColors[v.index()].cnt;
65 v.color() = avgColors[v.index()].c.template cast<uint8_t>();
66 }
67 }
68}

70template<ElemId ELEM_ID, MeshConcept MeshType>
71void setElemColorFromMaterial(MeshType& m)
72{
73 static_assert(
74 HasMaterials<MeshType>,
75 "The input Mesh must have the Materials component.");
76 requireElementComponent<ELEM_ID, CompId::COLOR>(m);
77 requireElementComponent<ELEM_ID, CompId::MATERIAL_INDEX>(m);
78
79 for (auto e : m.template elements<ELEM_ID>()) {
80 ushort matIndex = e.materialIndex();
81 if (matIndex < m.materialCount()) {
82 e.color() = m.materials()[matIndex].baseColor();
83 }
84 }
85}

280template<MeshConcept MeshType, typename Q1 = float, typename Q2 = float>
281void setVertexColorFromQuality(
282 MeshType& m,
283 Color::ColorMap colorMap = Color::ColorMap::RedBlue,
284 Q1 minQuality = 0.f,
285 Q2 maxQuality = 0.f)
286{
287 requireVertexColor(m);
288 requireVertexQuality(m);
289
290 if (minQuality == maxQuality) {
291 auto pair = vertexQualityMinMax(m);
292 minQuality = pair.first;
293 maxQuality = pair.second;
294 }
295 for (auto v : m.vertices()) {
296 v.color() =
297 colorFromInterval(minQuality, maxQuality, v.quality(), colorMap);

416template<FaceMeshConcept MeshType>
417void setVertexColorFromFaceBorderFlag(
418 MeshType& m,
419 Color borderColor = Color::Blue,
420 Color internalColor = Color::White,
421 Color mixColor = Color::Cyan)
422{
423 requireVertexColor(m);
424
425 const Color baseColor = Color::Green;
426
427 setVertexColor(m, baseColor);
428
429 for (auto f : m.faces()) {
430 for (uint i = 0; i < f.vertexCount(); ++i) {
431 if (f.edgeOnBorder(i)) {
432 if (f.vertex(i).color() == baseColor)
433 f.vertex(i).color() = borderColor;
434 if (f.vertex(i).color() == internalColor)
435 f.vertex(i).color() = mixColor;
436 if (f.vertexMod(i + 1).color() == baseColor)
437 f.vertexMod(i + 1).color() = borderColor;
438 if (f.vertexMod(i + 1).color() == internalColor)
439 f.vertexMod(i + 1).color() = mixColor;
440 }
441 else {
442 if (f.vertex(i).color() == baseColor)
443 f.vertex(i).color() = internalColor;
444 if (f.vertex(i).color() == borderColor)
445 f.vertex(i).color() = mixColor;
446 if (f.vertexMod(i + 1).color() == baseColor)
447 f.vertexMod(i + 1).color() = internalColor;
448 if (f.vertexMod(i + 1).color() == borderColor)
449 f.vertexMod(i + 1).color() = mixColor;
450 }
451 }
452 }

476 const auto& connectedComponents)
477{
478 requireFaceColor(m);

544template<FaceMeshConcept MeshType>
545void setFaceColorScattering(
546 MeshType& m,
547 uint nColors = 50,
548 bool checkFauxEdges = true)
549{
550 requireFaceColor(m);
551
552 Color baseColor = Color::Black;
553 setFaceColor(m, baseColor);
554
555 std::vector<Color> vc = colorScattering(nColors);
556
557 for (auto f : m.faces()) {
558 if (f.color() == baseColor) {
559 f.color() = vc[f.index() % nColors];
560 }
561 if constexpr (HasFaceAdjacentFaces<MeshType>) {
562 if (checkFauxEdges && m.isFaceAdjacentFacesAvailable()) {
563 for (uint i = 0; i < f.vertexCount(); ++i) {
564 if constexpr (requires { f.edgeFaux(i); }) {
565 if (f.edgeFaux(i)) {
566 assert(f.adjFace(i) != nullptr);
567 f.adjFace(i).color() = f.color();
568 }
569 }
570 }
571 }
572 }
573 }

597template<MeshConcept MeshType>
598void setVertexColorPerlinNoise(
599 MeshType& m,
600 const PositionType<MeshType>& period,
601 const PositionType<MeshType>& offset = PositionType<MeshType>(0, 0, 0),
602 bool onSelected = false)
603{
604 requireVertexColor(m);
605
606 PositionType<MeshType> p[3];
607
608 for (auto v : m.vertices()) {
609 if (!onSelected || v.selected()) {
610 p[0] = (v.position() / period[0]) + offset;
611 p[1] = (v.position() / period[1]) + offset;
612 p[2] = (v.position() / period[2]) + offset;
613 v.color() = Color(
614 127 + 128.0 * perlinNoise(p[0][0], p[0][1], p[0][2]),
615 127 + 128.0 * perlinNoise(p[1][0], p[1][1], p[1][2]),
616 127 + 128.0 * perlinNoise(p[2][0], p[2][1], p[2][2]),
617 255);
618 }
619 }
620}

640template<MeshConcept MeshType>
641void setVertexPerlinColor(
642 MeshType& m,
643 double period,
644 const PositionType<MeshType>& offset = PositionType<MeshType>(0, 0, 0),
645 const Color& color1 = Color::Black,
646 const Color& color2 = Color::White,
647 bool onSelected = false)
648{
649 requireVertexColor(m);
650
651 for (auto v : m.vertices()) {
652 if (!onSelected || v.selected()) {
653 PositionType<MeshType> p = v.position() / period + offset;
654
655 double factor = (perlinNoise(p[0], p[1], p[2]) + 1.0) / 2.0;
656
657 int rr = (color1[0] * factor) + (color2[0] * (1.0 - factor));
658 int gg = (color1[1] * factor) + (color2[1] * (1.0 - factor));
659 int bb = (color1[2] * factor) + (color2[2] * (1.0 - factor));
660 int aa = (color1[3] * factor) + (color2[3] * (1.0 - factor));
661
662 v.color() = Color(rr, gg, bb, aa);
663 }
664 }
