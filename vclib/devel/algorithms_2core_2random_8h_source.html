120template<BoxConcept BoxType>
121BoxType random(
122 DistConfig<typename BoxType::ScalarType> distConf = std::monostate(),
123 RandomConfig config = std::monostate())
124{
125 using PointType = BoxType::PointType;
126
127 return callWithDistribution(distConf, [&](auto&& distFunc) {
128 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
129 PointType minPoint, maxPoint;
130 for (uint i = 0; i < PointType::DIM; ++i) {
131 minPoint[i] = distFunc(gen);
132 maxPoint[i] = distFunc(gen);
133 if (minPoint[i] > maxPoint[i]) {
134 std::swap(minPoint[i], maxPoint[i]);
135 }
136 }
137 return BoxType(minPoint, maxPoint);
138 });
139 });
140}

230template<Point3Concept PointType>
231PointType randomTriangleBarycentricCoordinate(
232 RandomConfig config = std::monostate())
233{
234 using ScalarType = PointType::ScalarType;
235
236 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
237 PointType interp;
238 std::uniform_real_distribution<ScalarType> unif(0, 1);
239
240 interp[1] = unif(gen);
241 interp[2] = unif(gen);
242 if (interp[1] + interp[2] > 1.0) {
243 interp[1] = 1.0 - interp[1];
244 interp[2] = 1.0 - interp[2];
245 }
246
247 interp[0] = 1.0 - (interp[1] + interp[2]);
248 return interp;
249 });
250}

281template<typename ScalarType>
282std::vector<ScalarType> randomPolygonBarycentricCoordinate(
283 uint polySize,
284 RandomConfig config = std::monostate())
285{
286 return callWithRandomGenerator(config, [&](std::mt19937& gen) {
287 std::vector<ScalarType> barCoord(polySize);
288 ScalarType sum = static_cast<ScalarType>(0);
289
290 std::uniform_real_distribution<ScalarType> unif(0, 1);
291
292 for (uint i = 0; i < polySize; ++i) {
293 barCoord[i] = unif(gen);
294 sum += barCoord[i];
295 }
296
297 // Normalize so all weights sum to 1.
298 assert(sum != static_cast<ScalarType>(0));
299 for (uint i = 0; i < polySize; ++i) {
300 barCoord[i] /= sum;
301 }
302 return barCoord;
303 });
304}
