19template<MeshConcept OutMeshType, ElemId ELEM_ID, MeshConcept InMeshType>
20OutMeshType elementMeshFilter(
21 const InMeshType& m,
22 Range auto&& elemFilterRng,
23 bool saveBirthIndicesInCustomComponent = true)
24{
25 using OutElemType = OutMeshType::template ElementType<ELEM_ID>;
26 using IndexType = InMeshType::IndexType;
27
28 std::string ccname = "birth" + elementEnumString<ELEM_ID>();
29
30 OutMeshType res;
31 res.enableSameOptionalComponentsOf(m);
32
33 // enable the custom component birth element
34 if constexpr (comp::HasCustomComponents<OutElemType>) {
35 if (saveBirthIndicesInCustomComponent) {
36 res.template addElementCustomComponent<ELEM_ID, IndexType>(ccname);
37 }
38 }
39
40 for (const auto& [birthV, filter] :
41 std::views::zip(m.template elements<ELEM_ID>(), elemFilterRng)) {
42 if (filter) {
43 auto v = res.template add<ELEM_ID>();
44 // import all the components from the input mesh
45 res.template element<ELEM_ID>(v).importFrom(birthV, false);
46 // set the birth element
47 if constexpr (comp::HasCustomComponents<OutElemType>) {
48 if (saveBirthIndicesInCustomComponent) {
49 res.template element<ELEM_ID>(v)
50 .template customComponent<IndexType>(ccname) =
51 birthV.index();
52 }
53 }
54 }
55 }
56
57 return res;
58}

74template<MeshConcept OutMeshType, ElemId ELEM_ID, MeshConcept InMeshType>
75OutMeshType elementMeshFilterWithVRefs(
76 const InMeshType& m,
77 Range auto&& elemFilterRng,
78 bool saveBirthIndicesInCustomComponent = true)
79{
80 using IndexType = InMeshType::IndexType;
81 using OutIndexType = OutMeshType::IndexType;
82 using OutElemType = OutMeshType::template ElementType<ELEM_ID>;
83
84 std::string ccname = "birth" + elementEnumString<ELEM_ID>();
85
86 OutMeshType res;
87 res.enableSameOptionalComponentsOf(m);
88
89 // enable the custom component birthVertex
90 if constexpr (HasVertexCustomComponents<OutMeshType>) {
91 if (saveBirthIndicesInCustomComponent) {
92 res.template addVertexCustomComponent<IndexType>("birthVertex");
93 }
94 }
95
96 // enable the custom component birth element
97 if constexpr (comp::HasCustomComponents<OutElemType>) {
98 if (saveBirthIndicesInCustomComponent) {
99 res.template addElementCustomComponent<ELEM_ID, IndexType>(ccname);
100 }
101 }
102
103 std::vector<OutIndexType> vertexMapping(
104 m.vertexContainerSize(), OutMeshType::NULL_INDEX);
105
106 for (const auto& [birthF, filter] :
107 std::views::zip(m.template elements<ELEM_ID>(), elemFilterRng)) {
108 if (filter) {
109 std::vector<OutIndexType> verts(
110 birthF.vertexCount(), OutMeshType::NULL_INDEX);
111 uint vi = 0; // incremented with vertices of the element
112 // set all the vertex indices in the verts vector
113 // two cases here:
114 // - the ith vertex of the face has been already added, we need just
115 // to take its id in the out mesh from the vertexMapping vector
116 // - the ith vertex of the face has not been added: we need to add
117 // it and import all its components, and update the
118 // vertexMappingVector
119 for (auto&& v : birthF.vertices()) {
120 IndexType inVIdx = v.index();
121 // the vertex has not already added
122 if (vertexMapping[inVIdx] == OutMeshType::NULL_INDEX) {
123 // add the vertex to the out mesh
124 OutIndexType ov = res.addVertex();
125 // import all the components from the input mesh
126 res.vertex(ov).importFrom(v, false);
127 if constexpr (HasVertexCustomComponents<OutMeshType>) {
128 // set the birth vertex
129 if (saveBirthIndicesInCustomComponent) {
130 res.vertex(ov).template customComponent<IndexType>(
131 "birthVertex") = inVIdx;
132 }
133 }
134 vertexMapping[inVIdx] = ov;
135 verts[vi] = ov;
136 }
137 else {
138 verts[vi] = vertexMapping[inVIdx];
139 }
140 ++vi;
141 }
142
143 // now all the vertices of the elements are in the out mesh, we can
144 // add the actual element
145 auto f = res.template add<ELEM_ID>();
146 // import all the components from the input mesh
147 res.template element<ELEM_ID>(f).importFrom(birthF, false);
148
149 if constexpr (OutElemType::VERTEX_COUNT < 0) {
150 res.template element<ELEM_ID>(f).resizeVertices(verts.size());
151 }
152
153 res.template element<ELEM_ID>(f).setVertices(verts);
154
155 if constexpr (comp::HasCustomComponents<OutElemType>) {
156 // set the birth elements
157 if (saveBirthIndicesInCustomComponent) {
158 res.template element<ELEM_ID>(f)
159 .template customComponent<IndexType>(ccname) =
160 birthF.index();
161 }
162 }
163 }
164 }
165 return res;
166}

565template<EdgeMeshConcept OutMeshType, FaceMeshConcept InMeshType>
566OutMeshType faceEdgeMeshFilter(
567 const InMeshType& m,
568 const std::function<bool(typename InMeshType::ConstFace, uint)>&
569 faceEdgeFilter,
570 bool dontDuplicateEdges = true,
571 bool saveBirthIndicesInCustomComponent = true)
572{
573 using IndexType = InMeshType::IndexType;
574 using OutIndexType = OutMeshType::IndexType;
575
576 OutMeshType res;
577 // todo: enable only vertex same optional components of m
578 res.enableSameOptionalComponentsOf(m);
579
580 // enable the custom component birthVertex
581 if constexpr (HasVertexCustomComponents<OutMeshType>) {
582 if (saveBirthIndicesInCustomComponent) {
583 res.template addVertexCustomComponent<IndexType>("birthVertex");
584 }
585 }
586
587 std::vector<OutIndexType> vertexMapping(
588 m.vertexContainerSize(), OutMeshType::NULL_INDEX);
589
590 std::set<
591 std::pair<OutIndexType, OutIndexType>,
592 UnorderedPairComparator<OutIndexType>>
593 unorderedEdges;
594
595 for (const auto& f : m.faces()) {
596 for (uint ei = 0; ei < f.vertexCount(); ++ei) {
597 if (faceEdgeFilter(f, ei)) {
598 std::array<OutIndexType, 2> verts = {
599 OutMeshType::NULL_INDEX, OutMeshType::NULL_INDEX};
600 for (uint i = 0; i < 2; ++i) {
601 IndexType inVIdx = f.vertexIndexMod(ei + i);
602 if (vertexMapping[inVIdx] == OutMeshType::NULL_INDEX) {
603 OutIndexType ov = res.addVertex();
604 res.vertex(ov).importFrom(f.vertexMod(ei + i), false);
605 if constexpr (HasVertexCustomComponents<OutMeshType>) {
606 if (saveBirthIndicesInCustomComponent) {
607 res.vertex(ov)
608 .template customComponent<IndexType>(
609 "birthVertex") = inVIdx;
610 }
611 }
612 vertexMapping[inVIdx] = ov;
613 verts[i] = ov;
614 }
615 else {
616 verts[i] = vertexMapping[inVIdx];
617 }
618 }
619
620 std::pair<OutIndexType, OutIndexType> ep {verts[0], verts[1]};
621 if (!dontDuplicateEdges || !unorderedEdges.contains(ep)) {
622 OutIndexType e = res.addEdge(verts[0], verts[1]);
623 unorderedEdges.insert(ep);
624 }
625 }
626 }
627 }
628
629 return res;
630}
