52template<MeshConcept MeshType>
53void vertexPositionsToBuffer(
54 const MeshType& mesh,
55 auto* buffer,
56 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
57 uint numRows = UINT_NULL)
58{
59 using namespace detail;
60
61 const uint NUM_ROWS = numRows == UINT_NULL ? mesh.vertexCount() : numRows;
62 for (uint i = 0; const auto& p : mesh.vertices() | views::positions) {
63 at(buffer, i, 0, NUM_ROWS, 3, storage) = p.x();
64 at(buffer, i, 1, NUM_ROWS, 3, storage) = p.y();
65 at(buffer, i, 2, NUM_ROWS, 3, storage) = p.z();
66 ++i;
67 }

88template<MeshConcept MeshType>
89void vertexQuadIndicesToBuffer(const MeshType& mesh, auto* buffer)
90{
91 // creates indices for two triangles (quad) for each vertex
92 //
93 // 2-------3
94 // | \ |
95 // | \ |
96 // | \ |
97 // | \ |
98 // | \ |
99 // 0-------1
100 //
101 // - Triangle 1: [0, 1, 2]
102 // - Triangle 2: [1, 3, 2]
103 //
104 for (uint i = 0; const auto& v : mesh.vertices()) {
105 const uint baseIdx = i * 6;
106 const uint quadIdx = i * 4;
107 buffer[baseIdx + 0] = quadIdx + 0;
108 buffer[baseIdx + 1] = quadIdx + 1;
109 buffer[baseIdx + 2] = quadIdx + 2;
110 buffer[baseIdx + 3] = quadIdx + 1;
111 buffer[baseIdx + 4] = quadIdx + 3;
112 buffer[baseIdx + 5] = quadIdx + 2;
113 ++i;
114 }
115}

203template<FaceMeshConcept MeshType>
204void faceVertexIndicesToBuffer(
205 const MeshType& mesh,
206 auto* buffer,
207 bool getIndicesAsIfContainerCompact = true)
208{
209 const std::vector<uint> vertCompIndices =
210 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
211
212 // lambda to get the vertex index of a face (considering compact indices)
213 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
214
215 for (uint i = 0; const auto& f : mesh.faces()) {
216 for (uint j = 0; j < f.vertexCount(); ++j) {
217 buffer[i] = vIndex(f, j);
218 ++i;
219 }
220 }

343template<FaceMeshConcept MeshType>
344void triangulatedFaceVertexIndicesToBuffer(
345 const MeshType& mesh,
346 auto* buffer,
347 TriPolyIndexBiMap& indexMap = detail::indexMap,
348 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
349 uint numTriangles = UINT_NULL,
350 bool getIndicesAsIfContainerCompact = true)
351{
352 using namespace detail;
353
354 const std::vector<uint> vertCompIndices =
355 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
356
357 // lambda to get the vertex index of a face (considering compact indices)
358 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
359
360 // there will be at least a triangle for each polygon
361 indexMap.clear();
362 indexMap.reserve(mesh.faceCount(), mesh.faceContainerSize());
363
364 if constexpr (TriangleMeshConcept<MeshType>) {
365 // construct the indexMap, which maps each triangle to the face index
366 for (uint t = 0; const auto& f : mesh.faces()) {
367 // map the ith triangle to the f face
368 indexMap.insert(t, f.index());
369 ++t;
370 }
371
372 return faceVertexIndicesToBuffer(
373 mesh, buffer, 3, storage, getIndicesAsIfContainerCompact);
374 }
375 else {
376 // if the user did not give the number of triangles, and the buffer
377 // storage is column major, we need to compute the number of resulting
378 // triangles
379 if (numTriangles == UINT_NULL &&
380 storage == MatrixStorageType::COLUMN_MAJOR &&
381 mesh.faceCount() > 0) {
382 numTriangles = triangulatedFaceCount(mesh);
383 }
384 for (uint t = 0; const auto& f : mesh.faces()) {
385 std::vector<uint> vind = vcl::earCut(f);
386
387 // for each triangle of the triangulation (t is the triangle index)
388 for (uint vi = 0; vi < vind.size(); vi += 3) {
389 // map the t-th triangle to the f polygonal face
390 indexMap.insert(t, f.index());
391
392 for (uint k = 0; k < 3; ++k)
393 at(buffer, t, k, numTriangles, 3, storage) =
394 vIndex(f, vind[vi + k]);
395
396 ++t;
397 }
398 }
399 }
400
401 if (&indexMap == &detail::indexMap) {
402 // if the user did not give an indexMap, we clear it at the end of the
403 // function to keep the global one empty for the next call.
404 indexMap.clear();
405 }
406}

495void wireframeVertexIndicesToBuffer(
496 const MeshType& mesh,
497 auto* buffer,
498 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR,
499 bool getIndicesAsIfContainerCompact = true,
500 uint numRows = UINT_NULL)
501{
502 using namespace detail;
503
504 const std::vector<uint> vertCompIndices =
505 detail::vertCompactIndices(mesh, getIndicesAsIfContainerCompact);
506
507 // lambda to get the vertex index of a edge (considering compact indices)
508 auto vIndex = detail::vIndexLambda(mesh, vertCompIndices);
509
510 const uint NUM_ROWS =
511 numRows == UINT_NULL ? faceVertexReferencesCount(mesh) : numRows;
512
513 for (uint i = 0; const auto& f : mesh.faces()) {
514 for (uint j = 0; j < f.vertexCount(); ++j) {
515 uint v0 = vIndex(f, j);
516 uint v1 = vIndex(f, (j + 1) % f.vertexCount());
517
518 at(buffer, i, 0, NUM_ROWS, 2, storage) = v0;
519 at(buffer, i, 1, NUM_ROWS, 2, storage) = v1;
520
521 ++i;
522 }

1628 buffer[index] = 0;

1747void wedgeTexCoordsAsDuplicatedVertexTexCoordsToBuffer(
1748 const MeshType& mesh,
1749 const std::vector<std::pair<vcl::uint, vcl::uint>>& vertWedgeMap,
1750 const std::list<std::list<std::pair<vcl::uint, vcl::uint>>>&
1751 facesToReassign,
1752 auto* buffer,
1753 MatrixStorageType storage = MatrixStorageType::ROW_MAJOR)
1754{
1755 using namespace detail;
1756 using TexCoordType = typename MeshType::FaceType::WedgeTexCoordType;
1757
1758 requirePerFaceWedgeTexCoords(mesh);
1759
1760 const uint NUM_ROWS = mesh.vertexCount() + facesToReassign.size();
1761
1762 // first export the texcoords of the non-duplicated vertices, using the
1763 // vertWedgeMap to get the texcoord index in the face
1764 uint vi = 0; // current vertex (or current row in the matrix)
1765 for (const auto& v : mesh.vertices()) {
1766 TexCoordType w;
1767 uint fInd = vertWedgeMap[vi].first;
1768 uint wInd = vertWedgeMap[vi].second;
1769
1770 // check if the vi is referenced
1771 if (fInd != UINT_NULL && wInd != UINT_NULL) {
1772 w = mesh.face(fInd).wedgeTexCoord(wInd);
1773 }
1774 at(buffer, vi, 0, NUM_ROWS, 2, storage) = w.u();
1775 at(buffer, vi, 1, NUM_ROWS, 2, storage) = w.v();
1776
1777 ++vi;
1778 }
1779
1780 // then append the texcoords of the duplicated vertices, that can be found
1781 // by looking into the any of the facesToReassign element lists
1782 for (const auto& list : facesToReassign) {
1783 assert(list.begin() != list.end());
1784 const auto& p = list.front();
1785 uint fInd = p.first;
1786 uint wInd = p.second;
1787
1788 const auto& w = mesh.face(fInd).wedgeTexCoord(wInd);
1789 at(buffer, vi, 0, NUM_ROWS, 2, storage) = w.u();
1790 at(buffer, vi, 1, NUM_ROWS, 2, storage) = w.v();
