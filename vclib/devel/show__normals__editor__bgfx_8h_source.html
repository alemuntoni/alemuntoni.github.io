92 void updateNormalLines()
93 {
94 const uint currentSize = mVertexNormalsLines.size();
95 const uint targetSize = Base::drawList()->size();
96
97 // 1. Shrink if objects were removed
98 if (currentSize > targetSize) {
99 mVertexNormalsLines.resize(targetSize);
100 mFaceNormalsLines.resize(targetSize);
101 }
102
103 // 2. Incremental update for existing objects: compare linesCount()
104 // against expected vertex/face count. If matched, existing GPU buffers
105 // are preserved without churn.
106 const uint commonCount = std::min(currentSize, targetSize);
107 for (uint i = 0; i < commonCount; ++i) {
108 const auto& drawable = Base::drawList()->at(i);
109 const AbstractDrawableMesh* m =
110 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
111
112 if (m) {
113 const auto& provider = m->meshProvider();
114 uint expectedVLines =
115 provider.hasVertexNormals() ? provider.vertexCount() : 0;
116 uint expectedFLines =
117 provider.hasFaceNormals() ? provider.faceCount() : 0;
118
119 if (mVertexNormalsLines[i].linesCount() != expectedVLines) {
120 mVertexNormalsLines[i] = vertexNormalLines(provider);
121 }
122 if (mFaceNormalsLines[i].linesCount() != expectedFLines) {
123 mFaceNormalsLines[i] = faceNormalLines(provider);
124 }
125 }
126 else {
127 if (mVertexNormalsLines[i].linesCount() > 0) {
128 mVertexNormalsLines[i] = DrawableLines();
129 }
130 if (mFaceNormalsLines[i].linesCount() > 0) {
131 mFaceNormalsLines[i] = DrawableLines();
132 }
133 }
134 }
135
136 // 3. Append normal lines for newly added objects
137 mVertexNormalsLines.reserve(targetSize);
138 mFaceNormalsLines.reserve(targetSize);
139
140 for (uint i = commonCount; i < targetSize; ++i) {
141 const auto& drawable = Base::drawList()->at(i);
142 const AbstractDrawableMesh* m =
143 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
144 if (m) {
145 const auto& provider = m->meshProvider();
146 mVertexNormalsLines.push_back(vertexNormalLines(provider));
147 mFaceNormalsLines.push_back(faceNormalLines(provider));
148 }
149 else {
150 mVertexNormalsLines.emplace_back();
151 mFaceNormalsLines.emplace_back();
152 }
153 }
154
155 mDirty = false;
156 refreshSettings();
157 }
158
159 void refreshSettings() override
160 {
161 uint i = 0;
162 for (const auto& drawable : *Base::drawList()) {
163 const AbstractDrawableMesh* m =
164 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
165 if (m && i < mVertexNormalsLines.size()) {
166 double diag = m->boundingBox().diagonal();
167 double lineLength = diag * mSettings.normalLengthRatio;
168
169 mVertexNormalsLines[i].setGeneralColor(
170 mSettings.vertexNormalColor);
171 mVertexNormalsLines[i].setWidth(mSettings.thickness);
172 mVertexNormalsLines[i].setVectorLength(lineLength);
173
174 mFaceNormalsLines[i].setGeneralColor(mSettings.faceNormalColor);
175 mFaceNormalsLines[i].setWidth(mSettings.thickness);
176 mFaceNormalsLines[i].setVectorLength(lineLength);
177 }
178 i++;
179 }
180 Base::viewerUpdate();
181 }

183 void drawContent(uint viewId) override
184 {
185 // Ensure normal lines are synchronized before drawing
186 if (mDirty) {
187 updateNormalLines();
188 }
189
190 using enum EditorSettings::EditMode;
191
192 DrawObjectSettings settings;
193 settings.viewId = viewId;
194
195 if (mSettings.editMode == NONE)
196 return;
197
198 if (mSettings.editMode == CURRENT_OBJECT) {
199 uint id = Base::drawList()->selectedObjectId();
200 if (id < mVertexNormalsLines.size()) {
201 auto m = dynamic_cast<AbstractDrawableMesh*>(
202 Base::drawList()->at(id).get());
203 if (m) {
204 vcl::Matrix44f model =
205 m->meshProvider().transformMatrix().cast<float>();
206 if (mSettings.showVertexNormals) {
207 bgfx::setTransform(model.data());
208 mVertexNormalsLines[id].draw(settings);
209 }
210 if (mSettings.showFaceNormals) {
211 bgfx::setTransform(model.data());
212 mFaceNormalsLines[id].draw(settings);
213 }
214 }
215 }
216 }
217 else {
218 for (uint i = 0; i < mVertexNormalsLines.size(); ++i) {
219 bool show = (mSettings.editMode == VISIBLE_OBJECTS) ?
220 Base::drawList()->at(i)->isVisible() :
221 true;
222
223 if (show) {
224 auto m = dynamic_cast<AbstractDrawableMesh*>(
225 Base::drawList()->at(i).get());
226 if (m) {
227 vcl::Matrix44f model =
228 m->meshProvider().transformMatrix().cast<float>();
229 if (mSettings.showVertexNormals) {
230 bgfx::setTransform(model.data());
231 mVertexNormalsLines[i].draw(settings);
232 }
233 if (mSettings.showFaceNormals) {
234 bgfx::setTransform(model.data());
235 mFaceNormalsLines[i].draw(settings);
236 }
237 }
238 }
239 }
240 }

243private:
244 DrawableLines vertexNormalLines(const AbstractMeshProvider& provider)
245 {
246 DrawableLines vLines;
247 if (provider.hasVertexNormals()) {
248 uint vCount = provider.vertexCount();
249 std::vector<vcl::Point3d> pts;
250 pts.reserve(vCount * 2);
251
252 // Compute normals and lines for each vertex.
253 // Using a zero vector for degenerate normals guarantees
254 // linesCount() == vCount for reliable synchronization without
255 // rendering artifacts.
256 uint cSize = provider.vertexContainerSize();
257 for (uint i = 0; i < cSize; ++i) {
258 if (provider.isVertexDeleted(i)) {
259 continue;
260 }
261 vcl::Point3d pos = provider.vertexPosition(i);
262 vcl::Point3d n = provider.vertexNormal(i);
263 if (n.squaredNorm() > 0) {
264 n.normalize();
265 }
266 else {
267 n = vcl::Point3d(0, 0, 0);
268 }
269 pts.push_back(pos);
270 pts.push_back(n); // Vector direction
271 }
272 vLines.setVertices(pts);
273 vLines.setTopology(DrawableLines::Topology::VECTORS);
274 }
275
276 return vLines;
277 }
278
279 DrawableLines faceNormalLines(const AbstractMeshProvider& provider)
280 {
281 DrawableLines fLines;
282 if (provider.hasFaceNormals()) {
283 uint fCount = provider.faceCount();
284 std::vector<vcl::Point3d> pts;
285 pts.reserve(fCount * 2);
286
287 // Compute normals and lines for each face.
288 // Using a zero vector for degenerate normals guarantees
289 // linesCount() == fCount for reliable synchronization without
290 // rendering artifacts.
291 uint cSize = provider.faceContainerSize();
292 for (uint i = 0; i < cSize; ++i) {
293 if (provider.isFaceDeleted(i)) {
294 continue;
295 }
296 vcl::Point3d barycenter = provider.faceBarycenter(i);
297 vcl::Point3d n = provider.faceNormal(i);
298 if (n.squaredNorm() > 0) {
299 n.normalize();
300 }
301 else {
302 n = vcl::Point3d(0, 0, 0);
303 }
304 pts.push_back(barycenter);
305 pts.push_back(n); // Vector direction
306 }
307 fLines.setVertices(pts);
308 fLines.setTopology(DrawableLines::Topology::VECTORS);
309 }
310
311 return fLines;
312 }
