45template<bool MSB = false>
46class BitVector
47{
48 std::vector<uint8_t> mBytes;
49 std::size_t mSize = 0;
50
51 static constexpr uint bitIndex(uint i)
52 {
53 return MSB ? (7 - (i % 8)) : (i % 8);
54 }
55
56public:
57 BitVector() = default;
58
64 explicit BitVector(std::size_t size, bool value = false) :
65 mBytes((size + 7) / 8, value ? 0xFF : 0x00), mSize(size)
66 {
67 }
68
72 std::size_t size() const { return mSize; }
73
77 bool empty() const { return mSize == 0; }
78
85 void resize(std::size_t count, bool value = false)
86 {
87 mBytes.resize((count + 7) / 8, value ? 0xFF : 0x00);
88 mSize = count;
89 }
90
94 void clear()
95 {
96 mBytes.clear();
97 mSize = 0;
98 }
99
104 void push_back(bool value)
105 {
106 if (mSize % 8 == 0) {
107 mBytes.push_back(0);
108 }
109 BitProxy<uint8_t>(mBytes.back(), bitIndex(mSize)) = value;
110 mSize++;
111 }
112
118 bool operator[](std::size_t i) const
119 {
120 return mBytes[i / 8] & (1 << bitIndex(i));
121 }
122
129 BitProxy<uint8_t> operator[](std::size_t i)
130 {
131 return BitProxy<uint8_t>(mBytes[i / 8], bitIndex(i));
132 }
133
138 const std::vector<uint8_t>& bytes() const { return mBytes; }
139
144 const uint8_t* data() const { return mBytes.data(); }
145
149 std::size_t byteSize() const { return mBytes.size(); }
150
158 void setBytes(
159 const std::vector<uint8_t>& bytes,
160 std::size_t exactSizeInBits = 0)
161 {
162 mBytes = bytes;
163 mSize = exactSizeInBits > 0 ? exactSizeInBits : bytes.size() * 8;
164 }
165
173 void setBytes(std::vector<uint8_t>&& bytes, std::size_t exactSizeInBits = 0)
174 {
175 mBytes = std::move(bytes);
176 mSize = exactSizeInBits > 0 ? exactSizeInBits : bytes.size() * 8;
177 }
178
179 bool operator==(const BitVector& other) const
180 {
181 return mSize == other.mSize && mBytes == other.mBytes;
182 }
183
184 bool operator!=(const BitVector& other) const { return !(*this == other); }
185};
