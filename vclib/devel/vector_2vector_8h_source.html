119 Vector(std::size_t size, const T& value = T())

415 template<Range Rng>
416 void set(Rng&& r) requires InputRange<Rng, T>
417 {
418 if constexpr (N >= 0) {
419 uint n = std::min(
420 N,
421 (int) std::distance(
422 std::ranges::begin(r), std::ranges::end(r)));
423 std::copy_n(std::ranges::begin(r), n, mContainer.begin());
424 }
425 else {
426 mContainer =
427 std::vector<T>(std::ranges::begin(r), std::ranges::end(r));
428 }
429 }
