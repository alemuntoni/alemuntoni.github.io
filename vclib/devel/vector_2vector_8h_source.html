119 Vector(std::size_t size, const T& value = T())
120 {
121 if constexpr (N >= 0) {
122 if (size != N) {
123 throw WrongSizeException(
124 "Vector must have " + std::to_string(N) + " size.");

126 fill(value);

355 template<Range Rng>
356 void set(Rng&& r) requires InputRange<Rng, T>
357 {
358 if constexpr (N >= 0) {
359 uint n = std::min(
360 N,
361 (int) std::distance(
362 std::ranges::begin(r), std::ranges::end(r)));
363 std::copy_n(std::ranges::begin(r), n, mContainer.begin());
364 }
365 else {
366 mContainer =
367 std::vector<T>(std::ranges::begin(r), std::ranges::end(r));
368 }
369 }
