114template<MeshConcept MeshType>
115bool isTriangleMesh(const MeshType& m)
116{
117 if constexpr (HasTriangles<MeshType>) {
118 return true;
119 }
120 else if constexpr (HasFaces<MeshType>) {
121 using F = MeshType::FaceType;
122 for (const F& f : m.faces()) {
123 if (f.vertexCount() != 3)
124 return false;
125 }
126 return true;
127 }
128 else {
129 return false;
130 }
131}

209template<FaceMeshConcept MeshType>
210void requireTriangleMesh(const MeshType& m)
211{
212 if constexpr (!HasTriangles<MeshType>) {
213 for (const auto& f : m.faces()) {
214 if (f.vertexCount() != 3) {
215 throw MissingTriangularRequirementException(
216 "Triangle Mesh Required.");
217 }
218 }
219 }
220}
