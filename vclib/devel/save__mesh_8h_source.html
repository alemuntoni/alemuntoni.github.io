75template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
76void saveMesh(
77 const MeshType& m,
78 const std::string& filename,
79 const SaveSettings& settings,
80 LogType& log = nullLogger)
81{
82 FileFormat ff = FileInfo::fileFormat(filename);
83
84 if (ff == objFileFormat()) {
85 saveObj(m, filename, settings, log);
86 }
87 else if (ff == offFileFormat()) {
88 saveOff(m, filename, settings, log);
89 }
90 else if (ff == plyFileFormat()) {
91 savePly(m, filename, settings, log);
92 }
93 else if (ff == stlFileFormat()) {
94 saveStl(m, filename, settings, log);
95 }
96#ifdef VCLIB_WITH_TINYGLTF
97 else if (ff == gltfFileFormat()) {
98 saveGltf(m, filename, settings, log);
99 }
100#endif
101 else {
102 throw UnknownFileFormatException(ff.extensions().front());
103 }
104}
