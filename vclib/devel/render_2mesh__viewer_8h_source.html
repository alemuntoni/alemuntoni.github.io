74template<MeshConcept... MeshTypes>
75void showOnMeshViewer(
76 int argc,
77 char** argv,
78 MeshViewerConcept auto& viewer,
79 MeshTypes&&... meshes)
80{
81 auto prepare = [](auto&& m) -> decltype(auto) {
82 using MType = std::remove_cvref_t<decltype(m)>;
83 if constexpr (std::is_base_of_v<vcl::AbstractDrawableMesh, MType>) {
84 return std::forward<decltype(m)>(m);
85 }
86 else {
87 return vcl::makeDrawable(std::forward<decltype(m)>(m));
88 }
89 };
90
91 (viewer.pushDrawableObject(prepare(std::forward<MeshTypes>(meshes))), ...);
92
93 viewer.fitScene();
94
95 viewer.showMaximized();
96}

98template<MeshConcept MeshTypes>
99void showOnMeshViewer(
100 int argc,
101 char** argv,
102 MeshViewerConcept auto& viewer,
103 std::vector<MeshTypes>&& meshes,
104 RenderMode renderMode = RenderMode::CLASSIC,
105 const std::string& panorama = "")
106{
107 for (auto&& mesh : meshes) {
108 if constexpr (std::is_base_of_v<vcl::AbstractDrawableMesh, MeshTypes>) {
109 viewer.pushDrawableObject(std::move(mesh));
110 }
111 else {
112 viewer.pushDrawableObject(vcl::makeDrawable(std::move(mesh)));
113 }
114 }
115
116 auto sts = viewer.viewerSettings();
117
118 if (!panorama.empty()) {
119 sts.panoramaPath = panorama;
120 sts.imageBasedLighting = true;
121 sts.renderBackgroundPanorama = true;
122 }
123 sts.renderMode = renderMode;
124 viewer.setViewerSettings(sts);
125
126 viewer.fitScene();
127
128 viewer.showMaximized();
129}

131template<MeshConcept... MeshTypes>
132int showOnMeshViewer(int argc, char** argv, MeshTypes&&... meshes)
133{
134 Application app(argc, argv);
135
136 MeshViewer viewer;
137
138 pushDefaultEditors(viewer);
139
140 showOnMeshViewer(argc, argv, viewer, std::forward<MeshTypes>(meshes)...);
141
142 return app.exec();
143}
