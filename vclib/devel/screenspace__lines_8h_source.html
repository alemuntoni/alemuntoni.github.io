158 template<Range R>
159 requires Point2Concept<std::ranges::range_value_t<R>>
160 void setVertices(R&& verts)
161 {
162 mVerPosCount = std::ranges::size(verts);
163
164 // Move to the nearest multiple of 2 to ensure we have complete line
165 // pairs
166 uint nv = mVerPosCount + (mVerPosCount % 2);
167
168 VertexBuffer vertBuff;
169 auto [buffer, releaseFn] =
170 Context::getAllocatedBufferAndReleaseFn<float>(nv * 2);
171
172 size_t i = 0;
173 for (const auto& v : verts) {
174 buffer[i * 2 + 0] = v.x();
175 buffer[i * 2 + 1] = v.y();
176 ++i;
177 }
178
179 // Use createForCompute to enable SSBO binding for vertex pulling
180 vertBuff.create(
181 buffer,
182 nv,
183 bgfx::Attrib::Position,
184 2,
185 PrimitiveType::FLOAT,
186 releaseFn);
187 mVertexPositions.setOwned(std::move(vertBuff));
188 mIsUpdateProgramNeeded = true;
189 }
190
204 template<Range R>
205 requires std::integral<std::ranges::range_value_t<R>>
206 void setIndices(R&& indices)
207 {
208 mIndexCount = std::ranges::size(indices);
209
210 IndexBuffer indexBuff;
211 auto [buffer, releaseFn] =
212 Context::getAllocatedBufferAndReleaseFn<uint>(mIndexCount);
213
214 size_t i = 0;
215 for (const auto& idx : indices) {
216 buffer[i] = idx;
217 ++i;
218 }
219
220 indexBuff.create(buffer, mIndexCount, releaseFn);
221
222 mIndices.setOwned(std::move(indexBuff));
223 mIsUpdateProgramNeeded = true;
224 }

244
245 uint i = 0;

254 mVerPosCount,

258 true,
259 releaseFn);
