175 template<Range R>
176 requires Point2Concept<std::ranges::range_value_t<R>>
177 void setVertices(R&& verts)
178 {
179 mVerPosCount = std::ranges::size(verts);
180
181 // Move to the nearest multiple of 2 to ensure we have complete line
182 // pairs
183 uint nv = mVerPosCount + (mVerPosCount % 2);
184
185 VertexBuffer vertBuff;
186 auto [buffer, releaseFn] =
187 Context::getAllocatedBufferAndReleaseFn<float>(nv * 2);
188
189 size_t i = 0;
190 for (const auto& v : verts) {
191 buffer[i * 2 + 0] = v.x();
192 buffer[i * 2 + 1] = v.y();
193 ++i;
194 }
195
196 // Use createForCompute to enable SSBO binding for vertex pulling
197 vertBuff.create(
198 buffer,
199 nv,
200 bgfx::Attrib::Position,
201 2,
202 PrimitiveType::FLOAT,
203 releaseFn);
204 mVertexPositions.setOwned(std::move(vertBuff));
205 mIsUpdateProgramNeeded = true;
206 }
207
221 template<Range R>
222 requires std::integral<std::ranges::range_value_t<R>>
223 void setIndices(R&& indices)
224 {
225 mIndexCount = std::ranges::size(indices);
226
227 IndexBuffer indexBuff;
228 auto [buffer, releaseFn] =
229 Context::getAllocatedBufferAndReleaseFn<uint>(mIndexCount);
230
231 size_t i = 0;
232 for (const auto& idx : indices) {
233 buffer[i] = idx;
234 ++i;
235 }
236
237 indexBuff.create(buffer, mIndexCount, releaseFn);
238
239 mIndices.setOwned(std::move(indexBuff));
240 mIsUpdateProgramNeeded = true;
241 }
