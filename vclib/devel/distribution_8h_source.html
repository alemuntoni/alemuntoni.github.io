26template<typename Scalar>
27class Distribution
28{
29 std::multiset<Scalar> mSet;
30
31 Scalar mMin = std::numeric_limits<Scalar>::max();
32 Scalar mMax = std::numeric_limits<Scalar>::lowest();
33
34 Scalar mSum = 0;
35 Scalar mSqrdSum = 0;
36 Scalar mAvg = 0;
37 Scalar mSqrdAvg = 0;
38 Scalar mRMS = 0;

40public:
44 Distribution() = default;
45
49 void clear()
50 {
51 mSet.clear();
52 mMin = std::numeric_limits<Scalar>::max();
53 mMax = std::numeric_limits<Scalar>::lowest();
54 mSum = 0;
55 mSqrdSum = 0;
56 mAvg = 0;
57 mSqrdAvg = 0;
58 mRMS = 0;
59 }
60
65 void add(Scalar v)
66 {
67 mSet.insert(v);
68 if (v < mMin)
69 mMin = v;
70 if (v > mMax)
71 mMax = v;
72 mSum += v;
73 mSqrdSum += v * v;
74 mAvg = mSum / mSet.size();
75 mSqrdAvg = mSqrdSum / mSet.size();
76 mRMS = std::sqrt(mSqrdAvg);
77 }

85 Scalar min() const { return mMin; }
86
93 Scalar max() const { return mMax; }
94
99 uint size() const { return mSet.size(); }
100
105 Scalar sum() const { return mSum; }
106
111 Scalar average() const { return mAvg; }
112
117 Scalar rootMeanSquare() const { return mRMS; }
118
123 Scalar variance() const { return mSqrdAvg - mAvg * mAvg; }
124
129 Scalar standardDeviation() const { return std::sqrt(variance()); }
130
137 Scalar percentile(Scalar perc) const
138 {
139 assert(!mSet.empty());
140 assert(perc >= 0 && perc <= 1);
141
142 int index = mSet.size() * perc - 1;
143 if (index < 0)
144 index = 0;
145 auto it = mSet.begin();
146 std::advance(it, index);
147 return *it;
148 }
149};
