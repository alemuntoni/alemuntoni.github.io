32template<MatrixConcept MatrixType, Point3Concept PointType, typename ScalarType>
33void setTransformMatrixRotation(
34 MatrixType& matrix,
35 PointType axis,
36 const ScalarType& angleRad)
37{
38 ScalarType c = std::cos(angleRad);
39 ScalarType s = std::sin(angleRad);
40 ScalarType q = 1 - c;
41 axis.normalize();
42 matrix(0, 0) = axis[0] * axis[0] * q + c;
43 matrix(0, 1) = axis[0] * axis[1] * q - axis[2] * s;
44 matrix(0, 2) = axis[0] * axis[2] * q + axis[1] * s;
45 matrix(1, 0) = axis[1] * axis[0] * q + axis[2] * s;
46 matrix(1, 1) = axis[1] * axis[1] * q + c;
47 matrix(1, 2) = axis[1] * axis[2] * q - axis[0] * s;
48 matrix(2, 0) = axis[2] * axis[0] * q - axis[1] * s;
49 matrix(2, 1) = axis[2] * axis[1] * q + axis[0] * s;
50 matrix(2, 2) = axis[2] * axis[2] * q + c;
51}

92template<MatrixConcept MatrixType, Point3Concept PointType>
93void setTransformMatrixRotation(
94 MatrixType& matrix,
95 const PointType& fromVector,
96 const PointType& toVector)
97{
98 if (fromVector == toVector) {
99 matrix.block(0, 0, 3, 3).setIdentity();
100 return;
101 }
102 PointType axis = fromVector.cross(toVector);
103 auto angle = std::acos(
104 fromVector.dot(toVector) / (fromVector.norm() * toVector.norm()));
105 setTransformMatrixRotation(matrix, axis, angle);
106}

222template<Matrix33Or44Concept MatrixType>
223void removeScalingFromMatrixInPlace(MatrixType& matrix)
224{
225 using ScalarType = typename MatrixType::Scalar;
226 ScalarType scaleX = std::sqrt(
227 matrix(0, 0) * matrix(0, 0) + matrix(0, 1) * matrix(0, 1) +
228 matrix(0, 2) * matrix(0, 2));
229 ScalarType scaleY = std::sqrt(
230 matrix(1, 0) * matrix(1, 0) + matrix(1, 1) * matrix(1, 1) +
231 matrix(1, 2) * matrix(1, 2));
232 ScalarType scaleZ = std::sqrt(
233 matrix(2, 0) * matrix(2, 0) + matrix(2, 1) * matrix(2, 1) +
234 matrix(2, 2) * matrix(2, 2));
235 for (int i = 0; i < 3; ++i) {
236 matrix(0, i) /= scaleX;
237 matrix(1, i) /= scaleY;
238 matrix(2, i) /= scaleZ;
239 }
240}

355template<BoxConcept BoxType, MatrixConcept MatrixType>
356BoxType transformBox(const BoxType& box, const MatrixType& mat)
357{
358 using PointType = typename BoxType::PointType;
359 using ScalarType = typename PointType::ScalarType;
360
361 auto m = mat.template cast<ScalarType>();
362
363 BoxType result;
364 for (uint i = 0; i < (1 << BoxType::DIM); ++i) {
365 PointType corner;
366 // Construct the i-th corner of the bounding box by using the bits of i
367 // to choose between the min and max coordinates along each dimension.
368 for (uint d = 0; d < BoxType::DIM; ++d) {
369 corner[d] = (i & (1 << d)) ? box.max()[d] : box.min()[d];
370 }
371 corner *= m;
372 result.add(corner);
373 }
374 return result;
375}
