269 template<typename... I>
270 auto data(this auto&& self, I... indices) requires (sizeof...(indices) < N)
271 {
272 constexpr std::size_t n = sizeof...(indices);
273 if constexpr (n == 0) {
274 return self.mVec.data();
275 }
276 else {
277 std::size_t args[] = {static_cast<std::size_t>(indices)...};
278 std::size_t ind = args[0];
279 assert(args[0] < self.mSizes[0]);
280 uint i;
281 for (i = 1; i < n; i++) {
282 assert(args[i] < self.mSizes[i]);
283 ind *= self.mSizes[i];
284 ind += args[i];
285 }
286 for (; i < N; i++) {
287 ind *= self.mSizes[i];
288 }
289 return &self.mVec[ind];
290 }
291 }

387 template<typename... Sizes>
388 void conservativeResize(Sizes... s) requires (sizeof...(s) == N)
389 {
390 std::size_t newSizes[N] = {static_cast<std::size_t>(s)...};
391 std::size_t newTotalSize = 1;
392 for (uint i = 0; i < N; i++)
393 newTotalSize *= newSizes[i];
394 std::vector<T> newVector(newTotalSize);
395
396 for (std::size_t i = 0; i < mVec.size(); i++) {
397 std::array<std::size_t, N> indices = reverseIndex(i);
398 bool outOfBound = false;
399 for (std::size_t j = 0; j < N; j++)
400 if (indices[j] >= newSizes[j] || indices[j] >= mSizes[j])
401 outOfBound = true;
402 if (!outOfBound) {
403 newVector[getIndex(indices.data(), newSizes)] = mVec[i];
404 }
405 }
406
407 for (uint i = 0; i < mSizes.size(); i++) {
408 mSizes[i] = newSizes[i];
409 }
410
411 mVec = std::move(newVector);
412 }

447 Array<T, N - 1> subArray(uint r) const requires (N > 1)
448 {
449 assert(r < mSizes[0]);
450 Array<T, N - 1> sub;
451 size_t size = 1;
452 for (uint i = 0; i < mSizes.size() - 1; i++) {
453 sub.mSizes[i] = mSizes[i + 1];
454 size *= sub.mSizes[i];
455 }
456 sub.mVec = std::vector<T>(
457 mVec.begin() + r * size, mVec.begin() + (r + 1) * size);
458 return sub;
459 }

492 std::size_t getIndex(const std::size_t indices[]) const
493 {
494#ifndef NDEBUG
495 for (uint i = 0; i < N; i++) {
496 assert(indices[i] < mSizes[i]);
497 }
498#endif

502 std::array<std::size_t, N> reverseIndex(uint index)
503 {
504 std::array<std::size_t, N> indices;
505 for (long int i = N - 1; i >= 0; i--) {
506 indices[i] = index % mSizes[i];
507 index /= mSizes[i];
508 }
509 return indices;
510 }

512 static std::size_t getIndex(
513 const std::size_t indices[],
514 const std::size_t sizes[])
515 {
516#ifndef NDEBUG
517 for (uint i = 0; i < N; i++) {
518 assert(indices[i] < sizes[i]);
519 }
520#endif
521 return linearizeIndex(sizes, indices, N);
522 }

524 void initializeNestedLists(NestedInitializerLists<T, N> values)
525 {
526 std::list<std::size_t> szs =
527 NestedInitializerListsProcessor<T, N>::maxDimensionsLevels(values);
528
529 uint i = 0;
530 size_t totalSize = 1;
531 for (std::size_t s : szs) {
532 totalSize *= s;
533 mSizes[i++] = s;
534 }
535 mVec.resize(totalSize);
536
537 typename std::vector<T>::iterator iterator = mVec.begin();
538 NestedInitializerListsProcessor<T, N>::processElements(
539 values, [&iterator](T value) { *(iterator++) = value; }, szs);
540 }
