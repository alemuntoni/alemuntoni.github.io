95template<PointConcept PointType, SegmentConcept SegmentType>
96auto distance(
97 const PointType& point,
98 const SegmentType& segment,
99 PointType& closestPoint) requires (PointType::DIM == SegmentType::DIM)
100{
101 using ScalarType = PointType::ScalarType;
102
103 ScalarType dist;
104
105 PointType dir = segment.direction();
106 ScalarType esn = dir.squaredNorm();
107
108 if (esn < std::numeric_limits<ScalarType>::min()) {
109 closestPoint = segment.midPoint();
110 }
111 else {
112 ScalarType t = ((point - segment.p0()).dot(dir)) / esn;
113 if (t < 0)
114 t = 0;
115 else if (t > 1)
116 t = 1;
117
118 closestPoint = segment.p0() * (1 - t) + segment.p1() * t;
119 }
120 dist = point.dist(closestPoint);
121 return dist;
122}

184template<
185 Point3Concept PointType,
186 Triangle3Concept TriangleType,
187 typename ScalarType>
188auto boundedDistance(
189 const PointType& p,
190 const TriangleType& triangle,
191 ScalarType maxDist,
192 PointType& closest,
193 bool signedDist = false)
194{
195 using TPointType = TriangleType::PointType;
196
197 ScalarType dist;
198
199 // Extract the positions of the vertices of the face.
200 const TPointType& tp0 = triangle.point(0);
201 const TPointType& tp1 = triangle.point(1);
202 const TPointType& tp2 = triangle.point(2);
203
204 // If the face is degenerate (i.e. its normal vector has zero length),
205 // consider it as a segment.
206 if (triangle.normal().norm() == 0) {
207 // Calculate the bounding box of the face.
208 Box3<ScalarType> box = boundingBox(triangle);
209 Segment3<ScalarType> s(box.min(), box.max());
210 // If the diagonal of the bounding box is greater than zero, calculate
211 // the distance between the query point and the segment.
212 if (box.diagonal() > 0) {
213 dist = distance(p, s, closest);
214 }
215 else {
216 // If the diagonal of the bounding box is zero, set the closest
217 // point to be the minimum corner of the bounding box and calculate
218 // the distance between the query point and this point.
219 closest = box.min();
220 dist = p.dist(closest);
221 }
222 }
223 else {
224 // Calculate the distance between the query point and the plane of the
225 // triangle.
226 Plane<ScalarType> fPlane(tp0, triangle.normal());
227 dist = distance(p, fPlane, true);
228
229 if (std::abs(dist) >= maxDist)
230 return std::abs(dist);
231
232 // Project the query point onto the triangle plane to obtain the closest
233 // point on the triangle.
234 closest = p - fPlane.direction() * dist;
235
236 // Calculate the three edges of the triangle.
237 Point3<ScalarType> fEdge[3];
238 fEdge[0] = tp1 - tp0;
239 fEdge[1] = tp2 - tp1;
240 fEdge[2] = tp0 - tp2;
241
242 // Determine the best axis to use for projection by finding the axis
243 // with the largest component of the normal vector.
244 int bestAxis;
245 if (std::abs(triangle.normal()[0]) > std::abs(triangle.normal()[1])) {
246 if (std::abs(triangle.normal()[0]) > std::abs(triangle.normal()[2]))
247 bestAxis = 0;
248 else
249 bestAxis = 2;
250 }
251 else {
252 if (std::abs(triangle.normal()[1]) > std::abs(triangle.normal()[2]))
253 bestAxis = 1; /* 1 > 0 ? 2 */
254 else
255 bestAxis = 2; /* 2 > 1 ? 2 */
256 }
257
258 // Scale the edges by the inverse of the projection direction on the
259 // best axis.
260 ScalarType scaleFactor = 1 / fPlane.direction()[bestAxis];
261 fEdge[0] *= scaleFactor;
262 fEdge[1] *= scaleFactor;
263 fEdge[2] *= scaleFactor;
264
265 // Compute barycentric coordinates of closest point
266 ScalarType b0, b1, b2;
267 int ba = (bestAxis + 2) % 3;
268 int bb = (bestAxis + 1) % 3;
269
270 Segment3<ScalarType> s0(tp1, tp2), s1(tp2, tp0), s2(tp0, tp1);
271 b0 = fEdge[1][bb] * (closest[ba] - tp1[ba]) -
272 fEdge[1][ba] * (closest[bb] - tp1[bb]);
273 if (b0 <= 0) {
274 return distance(p, s0, closest);
275 }
276 b1 = fEdge[2][bb] * (closest[ba] - tp2[ba]) -
277 fEdge[2][ba] * (closest[bb] - tp2[bb]);
278 if (b1 <= 0) {
279 return distance(p, s1, closest);
280 }
281 b2 = fEdge[0][bb] * (closest[ba] - tp0[ba]) -
282 fEdge[0][ba] * (closest[bb] - tp0[bb]);
283 if (b2 <= 0) {
284 return distance(p, s2, closest);
285 }
286
287 const ScalarType EPS = ScalarType(0.000001);
288 ScalarType b = min(b0, b1, b2);
289 if (b < EPS * triangle.area()) {
290 ScalarType bt;
291 if (b == b0)
292 bt = distance(p, s0, closest);
293 else if (b == b1)
294 bt = distance(p, s1, closest);
295 else {
296 assert(b == b2);
297 bt = distance(p, s2, closest);
298 }
299 dist = bt;
300 }
301
302 if (!signedDist)
303 dist = std::abs(dist);
304 }
305
306 return dist;
307}
