19template<MeshConcept MeshType, typename TargetCont, typename SourceCont>
20void checkContainerReferences(const MeshType& mesh)
21{
22 constexpr ElemId TARGET_ELEM_ID = TargetCont::ELEMENT_ID;
23 constexpr ElemId SOURCE_ELEM_ID = SourceCont::ELEMENT_ID;
24 using SourceElem = typename SourceCont::ElementType;
25 using SourceComponents = typename SourceElem::Components;
26
27 auto checkComp = [&]<typename Comp>() {
28 if constexpr (comp::HasReferencesTo<TARGET_ELEM_ID, Comp>) {
29 // Check if component is enabled if optional
30 bool isAvailable = true;
31 if constexpr (comp::IsOptionalComponent<Comp>) {
32 isAvailable = mesh.template isElementComponentAvailable<
33 SOURCE_ELEM_ID,
34 Comp::COMPONENT_ID>();
35 }
36 if (isAvailable) {
37 const auto targetContainerSize =
38 mesh.template containerSize<TARGET_ELEM_ID>();
39
40 for (const auto& el : mesh.template elements<SOURCE_ELEM_ID>()) {
41 const uint count = Comp::refCount(el);
42 for (uint i = 0; i < count; ++i) {
43 const auto ref = Comp::refIndex(el, i);
44 if (ref == MeshType::NULL_INDEX) {
45 if constexpr (Comp::COMPONENT_ID ==
46 CompId::VERTEX_REFERENCES) {
47 throw InconsistentMeshException(
48 elementEnumString<SOURCE_ELEM_ID>() + " " +
49 std::to_string(el.index()) + " at vertex " +
50 std::to_string(i) +
51 " references an unassigned/null vertex (NULL_INDEX).");
52 }
53 }
54 else {
55 if (ref >= targetContainerSize) {
56 throw InconsistentMeshException(
57 elementEnumString<SOURCE_ELEM_ID>() + " " +
58 std::to_string(el.index()) +
59 " has a reference to " +
60 elementEnumString<TARGET_ELEM_ID>() +
61 " out of range (" + std::to_string(ref) +
62 " >= container size " +
63 std::to_string(targetContainerSize) + ").");
64 }
65 if (mesh.template element<TARGET_ELEM_ID>(ref).deleted()) {
66 throw InconsistentMeshException(
67 elementEnumString<SOURCE_ELEM_ID>() + " " +
68 std::to_string(el.index()) +
69 " references a deleted " +
70 elementEnumString<TARGET_ELEM_ID>() +
71 " (index " + std::to_string(ref) + ").");
72 }
73 }
74 }
75 }
76 }
77 }
78 };
79
80 [&]<typename... Comps>(TypeWrapper<Comps...>) {
81 (checkComp.template operator()<Comps>(), ...);
82 }(SourceComponents{});
83}
84
85} // namespace detail

108template<MeshConcept MeshType>
109bool checkMeshReferences(const MeshType& mesh)
110{
111 using Containers = typename MeshType::ElementContainers;
112
113 auto checkTarget = [&]<typename TargetCont>() {
114 auto checkSource = [&]<typename SourceCont>() {
115 detail::checkContainerReferences<MeshType, TargetCont, SourceCont>(
116 mesh);
117 };
118
119 [&]<typename... SConts>(TypeWrapper<SConts...>) {
120 (checkSource.template operator()<SConts>(), ...);
121 }(Containers{});
122 };
123
124 [&]<typename... TConts>(TypeWrapper<TConts...>) {
125 (checkTarget.template operator()<TConts>(), ...);
126 }(Containers{});
127
128 return true;
129}
