20template<typename ViewerDrawer>
21class BoundingBoxEditorBGFX : public Editor<ViewerDrawer>
22{
23 using Base = Editor<ViewerDrawer>;
24
25 std::vector<DrawableBox3> mBoxes;
26
27 BoundingBoxEditorSettings mSettings;
28
29 bool mDirty = true; // Indicates whether GPU box buffers need updating
30
31public:
32 BoundingBoxEditorBGFX() = default;
33
34 std::string name() const override { return "Bounding Box"; }
35
36 void onViewerSet() override
37 {
38 Base::viewerRegisterGlobalAction(
39 "Toggle Bounding Box",
40 {Key::B, {KeyModifier::NO_MODIFIER}},
41 [this]() { this->setActive(!this->isActive()); });
42 }
43

45
46 const BoundingBoxEditorSettings& settings() const override
47 {
48 return mSettings;
49 }
50
51 void loadSettings(const nlohmann::json& j) override
52 {
53 mSettings.loadSettings(j);
54 }
55
56 void saveSettings(nlohmann::json& j) const override
57 {
58 mSettings.saveSettings(j);
59 }
60
61 void setActive(bool active) override
62 {
63 Base::setActive(active);
64 // Lazily allocate/synchronize box buffers only when activated
65 if (active && mDirty) {
66 updateBoxes();
67 }
68 Base::viewerUpdate();
69 }
70
71 void refresh() override
72 {
73 // When inactive, release GPU buffers immediately and defer updates
74 if (!Base::isActive()) {
75 mBoxes.clear();
76 mDirty = true;
77 return;
78 }
79
80 updateBoxes();
81 }
82
83 void updateBoxes()
84 {
85 const uint currentSize = mBoxes.size();
86 const uint targetSize = Base::drawList()->size();
87
88 // 1. Shrink if objects were removed
89 if (currentSize > targetSize) {
90 mBoxes.resize(targetSize);
91 }
92
93 Color c = mSettings.color;
94 float thickness = mSettings.thickness;
95
96 // 2. Incremental update for existing objects (avoids GPU buffer churn
97 // if unchanged)
98 const uint commonCount = std::min(currentSize, targetSize);
99 for (uint i = 0; i < commonCount; ++i) {
100 const auto& drawable = Base::drawList()->at(i);
101 const AbstractDrawableMesh* m =
102 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
103 Box3d targetBox = m ? m->meshProvider().boundingBox() : Box3d();
104 if (mBoxes[i].boundingBox() != targetBox) {
105 mBoxes[i].setBox(targetBox);
106 }
107 }
108
109 // 3. Append bounding boxes for newly added objects
110 mBoxes.reserve(targetSize);
111 for (uint i = commonCount; i < targetSize; ++i) {
112 const auto& drawable = Base::drawList()->at(i);
113 const AbstractDrawableMesh* m =
114 dynamic_cast<const AbstractDrawableMesh*>(drawable.get());
115 if (m) {
116 mBoxes.push_back(DrawableBox3(
117 m->meshProvider().boundingBox(), c, thickness));
118 }
119 else {
120 mBoxes.push_back(DrawableBox3(Box3f(), c, thickness));
121 }
122 }
123
124 mDirty = false;
125 Base::viewerUpdate();
126 }
127
128 void refreshSettings() override
129 {
130 Color c = mSettings.color;
131 float thickness = mSettings.thickness;
132
133 for (auto& b : mBoxes) {
134 b.setColor(c);
135 b.setThickness(thickness);
136 }
137 Base::viewerUpdate();
138 }
139
140 void drawContent(uint viewId) override
141 {
142 // Ensure box buffers are synchronized before drawing
143 if (mDirty) {
144 updateBoxes();
145 }
146
147 using enum EditorSettings::EditMode;
148
149 DrawObjectSettings settings;
150 settings.viewId = viewId;
151
152 if (mSettings.editMode == NONE)
153 return;
154
155 if (mSettings.editMode == CURRENT_OBJECT) {
156 uint id = Base::drawList()->selectedObjectId();
157 if (id < mBoxes.size()) {
158 const AbstractDrawableMesh* m =
159 dynamic_cast<const AbstractDrawableMesh*>(
160 Base::drawList()->at(id).get());
161 if (m) {
162 Matrix44f transform = m->meshProvider()
163 .transformMatrix()
164 .template cast<float>();
165 bgfx::setTransform(transform.data());
166 }
167 mBoxes[id].draw(settings);
168 }
169 }
170 else {
171 for (uint i = 0; i < mBoxes.size(); ++i) {
172 bool show = (mSettings.editMode == VISIBLE_OBJECTS) ?
173 Base::drawList()->at(i)->isVisible() :
174 true;
175
176 if (show) {
177 const AbstractDrawableMesh* m =
178 dynamic_cast<const AbstractDrawableMesh*>(
179 Base::drawList()->at(i).get());
180 if (m) {
181 Matrix44f transform = m->meshProvider()
182 .transformMatrix()
183 .template cast<float>();
184 bgfx::setTransform(transform.data());
185 }
186 mBoxes[i].draw(settings);
187 }
188 }
189 }
190 }
191};
