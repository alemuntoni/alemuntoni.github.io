198 void bindVertexBuffers(const MeshRenderSettings& mrs) const
199 {
200 // TODO: streams cannot be higher than
201 // "bgfx::getCaps()->limits.maxVertexStreams". Buffers must be bound
202 // only if necessary (using MeshRenderSettings) right now, this is
203 // managed only for uvs (per vertex or per wedge, not both).
204 // We MUST be sure that the limit is not exceeded.
205
206 using enum MeshRenderInfo::Surface;
207
208 uint stream = 0;
209
210 // streams MUST be consecutive starting from 0
211 // otherwise on metal it won't work
212 mVertexPositionsBuffer.bind(stream++);
213
214 if (mVertexNormalsBuffer.isValid()) {
215 // bgfx limitation
216 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
217 mVertexNormalsBuffer.bind(stream++);
218 }
219
220 if (mVertexTangentsBuffer.isValid()) {
221 // bgfx limitation
222 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
223 mVertexTangentsBuffer.bind(stream++);
224 }
225
226 if (mVertexColorsBuffer.isValid()) {
227 // bgfx limitation
228 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
229 mVertexColorsBuffer.bind(stream++);
230 }
231
232 if (mVertexUVBuffer.isValid() && mrs.isSurface(COLOR_VERTEX_TEX)) {
233 // bgfx limitation
234 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
235 mVertexUVBuffer.bind(stream++);
236 }
237
238 if (mVertexWedgeUVBuffer.isValid() && mrs.isSurface(COLOR_WEDGE_TEX)) {
239 // bgfx limitation
240 assert(stream < bgfx::getCaps()->limits.maxVertexStreams);
241 mVertexWedgeUVBuffer.bind(stream++);
242 }
243 }

289 uint bindTextures(
290 const MeshRenderSettings& mrs,
291 uint chunkNumber,
292 const MeshType& m) const
293 {
294 uint materialId = Base::materialIndex(mrs, chunkNumber);
295
296 uint boundTextures = 0;
297
298 DrawableMeshUniforms::TextureType tt =
299 DrawableMeshUniforms::TextureType::BASE_COLOR;
300
301 if (materialId != UINT_NULL) {
302 for (uint j = 0; j < N_TEXTURE_TYPES; ++j) {
303 const auto& td = m.material(materialId).textureDescriptor(j);
304 const std::string& path = td.path();
305 if (!path.empty()) {
306 const Texture& tex = mMaterialTextures.at(path);
307 if (tex.isValid()) {
308 uint flags = Texture::samplerFlagsFromTexture(td);
309 tex.bind(
310 boundTextures,
311 sTextureSamplerUniforms[j].get().handle(),
312 flags);
313
314 tt = static_cast<DrawableMeshUniforms::TextureType>(j);
315 DrawableMeshUniforms::setTextureStage(
316 tt, boundTextures);
317 boundTextures++;
318 }
319 }
320 }
321 }
322 return boundTextures;
323 }

325 void updateEdgeSettings(const MeshRenderSettings& mrs)
326 {
327 using enum MeshRenderInfo::Edges;
328 using enum Lines::ColorSetting;
329
330 mEdgeLines.setWidth(mrs.edgesWidth());
331
332 Lines::Shading sh = Lines::Shading::NONE;
333 if (mrs.isEdges(SHADING_SMOOTH)) {
334 sh = Lines::Shading::PER_VERTEX;
335 }
336 else if (mrs.isEdges(SHADING_FLAT)) {
337 sh = Lines::Shading::PER_LINE;
338 }
339 mEdgeLines.setShading(sh);
340
341 if (mrs.isEdges(COLOR_USER)) {
342 mEdgeLines.setGeneralColor(mrs.edgesUserColor());
343 mEdgeLines.setColorSetting(GENERAL);
344 }
345 else if (mrs.isEdges(COLOR_MESH)) {
346 mEdgeLines.setGeneralColor(mMeshColor);
347 mEdgeLines.setColorSetting(GENERAL);
348 }
349 else if (mrs.isEdges(COLOR_VERTEX)) {
350 mEdgeLines.setColorSetting(PER_VERTEX);
351 }
352 else if (mrs.isEdges(COLOR_EDGE)) {
353 mEdgeLines.setColorSetting(PER_LINE);
354 }
355
356 mEdgeLines.setSelectionVisibility(mrs.isEdges(SELECTION));
357 mEdgeLines.setSelectionColor(mrs.edgesSelectionColor());
358 }

670 void setTextures(const MeshType& mesh) // override
671 {
672 // lambda that sets a texture
673 auto setTexture = [&](const Image& img,
674 const std::string& path,
675 bool generateMips) {
676 const uint size = img.width() * img.height();
677 assert(size > 0);
678
679 uint sizeWithMips = bimg::imageGetSize(
680 nullptr,
681 img.width(),
682 img.height(),
683 1,
684 false,
685 generateMips,
686 1,
687 bimg::TextureFormat::RGBA8) /
688 4; // in uints
689
690 uint numMips = 1;
691 if (generateMips)
692 numMips = bimg::imageGetNumMips(
693 bimg::TextureFormat::RGBA8, img.width(), img.height());
694
695 auto [buffer, releaseFn] =
696 Context::getAllocatedBufferAndReleaseFn<uint>(sizeWithMips);
697
698 const uint* tdata = reinterpret_cast<const uint*>(img.data());
699
700 std::copy(tdata, tdata + size, buffer); // mip level 0
701
702 if (numMips > 1) {
703 uint* source = buffer;
704 uint* dest;
705 uint offset = size;
706 for (uint mip = 1; mip < numMips; mip++) {
707 dest = source + offset;
708 uint mipSize = (img.width() >> mip) * (img.height() >> mip);
709 bimg::imageRgba8Downsample2x2(
710 dest, // output location
711 img.width() >> (mip - 1), // input width
712 img.height() >> (mip - 1), // input height
713 1, // depth, always 1 for 2D textures
714 (img.width() >> (mip - 1)) * 4, // input pitch
715 (img.width() >> mip) * 4, // output pitch
716 source // input location
717 );
718 source = dest;
719 offset = mipSize;
720 }
721 }
722
723 uint64_t flags = BGFX_TEXTURE_NONE | BGFX_SAMPLER_NONE;
724
725 if (img.colorSpace() == Image::ColorSpace::SRGB)
726 flags |= BGFX_TEXTURE_SRGB;
727
728 Texture tex;
729 tex.set(
730 buffer,
731 vcl::Point2i(img.width(), img.height()),
732 generateMips,
733 flags,
734 bgfx::TextureFormat::RGBA8,
735 false,
736 releaseFn);
737
738 // at() does not insert if already present, thus safe in parallel
739 mMaterialTextures.at(path) = std::move(tex);
740 };

742 auto loadImageAndSetTexture =
743 [&](const std::pair<std::string, uint>& pathPair) {
744 const std::string& path = pathPair.first;
745
746 uint materialId = pathPair.second / N_TEXTURE_TYPES;
747 uint textureType = pathPair.second % N_TEXTURE_TYPES;
748 // copy the image because it could be not loaded,
749 // and at the end it needs to be mirrored.
750 vcl::Image txtImg = mesh.textureImage(path);
751 if (txtImg.isNull()) { // try to load it just for rendering
752 try {
753 txtImg = vcl::loadImage(mesh.meshBasePath() + path);
754 txtImg.colorSpace() = Material::textureTypeToColorSpace(
755 static_cast<Material::TextureType>(textureType));
756 }
757 catch (...) {
758 // do nothing
759 }
760 if (txtImg.isNull()) {
761 // still null, use a dummy texture
762 txtImg = createCheckBoardImage(512);
763 }
764 }
765
766 // if loading succeeded (or dummy texture has been created)
767 if (!txtImg.isNull()) {
768 const TextureDescriptor& tex =
769 mesh.material(materialId)
770 .textureDescriptor(textureType);
771 using enum TextureDescriptor::MinificationFilter;
772
773 TextureDescriptor::MinificationFilter minFilter =
774 tex.minFilter();
775
776 bool hasMips =
777 minFilter >= NEAREST_MIPMAP_NEAREST ||
778 minFilter == NONE; // default LINEAR_MIPMAP_LINEAR
779
780 txtImg.mirror();
781 setTexture(txtImg, path, hasMips);
782 }
783 };

787 if constexpr (vcl::HasMaterials<MeshType>) {
788 // textures could be missing from the textureImages of the mesh
789 // setting the texture paths in a map - key is the path and value is
790 // an uint where materialIndex and textureType are encoded
791 // map is used to avoid duplicates, then is moved to a vector for
792 // parallel processing
793 std::map<std::string, uint> texturePaths;
794 for (uint i = 0; i < mesh.materialCount(); ++i) {
795 for (uint j = 0; j < N_TEXTURE_TYPES; ++j) {
796 const vcl::TextureDescriptor& td =
797 mesh.material(i).textureDescriptor(j);
798 if (!td.path().empty()) {
799 texturePaths[td.path()] = i * N_TEXTURE_TYPES + j;
800
801 // create a null texture in the map
802 // this is crucial to avoid insertions during the
803 // actual creation, that is done in parallel
804 mMaterialTextures[td.path()] = Texture();
805 }
806 }
807 }
808
809 // move to vector for parallel processing
810 std::vector<std::pair<std::string, uint>> texturePathVec;
811 texturePathVec.reserve(texturePaths.size());
812 for (const auto& tp : texturePaths) {
813 texturePathVec.push_back(tp);
814 }
815
816 parallelFor(texturePathVec, loadImageAndSetTexture);
817 }
818 }
