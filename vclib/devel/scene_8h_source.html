100 template<FaceMeshConcept MeshType>
101 Scene(const MeshType& m)
102 {
103 mDevice = rtcNewDevice(nullptr);
104
105 if (!mDevice)
106 throw std::runtime_error("Error creating Embree device.");
107
108 mScene = rtcNewScene(mDevice);
109
110 if (!mScene) {
111 rtcReleaseDevice(mDevice);
112 throw std::runtime_error("Error creating Embree scene.");
113 }
114
115 RTCGeometry geometry =
116 rtcNewGeometry(mDevice, RTC_GEOMETRY_TYPE_TRIANGLE);
117
118 if (!geometry) {
119 rtcReleaseScene(mScene);
120 rtcReleaseDevice(mDevice);
121 throw std::runtime_error("Error creating Embree geometry.");
122 }
123
124 float* vb = (float*) rtcSetNewGeometryBuffer(
125 geometry,
126 RTC_BUFFER_TYPE_VERTEX,
127 0,
128 RTC_FORMAT_FLOAT3,
129 3 * sizeof(float),
130 m.vertexCount());
131
132 vertexPositionsToBuffer(m, vb);
133
134 uint numTris = vcl::triangulatedFaceCount(m);
135
136 uint* ib = (uint*) rtcSetNewGeometryBuffer(
137 geometry,
138 RTC_BUFFER_TYPE_INDEX,
139 0,
140 RTC_FORMAT_UINT3,
141 3 * sizeof(unsigned),
142 numTris);
143
144 triangulatedFaceVertexIndicesToBuffer(
145 m, ib, mIndexMap, MatrixStorageType::ROW_MAJOR, numTris);
146
147 rtcCommitGeometry(geometry);
148 rtcAttachGeometry(mScene, geometry);
149
150 rtcReleaseGeometry(geometry);
151
152 rtcCommitScene(mScene);
153 }

200 template<typename ScalarType>
201 HitResult firstFaceIntersectedByRay(
202 const Point3<ScalarType>& origin,
203 const Point3<ScalarType>& direction,
204 float tNear = 0.f,
205 float tFar = std::numeric_limits<float>::infinity()) const
206 {
207 if (mScene) {
208 RTCRayHit rayhit = initRayHitValues(origin, direction, tNear, tFar);
209
210 rtcIntersect1(mScene, &rayhit);
211
212 if (rayhit.hit.geomID != RTC_INVALID_GEOMETRY_ID) {
213 uint fID = mIndexMap.polygon(rayhit.hit.primID);
214 float w = 1.f - rayhit.hit.u - rayhit.hit.v;
215 Point3f barycentricCoords(w, rayhit.hit.u, rayhit.hit.v);
216 uint triID = rayhit.hit.primID - mIndexMap.triangleBegin(fID);
217 float t = rayhit.ray.tfar;
218 return std::make_tuple(fID, barycentricCoords, triID, t);
219 }
220 }
221
222 return std::make_tuple(UINT_NULL, Point3f(), uint(0), float(0));
223 }

308 template<RandomAccessRange R1, RandomAccessRange R2>
309 std::vector<HitResult> firstFaceIntersectedByRays(
310 R1&& origins,
311 R2&& directions,
312 float tNear = 0.f,
313 float tFar = std::numeric_limits<float>::infinity()) const
314 requires Point3Concept<std::ranges::range_value_t<R1>> &&
315 Point3Concept<std::ranges::range_value_t<R2>>
316 {
317 using PointType = std::ranges::range_value_t<R1>;
318 using ScalarType = typename PointType::ScalarType;
319
320 assert(
321 std::ranges::size(origins) == std::ranges::size(directions) &&
322 "Origins and directions ranges must have the same size.");
323
324 std::size_t sz = std::ranges::size(origins);
325
326 std::vector<HitResult> results(sz);
327
328 if (mScene) {
329 // some cpus fail to guarantee thread safety for rtcIntersect16
330#ifndef VCL_EMBREE_FORCE_CHUNK_16
331 auto computeRay = [&](uint i) {
332 results[i] = firstFaceIntersectedByRay(
333 origins[i], directions[i], tNear, tFar);
334 };
335
336 std::vector<std::size_t> rayIndices(sz);
337 std::iota(rayIndices.begin(), rayIndices.end(), 0);
338
339 vcl::parallelFor(rayIndices, computeRay);
340#else
341 std::size_t chunks = sz / 16;
342 std::size_t rem = sz % 16;
343
344 if (rem != 0) {
345 ++chunks;
346 }
347
348 auto computeChunk = [&](uint chunk) {
349 std::array<int, 16> validMask;
350 std::ranges::fill(validMask, -1);
351
352 RTCRayHit16 rayHits;
353
354 std::size_t first = chunk * 16;
355 std::size_t last = first + 16;
356 if (last > sz) {
357 for (std::size_t i = sz % 16; i < 16; ++i) {
358 validMask[i] = 0;
359 }
360 last = sz;
361 }
362
363 for (std::size_t i = first; i < last; ++i) {
364 std::size_t idx = i - first;
365 initRayHitsValues(
366 rayHits, idx, origins[i], directions[i], tNear, tFar);
367 }
368
369 rtcIntersect16(validMask.data(), mScene, &rayHits);
370
371 for (std::size_t i = first; i < last; ++i) {
372 std::size_t idx = i - first;
373 if (rayHits.hit.geomID[idx] != RTC_INVALID_GEOMETRY_ID) {
374 uint fID = mIndexMap.polygon(rayHits.hit.primID[idx]);
375 float w = 1.f - rayHits.hit.u[idx] - rayHits.hit.v[idx];
376 Point3f barycentricCoords(
377 w, rayHits.hit.u[idx], rayHits.hit.v[idx]);
378 uint triID = rayHits.hit.primID[idx] -
379 mIndexMap.triangleBegin(fID);
380 float t = rayHits.ray.tfar[idx];
381 results[i] =
382 std::make_tuple(fID, barycentricCoords, triID, t);
383 }
384 else {
385 results[i] = std::make_tuple(
386 UINT_NULL, Point3f(), uint(0), float(0));
387 }
388 }
389 };
390
391 std::vector<std::size_t> chunkIndices(chunks);
392 std::iota(chunkIndices.begin(), chunkIndices.end(), 0);
393
394 vcl::parallelFor(chunkIndices, computeChunk);
395#endif
396 }
397 else {
398 std::fill(
399 results.begin(),
400 results.end(),
401 std::make_tuple(UINT_NULL, Point3f(), uint(0), float(0)));
402 }
403
404 return results;
405 }

430 template<RandomAccessRange R>
431 std::vector<HitResult> firstFaceIntersectedByRays(
432 R&& rays,
433 float tNear = 0.f,
434 float tFar = std::numeric_limits<float>::infinity()) const
435 requires Ray3Concept<std::ranges::range_value_t<R>>
436 {
437 auto getOriginView = [](const auto& r) { return r.origin(); };
438
439 auto getDirectionView = [](const auto& r) { return r.direction(); };
440
441 auto origins = rays | std::views::transform(getOriginView);
442 auto directions = rays | std::views::transform(getDirectionView);
443
444 return firstFaceIntersectedByRays(origins, directions, tNear, tFar);
445 }

466 template<typename ScalarType>
467 std::vector<HitResult> facesIntersectedByRay(
468 const Point3<ScalarType>& origin,
469 const Point3<ScalarType>& direction,
470 float eps) const
471 {
472 std::vector<HitResult> filteredHits;
473
474 if (!mScene || eps <= 0.f) {
475 return filteredHits;
476 }
477
478 constexpr uint MAX_RAY_STEPS = 2048;
479 const float nearAdvanceEps = eps;
480
481 float rayAdvance = eps;
482 float lastTHit = -std::numeric_limits<float>::infinity();
483
484 for (uint step = 0; step < MAX_RAY_STEPS; ++step) {
485 const float originAdvance = (step == 0) ? -eps : rayAdvance;
486 const float rayEps = (step == 0) ? 0.f : eps;
487
488 const Point3<ScalarType> currentOrigin =
489 origin + direction * originAdvance;
490
491 RTCRayHit rayhit =
492 initRayHitValues(currentOrigin, direction, rayEps);
493
494 rtcIntersect1(mScene, &rayhit);
495
496 if (rayhit.hit.geomID == RTC_INVALID_GEOMETRY_ID) {
497 break;
498 }
499
500 const float localTHit = rayhit.ray.tfar;
501 const float tHit = originAdvance + localTHit;
502
503 if (tHit < -eps) {
504 rayAdvance = std::max(eps, tHit + nearAdvanceEps);
505 continue;
506 }
507
508 if (tHit < lastTHit + eps) {
509 break;
510 }
511 lastTHit = tHit;
512
513 uint fID = mIndexMap.polygon(rayhit.hit.primID);
514 float w = 1.f - rayhit.hit.u - rayhit.hit.v;
515 Point3f barycentricCoords(w, rayhit.hit.u, rayhit.hit.v);
516 uint triID = rayhit.hit.primID - mIndexMap.triangleBegin(fID);
517
518 filteredHits.push_back(
519 std::make_tuple(fID, barycentricCoords, triID, tHit));
520
521 rayAdvance = tHit + eps;
522 }
523
524 return filteredHits;
525 }

547 template<typename ScalarType>
548 std::vector<HitResult> facesIntersectedByRay(
549 const Point3<ScalarType>& origin,
550 const Point3<ScalarType>& direction) const
551 {
552 if (!mScene) {
553 return {};
554 }
555
556 RTCBounds bounds;
557 rtcGetSceneBounds(mScene, &bounds);
558
559 const double dx = bounds.upper_x - bounds.lower_x;
560 const double dy = bounds.upper_y - bounds.lower_y;
561 const double dz = bounds.upper_z - bounds.lower_z;
562 const double diagonal = std::sqrt(dx * dx + dy * dy + dz * dz);
563 const double EPS = 1e-6 * diagonal;
564
565 if (!std::isfinite(EPS) || EPS <= 0.0) {
566 return {};
567 }
568
569 return facesIntersectedByRay(
570 origin, direction, static_cast<float>(EPS));
571 }

574 template<Point3Concept PointType>
575 static inline RTCRayHit initRayHitValues(
576 const PointType& origin,
577 const PointType& direction,
578 float tNear = 0.f,
579 float tFar = std::numeric_limits<float>::infinity())
580 {
581 RTCRayHit rayhit;
582 rayhit.ray.org_x = origin.x();
583 rayhit.ray.org_y = origin.y();
584 rayhit.ray.org_z = origin.z();
585 rayhit.ray.dir_x = direction.x();
586 rayhit.ray.dir_y = direction.y();
587 rayhit.ray.dir_z = direction.z();
588 rayhit.ray.tnear = tNear;
589 rayhit.ray.tfar = tFar;
590 rayhit.ray.mask = -1;
591 rayhit.ray.flags = 0;
592 rayhit.hit.geomID = RTC_INVALID_GEOMETRY_ID;
593 rayhit.hit.instID[0] = RTC_INVALID_GEOMETRY_ID;
594 return rayhit;
595 }
