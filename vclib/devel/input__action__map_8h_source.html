39template<StringConvertible Input, StringConvertible ActionID>
40class InputActionMap : public AbstractInputActionMap
41{
42 struct ActionDef
43 {
44 ActionID id;
45 std::string name;
46 std::vector<Input> inputs;
47 std::vector<Input> defaultInputs;
48 };
49
50 std::string mMapName;
51 std::vector<ActionDef> mDefs;
52 std::map<Input, ActionID> mBindings;
53
54public:
55 struct InitData
56 {
57 ActionID id;
58 std::string name;
59 std::vector<Input> defaultInputs = {};
60 };
61
62 InputActionMap(const std::string& name = "") : mMapName(name) {}
63
64 void registerAction(
65 const ActionID& id,
66 const std::string& name,
67 const std::vector<Input>& defaultInputs = {})
68 {
69 for (auto& def : mDefs) {
70 if (def.id == id) {
71 def.name = name;
72 def.defaultInputs = defaultInputs;
73 if (def.inputs.empty() && !defaultInputs.empty()) {
74 def.inputs = defaultInputs;
75 }
76 updateBindings();
77 return;
78 }
79 }
80 mDefs.push_back({id, name, defaultInputs, defaultInputs});
81 updateBindings();
82 }
83
84 void registerActions(std::initializer_list<InitData> actions)
85 {
86 for (const auto& a : actions) {
87 registerAction(a.id, a.name, a.defaultInputs);
88 }
89 }
90
91 void resetToDefaults() override
92 {
93 for (auto& def : mDefs) {
94 def.inputs = def.defaultInputs;
95 }
96 updateBindings();
97 }
98
99 std::string mapName() const override { return mMapName; }
100
101 InputType inputType() const override
102 {
103 if constexpr (
104 std::is_same_v<Input, std::pair<Key::Enum, KeyModifiers>> ||
105 std::is_same_v<Input, Key::Enum>) {
106 return InputType::KEY;
107 }
108 else if constexpr (
109 std::is_same_v<Input, std::pair<MouseButton::Enum, KeyModifiers>> ||
110 std::is_same_v<Input, MouseInput> ||
111 std::is_same_v<Input, MouseButton::Enum>) {
112 return InputType::MOUSE_BUTTON;
113 }
114 else if constexpr (
115 std::is_same_v<Input, std::pair<ScrollAxis::Enum, KeyModifiers>> ||
116 std::is_same_v<Input, ScrollAxis::Enum>) {
117 return InputType::SCROLL_AXIS;
118 }
119 else {
120 return InputType::UNKNOWN;
121 }
122 }
123
124 std::vector<AbstractInputActionMap::ActionInfo> actions() const override
125 {
126 std::vector<AbstractInputActionMap::ActionInfo> res;
127 res.reserve(mDefs.size());
128 for (const auto& def : mDefs) {
129 AbstractInputActionMap::ActionInfo info;
130 info.id = toString(def.id);
131 info.name = def.name;
132 for (const auto& in : def.inputs) {
133 info.inputs.push_back(toString(in));
134 }
135 for (const auto& in : def.defaultInputs) {
136 info.defaultInputs.push_back(toString(in));
137 }
138 res.push_back(info);
139 }
140 return res;
141 }
142
143 void setBindings(
144 const std::string& actionId,
145 const std::vector<std::string>& inputStrs) override
146 {
147 ActionID id;
148 try {
149 id = vcl::fromString<ActionID>(actionId);
150 }
151 catch (...) {
152 return;
153 }
154
155 for (auto& def : mDefs) {
156 if (def.id == id) {
157 def.inputs.clear();
158 for (const auto& inStr : inputStrs) {
159 if (!inStr.empty()) {
160 try {
161 def.inputs.push_back(vcl::fromString<Input>(inStr));
162 }
163 catch (...) {
164 }
165 }
166 }
167 updateBindings();
168 return;
169 }
170 }
171 }
172
173 bool hasBinding(const Input& in) const
174 {
175 return mBindings.find(in) != mBindings.end();
176 }
177
178 std::optional<ActionID> action(const Input& in) const
179 {
180 auto it = mBindings.find(in);
181 if (it != mBindings.end()) {
182 return it->second;
183 }
184 return std::nullopt;
185 }
186
187 std::vector<Input> inputs(const ActionID& id) const
188 {
189 for (const auto& def : mDefs) {
190 if (def.id == id) {
191 return def.inputs;
192 }
193 }
194 return {};
195 }
196
197 void loadSettings(const nlohmann::json& j) override
198 {
199 if (j.contains(mMapName)) {
200 for (const auto& [actionId, jValue] : j[mMapName].items()) {
201 std::vector<std::string> inputStrs;
202 if (jValue.is_array()) {
203 for (const auto& item : jValue) {
204 inputStrs.push_back(item.template get<std::string>());
205 }
206 }
207 else if (jValue.is_string()) {
208 std::string str = jValue.template get<std::string>();
209 if (!str.empty()) {
210 inputStrs.push_back(str);
211 }
212 }
213 setBindings(actionId, inputStrs);
214 }
215 }
216 }
217
218 void saveSettings(nlohmann::json& j) const override
219 {
220 auto& mapJson = j[mMapName];
221 for (const auto& def : mDefs) {
222 std::vector<std::string> strs;
223 for (const auto& in : def.inputs) {
224 strs.push_back(toString(in));
225 }
226 mapJson[toString(def.id)] = strs;
227 }
228 }
229
230private:
231 // rebuilds the reverse (input -> action) lookup table used by action().
232 // Each action can be bound to multiple Inputs; if two actions are bound to
233 // the same Input, the last one in mDefs silently wins the lookup (the UI
234 // is expected to warn about such conflicts, see checkConflicts()).
235 void updateBindings()
236 {
237 mBindings.clear();
238 for (const auto& def : mDefs) {
239 for (const auto& in : def.inputs) {
240 mBindings[in] = def.id;
241 }
242 }
243 }
244};
