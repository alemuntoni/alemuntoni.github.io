256template<typename T>
257concept TriangleConcept =
258 std::derived_from< // same type or derived type
259 std::remove_cvref_t<T>,
260 Triangle<typename RemoveRef<T>::PointType>> ||
261 std::derived_from< // same type or derived type
262 std::remove_cvref_t<T>,
263 TriangleWrapper<typename RemoveRef<T>::PointType>>;
