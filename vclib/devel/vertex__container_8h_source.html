178 template<typename... VC>
179 uint addVertices(const typename T::PositionType& p, const VC&... v)
180 {
181 uint vid = vertexContainerSize();
182 // reserve the new number of vertices
183 reserveVertices(vid + sizeof...(VC) + 1);
184 addVertex(p);
185 // pack expansion: will be translated at compile time as an addVertex()
186 // call for each argument of the addVertices member function
187 (addVertex(v), ...);
188 return vid;
189 }

204 {
205 uint vid = vertexContainerSize();

207 for (const auto& v : range) {
208 addVertex(v);
209 }
210 return vid;
