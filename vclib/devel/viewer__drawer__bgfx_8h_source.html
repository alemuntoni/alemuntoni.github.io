29template<typename DerivedRenderApp>
30class ViewerDrawerBGFX : public AbstractViewerDrawer<DerivedRenderApp>
31{
32 using Base = AbstractViewerDrawer<DerivedRenderApp>;
33 using DRA = DerivedRenderApp;
34
35 bool mStatsEnabled = false;
36
37 DrawableEnvironment mEnvironment;

39public:
40 ViewerDrawerBGFX(uint width = 1024, uint height = 768) : Base(width, height)
41 {
42 this->onResize(width, height);
43 }
44
45 ~ViewerDrawerBGFX() = default;
46
47 void setViewerSettings(const ViewerSettings& settings)
48 {
49 bool panoramaChanged =
50 (Base::viewerSettings().panoramaPath != settings.panoramaPath);
51 Base::setViewerSettings(settings);
52 if (panoramaChanged) {
53 setPanorama(settings.panoramaPath);
54 }
55 }
56
57 std::string panoramaFileName() const
58 {
59 return FileInfo::fileNameWithExtension(
60 Base::viewerSettings().panoramaPath);
61 }
62
63 void setPanorama(const std::string& panorama)
64 {
65 Panorama p(panorama);
66 mEnvironment = DrawableEnvironment(p, Base::canvasViewId());
67 }
68
69 void onResize(uint width, uint height) override
70 {
71 Base::onResize(width, height);
72 }
73
74 void onDrawContent(uint viewId) override
75 {
76 auto fbh = DRA::DRW::canvasFrameBuffer(derived());
77
78 DrawObjectSettings settings;
79 settings.viewId = viewId;
80
81 settings.renderMode = Base::viewerSettings().renderMode;
82 settings.imageBasedLighting = Base::viewerSettings().imageBasedLighting;
83
84 settings.environment = &mEnvironment;
85
86 setViewTransform(viewId);
87
88 DirectionalLightUniforms::setLight(Base::light());
89 DirectionalLightUniforms::bind();
90
91 ViewerDrawerUniforms::setExposure(Base::viewerSettings().exposure);
92 ViewerDrawerUniforms::setToneMapping(
93 Base::viewerSettings().toneMapping);
94 ViewerDrawerUniforms::setSpecularMipsLevels(
95 mEnvironment.specularMipLevels());
96 ViewerDrawerUniforms::setImageBasedLighting(
97 settings.imageBasedLighting && mEnvironment.canDraw());
98 ViewerDrawerUniforms::bind();
99
100 // background will be drawn only if settings allow it
101 mEnvironment.drawBackground(settings.viewId, Base::viewerSettings());
102
103 Base::drawableObjectVector().draw(settings);
104
105 Base::onDrawContent(viewId);
106 }
107
108 void onDrawId(uint viewId) override
109 {
110 DrawObjectSettings settings;
111 settings.objectId = Base::id();
112 settings.viewId = viewId;
113
114 setViewTransform(viewId);
115
116 Base::drawableObjectVector().drawId(settings);
117 }
118
119 bool onKeyPress(Key::Enum key, const KeyModifiers& modifiers) override
120 {
121 bool block = Base::onKeyPress(key, modifiers);
122
123 if (!block) {
124 switch (key) {
125 case Key::F1:
126 if (modifiers[KeyModifier::NO_MODIFIER]) {
127 mStatsEnabled = !mStatsEnabled;
128 bgfx::setDebug(
129 mStatsEnabled ? BGFX_DEBUG_STATS : BGFX_DEBUG_NONE);
130 }
131 break;
132
133 default: break;
134 }
135 }
136
137 return block;
138 }
139
140 // shadows Base::readDepthRequest: the requested homogeneousNDC is always
141 // overridden with the value required by the current bgfx backend
142 void readDepthRequest(double x, double y, bool homogeneousNDC = true)
143 {
144 homogeneousNDC = Context::instance().capabilites().homogeneousDepth;
145 Base::readDepthRequest(x, y, homogeneousNDC);
146 }
147
148private:
149 void setViewTransform(uint viewId)
150 {
151 // need to store the matrices
152 // parent viewer returns by value
153 Matrix44f vm = Base::viewMatrix();
154 Matrix44f pm = Base::projectionMatrix();
155
156 bgfx::setViewTransform(viewId, vm.data(), pm.data());
157 }
158
159 auto* derived() { return static_cast<DRA*>(this); }
160
161 const auto* derived() const { return static_cast<const DRA*>(this); }
162};
