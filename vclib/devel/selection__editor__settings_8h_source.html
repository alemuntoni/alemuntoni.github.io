22struct SelectionEditorSettings : public EditorSettings
23{
24 using KeyMap = InputActionMap<
25 std::pair<Key::Enum, KeyModifiers>,
26 SelectionAtomicAction>;
27 using MouseMap = InputActionMap<MouseInput, SelectionDragAction>;
28
29 bool selectVertices = false;

31 bool onlyVisible = false;
32 vcl::Color selectionBoxColor = vcl::Color(27, 120, 249, 64);
33
34 KeyMap keyBindings = defaultKeyMap();
35 MouseMap mouseBindings = defaultMouseMap();
36
40 void resetDefaults()
41 {
42 onlyVisible = false;
43 selectionBoxColor = vcl::Color(27, 120, 249, 64);
44 editMode = EditMode::CURRENT_OBJECT;
45 keyBindings.resetToDefaults();
46 mouseBindings.resetToDefaults();
47 }
48
52 std::vector<std::reference_wrapper<AbstractInputActionMap>> actionMaps()
53 override
54 {
55 return {keyBindings, mouseBindings};
56 }
57
61 std::vector<std::reference_wrapper<const AbstractInputActionMap>>

63 {
64 return {keyBindings, mouseBindings};

66
71 void loadSettings(const nlohmann::json& j) override
72 {
73 EditorSettings::loadSettings(j);
74
75 if (j.contains("Selection Editor")) {
76 const auto& jSel = j["Selection Editor"];
77 onlyVisible = jSel.value("onlyVisible", onlyVisible);
78 selectionBoxColor =
79 jSel.value("selectionBoxColor", selectionBoxColor);
80 editMode = static_cast<EditMode>(
81 jSel.value("editMode", static_cast<int>(editMode)));
82 }
83 }
84
89 void saveSettings(nlohmann::json& j) const override
90 {
91 EditorSettings::saveSettings(j);
92
93 auto& jSel = j["Selection Editor"];
94 jSel["onlyVisible"] = onlyVisible;
95 jSel["selectionBoxColor"] = selectionBoxColor;
96 jSel["editMode"] = static_cast<int>(editMode);
97 }
98
99private:
100 static KeyMap defaultKeyMap()
101 {
102 using enum Key::Enum;
103 using enum KeyModifier::Enum;
104 using Input = std::pair<Key::Enum, KeyModifiers>;
105
106 KeyMap map("Selection Atomic Actions");
107 map.registerActions({
108 {SelectionAtomicAction::ALL, "Select All", {Input {A, {CONTROL}}}},
109 {SelectionAtomicAction::NONE,
110 "Deselect All", {Input {D, {CONTROL}}}},
111 {SelectionAtomicAction::INVERT,
112 "Invert Selection", {Input {I, {CONTROL}}}}
113 });
114 return map;
115 }
116
117 static MouseMap defaultMouseMap()
118 {
119 using enum MouseButton::Enum;
120 using enum KeyModifier::Enum;
121
122 MouseMap map("Selection Drag Actions");
123 map.registerActions({
124 {SelectionDragAction::REGULAR,
125 "Regular Selection", {MouseInput {LEFT, {NO_MODIFIER}, false}} },
126 {SelectionDragAction::ADD,
127 "Add to Selection", {MouseInput {LEFT, {CONTROL}, false}} },
128 {SelectionDragAction::SUBTRACT,
129 "Subtract from Selection", {MouseInput {LEFT, {CONTROL, SHIFT}, false}}}
130 });
131 return map;
132 }
133};
