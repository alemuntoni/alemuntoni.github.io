35template<bool THROW = true>
36inline std::string readNextNonEmptyLine(std::istream& file)
37{
38 std::string line;
39 do {
40 std::getline(file, line);
41 if constexpr (THROW) {
42 if (!file) {
43 throw MalformedFileException("Unexpected end of file.");
44 }
45 }
46 if (file && line.size() > 0) {
47 removeCarriageReturn(line);
48 }
49 } while (file && line.size() == 0);
50 return line;
51}

65inline std::ifstream openInputFileStream(
66 const std::string& filename,
67 const std::string& ext = "")
68{
69 setlocale(LC_ALL, "C");
70
71 std::string actualfilename = filename;
72 if (!ext.empty()) {
73 actualfilename = FileInfo::addExtensionIfNeeded(filename, ext);
74 }
75 // need to set binary or windows will fail
76 std::ifstream fp(actualfilename, std::ifstream::binary);
77 fp.imbue(std::locale().classic());
78
79 if (!fp.is_open()) {
80 throw CannotOpenFileException(filename);
81 }
82 return fp;
83}

123inline Tokenizer readAndTokenizeNextNonEmptyLine(
124 std::istream& file,

180inline Tokenizer readAndTokenizeNextNonCommentLine(
181 std::istream& file,
182 const std::string& commentPrefix = "#",
183 const std::vector<char> separators = {' ', '\t'})
184{
185 std::string line;
186 Tokenizer tokenizer;
187
188 do {
189 line = readNextNonEmptyLine(file);
190 // Skip comment lines (lines starting with commentPrefix)
191 if (!line.empty() && line.size() >= commentPrefix.size() &&
192 line.substr(0, commentPrefix.size()) == commentPrefix) {
193 continue;
194 }
195 tokenizer = Tokenizer(line, separators);
196 } while (tokenizer.begin() == tokenizer.end());
197
198 return tokenizer;
199}

426template<typename T>
427T readPrimitiveType(
428 std::istream& file,
429 PrimitiveType type,
430 std::endian end = std::endian::native,
431 bool isColor = false)
432{
433 T p;
434 switch (type) {
435 case PrimitiveType::CHAR: p = readChar<T>(file, end); break;
436 case PrimitiveType::UCHAR: p = readUChar<T>(file, end); break;
437 case PrimitiveType::SHORT: p = readShort<T>(file, end); break;
438 case PrimitiveType::USHORT: p = readUShort<T>(file, end); break;
439 case PrimitiveType::INT: p = readInt<T>(file, end); break;
440 case PrimitiveType::UINT: p = readUInt<T>(file, end); break;
441 case PrimitiveType::FLOAT: p = readFloat<T>(file, end, isColor); break;
442 case PrimitiveType::DOUBLE: p = readDouble<T>(file, end, isColor); break;
443 default: assert(0); p = 0;
444 }
445 // if I read a color that must be returned as a float or double
446 if (isColor && !std::is_integral<T>::value)
447 p /= 255.0;
448 return p;
449}

554template<typename T>
555T readPrimitiveType(
556 Tokenizer::iterator& token,
557 PrimitiveType type,
558 std::endian = std::endian::native,
559 bool isColor = false)
560{
561 T p;
562 switch (type) {
563 case PrimitiveType::CHAR:
564 case PrimitiveType::UCHAR:
565 case PrimitiveType::SHORT:
566 case PrimitiveType::USHORT:
567 case PrimitiveType::INT:
568 case PrimitiveType::UINT: p = std::stoi(*token++); break;
569 case PrimitiveType::FLOAT:
570 case PrimitiveType::DOUBLE:
571 if (isColor) {
572 p = std::stod(*token++) * 255;
573 }
574 else {
575 p = std::stod(*token++);
576 }
577 break;
578 default: assert(0); p = 0;
579 }
580 // if I read a color that must be returned as a float or double
581 if (isColor && !std::is_integral<T>::value)
582 p /= 255.0;
583 return p;
584}
