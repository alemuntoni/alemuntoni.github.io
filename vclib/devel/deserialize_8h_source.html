69template<typename T>
70requires (HasMemberDeserialize<T> || IsNotClass<T>)
71void deserialize(std::istream& is, T& data)
72{
73 if constexpr (HasMemberDeserialize<T>) {
74 data.deserialize(is);
75 }
76 else {
77 deserialize(is, data, std::endian::little);
