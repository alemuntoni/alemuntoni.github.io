34template<CompId COMP_ID, typename Container>
35class ComponentView :
36 public std::ranges::view_interface<ComponentView<COMP_ID, Container>>
37{
38private:
39 Container* mContainer = nullptr;
40 bool mJumpDeleted = true;
41
42public:
43 using ContainerType = std::remove_cv_t<Container>;
44 using IndexType = ContainerType::IndexType;
45 using CompType = comp::ComponentTypeT<COMP_ID, Container>;
46 using RawDataType = CompType::DataType;
47 using DataType = std::conditional_t<
48 std::is_const_v<Container>,
49 const RawDataType,
50 RawDataType>;
51
52 using value_type = RawDataType;
53 using reference = DataType&;
54 using const_reference = const RawDataType&;
55 using pointer = DataType*;
56 using const_pointer = const RawDataType*;
57
58 using Iterator = ComponentIterator<COMP_ID, Container>;
59 using ConstIterator = ComponentIterator<COMP_ID, const Container>;
60
61public:
62 ComponentView() = default;
63
64 ComponentView(Container& container, bool jumpDeleted = true) :
65 mContainer(&container), mJumpDeleted(jumpDeleted)
66 {
67 }
68
69 template<CompId, typename>
70 friend class ComponentView;
71
72 // Implicit conversion from mutable ComponentView to const ComponentView
73 template<typename OtherContainerType>
74 requires std::is_same_v<Container, const OtherContainerType>
75 ComponentView(const ComponentView<COMP_ID, OtherContainerType>& other) :
76 mContainer(other.mContainer), mJumpDeleted(other.mJumpDeleted)
77 {
78 }
79
83 bool isAvailable() const
84 {
85 return mContainer && mContainer->template isComponentAvailable<COMP_ID>();
86 }
87
88
89 Iterator begin() const
90 {
91 if (!mContainer || !isAvailable()) {
92 return Iterator();
93 }
94 return Iterator(*mContainer, 0, needJump());
95 }
96
97 Iterator end() const
98 {
99 if (!mContainer || !isAvailable()) {
100 return Iterator();
101 }
102 return Iterator(
103 *mContainer, mContainer->elementContainerSize(), needJump());
104 }
105
106 reference operator[](IndexType i) const
107 {
108 assert(mContainer != nullptr);
109 assert(isAvailable() && "Component is not available!");
110 assert(i < mContainer->elementContainerSize());
111 return mContainer->template vector<COMP_ID>()[i];
112 }
113
114 pointer data() const
115 {
116 if (!mContainer || !isAvailable()) {
117 return nullptr;
118 }
119 return mContainer->template vector<COMP_ID>().data();
120 }
121
122 std::size_t size() const
123 {
124 if (!mContainer || !isAvailable()) {
125 return 0;
126 }
127 return mJumpDeleted ? mContainer->elementCount() :
128 mContainer->elementContainerSize();
129 }
130
131 bool empty() const { return size() == 0; }
132
133 bool isCompact() const
134 {
135 return mContainer && mContainer->isCompact();
136 }
137
138private:
139 // begin() and end() must agree on this flag
140 bool needJump() const
141 {
142 return mJumpDeleted &&
143 (mContainer->size() != mContainer->elementCount());
144 }
145};
