261 Mesh(const Mesh& oth) :
262 Args(oth)... // call auto copy constructors for all the container
263 // elements and components

269 // will need it to update all the pointers of this mesh
270 constexpr uint N_CONTAINERS =
271 NumberOfTypes<typename Mesh<Args...>::Containers>::value;
272 std::array<const void*, N_CONTAINERS> othBases =
273 Mesh<Args...>::getContainerBases(oth);

275 // update all the pointers contained on each container
276 // use the base pointer of each container of oth to update all the
277 // pointers in this mesh. Each pointer of oth that was copied in this
278 // mesh, will be updated computing its offset wrt the base of oth, and
279 // then adding that offset to the new base pointer of this mesh
280 (updateReferencesOfContainerType<Args>(*this, othBases), ...);

414 void append(const Mesh& m)
415 {
416 // for each container of this mesh, save its size
417 // will need it to update all the pointers/indices of this mesh
418 constexpr uint N_CONTAINERS =
419 NumberOfTypes<typename Mesh<Args...>::Containers>::value;
420 std::array<std::size_t, N_CONTAINERS> sizes =
421 Mesh<Args...>::getContainerSizes(*this);
422
423 // for each container of the other mesh m, save its bases
424 // will need it to update all the pointers of this mesh
425 std::array<const void*, N_CONTAINERS> bases =
426 Mesh<Args...>::getContainerBases(m);
427
428 // call the append function for each container
429 (appendContainer<Args>(m), ...);
430
431 // update all the pointers/indices contained on each container
432 (updateReferencesOfContainerTypeAfterAppend<Args>(*this, bases, sizes),
433 ...);
434
435 // manage transform matrix
436 if constexpr (mesh::HasTransformMatrix<Mesh<Args...>>) {
437 using Matrixtype = typename Mesh<Args...>::TransformMatrixType;
438
439 Matrixtype matrix = this->transformMatrix();
440 matrix = matrix.inverse().eval();
441 matrix *= m.transformMatrix();
442
443 if (matrix != Matrixtype::Identity()) {
444 (updatePosAndNormalsOfContainerTypeAfterAppend<Args>(
445 *this, sizes, matrix),
446 ...);
447 }
448 }
449
450 if constexpr (mesh::HasMaterials<Mesh<Args...>>) {
451 uint nMaterials = this->materialCount();
452
453 // mapping from material indices of m to material indices of this
454 std::vector<uint> mapping(m.materialCount());
455
456 // for each material of the other mesh, add it to this mesh
457 // if it does not exist yet
458 for (uint i = 0; i < m.materialCount(); ++i) {
459 auto it = std::find(
460 this->materialBegin(), this->materialEnd(), m.material(i));
461
462 if (it == this->materialEnd()) {
463 this->pushMaterial(m.material(i));
464 mapping[i] = nMaterials++;
465 }
466 else {
467 mapping[i] = std::distance(this->materialBegin(), it);
468 }
469 }
470
471 // update all the material indices in this mesh and add the texture
472 // images
473 if (nMaterials > 0 || mapping.size() > 0) {
474 for (const auto& p : m.textureImages()) {
475 this->pushTextureImage(p.first, p.second);
476 }
477
478 (updateMaterialIndicesOfContainerTypeAfterAppend<Args>(
479 *this, sizes, mapping),
480 ...);
481 }
482 }

499 template<typename OtherMeshType>
500 void importFrom(const OtherMeshType& m)
501 {
502 // This function will first:
503 // Call, for each Container and Component of the mesh, its importFrom
504 // function. In case of containers, it first creates the same number of
505 // elements in the container, and then calls the importFrom function for
506 // each new element.
507
508 (Args::importFrom(m), ...);
509
510 // Set to all element containers their parent mesh (this)
511 updateAllParentMeshPointers();
512
513 if constexpr (mesh::HasFaceContainer<Mesh<Args...>>) {
514 // Now I need to manage imports between different types of meshes
515 // (same type of meshes are already managed from importFrom member
516 // function).
517 //
518 // Generally speaking, Polygon meshes can import from any other type
519 // of mesh. We need to take care when this mesh has static vertex
520 // references number in the face container (VERTEX_COUNT >= 3).
521 //
522 // The follwing case don't need to be managed:
523 // - import polygon mesh from triangle mesh
524 //
525 // I can manage the following cases:
526 // - import triangle mesh from polygon mesh: need triangulation
527 //
528 // I cannot manage the follwing cases:
529 // - import static non-triangle mesh from polygon mesh or from a
530 // mesh with different VERTEX_COUNT
531
532 // in case of import from poly to triangle mesh, I need to manage
533 // triangulation of polygons and create additional triangle faces
534 // for each of the imported polygons. This function statically
535 // asserts that the import can be done.
536 if constexpr (mesh::HasFaceContainer<OtherMeshType>) {
537 using FaceContainer = typename Mesh<Args...>::FaceContainer;
538 FaceContainer::manageImportTriFromPoly(m);
539 }
540 }

547 void swap(Mesh& m2)
548 {
549 Mesh<Args...>& m1 = *this;
550
551 constexpr uint N_CONTAINERS =
552 NumberOfTypes<typename Mesh<Args...>::Containers>::value;
553 static_assert(N_CONTAINERS != 0);
554
555 // container bases of each container for m1 and m2
556 // we save the bases of the containers before swap
557 std::array<const void*, N_CONTAINERS> m1Bases =
558 Mesh<Args...>::getContainerBases(m1);
559 std::array<const void*, N_CONTAINERS> m2Bases =
560 Mesh<Args...>::getContainerBases(m2);
561
562 // actual swap of all the containers and the components of the mesh
563 // using pack expansion: swap will be called for each of the containers
564 // (or components!) that compose the Mesh
565 using std::swap;
566 (swap((Args&) m1, (Args&) m2), ...);
567
568 // Set to all elements their parent mesh
569 m1.updateAllParentMeshPointers();
570 m2.updateAllParentMeshPointers();
571
572 // update all the refs to m1 and m2: old base of m1 is now "old
573 // base" of m2, and viceversa
574 (updateReferencesOfContainerType<Args>(m1, m2Bases), ...);
575 (updateReferencesOfContainerType<Args>(m2, m1Bases), ...);
576 }

1759 const Element* oldBase,
1760 const std::array<std::size_t, N>& sizes = std::array<std::size_t, 0>(),

1768 Cont::updateReferences(oldBase, sizes[I], offset);
1769 }
1770 else {
1771 Cont::updateReferences(oldBase);
1772 }
1773 }
