35 struct MeshState
36 {
37 std::weak_ptr<vcl::DrawableObject> obj;
38 vcl::BitVector<true> vertexSelection;
39 vcl::BitVector<true> faceSelection;
40 };
41
42private:
43 std::vector<MeshState> mStates;
44
45public:
52 SelectionUndoRedoAction(std::vector<MeshState> states) :
53 mStates(std::move(states))
54 {
55 }
56
58 void undo() override { swapState(); }
59
61 void redo() override { swapState(); }
62
64 std::string name() const override { return "Selection Changed"; }
65
66private:
67 void swapState()
68 {
69 for (auto& state : mStates) {
70 if (auto lock = state.obj.lock()) {
71 if (auto* mesh =
72 dynamic_cast<AbstractDrawableMesh*>(lock.get())) {
73 if (!state.vertexSelection.empty() ||
74 !mesh->vertexSelectionBitVector().empty()) {
75 auto current = mesh->vertexSelectionBitVector();
76 mesh->setVertexSelectionBitVector(
77 state.vertexSelection);
78 state.vertexSelection = std::move(current);
79 }
80 if (!state.faceSelection.empty() ||
81 !mesh->faceSelectionBitVector().empty()) {
82 auto current = mesh->faceSelectionBitVector();
83 mesh->setFaceSelectionBitVector(state.faceSelection);
84 state.faceSelection = std::move(current);
85 }
86 }
87 }
88 }
89 }
90};
