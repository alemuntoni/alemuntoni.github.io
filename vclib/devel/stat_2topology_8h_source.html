24// function called for the Container Cont of the Mesh
25template<typename Cont>
26void setReferencedVertices(const auto& mesh, auto& refs, uint& nRefs)
27{
28 // check if the Cont container of the Mesh has vertex references
29 if constexpr (comp::HasVertexReferences<typename Cont::ElementType>) {
30 // if there are still some vertices non-referenced
31 if (nRefs < mesh.vertexCount()) {
32 constexpr uint ELEM_ID = Cont::ElementType::ELEMENT_ID;
33 // for eache element of the Cont container
34 for (const auto& el : mesh.template elements<ELEM_ID>()) {
35 // for each vertex of the element
36 for (uint vi : el.vertexIndices()) {
37 // vertex references shoule never be null
38 assert(vi != UINT_NULL);
39 if (!refs[vi]) {
40 // set the vertex as referenced
41 refs[vi] = true;
42 nRefs++;
43 }
44 }
45 }
46 }
47 }
48}

63// struct to store the information of the wedge texcoords
64template<typename WedgeTexCoordType>
65struct WedgeTexCoordsInfo
66{
67 WedgeTexCoordType texCoord;
68 uint texCoordIndex;
69
70 bool operator<(const WedgeTexCoordsInfo& other) const
71 {
72 if (texCoordIndex == other.texCoordIndex) {
73 return texCoord < other.texCoord;
74 }
75 return texCoordIndex < other.texCoordIndex;
76 }
77};

79template<FaceMeshConcept MeshType>
80std::vector<bool> nonManifoldVerticesVectorBool(const MeshType& m)
81 requires HasPerFaceAdjacentFaces<MeshType>
82{
83 requirePerFaceAdjacentFaces(m);
84
85 using FaceType = MeshType::FaceType;
86
87 std::vector<bool> nonManifoldVertices(m.vertexContainerSize(), false);
88
89 std::vector<uint> TD(m.vertexContainerSize(), 0);
90 std::vector<bool> nonManifoldInc(m.vertexContainerSize(), false);
91 // First Loop, count how many faces are incident on a vertex and store it in
92 // TD, and flag how many vertices are incident on non manifold edges.
93 for (const FaceType& f : m.faces()) {
94 for (uint i = 0; i < f.vertexCount(); ++i) {
95 TD[m.index(f.vertex(i))]++;
96 if (!isFaceManifoldOnEdge(f, i)) {
97 nonManifoldInc[m.index(f.vertex(i))] = true;
98 nonManifoldInc[m.index(f.vertexMod(i + 1))] = true;
99 }
100 }
101 }
102
103 std::vector<bool> visited(m.vertexContainerSize(), false);
104 for (const FaceType& f : m.faces()) {
105 for (uint i = 0; i < f.vertexCount(); ++i) {
106 if (!visited[m.index(f.vertex(i))]) {
107 visited[m.index(f.vertex(i))] = true;
108 MeshPos pos(&f, i);
109 uint starSize = pos.numberOfAdjacentFacesToV();
110 if (starSize != TD[m.index(f.vertex(i))])
111 nonManifoldVertices[m.index(f.vertex(i))] = true;
112 }
113 }
114 }
115
116 return nonManifoldVertices;
117}

119template<FaceMeshConcept MeshType>
120uint edgeCount(
121 const MeshType& m,
122 uint& numBoundaryEdges,
123 uint& numNonManifoldEdges)
124{
125 std::vector<ConstMeshEdgeUtil<MeshType>> edgeVec =
126 fillAndSortMeshEdgeUtilVector(m);
127
128 uint numEdges = 0;
129 numBoundaryEdges = 0;
130 numNonManifoldEdges = 0;
131
132 size_t f_on_cur_edge = 1;
133 for (size_t i = 0; i < edgeVec.size(); ++i) {
134 if (((i + 1) == edgeVec.size()) || !(edgeVec[i] == edgeVec[i + 1])) {
135 ++numEdges;
136 if (f_on_cur_edge == 1)
137 ++numBoundaryEdges;
138 if (f_on_cur_edge > 2)
139 ++numNonManifoldEdges;
140 f_on_cur_edge = 1;
141 }
142 else {
143 ++f_on_cur_edge;
144 }
145 }
146 return numEdges;
147}

455template<FaceMeshConcept MeshType>
456uint verticesToDuplicateByWedgeTexCoordsCount(
457 const MeshType& mesh,
458 std::vector<std::pair<uint, uint>>& vertWedgeMap =
459 detail::dummyVectorOfPairs,
460 std::list<uint>& vertsToDuplicate = detail::dummyUintList,
461 std::list<std::list<std::pair<uint, uint>>>& facesToReassign =
462 detail::dummyListOfLists)
463{
464 vcl::requirePerFaceWedgeTexCoords(mesh);
465
466 bool hasMaterials = false;
467 if constexpr (HasPerFaceMaterialIndex<MeshType>) {
468 hasMaterials = mesh.isPerFaceMaterialIndexEnabled();
469 }
470
471 using WedgeTexCoordType = MeshType::FaceType::WedgeTexCoordType;
472 using WedgeTexCoordsInfo = detail::WedgeTexCoordsInfo<WedgeTexCoordType>;
473
474 // list of faces that reference a wedge texcoord, with the index of the
475 // vertex in the face
476 using FaceList = std::list<std::pair<uint, uint>>;
477
478 vertWedgeMap.resize(mesh.vertexContainerSize());
479 vertsToDuplicate.clear();
480 facesToReassign.clear();
481
482 uint count = 0;
483
484 // for each vertex, I'll store a map of WedgeTexCoordsInfo
485 // each element of the map represent a unique wedge texcoord(the texcoord
486 // itself and the index of the texcoord), and for each element it maps a
487 // list of faces that reference the texcoord
488 std::vector<std::map<WedgeTexCoordsInfo, FaceList>> wedges(
489 mesh.vertexContainerSize());
490
491 for (const auto& f : mesh.faces()) {
492 for (uint i = 0; i < f.vertexCount(); ++i) {
493 uint vi = f.vertexIndex(i);
494
495 // check if the i-th wedge texcoord of the face already exists
496 uint matIdx = 0;
497 if constexpr (HasPerFaceMaterialIndex<MeshType>) {
498 if (hasMaterials) {
499 matIdx = f.materialIndex();
500 }
501 }
502 WedgeTexCoordsInfo wi = {f.wedgeTexCoord(i), matIdx};
503 auto it = wedges[vi].find(wi);
504 if (it == wedges[vi].end()) { // if it doesn't exist, add it
505 // if there was already a texcoord for the vertex, it means that
506 // the vertex will be duplicated
507 if (!wedges[vi].empty()) {
508 count++;
509 }
510 wedges[vi][wi].emplace_back(f.index(), i);
511 }
512 else {
513 // if it exists, add the face to the list of faces that
514 // reference the texcoord
515 it->second.emplace_back(f.index(), i);
516 }
517 }
518 }
519
520 // for each vertex, check if there are multiple texcoords
521 // note: here we will modify the maps for convenience: at the end of this
522 // loop, the info will be contained in the two lists vertsToDuplicate and
523 // facesToReassign (the wedges vector of map will be inconsistent)
524 for (uint vi = 0; auto& map : wedges) {
525 if (map.size() > 1) {
526 // there are multiple texcoords for the vertex vi, so it will be
527 // duplicated
528
529 // remove from the map the element having the higher number of
530 // faces referencing the texcoord (we do this to reassign the less
531 // number of faces)
532 auto it = std::max_element(
533 map.begin(), map.end(), [](const auto& a, const auto& b) {
534 return a.second.size() < b.second.size();
535 });
536 // store the reference of the texcoord to keep (pair face/vertex
537 // index in the face)
538 vertWedgeMap[vi] = it->second.front();
539 map.erase(it);
540
541 // store the vertex to duplicate, and the faces to reassign
542 for (auto& [wi, fl] : map) {
543 vertsToDuplicate.push_back(vi);
544 facesToReassign.push_back(std::move(fl));
545 }
546 }
547 else {
548 if (map.size() == 1) {
549 // there is only one texcoord for the vertex vi, so store its
550 // reference (pair face/vertex index in the face)
551 vertWedgeMap[vi] = map.begin()->second.front();
552 }
553 else { // the vertex was unreferenced
554 vertWedgeMap[vi] = {UINT_NULL, UINT_NULL};
555 }
556 }
557 ++vi;
558 }
559
560 assert(vertWedgeMap.size() == mesh.vertexContainerSize());
561 assert(vertsToDuplicate.size() == count);
562 assert(facesToReassign.size() == count);
563
564 return count;
565}

595template<typename Container, MeshConcept MeshType>
596Container referencedVertices(
597 const MeshType& mesh,
598 uint& nUnref,
599 bool onlyFaces = false)
600{
601 using VertexType = MeshType::VertexType;
602
603 uint nRefs = 0;
604
605 Container refVertices(mesh.vertexContainerSize(), false);
606
607 if (onlyFaces) {
608 if constexpr (FaceMeshConcept<MeshType>) {
609 using FaceContainer = MeshType::FaceContainer;
610
611 detail::setReferencedVertices(
612 mesh, refVertices, nRefs, TypeWrapper<FaceContainer>());
613 }
614 }
615 else {
616 detail::setReferencedVertices(
617 mesh, refVertices, nRefs, typename MeshType::Containers());
618 }
619
620 nUnref = mesh.vertexCount() - nRefs;
621
622 return refVertices;
623}

728template<FaceMeshConcept MeshType>
729uint holeCount(const MeshType& m) requires HasPerFaceAdjacentFaces<MeshType>
730{
731 requirePerFaceAdjacentFaces(m);
732
733 using VertexType = MeshType::VertexType;
734 using FaceType = MeshType::FaceType;
735
736 uint loopNum = 0;
737
738 // create a vector of bools to keep track of visited faces.
739 std::vector<bool> visitedFaces(m.faceContainerSize(), false);
740
741 // Traverse the mesh using a depth-first search algorithm to find all the
742 // holes.
743 for (const FaceType& f : m.faces()) {
744 uint e = 0;
745 for (const VertexType* v : f.vertices()) {
746 if (!visitedFaces[m.index(f)] && f.adjFace(e) == nullptr) {
747 MeshPos<FaceType> startPos(&f, e);
748 MeshPos<FaceType> curPos = startPos;
749 do {
750 curPos.nextEdgeOnBorderAdjacentToV();
751 curPos.flipVertex();
752 visitedFaces[m.index(curPos.face())] = true;
753 } while (curPos != startPos);
754 ++loopNum;
755 }
756 ++e;
757 }
758 }
759 return loopNum;
760}

785template<FaceMeshConcept MeshType>
786std::vector<std::set<uint>> connectedComponents(const MeshType& m)
787 requires HasPerFaceAdjacentFaces<MeshType>
788{
789 requirePerFaceAdjacentFaces(m);
790
791 using FaceType = MeshType::FaceType;
792
793 std::vector<std::set<uint>> cc;
794
795 // create a vector of bools to keep track of visited faces.
796 std::vector<bool> visitedFaces(m.faceContainerSize(), false);
797
798 // create a stack to hold the faces that need to be visited during the
799 // depth-first search.
800 std::stack<const FaceType*> sf;
801
802 // traverse the mesh using a depth-first search algorithm to find the
803 // connected components.
804 for (const FaceType& f : m.faces()) {
805 if (!visitedFaces[m.index(f)]) { // first time I see this face
806 visitedFaces[m.index(f)] = true;
807
808 // new connected component
809 cc.emplace_back();
810 std::set<uint>& ccf = cc[cc.size() - 1];
811 ccf.insert(m.index(f));
812
813 // while the stack is empty, visit the adjacent faces of the top
814 // face of the stack
815 sf.push(&f);
816 while (!sf.empty()) {
817 const FaceType* fpt = sf.top();
818 // remove the top face and add it to the connected component
819 sf.pop();
820 ccf.insert(m.index(fpt));
821
822 // add the adjacent faces of the current visited in the stack
823 for (uint j = 0; j < fpt->vertexCount(); ++j) {
824 const FaceType* adjf = fpt->adjFace(j);
825 // if there is an adj face and it has not been visited
826 if (adjf != nullptr && !visitedFaces[m.index(adjf)]) {
827 sf.push(adjf);
828 visitedFaces[m.index(adjf)] = true;
829 }
830 }
831 }
832 }
833 }
834 return cc;
835}
