17template<FaceMeshConcept MeshType>
18void fillHexahedronTriangles(MeshType& m)
19{
20 m.reserveFaces(12);

22 m.addFace(3, 1, 2);

82template<FaceMeshConcept MeshType, Point3Concept PositionType>
83MeshType createHexahedron(const PositionType& min, const PositionType& max)
84{
85 MeshType m;
86
87 // fill vertices...
88 m.addVertices(
89 PositionType(min(0), min(1), min(2)),
90 PositionType(max(0), min(1), min(2)),
91 PositionType(min(0), max(1), min(2)),
92 PositionType(max(0), max(1), min(2)),
93 PositionType(min(0), min(1), max(2)),
94 PositionType(max(0), min(1), max(2)),
95 PositionType(min(0), max(1), max(2)),
96 PositionType(max(0), max(1), max(2)));
97
98 // fill faces
99 if constexpr (HasTriangles<MeshType>) {
100 detail::fillHexahedronTriangles(m);
101 }
102 else {
103 detail::fillHexahedronQuads(m);
104 }
105 return m;
106}
