50class WedgeProxy

55public:

63 WedgeProxy() = default;

65 WedgeProxy(FaceType face, uint index) : mFace(face), mIndex(index) {}
66
67 // Converting constructor from mutable to const (or convertible) FaceType
68 template<typename OtherFace>
69 requires std::convertible_to<OtherFace, FaceType>
70 WedgeProxy(const WedgeProxy<OtherFace>& other) :
71 mFace(other.parentFace()), mIndex(other.index())
72 {
73 }
74
75 template<typename OtherFace>
76 bool operator==(const WedgeProxy<OtherFace>& other) const
77 {
78 return mFace == other.parentFace() && mIndex == other.index();
79 }
80
81 template<typename OtherFace>
82 auto operator<=>(const WedgeProxy<OtherFace>& other) const
83 {
84 if (auto cmp = mFace <=> other.parentFace(); cmp != 0) {
85 return cmp;
86 }
87 return mIndex <=> other.index();
88 }

93 void swap(WedgeProxy& other) noexcept
94 {
95 using std::swap;
96 swap(mFace, other.mFace);
97 swap(mIndex, other.mIndex);
98 }
