24 TextDrawer(uint width, uint height)

33 void onResize(uint width, uint height) override final

40 auto fbh = DerivedRenderApp::DRW::canvasFrameBuffer(derived());

43
44 // text
45 void enableText(bool b = true) { mTextView.enableText(b); }
46
47 bool isTextEnabled() const { return mTextView.isTextEnabled(); }
48
49 void setTextFont(VclFont::Enum font, uint fontSize)

51 mTextView.setTextFont(font, fontSize);

53
54 void setTextFont(const std::string& fontName, uint fontSize)
55 {
56 mTextView.setTextFont(fontName, fontSize);
57 }
58
59 void clearText() { mTextView.clearText(); }
60
61 void appendStaticText(
62 const Point2f& pos,
63 const std::string& text,

66 mTextView.appendStaticText(pos, text, color);
67 }
68
69 void appendTransientText(
70 const Point2f& pos,
71 const std::string& text,

74 mTextView.appendTransientText(pos, text, color);
75 }
76
77private:

80 const auto* derived() const
