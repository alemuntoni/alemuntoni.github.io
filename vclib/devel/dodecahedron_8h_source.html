25template<PolygonMeshConcept MeshType, LoggerConcept LogType = NullLogger>
26MeshType createDodecahedron(LogType& log = nullLogger)
27{
28 using PositionType = PositionType<MeshType>;
29 using ScalarType = ScalarType<MeshType>;
30
31 MeshType mesh;
32
33 const ScalarType fi = (1 + std::sqrt(5)) / 2;
34 const ScalarType s0 = 0;
35 const ScalarType s1 = fi;
36 const ScalarType s2 = std::pow(fi, 2);
37 const ScalarType s3 = 1;
38
39 log.log(0, "Adding vertices to PolyMesh...");
40
41 mesh.addVertices(
42 PositionType(-s1, -s1, s1),
43 PositionType(s2, s3, s0),
44 PositionType(s2, -s3, s0),
45 PositionType(-s2, s3, s0),
46 PositionType(-s2, -s3, s0),
47 PositionType(s0, s2, s3),
48 PositionType(s0, s2, -s3),
49 PositionType(s3, s0, -s2),
50 PositionType(-s3, s0, -s2),
51 PositionType(s0, -s2, -s3),
52 PositionType(s0, -s2, s3),
53 PositionType(s3, s0, s2),
54 PositionType(-s3, s0, s2),
55 PositionType(s1, s1, -s1),
56 PositionType(s1, s1, s1),
57 PositionType(-s1, s1, -s1),
58 PositionType(-s1, s1, s1),
59 PositionType(s1, -s1, -s1),
60 PositionType(s1, -s1, s1),
61 PositionType(-s1, -s1, -s1));
62
63 log.log(50, "Vertices added to PolyMesh.");
64 log.log(50, "Adding faces to PolyMesh...");
65
66 mesh.reserveFaces(12);
67 mesh.addFace(14, 11, 18, 2, 1);
68 mesh.addFace(2, 17, 7, 13, 1);
69 mesh.addFace(15, 8, 19, 4, 3);
70 mesh.addFace(4, 0, 12, 16, 3);
71 mesh.addFace(16, 5, 6, 15, 3);
72 mesh.addFace(13, 6, 5, 14, 1);
73 mesh.addFace(18, 10, 9, 17, 2);
74 mesh.addFace(19, 9, 10, 0, 4);
75 mesh.addFace(17, 9, 19, 8, 7);
76 mesh.addFace(13, 7, 8, 15, 6);
77 mesh.addFace(16, 12, 11, 14, 5);
78 mesh.addFace(18, 11, 12, 0, 10);
79
80 log.log(100, "Faces added to PolyMesh.");
81
82 return mesh;
83}

97template<TriangleMeshConcept MeshType, LoggerConcept LogType = NullLogger>
98MeshType createDodecahedron(LogType& log = nullLogger)
99{
100 log.startNewTask(0, 75, "Create Polygonal Dodecahedron.");
101
102 detail::TMPSimplePolyMesh pmesh =
103 createDodecahedron<detail::TMPSimplePolyMesh>(log);
104
105 log.endTask("Create Polygonal Dodecahedron.");
106 log.log(75, "Copying vertices into TriMesh...");
107
108 MeshType mesh;
109 using ST = ScalarType<MeshType>;
110 mesh.reserveVertices(pmesh.vertexCount());
111 for (const auto& v : pmesh.vertices()) {
112 mesh.addVertex(v.position().cast<ST>());
113 }
114
115 log.log(80, "Vertices copied into TriMesh.");
116 log.log(80, "Triangularize and copy Faces into TriMesh...");
117
118 for (const auto& f : pmesh.faces()) {
119 std::vector<uint> ind = earCut(f);
120 for (uint i = 0; i < ind.size(); i += 3) {
121 mesh.addFace(
122 f.vertexIndex(ind[i]),
123 f.vertexIndex(ind[i + 1]),
124 f.vertexIndex(ind[i + 2]));
125 }
126 }
127
128 log.log(100, "Faces triangularized and copied into TriMesh.");
129
130 return mesh;
131}
