404 static Image::ColorSpace textureTypeToColorSpace(TextureType type)
405 {
406 switch (type) {
407 case TextureType::BASE_COLOR:
408 case TextureType::EMISSIVE:
409 case TextureType::SPECULAR_COLOR: return Image::ColorSpace::SRGB;
410 case TextureType::METALLIC_ROUGHNESS:
411 case TextureType::NORMAL:
412 case TextureType::OCCLUSION:
413 case TextureType::CLEARCOAT:
414 case TextureType::CLEARCOAT_ROUGHNESS:
415 case TextureType::CLEARCOAT_NORMAL:
416 case TextureType::ANISOTROPY:
417 case TextureType::SPECULAR:
418 default: return Image::ColorSpace::LINEAR;
419 }
420 }
