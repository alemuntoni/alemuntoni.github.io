238 template<Range R>
239 requires Point3Concept<std::ranges::range_value_t<R>>
240 void setVertices(R&& verts)
241 {
242 mVerPosCount = std::ranges::size(verts);
243
244 // Compute padding to ensure the buffer size is a multiple of 4 floats
245 // (16 bytes). This is required because the vertex shader reads the
246 // buffer as vec4 elements.
247 uint padding = (4 - (mVerPosCount % 4)) % 4;
248 uint nv = mVerPosCount + padding;
249
250 VertexBuffer vertBuff;
251 auto [buffer, releaseFn] =
252 Context::getAllocatedBufferAndReleaseFn<float>(nv * 3);
253
254 for (size_t i = 0; const auto& v : verts) {
255 buffer[i * 3 + 0] = v.x();
256 buffer[i * 3 + 1] = v.y();
257 buffer[i * 3 + 2] = v.z();
258 ++i;
259 }
260
261 vertBuff.create(
262 buffer,
263 nv,
264 bgfx::Attrib::Position,
265 3,
266 PrimitiveType::FLOAT,
267 releaseFn);
268 mVertexPositions.setOwned(std::move(vertBuff));
269 mIsUpdateProgramNeeded = true;
270 }

326 template<Range R>

328 void setVertexColors(R&& vertColors)

334 uint nc = mVerColCount + padding;

364 template<Range R>
365 requires std::convertible_to<std::ranges::range_value_t<R>, bool>
366 void setVertexSelection(R&& selections)
367 {
368 mVerSelCount = std::ranges::size(selections);
369 BooleanBuffer buf;
370 buf.init(mVerSelCount);
371
372 vcl::BitVector<true> backup(mVerSelCount, false);
373
374 uint vidx = 0;
375 for (bool sel : selections) {
376 backup[vidx++] = sel;
377 }
378
379 buf.setFromCPUBuffer(backup);
380 mSelectionBuffer.setOwned(std::move(buf));
381 mIsUpdateProgramNeeded = true;
382 }
