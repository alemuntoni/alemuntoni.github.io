121 static bool isValid(ConstFaceType f, ConstVertexType v, short e)
122 {
123 if (!f || !v || e < 0)

201 bool flipFace()
202 {
203 ConstFaceType nf = mFace.adjFace(mEdge);
204 if (nf) {
205 // The edge of the adjacent face must be found using the vertices
206 // of the current edge: searching the position of the current face
207 // in the adjacent faces of nf is not enough, because nf could be
208 // adjacent to the current face along more than one edge (e.g. two
209 // faces with the same vertices), and in this case the first match
210 // could be the wrong edge, leading to wrong (and possibly endless)
211 // navigations.
212 auto v0 = mFace.vertex(mEdge);
213 auto v1 = mFace.vertexMod(mEdge + 1);
214 uint newEdge = nf.indexOfEdge(v0, v1);
215 assert(newEdge != UINT_NULL);
216 if (newEdge == UINT_NULL) {
217 *this = MeshPos();
218 return false;
219 }
220 mEdge = newEdge;
221 mFace = nf;
222 return true;
223 }
224 else {
225 return false;
226 }
227 }
228
233 void flipVertex()
234 {
235 if (mFace.vertexMod(mEdge) == mVertex) {
236 mVertex = mFace.vertexMod(mEdge + 1);
237 }
238 else {
239 mVertex = mFace.vertexMod(mEdge);
240 }
241 }
242
247 void flipEdge()
248 {
249 if (mFace.vertexMod(mEdge + 1) == mVertex) {
250 mEdge = (mEdge + 1) % (short) mFace.vertexCount();
251 }
252 else {
253 short n = mFace.vertexCount();
254 mEdge = ((mEdge - 1) % n + n) %
255 n; // be sure to get the right index of the previous edge
256 }
257 }

270 void nextEdgeAdjacentToV()

284 bool nextEdgeOnBorderAdjacentToV()
285 {
286 MeshPos startPos = *this;
287 do {
288 nextEdgeAdjacentToV();
289 } while (*this != startPos && !isEdgeOnBorder());
290 return (*this != startPos);
291 }

301 uint numberOfAdjacentFacesToV() const
302 {
303 bool onBorder = false;
304 uint count = countAdjacentFacesToV(onBorder);
305 // if we visited a border, it means that we visited the all faces
306 // adjacent to v twice to reach the same starting MeshPos.
307 if (onBorder)
308 count /= 2;
309 return count;
310 }

312 bool operator==(const MeshPos& op) const
313 {
314 return mFace == op.mFace && mVertex == op.mVertex && mEdge == op.mEdge;
315 }
316
317 bool operator!=(const MeshPos& op) const { return !(*this == op); }
318
319 bool operator<(const MeshPos& op) const
320 {
321 if (mFace == op.mFace) {
322 if (mEdge == op.mEdge)
323 return mVertex.index() < op.mVertex.index();
324 else
325 return mEdge < op.mEdge;
326 }
327 else {
328 return mFace.index() < op.mFace.index();
329 }
330 }
331
332private:
333 uint countAdjacentFacesToV(bool& onBorder) const
334 {
335 uint count = 0;
336 onBorder = false;
337 // start from this MeshPos
338 MeshPos mp = *this;
339 do {
340 // go to the next edge in the star of v (if face is on border, will
341 // change just edge)
342 mp.nextEdgeAdjacentToV();
343 ++count;
344 // if we are visiting a border, it means that we will start to
345 // navigate in the opposite sense as we were before.
346 if (mp.isEdgeOnBorder()) // flag that we visited a border
347 onBorder = true;
348 }
349 // we end when we will reach the same pos (same triplet f-v-e) of start
350 while (mp != *this);
351 return count;
352 }
