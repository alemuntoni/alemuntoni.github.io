45template<FaceConcept FaceType, PointConcept PointType>
46bool intersect(const FaceType& f, const Box<PointType>& box)
47{
48 if constexpr (TriangleFaceConcept<FaceType>) {
49 return intersect(
50 TriangleWrapper(
51 f.vertex(0).position(),
52 f.vertex(1).position(),
53 f.vertex(2).position()),
54 box);
55 }
56 else {
57 if (f.vertexCount() == 3) {
58 return intersect(
59 TriangleWrapper(
60 f.vertex(0).position(),
61 f.vertex(1).position(),
62 f.vertex(2).position()),
63 box);
64 }
65 else {
66 bool b = false;
67
68 std::vector<uint> tris = earCut(f);
69 for (uint i = 0; i < tris.size() && !b; i += 3) {
70 b |= intersect(
71 TriangleWrapper(
72 f.vertex(tris[i]).position(),
73 f.vertex(tris[i + 1]).position(),
74 f.vertex(tris[i + 2]).position()),
75 box);
76 }
77 return b;
78 }
79 }
80}

113template<FaceConcept FaceType, PointConcept PointType, typename SScalar>
114bool intersect(
115 const FaceType& f,
116 const Sphere<SScalar>& sphere,
117 PointType& witness,
118 std::pair<SScalar, SScalar>& res)
119{
120 if constexpr (TriangleFaceConcept<FaceType>) {
121 return intersect(
122 TriangleWrapper(
123 f.vertex(0).position(),
124 f.vertex(1).position(),
125 f.vertex(2).position()),
126 sphere,
127 witness,
128 res);
129 }
130 else {
131 if (f.vertexCount() == 3) {
132 return intersect(
133 TriangleWrapper(
134 f.vertex(0).position(),
135 f.vertex(1).position(),
136 f.vertex(2).position()),
137 sphere,
138 witness,
139 res);
140 }
141 else {
142 res.first = std::numeric_limits<SScalar>::max();
143 std::pair<SScalar, SScalar> r;
144
145 bool b = false;
146 PointType w;
147
148 std::vector<uint> tris = earCut(f);
149 for (uint i = 0; i < tris.size() && !b; i += 3) {
150 b |= intersect(
151 TriangleWrapper(
152 f.vertex(tris[i]).position(),
153 f.vertex(tris[i + 1]).position(),
154 f.vertex(tris[i + 2]).position()),
155 sphere,
156 w,
157 r);
158
159 if (r.first < res.first) {
160 res = r;
161 witness = w;
162 }
163 }
164 return b;
165 }
166 }
167}

244template<Ray3Concept RayType, FaceConcept FaceType>
245std::optional<typename RayType::PointType> intersection(
246 const RayType& ray,
247 const FaceType& face,
248 std::optional<std::reference_wrapper<typename RayType::ScalarType>> t = {})
249 requires std::same_as<
250 typename RayType::ScalarType,
251 ScalarType<VertexType<FaceType>>>
252{
253 using PointType = typename RayType::PointType;
254 using ScalarType = typename PointType::ScalarType;
255
256 auto triangleIntersection = [](const FaceType& f,
257 const RayType& r,
258 auto t) -> std::optional<PointType> {
259 return intersection(
260 r,
261 TriangleWrapper(
262 f.vertex(0).position(),
263 f.vertex(1).position(),
264 f.vertex(2).position()),
265 t);
266 };
267
268 if constexpr (TriangleFaceConcept<FaceType>) {
269 return triangleIntersection(face, ray, t);
270 }
271 else {
272 if (face.vertexCount() == 3) {
273 return triangleIntersection(face, ray, t);
274 }
275 else {
276 std::vector<uint> tris = earCut(face);
277 for (uint i = 0; i < tris.size(); i += 3) {
278 std::optional<PointType> res = intersection(
279 ray,
280 TriangleWrapper(
281 face.vertex(tris[i]).position(),
282 face.vertex(tris[i + 1]).position(),
283 face.vertex(tris[i + 2]).position()),
284 t);
285 if (res.has_value())
286 return res;
287 }
288 return std::nullopt;
289 }
290 }
291}
