38template<FaceConcept FaceType, PointConcept PointType>
39bool intersect(const FaceType& f, const Box<PointType>& box)
40{
41 if constexpr (TriangleFaceConcept<FaceType>) {
42 return intersect(
43 TriangleWrapper(
44 f.vertex(0)->position(),
45 f.vertex(1)->position(),
46 f.vertex(2)->position()),
47 box);
48 }
49 else {
50 bool b = false;
51
52 std::vector<uint> tris = earCut(f);
53 for (uint i = 0; i < tris.size() && !b; i += 3) {
54 b |= intersect(
55 TriangleWrapper(
56 f.vertex(tris[i])->position(),
57 f.vertex(tris[i + 1])->position(),
58 f.vertex(tris[i + 2])->position()),
59 box);
60 }
61 return b;
62 }
63}

96template<FaceConcept FaceType, PointConcept PointType, typename SScalar>
97bool intersect(
98 const FaceType& f,
99 const Sphere<SScalar>& sphere,
100 PointType& witness,
101 std::pair<SScalar, SScalar>& res)
102{
103 if constexpr (TriangleFaceConcept<FaceType>) {
104 return intersect(
105 TriangleWrapper(
106 f.vertex(0)->position(),
107 f.vertex(1)->position(),
108 f.vertex(2)->position()),
109 sphere,
110 witness,
111 res);
112 }
113 else {
114 if (f.vertexCount() == 3) {
115 return intersect(
116 TriangleWrapper(
117 f.vertex(0)->position(),
118 f.vertex(1)->position(),
119 f.vertex(2)->position()),
120 sphere,
121 witness,
122 res);
123 }
124 else {
125 res.first = std::numeric_limits<SScalar>::max();
126 std::pair<SScalar, SScalar> r;
127
128 bool b = false;
129 PointType w;
130
131 std::vector<uint> tris = earCut(f);
132 for (uint i = 0; i < tris.size() && !b; i += 3) {
133 b |= intersect(
134 TriangleWrapper(
135 f.vertex(tris[i])->position(),
136 f.vertex(tris[i + 1])->position(),
137 f.vertex(tris[i + 2])->position()),
138 sphere,
139 w,
140 r);
141
142 if (r.first < res.first) {
143 res = r;
144 witness = w;
145 }
146 }
147 return b;
148 }
149 }
150}

212template<Ray3Concept RayType, FaceConcept FaceType>
213std::optional<typename RayType::PointType> intersection(
214 const RayType& ray,
215 const FaceType& face,
216 std::optional<std::reference_wrapper<typename RayType::ScalarType>> t = {})
217 requires std::same_as<
218 typename RayType::ScalarType,
219 typename FaceType::VertexType::PositionType::ScalarType>
220{
221 using PointType = typename RayType::PointType;
222 using ScalarType = typename PointType::ScalarType;
223
224 auto triangleIntersection = [](const FaceType& f,
225 const RayType& r,
226 auto t) -> std::optional<PointType> {
227 return intersection(
228 r,
229 TriangleWrapper(
230 f.vertex(0)->position(),
231 f.vertex(1)->position(),
232 f.vertex(2)->position()),
233 t);
234 };
235
236 if constexpr (TriangleFaceConcept<FaceType>) {
237 return triangleIntersection(face, ray, t);
238 }
239 else {
240 if (face.vertexCount() == 3) {
241 return triangleIntersection(face, ray, t);
242 }
243 else {
244 std::vector<uint> tris = earCut(face);
245 for (uint i = 0; i < tris.size(); i += 3) {
246 std::optional<PointType> res = intersection(
247 ray,
248 TriangleWrapper(
249 face.vertex(tris[i])->position(),
250 face.vertex(tris[i + 1])->position(),
251 face.vertex(tris[i + 2])->position()),
252 t);
253 if (res.has_value())
254 return res;
255 }
256 return std::nullopt;
257 }
258 }
259}
