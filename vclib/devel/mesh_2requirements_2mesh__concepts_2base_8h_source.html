37template<typename MeshType, ElemId ELEM_ID>
38concept HasContainerOf = requires {
39 requires std::remove_cvref_t<MeshType>::template hasContainerOf<ELEM_ID>();

79template<typename MeshType, ElemId ELEM_ID, CompId COMP_ID>
80concept HasElementComponent =
81 HasContainerOf<MeshType, ELEM_ID> &&
82 comp::HasComponent<
83 COMP_ID,
84 typename std::remove_cvref_t<MeshType>::template ContainerType<
85 ELEM_ID>>;
