401 uint indexOfEdge(const Vertex* v1, const Vertex* v2) const
402 {
403 if constexpr (STORE_INDICES) {
404 return indexOfEdge(
405 Base::indexFromPointer(v1), Base::indexFromPointer(v2));
406 }
407 else {
408 uint vid = indexOfVertex(v1);
409 if (vid == UINT_NULL) {
410 return UINT_NULL;
411 }
412 else if (vertexMod(vid + 1) == v2) {
413 return vid;
414 }
415 else if (vertexMod((int) vid - 1) == v2) {
416 int n = vertexCount(); // n must be int to avoid unwanted casts
417 return (((int) vid - 1) % n + n) % n;
418 }
419 else {
420 return UINT_NULL;
421 }
422 }
423 }

441 uint indexOfEdge(uint vi1, uint vi2) const
442 {
443 if constexpr (STORE_INDICES) {
444 uint vid = indexOfVertex(vi1);
445 if (vid == UINT_NULL) {
446 return UINT_NULL;
447 }
448 else if (vertexIndexMod(vid + 1) == vi2) {
449 return vid;
450 }
451 else if (vertexIndexMod((int) vid - 1) == vi2) {
452 int n = vertexCount(); // n must be int to avoid unwanted casts
453 return (((int) vid - 1) % n + n) % n;
454 }
455 else {
456 return UINT_NULL;
457 }
458 }
459 else {
460 return indexOfEdge(
461 Base::elemFromParent(vi1), Base::elemFromParent(vi2));
462 }

700/* importFrom function */
701
702template<
703 bool STORE_INDICES,
704 typename Vertex,
705 int N,
706 typename ParentElemType,
707 bool VERT>
708template<typename Element>
709void VertexReferences<STORE_INDICES, Vertex, N, ParentElemType, VERT>::
710 importFrom(const Element& e, bool importRefs)
711{
712 if (importRefs) {
713 if constexpr (HasVertexReferences<Element>) {
714 if constexpr (N > 0) {
715 // same size non-polygonal faces
716 if constexpr (N == Element::VERTEX_COUNT) {
717 importIndicesFrom(e);
718 }
719 // from polygonal to fixed size, but the polygon size == the
720 // fixed face size
721 else if constexpr (Element::VERTEX_COUNT < 0) {
722 if (e.vertexCount() == N) {
723 importIndicesFrom(e);
724 }
725 }
726 else {
727 // do not import in this case: cannot import from a face
728 // of different size
729 }
730 }
731 else {
732 // from fixed to polygonal size: need to resize first, then
733 // import
734 resizeVertices(e.vertexCount());
735 importIndicesFrom(e);
736 }
737 }
738 }
