44template<typename Scalar, uint N>
45class OrthoFrame
46{
47 static_assert(N > 0, "Number of dimensions of OrthoFrame must be > 0.");
48
49 Point<Scalar, N> mOrigin;
50 OrthoBasis<Scalar, N> mBasis;
51
52public:
53 using ScalarType = Scalar;
54 using PointType = Point<Scalar, N>;
55 using BasisType = OrthoBasis<Scalar, N>;
56
57 static const uint DIM = N;
58
65 OrthoFrame() = default;
66
77 template<typename... Axes>
78 OrthoFrame(const PointType& origin, Axes&&... axes)
79 requires (sizeof...(Axes) == N)
80 : mOrigin(origin), mBasis(std::forward<Axes>(axes)...)
81 {
82 }

91 OrthoFrame(const PointType& origin, const std::array<PointType, N>& axes) :
92 mOrigin(origin), mBasis(axes)

94 }

103 OrthoFrame(const PointType& origin, const BasisType& basis) :
104 mOrigin(origin), mBasis(basis)
105 {
106 }
107
122 static OrthoFrame fromNormal(
123 const PointType& origin,
124 const PointType& normal) requires (N == 3)
125 {
126 return OrthoFrame(origin, BasisType::fromNormal(normal));
127 }
128
133 const PointType& origin() const { return mOrigin; }
134
139 PointType& origin() { return mOrigin; }
140
145 const BasisType& basis() const { return mBasis; }
146
151 BasisType& basis() { return mBasis; }
152
161 const PointType& axis(uint i) const { return mBasis.axis(i); }
162
171 PointType& axis(uint i) { return mBasis.axis(i); }
172
185 PointType toWorld(const PointType& local) const
186 {
187 PointType p = mOrigin;
188 p += mBasis.toWorld(local);
189 return p;
190 }
191
203 PointType toLocal(const PointType& world) const
204 {
205 return mBasis.toLocal(world - mOrigin);
206 }
207
218 Eigen::Transform<Scalar, N, Eigen::Affine> transformMatrix() const
219 {
220 Eigen::Transform<Scalar, N, Eigen::Affine> t;
221 t.linear() = mBasis.rotationMatrix();
222 t.translation() = mOrigin;
223 return t;
224 }
225
231 template<typename S>
232 auto cast() const
233 {
234 if constexpr (std::is_same_v<Scalar, S>) {
235 return *this;
236 }
237 else {
238 return OrthoFrame<S, N>(
239 mOrigin.template cast<S>(), mBasis.template cast<S>());
240 }
241 }
242
247 void serialize(std::ostream& os) const
248 {
249 mOrigin.serialize(os);
250 mBasis.serialize(os);
251 }
252
257 void deserialize(std::istream& is)
258 {
259 mOrigin.deserialize(is);
260 mBasis.deserialize(is);
261 }
262
263 bool operator==(const OrthoFrame& f) const
264 {
265 return mOrigin == f.mOrigin && mBasis == f.mBasis;
266 }
267
268 bool operator!=(const OrthoFrame& f) const { return !(*this == f); }
269};
