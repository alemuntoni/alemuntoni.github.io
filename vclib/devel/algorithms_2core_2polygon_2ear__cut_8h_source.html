74template<Point2IteratorConcept Iterator>
75std::vector<uint> earCut(Iterator begin, Iterator end)
76{
77 using PointT = Iterator::value_type;
78 using Scalar = PointT::ScalarType;
79
80 // Convert the polygon represented as a sequence of vertices to a vector
81 // of vectors of points. This is necessary because the earcut library
82 // expects the polygon to be represented as a vector of contours, where
83 // each contour is a vector of points. In this case, there is only one
84 // contour, which is represented as a vector of points.
85 std::vector<std::vector<Point2<Scalar>>> poly;
86 poly.emplace_back(begin, end);
87
88 // Use the earcut library to triangulate the polygon and return the
89 // result.
90 return mapbox::earcut<uint>(poly);
91}

128template<Point3IteratorConcept Iterator>
129std::vector<uint> earCut(Iterator begin, Iterator end)
130{
131 // Project the 3D polygon onto a 2D plane defined by its normal.
132 auto poly2D = project(begin, end);
133
134 // Use the ear-cut algorithm to triangulate the polygon in the 2D plane
135 // and return the result.
136 return earCut(poly2D.begin(), poly2D.end());
137}
