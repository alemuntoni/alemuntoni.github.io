273 template<typename Iterator>
274 static PointT normal(Iterator begin, Iterator end) requires (
275 std::is_same_v<typename Iterator::value_type, PointT> &&
276 PointT::DIM == 3)
277 {
278 // compute the sum of normals for each triplet of consecutive points
279 PointT sum;
280 sum.setZero();
281 for (auto i = begin; i != end; ++i) {
282 auto i1 = i;
283 ++i1;
284 if (i1 == end)
285 i1 = begin;
286
287 auto i2 = i1;
288 ++i2;
289 if (i2 == end)
290 i2 = begin;
291
292 sum += (*i1 - *i).cross(*i2 - *i);
293 }
294 sum.normalize();
295 return sum;
296 }

323 template<typename Iterator>
324 static PointT barycenter(Iterator begin, Iterator end)
325 requires (std::is_same_v<typename Iterator::value_type, PointT>)
326 {
327 PointT bar;
328 bar.setZero();
329
330 uint cnt = 0;
331 for (; begin != end; ++begin) {
332 bar += *begin;
333 ++cnt;
334 }
335
336 assert(cnt);
337
338 return bar / cnt;
339 }

371 template<typename Iterator, typename WIterator>
372 static PointT weightedBarycenter(
373 Iterator begin,
374 Iterator end,
375 WIterator wBegin)
376 requires (std::is_same_v<typename Iterator::value_type, PointT>)
377 {
378 using ScalarType = WIterator::value_type;
379
380 PointT bar;
381 bar.setZero();
382 ScalarType wSum = 0;
383
384 for (; begin != end; ++begin, ++wBegin) {
385 bar += *begin * *wBegin;
386 wSum += *wBegin;
387 }
388
389 return bar / wSum;
390 }

429 template<typename Iterator>
430 static ScalarType perimeter(Iterator begin, Iterator end)
431 requires (std::is_same_v<typename Iterator::value_type, PointT>)
432 {
433 using Scalar = PointType::ScalarType;
434
435 Scalar per = 0;
436 for (auto i = begin; i != end; ++i) {
437 const PointT& p0 = *i;
438 auto i1 = i;
439 ++i;
440 if (i1 == end)
441 i1 = begin;
442 const PointT& p1 = *i1;
443 per += p0.dist(p1);
444 }
445 return per;
446 }

477 template<typename Iterator>
478 static ScalarType area(Iterator begin, Iterator end)
479 requires (std::is_same_v<typename Iterator::value_type, PointT>)
480 {
481 using Scalar = PointType::ScalarType;
482
483 PointType bar = barycenter(begin, end);
484 Scalar area = 0;
485 for (auto i = begin; i != end; ++i) {
486 const PointT& p0 = *i;
487 auto i1 = i;
488 ++i1;
489 if (i1 == end)
490 i1 = begin;
491 const PointT& p1 = *i1;
492
493 area += Triangle<PointT>::area(p0, p1, bar);
494 }
495 return area;
496 }
