130 void set(const bimg::ImageContainer& image, bool hasMips, uint64_t flags)
131 {
132 const bool imageHasAlreadyMips = image.m_numMips > 1;
133 const bool passDataDirectly =
134 !hasMips || (hasMips && imageHasAlreadyMips);
135
136 if (bgfx::isValid(mTextureHandle))
137 bgfx::destroy(mTextureHandle);
138
139 if (passDataDirectly) {
140 const uint8_t* imageData = (uint8_t*) image.m_data;
141
142 if (image.m_cubeMap) {
143 mTextureHandle = bgfx::createTextureCube(
144 image.m_width,
145 hasMips,
146 image.m_numLayers,
147 bgfx::TextureFormat::Enum(image.m_format),
148 flags,
149 bgfxCopyMemory(imageData, image.m_size));
150 }
151 else {
152 mTextureHandle = bgfx::createTexture2D(
153 image.m_width,
154 image.m_height,
155 hasMips,
156 image.m_numLayers,
157 bgfx::TextureFormat::Enum(image.m_format),
158 flags,
159 bgfxCopyMemory(imageData, image.m_size));
160 }
161 }
162 else {
163 // Image has not mips but we still want to allocate the related
164 // memory. For this we need to actually create an empty texture and
165 // update it manually to avoid undefined behavior.
166
167 if (image.m_cubeMap) {
168 mTextureHandle = bgfx::createTextureCube(
169 image.m_width,
170 hasMips,
171 image.m_numLayers,
172 bgfx::TextureFormat::Enum(image.m_format),
173 flags);
174
175 for (uint8_t face = 0; face < 6; face++)
176 copyMip0ToTexture(image, face);
177 }
178 else {
179 mTextureHandle = bgfx::createTexture2D(
180 image.m_width,
181 image.m_height,
182 hasMips,
183 image.m_numLayers,
184 bgfx::TextureFormat::Enum(image.m_format),
185 flags);
186
187 copyMip0ToTexture(image);
188 }
189 }
190 }

207 void set(
208 const void* data,
209 const vcl::Point2i& size,
210 bool hasMips = false,
211 uint64_t flags = BGFX_TEXTURE_NONE | BGFX_SAMPLER_NONE,
212 bgfx::TextureFormat::Enum format = bgfx::TextureFormat::RGBA8,
213 bool isCubemap = false,
214 bgfx::ReleaseFn releaseFn = nullptr)
215 {
216 uint32_t sz = bimg::imageGetSize(
217 nullptr,
218 size.x(),
219 size.y(),
220 1,
221 isCubemap,
222 hasMips,
223 1,
224 // there is correspondence between bimg and bgfx texture formats
225 static_cast<bimg::TextureFormat::Enum>(toUnderlying(format)));
226
227 const bgfx::Memory* mem = nullptr;
228 if (data != nullptr) {
229 mem = bgfx::makeRef(data, sz, releaseFn);
230 }
231
232 set(mem, size, hasMips, 1, format, isCubemap, flags);
233 }

321 static uint samplerFlagsFromTexture(const TextureDescriptor& tex)
322 {
323 using enum TextureDescriptor::MinificationFilter;
324 using enum TextureDescriptor::WrapMode;
325
326 uint flags = BGFX_SAMPLER_NONE;
327
328 // set minification filter - bgfx default is linear
329 if (tex.minFilter() == NEAREST ||
330 tex.minFilter() == NEAREST_MIPMAP_LINEAR ||
331 tex.minFilter() == NEAREST_MIPMAP_NEAREST)
332 flags |= BGFX_SAMPLER_MIN_POINT;
333
334 // set mipmap filter - bgfx default is linear
335 if (tex.minFilter() == NEAREST_MIPMAP_NEAREST ||
336 tex.minFilter() == LINEAR_MIPMAP_NEAREST)
337 flags |= BGFX_SAMPLER_MIP_POINT;
338
339 // set magnification filter - bgfx default is linear
340 if (tex.magFilter() == TextureDescriptor::MagnificationFilter::NEAREST)
341 flags |= BGFX_SAMPLER_MAG_POINT;
342
343 // set wrap modes - bgfx default is repeat
344 if (tex.wrapU() == CLAMP_TO_EDGE)
345 flags |= BGFX_SAMPLER_U_CLAMP;
346 else if (tex.wrapU() == MIRRORED_REPEAT)
347 flags |= BGFX_SAMPLER_U_MIRROR;
348
349 if (tex.wrapV() == CLAMP_TO_EDGE)
350 flags |= BGFX_SAMPLER_V_CLAMP;
351 else if (tex.wrapV() == MIRRORED_REPEAT)
352 flags |= BGFX_SAMPLER_V_MIRROR;
353
354 return flags;
355 }

357private:
358 void copyMip0ToTexture(const bimg::ImageContainer& image, uint8_t face = 0)
359 {
360 bimg::ImageMip mip;
361 bimg::imageGetRawData(image, face, 0, image.m_data, image.m_size, mip);
362
363 if (image.m_cubeMap) {
364 bgfx::updateTextureCube(
365 mTextureHandle,
366 0,
367 face,
368 0,
369 0,
370 0,
371 mip.m_width,
372 mip.m_height,
373 bgfxCopyMemory(mip.m_data, mip.m_size));
374 }
375 else {
376 bgfx::updateTexture2D(
377 mTextureHandle,
378 0,
379 0,
380 0,
381 0,
382 mip.m_width,
383 mip.m_height,
384 bgfxCopyMemory(mip.m_data, mip.m_size));
385 }
386 }

388 static const bgfx::Memory* bgfxCopyMemory(
389 const uint8_t* data,
390 uint32_t size)
391 {
392 auto [buffer, releaseFn] =
393 Context::getAllocatedBufferAndReleaseFn<uint8_t>(size);
394
395 std::copy(data, data + size, buffer);
396
397 return bgfx::makeRef(buffer, size, releaseFn);
398 }
