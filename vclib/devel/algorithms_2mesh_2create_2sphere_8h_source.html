105template<FaceMeshConcept MeshType>
106MeshType createSphereUV(
107 const SphereConcept auto& sp,
108 uint parallels = 10,
109 uint meridians = 20)
110{
111 using PositionType = PositionType<MeshType>;
112 using IndexType = IndexType<MeshType>;
113
114 MeshType mesh;
115
116 mesh.addVertex(PositionType(0, 1, 0));
117 for (uint j = 0; j < parallels - 1; ++j) {
118 const double polar = M_PI * double(j + 1) / double(parallels);
119 const double sp = std::sin(polar);
120 const double cp = std::cos(polar);
121 for (uint i = 0; i < meridians; ++i) {
122 const double azimuth = 2.0 * M_PI * double(i) / double(meridians);
123 const double sa = std::sin(azimuth);
124 const double ca = std::cos(azimuth);
125 const double x = sp * ca;
126 const double y = cp;
127 const double z = sp * sa;
128 mesh.addVertex(PositionType(x, y, z));
129 }
130 }
131 mesh.addVertex(PositionType(0, -1, 0));
132
133 for (uint i = 0; i < meridians; ++i) {
134 auto v = mesh.vertex(0);
135 auto a = mesh.vertex(i + 1);
136 auto b = mesh.vertex((i + 1) % meridians + 1);
137 mesh.addFace(0, b, a);
138 }
139
140 for (uint j = 0; j < parallels - 2; ++j) {
141 uint aStart = j * meridians + 1;
142 uint bStart = (j + 1) * meridians + 1;
143 for (uint i = 0; i < meridians; ++i) {
144 auto a = mesh.vertex(aStart + i);
145 auto a1 = mesh.vertex(aStart + (i + 1) % meridians);
146 auto b = mesh.vertex(bStart + i);
147 auto b1 = mesh.vertex(bStart + (i + 1) % meridians);
148 if constexpr (HasTriangles<MeshType>) {
149 mesh.addFace(a, a1, b1);
150 mesh.addFace(b1, b, a);
151 }
152 else {
153 mesh.addFace(a, a1, b1, b);
154 }
155 }
156 }
157
158 for (uint i = 0; i < meridians; ++i) {
159 auto a = mesh.vertex(i + meridians * (parallels - 2) + 1);
160 auto b =
161 mesh.vertex((i + 1) % meridians + meridians * (parallels - 2) + 1);
162 auto v = mesh.vertex(mesh.vertexCount() - 1);
163 mesh.addFace(v, a, b);
164 }
165
166 scale(mesh, sp.radius());
167 translate(mesh, sp.center());
168
169 return mesh;
170}

185template<FaceMeshConcept MeshType>
186MeshType createSphereNormalizedCube(
187 const SphereConcept auto& sp,
188 uint divisions)
189{
190 using PositionType = PositionType<MeshType>;
191 using ScalarType = ScalarType<MeshType>;
192
193 MeshType mesh;
194
195 const double step = 1.0 / double(divisions);
196 const PositionType step3(step, step, step);
197
198 for (uint face = 0; face < 6; ++face) {
199 const PositionType origin =
200 detail::cts::origins[face].cast<ScalarType>();
201 const PositionType right = detail::cts::rights[face].cast<ScalarType>();
202 const PositionType up = detail::cts::ups[face].cast<ScalarType>();
203 for (uint j = 0; j < divisions + 1; ++j) {
204 const PositionType j3(j, j, j);
205 for (uint i = 0; i < divisions + 1; ++i) {
206 const PositionType i3(i, i, i);
207 const PositionType p =
208 origin + step3.mul(i3.mul(right) + j3.mul(up));
209
210 mesh.addVertex(p.normalized());
211 }
212 }
213 }
214
215 const uint k = divisions + 1;
216 for (uint face = 0; face < 6; ++face) {
217 for (uint j = 0; j < divisions; ++j) {
218 const bool bottom = j < (divisions / 2);
219 for (uint i = 0; i < divisions; ++i) {
220 const bool left = i < (divisions / 2);
221 auto a = mesh.vertex((face * k + j) * k + i);
222 auto b = mesh.vertex((face * k + j) * k + i + 1);
223 auto c = mesh.vertex((face * k + j + 1) * k + i);
224 auto d = mesh.vertex((face * k + j + 1) * k + i + 1);
225
226 if constexpr (HasTriangles<MeshType>) {
227 if (bottom ^ left) {
228 mesh.addFace(a, c, b);
229 mesh.addFace(c, d, b);
230 }
231 else {
232 mesh.addFace(a, c, d);
233 mesh.addFace(a, d, b);
234 }
235 }
236 else {
237 mesh.addFace(a, c, d, b);
238 }
239 }
240 }
241 }
242
243 scale(mesh, sp.radius());
244 translate(mesh, sp.center());
245
246 return mesh;
247}

262template<FaceMeshConcept MeshType>
263MeshType createSphereSpherifiedCube(
264 const SphereConcept auto& sp,
265 uint divisions)
266{
267 using PositionType = PositionType<MeshType>;
268 using ScalarType = ScalarType<MeshType>;
269
270 MeshType mesh;
271
272 const double step = 1.0 / double(divisions);
273 const PositionType step3(step, step, step);
274
275 for (uint face = 0; face < 6; ++face) {
276 const PositionType origin =
277 detail::cts::origins[face].cast<ScalarType>();
278 const PositionType right = detail::cts::rights[face].cast<ScalarType>();
279 const PositionType up = detail::cts::ups[face].cast<ScalarType>();
280 for (uint j = 0; j < divisions + 1; ++j) {
281 const PositionType j3(j, j, j);
282 for (uint i = 0; i < divisions + 1; ++i) {
283 const PositionType i3(i, i, i);
284 const PositionType p =
285 origin + step3.mul(i3.mul(right) + j3.mul(up));
286 const PositionType p2 = p.mul(p);
287 const PositionType n(
288 p.x() * std::sqrt(
289 1.0 - 0.5 * (p2.y() + p2.z()) +
290 p2.y() * p2.z() / 3.0),
291 p.y() * std::sqrt(
292 1.0 - 0.5 * (p2.z() + p2.x()) +
293 p2.z() * p2.x() / 3.0),
294 p.z() * std::sqrt(
295 1.0 - 0.5 * (p2.x() + p2.y()) +
296 p2.x() * p2.y() / 3.0));
297 mesh.addVertex(n);
298 }
299 }
300 }
301
302 const uint k = divisions + 1;
303 for (uint face = 0; face < 6; ++face) {
304 for (uint j = 0; j < divisions; ++j) {
305 const bool bottom = j < (divisions / 2);
306 for (uint i = 0; i < divisions; ++i) {
307 const bool left = i < (divisions / 2);
308 auto a = mesh.vertex((face * k + j) * k + i);
309 auto b = mesh.vertex((face * k + j) * k + i + 1);
310 auto c = mesh.vertex((face * k + j + 1) * k + i);
311 auto d = mesh.vertex((face * k + j + 1) * k + i + 1);
312 if constexpr (HasTriangles<MeshType>) {
313 if (bottom ^ left) {
314 mesh.addFace(a, c, b);
315 mesh.addFace(c, d, b);
316 }
317 else {
318 mesh.addFace(a, c, d);
319 mesh.addFace(a, d, b);
320 }
321 }
322 else {
323 mesh.addFace(a, c, d, b);
324 }
325 }
326 }
327 }
328
329 scale(mesh, sp.radius());
330 translate(mesh, sp.center());
331
332 return mesh;
333}

348template<FaceMeshConcept MeshType>
349MeshType createSphereIcosahedron(const SphereConcept auto& sp, uint divisions)
350{
351 using PositionType = PositionType<MeshType>;
352 using ScalarType = ScalarType<MeshType>;
353 using IndexType = IndexType<MeshType>;
354
355 MeshType mesh = createIcosahedron<MeshType>(true);
356
357 for (uint d = 0; d < divisions; d++) {
358 IndexType nf = mesh.faceCount();
359 for (IndexType f = 0; f < nf; f++) {
360 auto f0 = mesh.face(f);
361 auto v0 = f0.vertex(0);
362 auto v1 = f0.vertex(1);
363 auto v2 = f0.vertex(2);
364 auto v1id = v1.index();
365 auto v2id = v2.index();
366
367 PositionType pa = (v0.position() + v1.position());
368 pa.normalize();
369 PositionType pb = (v1.position() + v2.position());
370 pb.normalize();
371 PositionType pc = (v2.position() + v0.position());
372 pc.normalize();
373 auto vaid = mesh.addVertex(pa);
374 auto vbid = mesh.addVertex(pb);
375 auto vcid = mesh.addVertex(pc);
376
377 f0.setVertex(1, vaid);
378 f0.setVertex(2, vcid);
379 mesh.addFace(vaid, v1id, vbid);
380 mesh.addFace(vcid, vbid, v2id);
381 mesh.addFace(vaid, vbid, vcid);
382 }
383 }
384
385 removeDuplicateVertices(mesh);
386
387 scale(mesh, sp.radius());
388 translate(mesh, sp.center());
389
390 return mesh;
391}

408template<FaceMeshConcept MeshType>
409MeshType createSphere(
410 const SphereConcept auto& sp,
411 const CreateSphereArgs& args = CreateSphereArgs())
412{
413 using enum CreateSphereArgs::CreateSphereMode;
414
415 MeshType m;
416 switch (args.mode) {
417 case UV:
418 m = createSphereUV<MeshType>(sp, args.parallels, args.meridians);
419 break;
420 case NORMALIZED_CUBE:
421 m = createSphereNormalizedCube<MeshType>(sp, args.divisions);
422 break;
423 case SPHERIFIED_CUBE:
424 m = createSphereSpherifiedCube<MeshType>(sp, args.divisions);
425 break;
426 case ICOSAHEDRON:
427 m = createSphereIcosahedron<MeshType>(sp, args.divisions);
428 break;
429 }
430
431 return m;
432}
