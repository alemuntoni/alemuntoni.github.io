105template<FaceMeshConcept MeshType>
106MeshType createSphereUV(
107 const SphereConcept auto& sp,
108 uint parallels = 10,
109 uint meridians = 20)
110{
111 using VertexType = MeshType::VertexType;
112 using PositionType = VertexType::PositionType;
113 using Facetype = MeshType::FaceType;
114
115 MeshType mesh;
116
117 mesh.addVertex(PositionType(0, 1, 0));
118 for (uint j = 0; j < parallels - 1; ++j) {
119 const double polar = M_PI * double(j + 1) / double(parallels);
120 const double sp = std::sin(polar);
121 const double cp = std::cos(polar);
122 for (uint i = 0; i < meridians; ++i) {
123 const double azimuth = 2.0 * M_PI * double(i) / double(meridians);
124 const double sa = std::sin(azimuth);
125 const double ca = std::cos(azimuth);
126 const double x = sp * ca;
127 const double y = cp;
128 const double z = sp * sa;
129 mesh.addVertex(PositionType(x, y, z));
130 }
131 }
132 mesh.addVertex(PositionType(0, -1, 0));
133
134 for (uint i = 0; i < meridians; ++i) {
135 VertexType* v = &mesh.vertex(0);
136 VertexType* a = &mesh.vertex(i + 1);
137 VertexType* b = &mesh.vertex((i + 1) % meridians + 1);
138 mesh.addFace(0, b, a);
139 }
140
141 for (uint32_t j = 0; j < parallels - 2; ++j) {
142 uint32_t aStart = j * meridians + 1;
143 uint32_t bStart = (j + 1) * meridians + 1;
144 for (uint32_t i = 0; i < meridians; ++i) {
145 VertexType* a = &mesh.vertex(aStart + i);
146 VertexType* a1 = &mesh.vertex(aStart + (i + 1) % meridians);
147 VertexType* b = &mesh.vertex(bStart + i);
148 VertexType* b1 = &mesh.vertex(bStart + (i + 1) % meridians);
149 if constexpr (HasTriangles<MeshType>) {
150 mesh.addFace(a, a1, b1);
151 mesh.addFace(b1, b, a);
152 }
153 else {
154 mesh.addFace(a, a1, b1, b);
155 }
156 }
157 }
158
159 for (uint32_t i = 0; i < meridians; ++i) {
160 VertexType* a = &mesh.vertex(i + meridians * (parallels - 2) + 1);
161 VertexType* b =
162 &mesh.vertex((i + 1) % meridians + meridians * (parallels - 2) + 1);
163 VertexType* v = &mesh.vertex(mesh.vertexCount() - 1);
164 mesh.addFace(v, a, b);
165 }
166
167 scale(mesh, sp.radius());
168 translate(mesh, sp.center());
169
170 return mesh;
171}

186template<FaceMeshConcept MeshType>
187MeshType createSphereNormalizedCube(
188 const SphereConcept auto& sp,
189 uint divisions)
190{
191 using VertexType = MeshType::VertexType;
192 using PositionType = VertexType::PositionType;
193 using Facetype = MeshType::FaceType;
194 using ScalarType = PositionType::ScalarType;
195
196 MeshType mesh;
197
198 const double step = 1.0 / double(divisions);
199 const PositionType step3(step, step, step);
200
201 for (uint face = 0; face < 6; ++face) {
202 const PositionType origin =
203 detail::cts::origins[face].cast<ScalarType>();
204 const PositionType right = detail::cts::rights[face].cast<ScalarType>();
205 const PositionType up = detail::cts::ups[face].cast<ScalarType>();
206 for (uint j = 0; j < divisions + 1; ++j) {
207 const PositionType j3(j, j, j);
208 for (uint i = 0; i < divisions + 1; ++i) {
209 const PositionType i3(i, i, i);
210 const PositionType p =
211 origin + step3.mul(i3.mul(right) + j3.mul(up));
212
213 mesh.addVertex(p.normalized());
214 }
215 }
216 }
217
218 const uint k = divisions + 1;
219 for (uint face = 0; face < 6; ++face) {
220 for (uint j = 0; j < divisions; ++j) {
221 const bool bottom = j < (divisions / 2);
222 for (uint i = 0; i < divisions; ++i) {
223 const bool left = i < (divisions / 2);
224 VertexType* a = &mesh.vertex((face * k + j) * k + i);
225 VertexType* b = &mesh.vertex((face * k + j) * k + i + 1);
226 VertexType* c = &mesh.vertex((face * k + j + 1) * k + i);
227 VertexType* d = &mesh.vertex((face * k + j + 1) * k + i + 1);
228
229 if constexpr (HasTriangles<MeshType>) {
230 if (bottom ^ left) {
231 mesh.addFace(a, c, b);
232 mesh.addFace(c, d, b);
233 }
234 else {
235 mesh.addFace(a, c, d);
236 mesh.addFace(a, d, b);
237 }
238 }
239 else {
240 mesh.addFace(a, c, d, b);
241 }
242 }
243 }
244 }
245
246 scale(mesh, sp.radius());
247 translate(mesh, sp.center());
248
249 return mesh;
250}

265template<FaceMeshConcept MeshType>
266MeshType createSphereSpherifiedCube(
267 const SphereConcept auto& sp,
268 uint divisions)
269{
270 using VertexType = MeshType::VertexType;
271 using PositionType = VertexType::PositionType;
272 using ScalarType = PositionType::ScalarType;
273 using Facetype = MeshType::FaceType;
274
275 MeshType mesh;
276
277 const double step = 1.0 / double(divisions);
278 const PositionType step3(step, step, step);
279
280 for (uint face = 0; face < 6; ++face) {
281 const PositionType origin =
282 detail::cts::origins[face].cast<ScalarType>();
283 const PositionType right = detail::cts::rights[face].cast<ScalarType>();
284 const PositionType up = detail::cts::ups[face].cast<ScalarType>();
285 for (uint j = 0; j < divisions + 1; ++j) {
286 const PositionType j3(j, j, j);
287 for (uint i = 0; i < divisions + 1; ++i) {
288 const PositionType i3(i, i, i);
289 const PositionType p =
290 origin + step3.mul(i3.mul(right) + j3.mul(up));
291 const PositionType p2 = p.mul(p);
292 const PositionType n(
293 p.x() * std::sqrt(
294 1.0 - 0.5 * (p2.y() + p2.z()) +
295 p2.y() * p2.z() / 3.0),
296 p.y() * std::sqrt(
297 1.0 - 0.5 * (p2.z() + p2.x()) +
298 p2.z() * p2.x() / 3.0),
299 p.z() * std::sqrt(
300 1.0 - 0.5 * (p2.x() + p2.y()) +
301 p2.x() * p2.y() / 3.0));
302 mesh.addVertex(n);
303 }
304 }
305 }
306
307 const uint k = divisions + 1;
308 for (uint face = 0; face < 6; ++face) {
309 for (uint j = 0; j < divisions; ++j) {
310 const bool bottom = j < (divisions / 2);
311 for (uint i = 0; i < divisions; ++i) {
312 const bool left = i < (divisions / 2);
313 VertexType* a = &mesh.vertex((face * k + j) * k + i);
314 VertexType* b = &mesh.vertex((face * k + j) * k + i + 1);
315 VertexType* c = &mesh.vertex((face * k + j + 1) * k + i);
316 VertexType* d = &mesh.vertex((face * k + j + 1) * k + i + 1);
317 if constexpr (HasTriangles<MeshType>) {
318 if (bottom ^ left) {
319 mesh.addFace(a, c, b);
320 mesh.addFace(c, d, b);
321 }
322 else {
323 mesh.addFace(a, c, d);
324 mesh.addFace(a, d, b);
325 }
326 }
327 else {
328 mesh.addFace(a, c, d, b);
329 }
330 }
331 }
332 }
333
334 scale(mesh, sp.radius());
335 translate(mesh, sp.center());
336
337 return mesh;
338}

353template<FaceMeshConcept MeshType>
354MeshType createSphereIcosahedron(const SphereConcept auto& sp, uint divisions)
355{
356 using VertexType = MeshType::VertexType;
357 using PositionType = VertexType::PositionType;
358 using FaceType = MeshType::FaceType;
359
360 MeshType mesh = createIcosahedron<MeshType>(true);
361
362 for (uint d = 0; d < divisions; d++) {
363 uint nf = mesh.faceCount();
364 for (uint f = 0; f < nf; f++) {
365 FaceType& f0 = mesh.face(f);
366 VertexType& v0 = *f0.vertex(0);
367 VertexType& v1 = *f0.vertex(1);
368 VertexType& v2 = *f0.vertex(2);
369 uint v1id = mesh.index(v1);
370 uint v2id = mesh.index(v2);
371
372 PositionType pa = (v0.position() + v1.position());
373 pa.normalize();
374 PositionType pb = (v1.position() + v2.position());
375 pb.normalize();
376 PositionType pc = (v2.position() + v0.position());
377 pc.normalize();
378 uint vaid = mesh.addVertex(pa);
379 uint vbid = mesh.addVertex(pb);
380 uint vcid = mesh.addVertex(pc);
381
382 f0.setVertex(1, vaid);
383 f0.setVertex(2, vcid);
384 mesh.addFace(vaid, v1id, vbid);
385 mesh.addFace(vcid, vbid, v2id);
386 mesh.addFace(vaid, vbid, vcid);
387 }
388 }
389
390 removeDuplicateVertices(mesh);
391
392 scale(mesh, sp.radius());
393 translate(mesh, sp.center());
394
395 return mesh;
396}

413template<FaceMeshConcept MeshType>
414MeshType createSphere(
415 const SphereConcept auto& sp,
416 const CreateSphereArgs& args = CreateSphereArgs())
417{
418 using enum CreateSphereArgs::CreateSphereMode;
419
420 MeshType m;
421 switch (args.mode) {
422 case UV:
423 m = createSphereUV<MeshType>(sp, args.parallels, args.meridians);
424 break;
425 case NORMALIZED_CUBE:
426 m = createSphereNormalizedCube<MeshType>(sp, args.divisions);
427 break;
428 case SPHERIFIED_CUBE:
429 m = createSphereSpherifiedCube<MeshType>(sp, args.divisions);
430 break;
431 case ICOSAHEDRON:
432 m = createSphereIcosahedron<MeshType>(sp, args.divisions);
433 break;
434 }
435
436 return m;
437}
