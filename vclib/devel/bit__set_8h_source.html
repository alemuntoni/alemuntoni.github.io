74 template<NonBoolIntegralOrEnum I>
75 BitSet(std::initializer_list<I> l)
76 {
77 for (const auto& i : l) {
78 if constexpr (std::is_enum_v<I>)
79 at(toUnderlying(i)) = true;
80 else
81 at(i) = true;
82 }

94 template<typename B>
95 BitSet(std::initializer_list<B> l) requires std::same_as<bool, B>
96 {
97 if (l.size() > SIZE)
98 throw std::invalid_argument(
99 "BitSet: list size is greater than the number of bits of the "
100 "BitSet");
101
102 for (uint i = 0; const auto& b : l)
103 at(i++) = b;
104 }

385private:
386 static constexpr uint bitIndex(uint i)
387 {
388 if constexpr (MSB)
389 return static_cast<uint>(SIZE - i - 1);
390 else
391 return i;
