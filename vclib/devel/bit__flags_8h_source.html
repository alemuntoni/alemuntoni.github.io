267 // Component interface functions
268 template<typename Element>
269 void importFrom(const Element& e, bool = true)
270 {
271 if constexpr (HasBitFlags<Element>) {
272 resetBitFlags();
273 if constexpr (
274 HasPolygonBitFlags<Element> || HasTriangleBitFlags<Element>) {
275 deletedBit() = e.deleted();
276 selected() = e.selected();
277 visited() = e.visited();
278 onBorder() = e.onBorder();
279 const uint UM = std::min(USER_BIT_COUNT, e.USER_BIT_COUNT);
280 for (uint i = 0; i < UM; ++i)
281 userBit(i) = e.userBit(i);
282 }
283 else {
284 flags() = e.flags();
285 }
286 }
287 }
