25template<typename T>
26inline constexpr bool IsRawSerializable =
27 std::endian::native == std::endian::little &&
28 std::is_trivially_copyable_v<T> &&
29 (std::has_unique_object_representations_v<T> ||
30 std::is_floating_point_v<T>) &&
31 !std::is_same_v<T, bool>;

281 template<typename Index>

283 {
284 if constexpr (vectorCount() > 0) {
285 vectorCompact<vectorCount() - 1>(newIndices);
286 }
287 std::size_t newSize = 0;

289 if (newIndices[i] != std::numeric_limits<Index>::max()) {

526 template<std::size_t I>
527 void serializeVector(std::ostream& os) const
528 {
529 if (mVecEnabled[I]) {
530 const auto& vec = std::get<I>(mVecTuple);
531 using IType = std::tuple_element_t<I, std::tuple<Types...>>;
532 if constexpr (detail::IsRawSerializable<IType>) {
533 if (mSize > 0) {
534 os.write(
535 reinterpret_cast<const char*>(vec.data()),
536 static_cast<std::streamsize>(mSize * sizeof(IType)));
537 }
538 }
539 else {
540 for (const auto& elem : vec) {
541 vcl::serialize(os, elem);
542 }
543 }
544 }
545 }
546
547 template<std::size_t... Is>
548 void deserializeVectors(std::istream& is, std::index_sequence<Is...>)
549 {
550 (deserializeVector<Is>(is), ...);
551 }
552
553 template<std::size_t I>
554 void deserializeVector(std::istream& is)
555 {
556 auto& vec = std::get<I>(mVecTuple);
557 if (mVecEnabled[I]) {
558 vec.resize(mSize);
559 using IType = std::tuple_element_t<I, std::tuple<Types...>>;
560 if constexpr (detail::IsRawSerializable<IType>) {
561 if (mSize > 0) {
562 is.read(
563 reinterpret_cast<char*>(vec.data()),
564 static_cast<std::streamsize>(mSize * sizeof(IType)));
565 }
566 }
567 else {
568 for (auto& elem : vec) {
569 vcl::deserialize(is, elem);
570 }
571 }
572 }
573 else {
574 vec.clear();
575 }
576 }
