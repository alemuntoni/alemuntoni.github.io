120inline double lnOfFactorial(int n)
121{
122 static const int FAK_LEN = 1024;
123
124 // define constants
125 static const double // coefficients in Stirling approximation
126 C0 = 0.918938533204672722, // ln(sqrt(2*pi))
127 C1 = 1. / 12., C3 = -1. / 360.;
128 // C5 = 1./1260., // use r^5 term if FAK_LEN < 50
129 // C7 = -1./1680.; // use r^7 term if FAK_LEN < 20
130 // static variables
131 static double fac_table[FAK_LEN]; // table of ln(n!):
132 // remember if fac_table has been initialized
133 static bool initialized = false;
134
135 if (n < FAK_LEN) {
136 if (n <= 1) {
137 if (n < 0)
138 assert(0); //("Parameter negative in LnFac function");
139 return 0;
140 }
141 if (!initialized) { // first time. Must initialize table
142 // make table of ln(n!)
143 double sum = fac_table[0] = 0.;
144 for (uint i = 1; i < FAK_LEN; i++) {
145 sum += std::log(double(i));
146 fac_table[i] = sum;
147 }
148 initialized = true;
149 }
150 return fac_table[n];
151 }
152 else {
153 // not found in table. use Stirling approximation
154 double n1, r;
155 n1 = n;
156 r = 1. / n1;
157 return (n1 + 0.5) * std::log(n1) - n1 + C0 + r * (C1 + r * r * C3);
158 }
159}
