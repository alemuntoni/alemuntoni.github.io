291 IndexType addElement()
292 {
293 std::size_t id = mElemComponents.size();
294 mElemComponents.resize(id + 1);
295 if constexpr (comp::HasCustomComponents<Components>) {
296 mCustomComponents.resize(id + 1);
297 }
298 ++mElemCount;
299 initDefaultComponents(id, id + 1);
300 return static_cast<IndexType>(id);

314 mElemComponents.resize(baseId + n);
315 if constexpr (comp::HasCustomComponents<Components>) {
316 mCustomComponents.resize(baseId + n);
317 }
318 mElemCount += n;
319 initDefaultComponents(baseId, baseId + n);
320 return static_cast<IndexType>(baseId);

354 void resizeElements(std::size_t newSize)
355 {
356 std::size_t oldSize = mElemComponents.size();
357 mElemComponents.resize(newSize);
358 if constexpr (comp::HasCustomComponents<Components>) {
359 mCustomComponents.resize(newSize);
360 }
361 if (newSize > oldSize) {
362 mElemCount += (newSize - oldSize);
363 initDefaultComponents(oldSize, newSize);
364 }
365 else if (newSize < oldSize) {
366 mElemCount = 0;
367 for (std::size_t i = 0; i < newSize; ++i) {
368 if (!isElementDeleted(i)) {
369 ++mElemCount;
370 }
371 }
372 }

505 IndexType elementIndexIfCompact(IndexType i) const
506 {
507 assert(i < size());
508 if (isElementDeleted(i)) {
509 return NULL_INDEX;
510 }
511 if (isElementContainerCompact()) {
512 return i;
513 }
514 else {

535 std::vector<IndexType> elementCompactIndices() const
536 {
537 std::vector<IndexType> newIndices(size());
538 if (isElementContainerCompact()) {
539 for (IndexType i = 0; i < size(); ++i) {
540 newIndices[i] = i;
541 }
542 return newIndices;
543 }
544
545 IndexType currentIndex = 0;
546 for (IndexType i = 0; i < size(); ++i) {
547 if (!isElementDeleted(i)) {
548 newIndices[i] = currentIndex++;
549 }
550 else {
551 newIndices[i] = NULL_INDEX;
552 }
553 }
554 return newIndices;

790 template<CompId COMP_ID>
791 void enableOptionalComponent()
792 {
793 static_assert(
794 comp::HasComponent<COMP_ID, Components>,
795 "Component not found in container");
796 if constexpr (comp::HasOptionalComponent<COMP_ID, Components>) {
797 if (!isComponentAvailable<COMP_ID>()) {
798 constexpr uint N =
799 comp::IndexOfCompId<COMP_ID, VectorComponents>::value;
800 mElemComponents.template enable<N>();
801 using Comp = comp::ComponentTypeT<COMP_ID, Components>;
802 if constexpr (comp::HasPolygonVertexReferences<Components>) {
803 if constexpr (comp::IsTiedToVertexCount<Comp>) {
804 for (auto e : elements(false)) {
805 e.Comp::resize(e.vertexCount());
806 }
807 }
808 }
809 if constexpr (comp::HasDefaultValue<Comp>) {
810 auto& vec = this->template vector<COMP_ID>();
811 std::fill(
812 vec.begin(), vec.end(), comp::defaultValue<Comp>());
813 }
814 }
815 }

1165 void importFrom(const OtherMesh& m)
1166 {
1167 if constexpr (OtherMesh::template hasContainerOf<ELEMENT_ID>()) {
1168 const auto& c = m.template container<ELEMENT_ID>();
1169 clearElements();
1170 const std::size_t n = c.size();

1176 if constexpr (
1177 comp::HasCustomComponents<Components> &&
1178 comp::HasCustomComponents<
1179 typename std::remove_cvref_t<decltype(c)>::Components>) {
1180 mCustomComponents = c.mCustomComponents;

1403 void initDefaultComponentsImpl(
1404 vcl::TypeWrapper<Comps...>,
1405 std::size_t begin,
1406 std::size_t end)
1407 {
1408 auto initComp = [&]<typename Comp>() {
1409 if constexpr (detail::HasVectorColumnPred<Comp>::value) {
1410 if constexpr (comp::HasDefaultValue<Comp>) {
1411 if (this->template isComponentAvailable<
1412 Comp::COMPONENT_ID>()) {

1460 template<typename... Comps>
1461 void appendImpl(vcl::TypeWrapper<Comps...>, const ElementContainer& other)
1462 {
1463 const std::size_t on = other.size();
1464 if (on == 0) {
1465 return;
1466 }
1467
1468 const std::size_t n = size();
1469
1470 // 1. Allocate space for 'on' new elements across all active component vectors.
1471 // Note: addElements(on) temporarily increments mElemCount by on (assuming all valid).
1472 addElements(on);
1473
1474 // 2. Fast-copy component data: for each component enabled in both containers,
1475 // copy elements from other's vector into this vector starting at offset n.
1476 auto copyComponent = [&]<typename Comp>() {
1477 if constexpr (Comp::COMPONENT_ID != CompId::CUSTOM_COMPONENTS) {
1478 if (this->template isComponentAvailable<Comp::COMPONENT_ID>() &&
1479 other.template isComponentAvailable<Comp::COMPONENT_ID>()) {
1480 auto& thisVec = this->template vector<Comp::COMPONENT_ID>();
1481 const auto& otherVec =
1482 other.template vector<Comp::COMPONENT_ID>();
1483 std::copy(
1484 otherVec.begin(), otherVec.end(), thisVec.begin() + n);
1485 }
1486 }
1487 };
1488
1489 (copyComponent.template operator()<Comps>(), ...);
1490
1491 // 3. Preserve TTVC (Tied-To-Vertex-Count) invariant on polygon meshes:
1492 // If an optional TTVC component (e.g., AdjacentFaces, AdjacentEdges, Wedges,
1493 // PolygonBitFlags) is enabled in 'this' container but was disabled in 'other',
1494 // addElements(on) initialized inner vectors to size 0. Since the vertex references
1495 // were copied in step 2, we now have the correct vertexCount() for each appended
1496 // polygon and can resize the missing TTVC component to match it.
1497 if constexpr (comp::HasPolygonVertexReferences<Components>) {
1498 auto resizeTTVCIfMissingInOther = [&]<typename Comp>() {
1499 if constexpr (comp::IsTiedToVertexCount<Comp>) {
1500 if (this->template isComponentAvailable<
1501 Comp::COMPONENT_ID>() &&
1502 !other.template isComponentAvailable<
1503 Comp::COMPONENT_ID>()) {
1504 for (std::size_t i = n; i < n + on; ++i) {
1505 auto p = element(i);
1506 p.Comp::resize(p.vertexCount());
1507 }
1508 }
1509 }
1510 };
1511
1512 (resizeTTVCIfMissingInOther.template operator()<Comps>(), ...);
1513 }
1514
1515 // 4. Append custom dynamic components (if supported).
1516 if constexpr (comp::HasCustomComponents<Components>) {
1517 mCustomComponents.append(other.mCustomComponents, n, on);
1518 }
1519
1520 // 5. Correct the valid element counter:
1521 // addElements(on) assumed all 'on' elements were valid (+on).
1522 // We deduct 'on' and add other.elementCount() so deleted elements in 'other'
1523 // are properly accounted for.
1524 mElemCount = (mElemCount - on) + other.elementCount();
1525 }
