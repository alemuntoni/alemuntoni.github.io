461 void importFlagsFromVCGFormat(int f)
462 {
463 if (f & 0x00000010)
464 visited() = true;
465 if (f & 0x00000020)
466 selected() = true;
467 if (f & 0x00000040)
468 edgeOnBorder(0) = true;
469 if (f & 0x00000080)
470 edgeOnBorder(1) = true;
471 if (f & 0x00000100)
472 edgeOnBorder(2) = true;
473 if (f & 0x00008000)
474 edgeSelected(0) = true;
475 if (f & 0x00010000)
476 edgeSelected(1) = true;
477 if (f & 0x00020000)
478 edgeSelected(2) = true;
479 if (f & 0x00040000)
480 edgeFaux(0) = true;
481 if (f & 0x00080000)
482 edgeFaux(1) = true;
483 if (f & 0x00100000)
484 edgeFaux(2) = true;
485 }

494 int exportFlagsToVCGFormat() const
495 {
496 int f = 0;
497 if (visited())
498 f |= 0x00000010;
499 if (selected())
500 f |= 0x00000020;
501 if (edgeOnBorder(0))
502 f |= 0x00000040;
503 if (edgeOnBorder(1))
504 f |= 0x00000080;
505 if (edgeOnBorder(2))
506 f |= 0x00000100;
507 if (edgeSelected(0))
508 f |= 0x00008000;
509 if (edgeSelected(1))
510 f |= 0x00010000;
511 if (edgeSelected(2))
512 f |= 0x00020000;
513 if (edgeFaux(0))
514 f |= 0x00040000;
515 if (edgeFaux(1))
516 f |= 0x00080000;
517 if (edgeFaux(2))
518 f |= 0x00100000;
519 return f;
520 }

525 // Component interface functions
526 template<typename Element>
527 void importFrom(const Element& e, bool = true)
528 {
529 if constexpr (HasBitFlags<Element>) {
530 resetBitFlags();
531 if constexpr (HasPolygonBitFlags<Element>) {
532 flags() = e.flags();
533 edgeFlags() = e.edgeFlags();
534 }
535 else {
536 // BitFlags
537 deletedBit() = e.deleted();
538 selected() = e.selected();
539 visited() = e.visited();
540 const uint UM = std::min(USER_BIT_COUNT, e.USER_BIT_COUNT);
541 for (uint i = 0; i < UM; ++i)
542 userBit(i) = e.userBit(i);
543 if constexpr (HasTriangleBitFlags<Element>) {
544 // TriangleBitFlags
545 for (uint i = 0; i < 3; ++i) {
546 edgeOnBorder(i) = e.edgeOnBorder(i);
547 edgeSelected(i) = e.edgeSelected(i);
548 edgeVisited(i) = e.edgeVisited(i);
549 edgeFaux(i) = e.edgeFaux(i);
550 }
551 }
552 }
553 }
