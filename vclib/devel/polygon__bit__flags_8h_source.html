29template<int N = -1>
30struct PolygonBitFlagsData
31{
32 BitSet<uint> flags;
33 vcl::Vector<BitSet<uchar>, N> edgeFlags;
34
35 PolygonBitFlagsData() = default;
36
37 explicit PolygonBitFlagsData(std::size_t n) requires (N < 0)
38 : edgeFlags(n)
39 {
40 }
41
42 decltype(auto) operator[](this auto&& self, std::size_t bit)
43 {
44 return self.flags[bit];
45 }
46
47 bool operator==(const PolygonBitFlagsData&) const = default;
48
49 void serialize(std::ostream& os) const
50 {
51 vcl::serialize(os, flags, edgeFlags);
52 }
53
54 void deserialize(std::istream& is)
55 {
56 vcl::deserialize(is, flags, edgeFlags);
57 }
58};

121template<int N = -1>
122class PolygonBitFlags :

127 static constexpr uint FIRST_USER_BIT = 15;
128 static constexpr uint FIRST_EDGE_USER_BIT = 3;
129
130 // indices of the bits
131 enum {
132 DELETED = 0, // bit 0
133 SELECTED = 1, // bit 1
134 VISITED = 2, // bit 2
135 // Faux edges, for portability with TriangleBits
136 FAUX0 = 3, // bits [3, 5]
137 };
138
139 // indices of the bits used for each edge of the polygon
140 enum { EDGEBORD = 0, EDGESEL = 1, EDGEVIS = 2 };
141
142public:
146 using FlagsType = uint;
147
151 using EdgeFlagsType = uchar;
152
156 static constexpr uint USER_BIT_COUNT =
157 sizeof(FlagsType) * 8 - FIRST_USER_BIT;
158
163 static constexpr uint EDGE_USER_BIT_COUNT =
164 sizeof(EdgeFlagsType) * 8 - FIRST_EDGE_USER_BIT;
165
169 static constexpr int EDGE_COUNT = N;
170
175 static constexpr bool TIED_TO_VERTEX_COUNT = (N < 0);

312 void resetBitFlags(this auto&& self)
313 {
314 bool isD = self.deleted();
315 self.flags().reset();
316 const uint n = self.edgeFlags().size();
317 for (uint i = 0; i < n; ++i) {
318 self.edgeFlags()[i].reset();

381 void importFlagsFromVCGFormat(this auto&& self, int f)
382 {
383 self.resetBitFlags();
384 if (f & 0x00000010)
385 self.visited() = true;
386 if (f & 0x00000020)
387 self.selected() = true;
388 if (self.edgeFlags().size() > 0 && (f & 0x00000040))
389 self.edgeOnBorder(0) = true;
390 if (self.edgeFlags().size() > 1 && (f & 0x00000080))
391 self.edgeOnBorder(1) = true;
392 if (self.edgeFlags().size() > 2 && (f & 0x00000100))
393 self.edgeOnBorder(2) = true;
394 if (self.edgeFlags().size() > 0 && (f & 0x00008000))
395 self.edgeSelected(0) = true;
396 if (self.edgeFlags().size() > 1 && (f & 0x00010000))
397 self.edgeSelected(1) = true;
398 if (self.edgeFlags().size() > 2 && (f & 0x00020000))
399 self.edgeSelected(2) = true;
400 if (f & 0x00040000)
401 self.edgeFaux(0) = true;
402 if (f & 0x00080000)
403 self.edgeFaux(1) = true;
404 if (f & 0x00100000)
405 self.edgeFaux(2) = true;
406 }

415 int exportFlagsToVCGFormat(this auto&& self)
416 {
417 int f = 0;
418 if (self.visited())
419 f |= 0x00000010;
420 if (self.selected())
421 f |= 0x00000020;
422 if (self.edgeFlags().size() > 0 && self.edgeOnBorder(0))
423 f |= 0x00000040;
424 if (self.edgeFlags().size() > 1 && self.edgeOnBorder(1))
425 f |= 0x00000080;
426 if (self.edgeFlags().size() > 2 && self.edgeOnBorder(2))
427 f |= 0x00000100;
428 if (self.edgeFlags().size() > 0 && self.edgeSelected(0))
429 f |= 0x00008000;
430 if (self.edgeFlags().size() > 1 && self.edgeSelected(1))
431 f |= 0x00010000;
432 if (self.edgeFlags().size() > 2 && self.edgeSelected(2))
433 f |= 0x00020000;
434 if (self.edgeFaux(0))
435 f |= 0x00040000;
436 if (self.edgeFaux(1))
437 f |= 0x00080000;
438 if (self.edgeFaux(2))
439 f |= 0x00100000;
440 return f;
441 }

510template<int N>
511template<typename OtherElement>
512void PolygonBitFlags<N>::importFrom(
513 this auto&& self,
514 const OtherElement& other,
515 bool)
516{
517 if constexpr (HasBitFlags<OtherElement>) {
518 const bool wasDeleted = self.deleted();
519 const bool isDeleted = other.deleted();
520 self.resetBitFlags();
521 if constexpr (HasPolygonBitFlags<OtherElement>) {
522 self.setUnderlyingBitFlags(other.underlyingBitFlags());
523 self.setDeleted(isDeleted);
524 const uint n = std::min(
525 (uint) self.edgeCount(),
526 (uint) other.edgeCount());
527 for (uint i = 0; i < n; ++i) {
528 self.setUnderlyingEdgeBitFlags(i, other.underlyingEdgeBitFlags(i));
529 }
530 }
531 else {
532 self.setDeleted(isDeleted);
533 self.selected() = other.selected();
534 self.visited() = other.visited();
535
536 const uint um =
537 std::min(USER_BIT_COUNT, (uint) OtherElement::USER_BIT_COUNT);
538 for (uint i = 0; i < um; ++i) {
539 self.userBit(i) = other.userBit(i);
540 }
541 if constexpr (HasTriangleBitFlags<OtherElement>) {
542 const uint numEdges =
543 std::min((uint) self.edgeCount(), 3u);
544 for (uint i = 0; i < numEdges; ++i) {
545 self.edgeOnBorder(i) = other.edgeOnBorder(i);
546 self.edgeSelected(i) = other.edgeSelected(i);
547 self.edgeVisited(i) = other.edgeVisited(i);
548 self.edgeFaux(i) = other.edgeFaux(i);
549 }
550 }
551 }
552 if constexpr (CanAdjustContainerElemCount<decltype(self)>) {
553 if (self.isValid()) {
554 self.parentContainer().adjustElemCountOnDeletedChange(
555 wasDeleted, isDeleted);
556 }
557 }
558 }
559}
