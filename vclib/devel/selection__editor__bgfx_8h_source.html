213 void draw(uint viewId) override
214 {
215 if (!isSelectionActive()) {
216 mSelectionBox.reset();
217 mSelectionAnchor.reset();
218 return;
219 }
220
221 // Re-configure visible selection views every frame — required by Qt
222 // (all other views follow this pattern; see context.cpp comment)
223 bgfx::setViewFrameBuffer(
224 mVisibleSelectionViewIds[0], mVisibleSelectionFB.handle());
225 bgfx::setViewClear(
226 mVisibleSelectionViewIds[0],
227 BGFX_CLEAR_COLOR | BGFX_CLEAR_DEPTH,
228 UINT_NULL);
229 bgfx::setViewRect(
230 mVisibleSelectionViewIds[0],
231 0,
232 0,
233 mVisibleFaceFBSize,
234 mVisibleFaceFBSize);
235 bgfx::touch(mVisibleSelectionViewIds[0]);
236
237 bgfx::setViewFrameBuffer(
238 mVisibleSelectionViewIds[1], mComputePassFB.handle());
239 bgfx::setViewClear(mVisibleSelectionViewIds[1], BGFX_CLEAR_NONE);
240 bgfx::setViewRect(mVisibleSelectionViewIds[1], 0, 0, 1, 1);
241 bgfx::touch(mVisibleSelectionViewIds[1]);
242
243 // 1) Draw the selection box overlay (only visible while dragging)
244 if (mSelectionInProgress) {
245 mScreenSpaceBox.setBox(mSelectionBox.value_or(Box2d {}));
246 mScreenSpaceBox.setVisible(true);
247 mScreenSpaceBox.draw(viewId);
248 }
249 else {
250 mScreenSpaceBox.setVisible(false);
251 }
252
253 // 2) Compute GPU selections with current parameters, if pending
254 if (mSelectionCalcRequired) {
255 computeSelections(viewId);
256 // Trigger continuous updates to initiate GPU-to-CPU selection
257 // readback
258 Base::viewerUpdate();
259
260 // Computation request done — reset flag
261 mSelectionCalcRequired = false;
262 }
263
264 // Request continuous updates until GPU-to-CPU readback completes for
265 // all meshes
266 bool readbackPending = false;
267 for (const auto& obj : *Base::drawList()) {
268 auto mesh = std::dynamic_pointer_cast<AbstractDrawableMesh>(obj);
269 if (mesh && mesh->isSelectionReadbackPending()) {
270 readbackPending = true;
271 Base::viewerUpdate();
272 break;
273 }
274 }
275
276 // Finalize undo/redo action once readback completes
277 if (!readbackPending && mActionCreationPending &&
278 !mSelectionInProgress) {
279 finalizeSelectionAction();
280 mActionCreationPending = false;
281 }
282 }

289 bool onKeyPress(Key::Enum key, const KeyModifiers& modifiers) override
290 {
291 if (!isSelectionActive() || mSelectionInProgress)
292 return false;
293
294 auto actionOpt = mSettings.keyBindings.action({key, modifiers});
295 if (actionOpt.has_value()) {
296 SelectionAtomicAction action = actionOpt.value();
297
298 if (!mActionCreationPending) {
299 savePreSelectionStates();
300 mActionCreationPending = true;
301 }
302
303 mCurrentSelectionModes = actionModesForSettings(action);
304 mSelectionCalcRequired = true;
305 return true;
306 }
307 return false;
308 }
309
310 bool onKeyRelease(Key::Enum key, const KeyModifiers& modifiers) override
311 {
312 return false;
313 }
314
315 bool onMouseMove(double x, double y, const KeyModifiers& modifiers) override
316 {
317 if (!isSelectionActive())
318 return false;
319 if (mSelectionInProgress) {
320 mSelectionBox = Box2d(mSelectionAnchor.value());
321 mSelectionBox->add(Point2d {x, y});
322
323 if (mCurrentMouseAction.has_value()) {
324 mCurrentSelectionModes =
325 actionModesForSettings(mCurrentMouseAction.value());
326 }
327
328 mSelectionCalcRequired = true;
329 return true; // Consume event only while dragging
330 }
331 return false; // Let it propagate if not dragging
332 }
333
334 bool onMousePress(
335 vcl::MouseButton::Enum button,
336 double x,
337 double y,
338 const vcl::KeyModifiers& modifiers) override
339 {
340 if (!isSelectionActive())
341 return false;
342
343 auto actionOpt = mSettings.mouseBindings.action({button, modifiers});
344 if (actionOpt.has_value() && !mSelectionInProgress) {
345 SelectionDragAction action = actionOpt.value();
346
347 if (!mActionCreationPending) {
348 savePreSelectionStates();
349 mActionCreationPending = true;
350 }
351 mSelectionInProgress = true;
352 mSelectionAnchor = Point2d {x, y};
353 mSelectionBox = Box2d({x, y});
354 mCurrentMouseAction = actionOpt.value();
355 mCurrentSelectionModes = actionModesForSettings(action);
356 return true; // Smart blocking
357 }
358 return false;
359 }
360
361 bool onMouseRelease(
362 MouseButton::Enum button,
363 double x,
364 double y,
365 const KeyModifiers& modifiers) override
366 {
367 if (mSelectionInProgress) {
368 // We don't check the button here to gracefully handle edge cases
369 // where release might happen differently, but we could enforce it.
370 mSelectionInProgress = false;
371 mCurrentMouseAction.reset();
372 mSelectionCalcRequired = true;
373 // Force a repaint so the selection box disappears immediately
374 Base::viewerUpdate();
375 return true; // We handled the release of our drag
376 }
377 return false;
378 }

383 return false; // SelectionEditor doesn't consume scroll events

455 bool setVisibleTrisSelectionProjViewMatrix(const Box2d& box)
456 {
457 if (box.isNull() || box.isDegenerate()) {
458 return false;
459 }
460 auto s = Base::viewerCanvasSize();
461 uint win_w = s.x();
462 uint win_h = s.y();
463 Point4f minNDC(
464 float(box.min().x()) / float(win_w) * 2.f - 1.f,
465 1.f - float(box.max().y()) / float(win_h) * 2.f,
466 0.f,
467 1.f);
468 Point4f maxNDC(
469 float(box.max().x()) / float(win_w) * 2.f - 1.f,
470 1.f - float(box.min().y()) / float(win_h) * 2.f,
471 0.f,
472 1.f);
473 float w = maxNDC.x() - minNDC.x();
474 float h = maxNDC.y() - minNDC.y();
475 assert(w > 0.f && h > 0.f);
476 // set the matrix to translate and scale (scale * translate)
477 // translate to the box center and scale to box size (relative to NDC)
478 // (translate by -minNDC + 1/2 * dim then scale by
479 // 2/dim to map the box to the whole viewport)
480 const Matrix44f scaleTranslate {
481 {2.f / w, 0.f, 0.f, -(2.f / w * minNDC.x() + 1.f)},
482 {0.f, 2.f / h, 0.f, -(2.f / h * minNDC.y() + 1.f)},
483 {0.f, 0.f, 1.f, 0.f },
484 {0.f, 0.f, 0.f, 1.f }
485 };
486 auto vm = Base::viewerViewMatrix();
487 auto pm = Base::viewerProjectionMatrix();
488 Matrix44f newProj = scaleTranslate * pm;
489 bgfx::setViewTransform(
490 mVisibleSelectionViewIds[0], vm.data(), newProj.data());
491 return true;
492 }
493
501 void computeSelections(uint viewId)
502 {
503 Box2d minMaxBox = mSelectionBox.value_or(Box2d {});
504
505 // If any active mode is a visible-selection mode, set up the
506 // restricted projection matrix once for all such modes.
507 bool hasVisibleMode = false;
508 for (const auto& mode : mCurrentSelectionModes) {
509 if (mode.isVisibleSelection()) {
510 hasVisibleMode = true;
511 break;
512 }
513 }
514
515 bool skipVisibleSelection = false;
516 if (hasVisibleMode && !minMaxBox.isNull()) {
517 skipVisibleSelection = !setVisibleTrisSelectionProjViewMatrix(
518 calculateWindowSpaceMeshBB().intersection(minMaxBox));
519 }
520
521 auto dl = Base::drawList();
522
523 for (const auto& mode : mCurrentSelectionModes) {
524 if (mode.isVisibleSelection() && skipVisibleSelection) {
525 // Box is outside the mesh screen-space BB.
526 // For REGULAR mode we must clear the current selection;
527 // for ADD/SUBTRACT we simply skip (no change).
528 if (mode.primitive == SelectionPrimitive::FACE &&
529 mode.isAction(SelectionDragAction::REGULAR) &&
530 mode.visible) {
531 SelectionParameters clearParams = {
532 viewId,
533 mVisibleSelectionViewIds[0],
534 mVisibleSelectionViewIds[1],
535 Box2d(),
536 SelectionMode {
537 SelectionPrimitive::FACE,
538 SelectionAtomicAction::NONE },
539 mSelectionInProgress,
540 bgfx::getTexture(mVisibleSelectionFB, 0),
541 bgfx::getTexture(mVisibleSelectionFB, 1),
542 {mVisibleFaceFBSize, mVisibleFaceFBSize},
543 0
544 };
545 for (size_t i = 0; i < dl->size(); i++) {
546 if (!shouldProcessObject(*dl, i)) {
547 continue;
548 }
549 clearParams.meshId = i;
550 auto el = dl->at(i);
551 if (auto p =
552 dynamic_cast<AbstractDrawableMesh*>(el.get())) {
553 p->computeSelection(clearParams);
554 }
555 }
556 }
557 continue; // skip this visible-selection mode
558 }
559
560 SelectionParameters params = {
561 viewId,
562 mVisibleSelectionViewIds[0],
563 mVisibleSelectionViewIds[1],
564 minMaxBox,
565 mode,
566 mSelectionInProgress,
567 bgfx::getTexture(mVisibleSelectionFB, 0),
568 bgfx::getTexture(mVisibleSelectionFB, 1),
569 {mVisibleFaceFBSize, mVisibleFaceFBSize},
570 0
571 };
572 for (size_t i = 0; i < dl->size(); i++) {
573 if (!shouldProcessObject(*dl, i)) {
574 continue;
575 }
576 params.meshId = i;
577 auto el = dl->at(i);
578 if (auto p = dynamic_cast<AbstractDrawableMesh*>(el.get())) {
579 p->computeSelection(params);
580 }
581 }
582 }
583
584 // Computation done — clear box if drag ended
585 if (!mSelectionInProgress) {
586 mSelectionBox.reset();
587 mSelectionAnchor.reset();
588 }
589 }

595 Box2d calculateWindowSpaceMeshBB()
596 {
597 auto vmf = Base::viewerViewMatrix();
598 auto pmf = Base::viewerProjectionMatrix();
599 Matrix44d vMat = vmf.template cast<double>();
600 Matrix44d pMat = pmf.template cast<double>();
601
602 Box3d frustumNDC;
603 frustumNDC.add(Point3d {-1.0, -1.0, 0.0});
604 frustumNDC.add(Point3d {1.0, 1.0, 1.0});
605
606 Box3d totalBB;
607 auto dl = Base::drawList();
608 for (size_t i = 0; i < dl->size(); i++) {
609 if (!shouldProcessObject(*dl, i)) {
610 continue;
611 }
612 std::shared_ptr<DrawableObject> el = dl->at(i);
613 if (!el->isVisible() ||
614 !dynamic_cast<AbstractDrawableMesh*>(el.get())) {
615 continue;
616 }
617 Box3d bb = el->boundingBox();
618 if (bb.isNull()) {
619 continue;
620 }
621 Point3d maxBB = bb.max();
622 Point3d minBB = bb.min();
623 Point4d boxPoints[8] = {
624 Point4d {minBB.x(), minBB.y(), minBB.z(), 1.0},
625 Point4d {maxBB.x(), minBB.y(), minBB.z(), 1.0},
626 Point4d {minBB.x(), maxBB.y(), minBB.z(), 1.0},
627 Point4d {maxBB.x(), maxBB.y(), minBB.z(), 1.0},
628 Point4d {minBB.x(), minBB.y(), maxBB.z(), 1.0},
629 Point4d {minBB.x(), maxBB.y(), maxBB.z(), 1.0},
630 Point4d {maxBB.x(), minBB.y(), maxBB.z(), 1.0},
631 Point4d {maxBB.x(), maxBB.y(), maxBB.z(), 1.0}
632 };
633 Box3d tempBox;
634 for (const Point4d& p : boxPoints) {
635 Point4d pNDC = pMat * vMat * p;
636 pNDC /= pNDC.w();
637 tempBox.add(pNDC.head<3>());
638 }
639 Box3d inters = tempBox.intersection(frustumNDC);
640 if (inters.isNull()) {
641 continue;
642 }
643 totalBB.add(inters);
644 }
645
646 Box2d box;
647 if (totalBB.isNull()) {
648 return box;
649 }
650 auto s = Base::viewerCanvasSize();
651 uint width = s.x();
652 uint height = s.y();
653 Point3d boxPts[2] = {totalBB.min(), totalBB.max()};
654 Point2d sSpace[2];
655 for (size_t i = 0; i < 2; i++) {
656 sSpace[i] = Point2d {
657 (boxPts[i].x() + 1) / 2 * double(width),
658 (1 - boxPts[i].y()) / 2 * double(height)};
659 }
660 box.add(sSpace[0]);
661 box.add(sSpace[1]);
662 return box;
663 }
664
665 // ---- Undo/Redo helpers ----
666
672 void savePreSelectionStates()
673 {
674 mPreSelectionStates.clear();
675 auto dl = Base::drawList();
676 for (size_t i = 0; i < dl->size(); i++) {
677 if (!shouldProcessObject(*dl, i))
678 continue;
679 auto el = dl->at(i);
680 if (auto p = dynamic_cast<AbstractDrawableMesh*>(el.get())) {
681 SelectionUndoRedoAction::MeshState state;
682 state.obj = el;
683 state.vertexSelection = p->vertexSelectionBitVector();
684 state.faceSelection = p->faceSelectionBitVector();
685 mPreSelectionStates.push_back(std::move(state));
686 }
687 }
688 }
689
695 void finalizeSelectionAction()
696 {
697 bool changed = false;
698 for (auto& state : mPreSelectionStates) {
699 if (auto lock = state.obj.lock()) {
700 if (auto* mesh =
701 dynamic_cast<AbstractDrawableMesh*>(lock.get())) {
702 if (state.vertexSelection !=
703 mesh->vertexSelectionBitVector()) {
704 changed = true;
705 break;
706 }
707 if (state.faceSelection != mesh->faceSelectionBitVector()) {
708 changed = true;
709 break;
710 }
711 }
712 }
713 }
714
715 if (changed) {
716 auto action = std::make_unique<SelectionUndoRedoAction>(
717 std::move(mPreSelectionStates));
718 Base::pushUndoRedoAction(std::move(action));
719 Base::viewerUpdate();
720 }
721 mPreSelectionStates.clear();
722 }
