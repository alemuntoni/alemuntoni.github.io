46template<
47 FaceMeshConcept MeshType,
48 FaceConcept FaceType,
49 typename IndexT = IndexType<MeshType>>
50void addTriangleFacesFromPolygon(
51 MeshType& m,
52 FaceType& f,
53 const std::vector<IndexT>& polygon)
54{
55 using PositionType = VertexType<MeshType>::PositionType;
56
57 // from the ids, create a polygon of positions
58 std::vector<PositionType> polPositions(polygon.size());
59 for (uint i = 0; i < polygon.size(); ++i) {
60 if (polygon[i] >= m.vertexContainerSize()) {
61 throw BadVertexIndexException(
62 "Index " + std::to_string(polygon[i]) +
63 " is out of range in Vertex Container.");
64 }
65 if (m.vertex(polygon[i]).deleted()) {
66 throw BadVertexIndexException(
67 "Vertex " + std::to_string(polygon[i]) + " is deleted.");
68 }
69 polPositions[i] = m.vertex(polygon[i]).position();
70 }
71
72 // compute earcut of the polygons
73 std::vector<uint> tris = earCut(polPositions);
74
75 // faux edges management: create a set of unordered edges of the polygon
76 // note: we use indices from 0 to polygon.size() because that are the output
77 // indices given by the earcut algorithm
78 std::set<std::pair<uint, uint>, UnorderedPairComparator<uint>>
79 unorderedEdges;
80 for (uint i = 0; i < polygon.size(); ++i)
81 unorderedEdges.emplace(i, (i + 1) % (uint) polygon.size());
82
83 if constexpr (FaceType::VERTEX_COUNT < 0) {
84 f.resizeVertices(3);
85 }
86
87 // set the first triangle of the loaded polygon
88 for (uint i = 0; i < f.vertexCount(); ++i) {
89 f.setVertex(i, polygon[tris[i]]);
90 }
91
92 if constexpr (face::HasFaceBitFlags<FaceType>) {
93 if (unorderedEdges.find(std::make_pair(tris[0], tris[1])) ==
94 unorderedEdges.end())
95 f.edgeFaux(0) = true;
96 if (unorderedEdges.find(std::make_pair(tris[1], tris[2])) ==
97 unorderedEdges.end())
98 f.edgeFaux(1) = true;
99 if (unorderedEdges.find(std::make_pair(tris[2], tris[0])) ==
100 unorderedEdges.end())
101 f.edgeFaux(2) = true;
102 }
103
104 // remaining triangles, need to create more faces in the mesh
105 for (uint i = 3; i < tris.size(); i += 3) {
106 auto ff = m.addFace();
107
108 if constexpr (FaceType::VERTEX_COUNT < 0) {
109 m.face(ff).resizeVertices(3);
110 }
111
112 for (uint j = 0; j < m.face(ff).vertexCount(); ++j) {
113 m.face(ff).setVertex(j, polygon[tris[i + j]]);
114 }
115
116 if constexpr (face::HasFaceBitFlags<FaceType>) {
117 if (unorderedEdges.find(std::make_pair(tris[i], tris[i + 1])) ==
118 unorderedEdges.end())
119 m.face(ff).edgeFaux(0) = true;
120 if (unorderedEdges.find(std::make_pair(tris[i + 1], tris[i + 2])) ==
121 unorderedEdges.end())
122 m.face(ff).edgeFaux(1) = true;
123 if (unorderedEdges.find(std::make_pair(tris[i + 2], tris[i + 0])) ==
124 unorderedEdges.end())
125 m.face(ff).edgeFaux(2) = true;
126 }
127 }
128}

146IndexType<MeshType> addTriangleFacesFromPolygon(
147 MeshType& m,
148 const std::vector<IndexT>& polygon)
149{
150 auto fid = m.addFace();
151 addTriangleFacesFromPolygon(m, m.face(fid), polygon);

171template<FaceConcept FaceType>
172bool isFaceManifoldOnEdge(const FaceType& f, uint edge)
173 requires comp::HasAdjacentFaces<FaceType>
174{
175 // Check if the AdjacentFaces component is available for the given face.
176 if (!f.isAdjacentFacesAvailable()) {
177 throw MissingComponentException(
178 "Face has no Adjacent Faces component.");
179 }
180
181 // Check if the edge is a boundary edge.
182 if (f.adjFace(edge) == nullptr) {
183 return true;
184 }
185 else { // Check if the edge is shared by exactly two faces.
186 return f.adjFace(edge).indexOfAdjFace(f) != UINT_NULL;
187 }
188}

252template<FaceConcept FaceType>
253bool checkFlipEdge(const FaceType& f, uint edge)
254 requires comp::HasAdjacentFaces<FaceType>
255{
256 if (!f.isAdjacentFacesAvailable()) {
257 throw MissingComponentException(
258 "Face has no Adjacent Faces component.");
259 }
260
261 if (f.vertexCount() > 3)
262 return false;
263
264 if (isFaceEdgeOnBorder(f, edge))
265 return false;
266
267 auto v0 = f.vertex(edge);
268 auto v1 = f.vertexMod(edge + 1);
269
270 auto of = f.adjFace(edge);
271 uint oe = of.indexOfAdjFace(f);
272 assert(oe != UINT_NULL);
273
274 // check if the vertices of the edge are the same
275 // e.g. the mesh has to be well oriented
276 if (of.vertex(oe) != v1 || of.vertexMod(oe + 1) != v0)
277 return false;
278
279 // check if the flipped edge is already present in the mesh
280 // f_v2 and of_v2 are the vertices of the new edge
281 auto f_v2 = f.vertexMod(edge + 2);
282 auto of_v2 = of.vertexMod(oe + 2);
283
284 MeshPos pos(f, f_v2);
285 MeshPos startPos = pos;
286 // loop in the one ring of f_v2
287 do {
288 pos.nextEdgeAdjacentToV();
289 if (pos.adjVertex() == of_v2)
290 return false;
291 } while (pos != startPos);
292
293 return true;
294}

385template<FaceConcept FaceType>
386auto faceDihedralAngleOnEdge(const FaceType& f, uint e)
387 requires comp::HasAdjacentFaces<FaceType>
388{
389 if (!f.isAdjacentFacesAvailable()) {
390 throw MissingComponentException(
391 "Face has no Adjacent Faces component.");
392 }
393
394 /*
395 * v0 ___________ vf1
396 * |\ |
397 * | e\ f1 |
398 * | \e1 |
399 * |f \ |
400 * | \ |
401 * |__________\|
402 * vf0 v1
403 */
404
405 assert(f.adjFace(e) != nullptr);
406 auto f1 = f.adjFace(e);
407
408 uint e1 = f1.indexOfAdjFace(f);
409 assert(e1 != UINT_NULL);
410
411 auto vf0 = f.vertexMod((int) e - 1);
412 auto vf1 = f1.vertexMod((int) e1 - 1);
413
414 auto n0 = faceNormal(f);
415 auto n1 = faceNormal(f1);
416
417 auto off0 = n0 * vf0.position();
418 auto off1 = n1 * vf1.position();
419
420 auto dist01 = off0 - n0 * vf1.position();
421 auto dist10 = off1 - n1 * vf0.position();
422
423 auto sign = std::abs(dist01) > std::abs(dist10) ? dist01 : dist10;
424
425 auto angleRad = n0.angle(n1);
426 return sign > 0 ? angleRad : -angleRad;
427}

454template<FaceConcept FaceType>
455void detachAdjacentFacesOnEdge(FaceType f, uint edge)
456 requires comp::HasAdjacentFaces<FaceType>
457{
458 if (!f.isAdjacentFacesAvailable()) {
459 throw MissingComponentException(
460 "Face has no Adjacent Faces component.");
461 }
462
463 FaceType nextFace = f.adjFace(edge);
464
465 // if nextFace == nullptr there is nothing to do
466 // the face is already detached on the edge
467 if (nextFace != nullptr) {
468 FaceType prevFace;
469 EdgeAdjFaceIterator<FaceType> begin(f, edge), end;
470 for (auto it = begin; it != end; ++it) {
471 prevFace = *it;
472 }
473
474 if (nextFace == prevFace) { // manifold case
475 uint en = nextFace.indexOfAdjFace(f);
476 assert(en != UINT_NULL);
477 nextFace.setAdjFace(en, FaceType::NULL_INDEX);
478 }
479 else { // non manifold case
480 // the prev face does not have to point to f anymore, but to
481 // nextFace
482 uint pn = prevFace.indexOfAdjFace(f);
483 assert(pn != UINT_NULL);
484 prevFace.setAdjFace(pn, nextFace);
485 }
486 f.setAdjFace(edge, FaceType::NULL_INDEX);
487 }
488}

511template<FaceConcept FaceType>
512void detachFace(FaceType f) requires comp::HasAdjacentFaces<FaceType>
513{
514 if (!f.isAdjacentFacesAvailable()) {
515 throw MissingComponentException(
516 "Face has no Adjacent Faces component.");
517 }
518
519 using VertexType = vcl::VertexType<FaceType>;
520
521 for (uint e = 0; e < f.vertexCount(); ++e) {
522 detachAdjacentFacesOnEdge(f, e);
523
524 // if the vertices have adjacent faces
525 if constexpr (comp::HasAdjacentFaces<VertexType>) {
526 if (f.vertex(e).isAdjacentFacesAvailable()) {
527 VertexType v = f.vertex(e);
528 uint vpos = v.indexOfAdjFace(f);
529 if (vpos != UINT_NULL) { // may happen if vertex adj faces are
530 // not initialized / updated
531 v.eraseAdjFace(vpos); // the vertex v has not anymore the
532 // adjacent face f
533 }
534 }
535 }
536 }

552template<FaceConcept FaceType>
553std::set<FaceType> floodFacePatch(FaceType seed, auto&& faceSelector)
554{
555 if (!seed.isAdjacentFacesAvailable()) {
556 throw MissingComponentException(
557 "Face has no Adjacent Faces component.");
558 }
559
560 std::set<FaceType> faces;
561 // only faces that satisfy the faceSelector will stay on the stack
562 std::vector<FaceType> stackFaces;
563
564 if (!faceSelector(seed))
565 return faces;
566
567 faces.insert(seed);
568
569 for (FaceType adjacent : seed.adjFaces()) {
570 // adding neighbor faces (if faceSelector returns true) to the stack
571 if (adjacent && faceSelector(adjacent))
572 stackFaces.push_back(adjacent);
573 }
574
575 // while there aren't other faces on the stack
576 while (stackFaces.size() > 0) {
577 FaceType fi = stackFaces[stackFaces.size() - 1];
578 stackFaces.pop_back(); // pop
579 faces.insert(fi);
580 for (FaceType adjacent : fi.adjFaces()) {
581 if (adjacent && faceSelector(adjacent)) {
582 if (faces.find(adjacent) == faces.end())
583 stackFaces.push_back(adjacent);
584 }
585 }
586 }
587 return faces;
588}
