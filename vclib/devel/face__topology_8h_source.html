46template<FaceMeshConcept MeshType, FaceConcept FaceType>
47void addTriangleFacesFromPolygon(
48 MeshType& m,
49 FaceType& f,
50 const std::vector<uint>& polygon)
51{
52 using VertexType = MeshType::VertexType;
53 using PositionType = VertexType::PositionType;
54
55 // from the ids, create a polygon of positions
56 std::vector<PositionType> polPositions(polygon.size());
57 for (uint i = 0; i < polygon.size(); ++i) {
58 if (polygon[i] >= m.vertexContainerSize()) {
59 throw BadVertexIndexException(
60 "Index " + std::to_string(polygon[i]) +
61 " is out of range in Vertex Container.");
62 }
63 if (m.vertex(polygon[i]).deleted()) {
64 throw BadVertexIndexException(
65 "Vertex " + std::to_string(polygon[i]) + " is deleted.");
66 }
67 polPositions[i] = m.vertex(polygon[i]).position();
68 }
69
70 // compute earcut of the polygons
71 std::vector<uint> tris = earCut(polPositions);
72
73 // faux edges management: create a set of unordered edges of the polygon
74 // note: we use indices from 0 to polygon.size() because that are the output
75 // indices given by the earcut algorithm
76 std::set<std::pair<uint, uint>, UnorderedPairComparator<uint>>
77 unorderedEdges;
78 for (uint i = 0; i < polygon.size(); ++i)
79 unorderedEdges.emplace(i, (i + 1) % (uint) polygon.size());
80
81 if constexpr (FaceType::VERTEX_COUNT < 0) {
82 f.resizeVertices(3);
83 }
84
85 // set the first triangle of the loaded polygon
86 for (uint i = 0; i < f.vertexCount(); ++i) {
87 f.setVertex(i, polygon[tris[i]]);
88 }
89
90 if constexpr (face::HasFaceBitFlags<FaceType>) {
91 if (unorderedEdges.find(std::make_pair(tris[0], tris[1])) ==
92 unorderedEdges.end())
93 f.edgeFaux(0) = true;
94 if (unorderedEdges.find(std::make_pair(tris[1], tris[2])) ==
95 unorderedEdges.end())
96 f.edgeFaux(1) = true;
97 if (unorderedEdges.find(std::make_pair(tris[2], tris[0])) ==
98 unorderedEdges.end())
99 f.edgeFaux(2) = true;
100 }
101
102 // remaining triangles, need to create more faces in the mesh
103 for (uint i = 3; i < tris.size(); i += 3) {
104 uint ff = m.addFace();
105
106 if constexpr (FaceType::VERTEX_COUNT < 0) {
107 m.face(ff).resizeVertices(3);
108 }
109
110 for (uint j = 0; j < m.face(ff).vertexCount(); ++j) {
111 m.face(ff).setVertex(j, polygon[tris[i + j]]);
112 }
113
114 if constexpr (face::HasFaceBitFlags<FaceType>) {
115 if (unorderedEdges.find(std::make_pair(tris[i], tris[i + 1])) ==
116 unorderedEdges.end())
117 m.face(ff).edgeFaux(0) = true;
118 if (unorderedEdges.find(std::make_pair(tris[i + 1], tris[i + 2])) ==
119 unorderedEdges.end())
120 m.face(ff).edgeFaux(1) = true;
121 if (unorderedEdges.find(std::make_pair(tris[i + 2], tris[i + 0])) ==
122 unorderedEdges.end())
123 m.face(ff).edgeFaux(2) = true;
124 }
125 }
126}

167template<FaceConcept FaceType>
168bool isFaceManifoldOnEdge(const FaceType& f, uint edge)
169 requires comp::HasAdjacentFaces<FaceType>
170{
171 // Check if the AdjacentFaces component is available for the given face.
172 if (!comp::isAdjacentFacesAvailableOn(f)) {
173 throw MissingComponentException(
174 "Face has no Adjacent Faces component.");
175 }
176
177 // Check if the edge is a boundary edge.
178 if (f.adjFace(edge) == nullptr) {
179 return true;
180 }
181 else { // Check if the edge is shared by exactly two faces.
182 return f.adjFace(edge)->indexOfAdjFace(&f) != UINT_NULL;
183 }
184}

248template<FaceConcept FaceType>
249bool checkFlipEdge(const FaceType& f, uint edge)
250 requires comp::HasAdjacentFaces<FaceType>
251{
252 if (!comp::isAdjacentFacesAvailableOn(f)) {
253 throw MissingComponentException(
254 "Face has no Adjacent Faces component.");
255 }
256
257 using VertexType = FaceType::VertexType;
258
259 if (f.vertexCount() > 3)
260 return false;
261
262 if (isFaceEdgeOnBorder(f, edge))
263 return false;
264
265 const VertexType* v0 = f.vertex(edge);
266 const VertexType* v1 = f.vertexMod(edge + 1);
267
268 const FaceType* of = f.adjFace(edge);
269 uint oe = of->indexOfAdjFace(&f);
270 assert(oe != UINT_NULL);
271
272 // check if the vertices of the edge are the same
273 // e.g. the mesh has to be well oriented
274 if (of->vertex(oe) != v1 || of->vertexMod(oe + 1) != v0)
275 return false;
276
277 // check if the flipped edge is already present in the mesh
278 // f_v2 and of_v2 are the vertices of the new edge
279 const VertexType* f_v2 = f.vertexMod(edge + 2);
280 const VertexType* of_v2 = of->vertexMod(oe + 2);
281
282 MeshPos<FaceType> pos(&f, f_v2);
283 MeshPos<FaceType> startPos = pos;
284 // loop in the one ring of f_v2
285 do {
286 pos.nextEdgeAdjacentToV();
287 if (pos.adjVertex() == of_v2)
288 return false;
289 } while (pos != startPos);
290
291 return true;
292}

383template<FaceConcept FaceType>
384auto faceDihedralAngleOnEdge(const FaceType& f, uint e)
385 requires comp::HasAdjacentFaces<FaceType>
386{
387 if (!comp::isAdjacentFacesAvailableOn(f)) {
388 throw MissingComponentException(
389 "Face has no Adjacent Faces component.");
390 }
391
392 /*
393 * v0 ___________ vf1
394 * |\ |
395 * | e\ f1 |
396 * | \e1 |
397 * |f \ |
398 * | \ |
399 * |__________\|
400 * vf0 v1
401 */
402
403 assert(f.adjFace(e) != nullptr);
404 const FaceType& f1 = *(f.adjFace(e));
405
406 uint e1 = f1.indexOfAdjFace(&f);
407 assert(e1 != UINT_NULL);
408
409 const auto& vf0 = *(f.vertexMod((int) e - 1));
410 const auto& vf1 = *(f1.vertexMod(e1 - 1));
411
412 auto n0 = faceNormal(f);
413 auto n1 = faceNormal(f1);
414
415 auto off0 = n0 * vf0.position();
416 auto off1 = n1 * vf1.position();
417
418 auto dist01 = off0 - n0 * vf1.position();
419 auto dist10 = off1 - n1 * vf0.position();
420
421 auto sign = std::abs(dist01) > std::abs(dist10) ? dist01 : dist10;
422
423 auto angleRad = n0.angle(n1);
424 return sign > 0 ? angleRad : -angleRad;
425}

452template<FaceConcept FaceType>
453void detachAdjacentFacesOnEdge(FaceType& f, uint edge)
454 requires comp::HasAdjacentFaces<FaceType>
455{
456 if (!comp::isAdjacentFacesAvailableOn(f)) {
457 throw MissingComponentException(
458 "Face has no Adjacent Faces component.");
459 }
460
461 FaceType* nextFace = f.adjFace(edge);
462
463 // if nextFace == nullptr there is nothing to do
464 // the face is already detached on the edge
465 if (nextFace != nullptr) {
466 FaceType* prevFace;
467 ConstEdgeAdjFaceIterator<FaceType> begin(f, edge), end;
468 for (auto it = begin; it != end; ++it) {
469 prevFace = *it;
470 }
471
472 if (nextFace == prevFace) { // manifold case
473 uint en = nextFace->indexOfAdjFace(&f);
474 assert(en != UINT_NULL);
475 nextFace->setAdjFace(en, nullptr);
476 }
477 else { // non manifold case
478 // the prev face does not have to point to f anymore, but to
479 // nextFace
480 uint pn = prevFace->indexOfAdjFace(&f);
481 assert(pn != UINT_NULL);
482 prevFace->setAdjFace(pn, nextFace);
483 }
484 f.setAdjFace(edge, nullptr);
485 }
486}

509template<FaceConcept FaceType>
510void detachFace(FaceType& f) requires comp::HasAdjacentFaces<FaceType>
511{
512 if (!comp::isAdjacentFacesAvailableOn(f)) {
513 throw MissingComponentException(
514 "Face has no Adjacent Faces component.");
515 }
516
517 using VertexType = FaceType::VertexType;
518
519 for (uint e = 0; e < f.vertexCount(); ++e) {
520 detachAdjacentFacesOnEdge(f, e);
521
522 // if the vertices have adjacent faces
523 if constexpr (comp::HasAdjacentFaces<VertexType>) {
524 if (comp::isAdjacentFacesAvailableOn(f.vertex(e))) {
525 VertexType* v = f.vertex(e);
526 uint vpos = v->indexOfAdjFace(&f);
527 if (vpos != UINT_NULL) { // may happen if vertex adj faces are
528 // not initialized / updated
529 v->eraseAdjFace(vpos); // the vertex v has not anymore the
530 // adjacent face f
531 }
532 }
533 }
534 }

550template<FaceConcept FaceType>
551std::set<const FaceType*> floodFacePatch(
552 const FaceType& seed,
553 auto&& faceSelector)
554{
555 if (!comp::isAdjacentFacesAvailableOn(seed)) {
556 throw MissingComponentException(
557 "Face has no Adjacent Faces component.");
558 }
559
560 std::set<const FaceType*> faces;
561 // only faces that satisfy the faceSelector will stay on the stack
562 std::vector<const FaceType*> stackFaces;
563
564 if (!faceSelector(seed))
565 return faces;
566
567 faces.insert(&seed);
568
569 for (const FaceType* adjacent : seed.adjFaces()) {
570 // adding neighbor faces (if faceSelector returns true) to the stack
571 if (adjacent && faceSelector(*adjacent))
572 stackFaces.push_back(adjacent);
573 }
574
575 // while there aren't other faces on the stack
576 while (stackFaces.size() > 0) {
577 const FaceType* fi = stackFaces[stackFaces.size() - 1];
578 stackFaces.pop_back(); // pop
579 faces.insert(fi);
580 for (const FaceType* adjacent : fi->adjFaces()) {
581 if (adjacent && faceSelector(*adjacent)) {
582 if (faces.find(adjacent) == faces.end())
583 stackFaces.push_back(adjacent);
584 }
585 }
586 }
587 return faces;
588}
