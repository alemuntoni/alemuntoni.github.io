97inline void* getDisplayId()
98{
99 void* displayId = nullptr;
100#ifdef Q_OS_LINUX
101 // Get the display ID depending on whether the system is using X11 or
102 // Wayland
103 if (auto x11App =
104 qApp->nativeInterface<QNativeInterface::QX11Application>()) {
105 displayId = x11App->display();
106 }
107 else if (
108 auto wayApp =
109 qApp->nativeInterface<QNativeInterface::QWaylandApplication>()) {
110 displayId = wayApp->display();
111 }
112#endif
113 return displayId;
114}
