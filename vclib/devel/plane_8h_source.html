30template<typename Scalar, bool NORM = true>
31class Plane
32{
33 Point3<Scalar> mDir;

36public:

40 static constexpr bool NORMED = NORM;

45 Plane() {}

52 Plane(const Point3<Scalar>& direction, Scalar offset) :
53 mDir(direction), mOffset(offset)
54 {
55 if constexpr (NORM) {
56 Scalar n = mDir.norm();
57 mDir /= n;
58 mOffset /= n;
59 }
60 }
61
68 Plane(const Point3<Scalar>& p0, const Point3<Scalar>& normal)
69 {
70 mDir = normal;
71 if constexpr (NORM) {
72 mDir.normalize();
73 }
74 mOffset = p0.dot(mDir);
75 }

83 Plane(
84 const Point3<Scalar>& p0,
85 const Point3<Scalar>& p1,
86 const Point3<Scalar>& p2) :
87 Plane<Scalar, NORM>(p0, (p2 - p0).cross(p1 - p0))
88 {
89 }
90
91 template<typename S>
92 auto cast() const
93 {
94 if constexpr (std::is_same<Scalar, S>::value) {
95 return *this;
96 }
97 else {
98 return Plane<S, NORM>(mDir.template cast<S>(), mOffset);
99 }
100 }
101
106 const Point3<Scalar>& direction() const { return mDir; }
107
112 Scalar offset() const { return mOffset; }
113
119 Point3<Scalar> projectPoint(const Point3<Scalar>& p) const
120 {
121 Scalar k = p.dot(mDir) - mOffset;
122 return p - mDir * k;
123 }
124
130 Point3<Scalar> mirrorPoint(const Point3<Scalar>& p) const
131 {
132 Point3<Scalar> mirr = projectPoint(p);
133 mirr += mirr - p;
134 return mirr;
135 }
136
141 void serialize(std::ostream& os) const
142 {
143 mDir.serialize(os);
144 vcl::serialize(os, mOffset);
145 }
146
151 void deserialize(std::istream& is)
152 {
153 mDir.deserialize(is);
154 vcl::deserialize(is, mOffset);
155 }
156
157 bool operator==(const Plane& p) const
158 {
159 return mOffset == p.mOffset && mDir == p.mDir;
160 }
161
162 bool operator!=(const Plane& p) const { return !(*this == p); }
163};
