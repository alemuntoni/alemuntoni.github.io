64template<typename ScalarType>
65Eigen::MatrixXi rebuildPolygons(
66 const Eigen::MatrixX3i& FR,
67 const Eigen::VectorXi& indices,
68 uint F0Rows,
69 uint m0FaceContainerSize,
70 const TriPolyIndexBiMap& m0BiMap,
71 const TriPolyIndexBiMap& m1BiMap,
72 Eigen::VectorXi& outIndices)
73{
74 // Step 1: Map each resulting triangle to its global polygon ID
75 // (m0 polygons first, then m1 polygons)
76 std::vector<int> globalPolyIds(FR.rows());
77
78 // number of polygons in m0 to offset the global IDs for m1 polygons
79 uint m0PolyCount = m0FaceContainerSize;
80
81 for (int i = 0; i < FR.rows(); ++i) {
82 int j = indices[i];
83 if (j < (int) F0Rows) {
84 globalPolyIds[i] = m0BiMap.polygon(j);
85 }
86 else {
87 globalPolyIds[i] = m0PolyCount + m1BiMap.polygon(j - F0Rows);
88 }
89 }
90
91 // Step 2: Compute triangle-triangle adjacency to find connected components
92 Eigen::MatrixXi TT, TTi;
93 ::igl::triangle_triangle_adjacency(FR, TT, TTi);
94
95 std::vector<bool> visited(FR.rows(), false);
96 std::vector<std::vector<int>> newPolygons;
97 std::vector<int> newPolygonsGlobalIds;
98 std::vector<int> keepTriangles;
99 std::vector<int> keepTrianglesGlobalIds;
100
101 // Step 3: Group adjacent triangles belonging to the same original polygon
102 for (int i = 0; i < FR.rows(); ++i) {
103 if (visited[i])
104 continue;
105
106 // Perform BFS to find all connected triangles from the same source
107 // polygon
108 std::vector<int> comp;
109 std::vector<int> queue;
110 queue.push_back(i);
111 visited[i] = true;
112
113 auto targetId = globalPolyIds[i];
114
115 while (!queue.empty()) {
116 int curr = queue.back();
117 queue.pop_back();
118 comp.push_back(curr);
119
120 for (int e = 0; e < 3; ++e) {
121 int adj = TT(curr, e);
122 // If adjacent triangle exists, is unvisited, and comes from the
123 // same polygon
124 if (adj != -1 && !visited[adj] &&
125 globalPolyIds[adj] == targetId) {
126 visited[adj] = true;
127 queue.push_back(adj);
128 }
129 }
130 }
131
132 // If the component is just a single triangle, keep it as is
133 if (comp.size() == 1) {
134 keepTriangles.push_back(comp[0]);
135 keepTrianglesGlobalIds.push_back(globalPolyIds[comp[0]]);
136 continue;
137 }
138
139 // Step 4: Extract the boundary edges of the grouped triangles
140 std::vector<std::pair<int, int>> boundEdges;
141 bool manifoldBoundary = true;
142 std::map<int, int> inDegree, outDegree;
143 std::map<int, int> nextVert;
144
145 for (int t : comp) {
146 for (int e = 0; e < 3; ++e) {
147 int adj = TT(t, e);
148 // An edge is on the boundary if it has no neighbor or the
149 // neighbor is from a different polygon
150 if (adj == -1 || globalPolyIds[adj] != targetId) {
151 int v0 = FR(t, e);
152 int v1 = FR(t, (e + 1) % 3);
153 boundEdges.push_back({v0, v1});
154
155 // Track vertex degrees to ensure the boundary is a simple
156 // manifold loop
157 outDegree[v0]++;
158 inDegree[v1]++;
159 nextVert[v0] = v1;
160 if (outDegree[v0] > 1 || inDegree[v1] > 1) {
161 manifoldBoundary = false;
162 }
163 }
164 }
165 }
166
167 // If boundary is not manifold (e.g. self-intersecting) or empty,
168 // fallback to triangles
169 if (!manifoldBoundary || boundEdges.empty()) {
170 for (int t : comp) {
171 keepTriangles.push_back(t);
172 keepTrianglesGlobalIds.push_back(globalPolyIds[t]);
173 }
174 continue;
175 }
176
177 // Step 5: Trace the boundary edges to form the new reconstructed
178 // polygon
179 int startVert = boundEdges[0].first;
180 int currVert = startVert;
181 std::vector<int> loop;
182 while (true) {
183 loop.push_back(currVert);
184 auto it = nextVert.find(currVert);
185 if (it == nextVert.end()) {
186 manifoldBoundary =
187 false; // Dead end, should not happen for a closed loop
188 break;
189 }
190 currVert = it->second;
191 if (currVert == startVert)
192 break; // Loop closed
193
194 // Prevent infinite loops in case of disjoint boundary components
195 if (loop.size() > boundEdges.size()) {
196 manifoldBoundary = false;
197 break;
198 }
199 }
200
201 // If a single closed loop was successfully traced and encompasses all
202 // boundary edges
203 if (manifoldBoundary && loop.size() == boundEdges.size()) {
204 newPolygons.push_back(loop);
205 newPolygonsGlobalIds.push_back(globalPolyIds[comp[0]]);
206 }
207 else {
208 // Otherwise, keep the original triangles
209 for (int t : comp) {
210 keepTriangles.push_back(t);
211 keepTrianglesGlobalIds.push_back(globalPolyIds[t]);
212 }
213 }
214 }
215
216 // Step 6: Format the output matrix
217 if (newPolygons.empty()) {
218 outIndices.resize(FR.rows());
219 for (int i = 0; i < FR.rows(); ++i) {
220 outIndices[i] = globalPolyIds[i];
221 }
222 return FR;
223 }
224
225 // Find the maximum polygon size for matrix padding
226 int maxPolySize = 3;
227 for (const auto& poly : newPolygons) {
228 if ((int) poly.size() > maxPolySize) {
229 maxPolySize = poly.size();
230 }
231 }
232
233 // Create the padded matrix, initialized to -1
234 Eigen::MatrixXi FRPoly(
235 keepTriangles.size() + newPolygons.size(), maxPolySize);
236 FRPoly.setConstant(-1);
237 outIndices.resize(keepTriangles.size() + newPolygons.size());
238
239 // Populate the matrix with retained triangles
240 int row = 0;
241 for (size_t i = 0; i < keepTriangles.size(); ++i) {
242 int t = keepTriangles[i];
243 FRPoly(row, 0) = FR(t, 0);
244 FRPoly(row, 1) = FR(t, 1);
245 FRPoly(row, 2) = FR(t, 2);
246 outIndices[row] = keepTrianglesGlobalIds[i];
247 row++;
248 }
249
250 // Populate the matrix with the newly reconstructed polygons
251 for (size_t i = 0; i < newPolygons.size(); ++i) {
252 const auto& poly = newPolygons[i];
253 for (size_t j = 0; j < poly.size(); ++j) {
254 FRPoly(row, j) = poly[j];
255 }
256 outIndices[row] = newPolygonsGlobalIds[i];
257 row++;
258 }
259
260 return FRPoly;
261}

263template<typename MeshType>
264void transferFaceComponents(
265 const MeshType& m0,
266 const MeshType& m1,
267 const Eigen::VectorXi& indices,
268 MeshType& out,
269 auto normalPlaceHolder,
270 Color colorPlaceHolder,
271 auto qualityPlaceHolder)
272{
273 constexpr ElemId F = vcl::ElemId::FACE;
274
275 auto transferComponent = [&]<CompId COMP_ID>(
276 auto& f,
277 int index,
278 uint m0Faces,
279 const auto& placeHolder) {
280 // avoid compile errors for meshes without the component COMP_ID
281 if constexpr (MeshType::template hasElementComponent<F, COMP_ID>()) {
282 if (index < (int) m0Faces) {
283 if (m0.template isElementComponentAvailable<F, COMP_ID>())
284 f.template componentValue<COMP_ID>() =
285 m0.face(index).template componentValue<COMP_ID>();
286 else
287 f.template componentValue<COMP_ID>() = placeHolder;
288 }
289 else {
290 int m1Idx = index - m0Faces;
291 if (m1.template isElementComponentAvailable<F, COMP_ID>())
292 f.template componentValue<COMP_ID>() =
293 m1.face(m1Idx).template componentValue<COMP_ID>();
294 else
295 f.template componentValue<COMP_ID>() = placeHolder;
296 }
297 }
298 };
299
300 bool transferNormals = false;
301 if constexpr (HasFaceNormal<MeshType>) {
302 transferNormals =
303 m0.isFaceNormalAvailable() || m1.isFaceNormalAvailable();
304 }
305 bool transferColors = false;
306 if constexpr (HasFaceColor<MeshType>) {
307 transferColors =
308 m0.isFaceColorAvailable() || m1.isFaceColorAvailable();
309 }
310 bool transferQuality = false;
311 if constexpr (HasFaceQuality<MeshType>) {
312 transferQuality =
313 m0.isFaceQualityAvailable() || m1.isFaceQualityAvailable();
314 }
315
316 if (transferNormals || transferColors || transferQuality) {
317 if (transferNormals)
318 enableIfFaceNormalOptional(out);
319 if (transferColors)
320 enableIfFaceColorOptional(out);
321 if (transferQuality)
322 enableIfFaceQualityOptional(out);
323
324 uint m0Faces = m0.faceContainerSize();
325 for (uint i = 0; auto f : out.faces()) {
326 if (transferNormals)
327 transferComponent.template operator()<CompId::NORMAL>(
328 f, indices[i], m0Faces, normalPlaceHolder);
329 if (transferColors)
330 transferComponent.template operator()<CompId::COLOR>(
331 f, indices[i], m0Faces, colorPlaceHolder);
332 if (transferQuality)
333 transferComponent.template operator()<CompId::QUALITY>(
334 f, indices[i], m0Faces, qualityPlaceHolder);
335 ++i;
336 }
337 }
338}

340template<typename MeshType, typename FMat>
341void transferVertexComponents(
342 const MeshType& m0,
343 const MeshType& m1,
344 const FMat& F0,
345 const FMat& F1,
346 const Eigen::MatrixXi& FRTri,
347 const Eigen::VectorXi& indicesTri,
348 MeshType& out,
349 auto normalPlaceHolder,
350 Color colorPlaceHolder,
351 auto qualityPlaceHolder)
352{
353 constexpr ElemId V = vcl::ElemId::VERTEX;
354 using ScalarType = ScalarType<MeshType>;
355
356 // Determine which vertex components need to be transferred to the output
357 // mesh based on their availability in the input meshes.
358 bool transferNormals = false;
359 if constexpr (HasVertexNormal<MeshType>) {
360 transferNormals =
361 m0.isVertexNormalAvailable() || m1.isVertexNormalAvailable();
362 }
363 bool transferColors = false;
364 if constexpr (HasVertexColor<MeshType>) {
365 transferColors =
366 m0.isVertexColorAvailable() || m1.isVertexColorAvailable();
367 }
368 bool transferQuality = false;
369 if constexpr (HasVertexQuality<MeshType>) {
370 transferQuality =
371 m0.isVertexQualityAvailable() || m1.isVertexQualityAvailable();
372 }
373
374 if (!transferNormals && !transferColors && !transferQuality)
375 return;
376
377 if (transferNormals)
378 enableIfVertexNormalOptional(out);
379 if (transferColors)
380 enableIfVertexColorOptional(out);
381 if (transferQuality)
382 enableIfVertexQualityOptional(out);
383
384 uint m0Triangles = F0.rows();
385
386 std::vector<Point4f> colorAccum;
387 std::vector<Point3<ScalarType>> normalAccum;
388 std::vector<double> qualityAccum;
389 std::vector<int> incidentCount(out.vertexCount(), 0);
390
391 if (transferColors)
392 colorAccum.resize(out.vertexCount(), Point4f());
393 if (transferNormals)
394 normalAccum.resize(out.vertexCount(), Point3<ScalarType>());
395 if (transferQuality)
396 qualityAccum.resize(out.vertexCount(), 0.0);
397
398 auto accumulateComponent = [&]<CompId COMP_ID, typename AccumType>(
399 std::vector<AccumType>& accum,
400 int outVIdx,
401 bool fromM0,
402 const Eigen::Vector3i& birthTri,
403 const auto& bcoords,
404 const auto& placeHolder) {
405 if constexpr (MeshType::template hasElementComponent<V, COMP_ID>()) {
406 auto c0 = placeHolder;
407 auto c1 = placeHolder;
408 auto c2 = placeHolder;
409
410 if (fromM0 && m0.template isElementComponentAvailable<V, COMP_ID>()) {
411 c0 = m0.vertex(birthTri[0]).template componentValue<COMP_ID>();
412 c1 = m0.vertex(birthTri[1]).template componentValue<COMP_ID>();
413 c2 = m0.vertex(birthTri[2]).template componentValue<COMP_ID>();
414 }
415 else if (
416 !fromM0 &&
417 m1.template isElementComponentAvailable<V, COMP_ID>()) {
418 c0 = m1.vertex(birthTri[0]).template componentValue<COMP_ID>();
419 c1 = m1.vertex(birthTri[1]).template componentValue<COMP_ID>();
420 c2 = m1.vertex(birthTri[2]).template componentValue<COMP_ID>();
421 }
422
423 if constexpr (COMP_ID == CompId::COLOR) {
424 Point4f v;
425 for (uint i = 0; i < 4; i++) {
426 v[i] = c0[i] * bcoords[0] + c1[i] * bcoords[1] +
427 c2[i] * bcoords[2];
428 }
429
430 accum[outVIdx] += v;
431 }
432 else {
433 auto v = c0 * bcoords[0] + c1 * bcoords[1] + c2 * bcoords[2];
434 accum[outVIdx] += v;
435 }
436 }
437 };
438
439 for (int i = 0; i < FRTri.rows(); ++i) {
440 int birthTriIdx = indicesTri[i];
441 bool fromM0 = birthTriIdx < (int) m0Triangles;
442
443 Eigen::Vector3i birthTri;
444 if (fromM0) {
445 birthTri = F0.row(birthTriIdx);
446 }
447 else {
448 birthTri = F1.row(birthTriIdx - m0Triangles);
449 }
450
451 Point3<ScalarType> p0 = fromM0 ? m0.vertex(birthTri[0]).position() :
452 m1.vertex(birthTri[0]).position();
453 Point3<ScalarType> p1 = fromM0 ? m0.vertex(birthTri[1]).position() :
454 m1.vertex(birthTri[1]).position();
455 Point3<ScalarType> p2 = fromM0 ? m0.vertex(birthTri[2]).position() :
456 m1.vertex(birthTri[2]).position();
457
458 for (int j = 0; j < 3; ++j) {
459 int outVIdx = FRTri(i, j);
460 auto outP = out.vertex(outVIdx).position();
461
462 auto bcoords = vcl::Triangle3<ScalarType>::barycentricCoordinates(
463 p0, p1, p2, outP);
464
465 if (transferColors) {
466 accumulateComponent.template operator()<CompId::COLOR>(
467 colorAccum,
468 outVIdx,
469 fromM0,
470 birthTri,
471 bcoords,
472 colorPlaceHolder);
473 }
474 if (transferNormals) {
475 accumulateComponent.template operator()<CompId::NORMAL>(
476 normalAccum,
477 outVIdx,
478 fromM0,
479 birthTri,
480 bcoords,
481 normalPlaceHolder);
482 }
483 if (transferQuality) {
484 accumulateComponent.template operator()<CompId::QUALITY>(
485 qualityAccum,
486 outVIdx,
487 fromM0,
488 birthTri,
489 bcoords,
490 qualityPlaceHolder);
491 }
492
493 incidentCount[outVIdx]++;
494 }
495 }
496
497 auto finalizeComponent = [&]<CompId COMP_ID, typename AccumType>(
498 const std::vector<AccumType>& accum,
499 int incCount,
500 int vIdx) {
501 if constexpr (MeshType::template hasElementComponent<V, COMP_ID>()) {
502 if constexpr (COMP_ID == CompId::COLOR) {
503 auto clampV = [](float v) {
504 return static_cast<uint8_t>(std::clamp(v, 0.0f, 255.0f));
505 };
506
507 float inv = 1.0f / incCount;
508 out.vertex(vIdx).template componentValue<COMP_ID>() = Color(
509 clampV(accum[vIdx][0] * inv),
510 clampV(accum[vIdx][1] * inv),
511 clampV(accum[vIdx][2] * inv),
512 clampV(accum[vIdx][3] * inv));
513 }
514 else if constexpr (COMP_ID == CompId::NORMAL) {
515 out.vertex(vIdx).template componentValue<COMP_ID>() =
516 accum[vIdx].normalized();
517 }
518 else {
519 out.vertex(vIdx).template componentValue<COMP_ID>() =
520 accum[vIdx] / incCount;
521 }
522 }
523 };
524
525 for (uint i = 0; i < out.vertexCount(); ++i) {
526 int incCount = incidentCount[i];
527 if (incCount > 0) {
528 if (transferColors) {
529 finalizeComponent.template operator()<CompId::COLOR>(
530 colorAccum, incCount, i);
531 }
532 if (transferNormals) {
533 finalizeComponent.template operator()<CompId::NORMAL>(
534 normalAccum, incCount, i);
535 }
536 if (transferQuality) {
537 finalizeComponent.template operator()<CompId::QUALITY>(
538 qualityAccum, incCount, i);
539 }
540 }
541 }
542}

570template<FaceMeshConcept MeshType>
571MeshType meshBoolean(const MeshType& m0, const MeshType& m1, MeshBoolean op)
572{
573 // TODO: allow to use non-compact meshes
574
575 using ScalarType = ScalarType<MeshType>;
576
577 using EigenMatrixX3m = Eigen::Matrix<ScalarType, Eigen::Dynamic, 3>;
578
579 // vcl to eigen meshes
580 TriPolyIndexBiMap m0BiMap;
581 TriPolyIndexBiMap m1BiMap;
582
583 auto V0 = vcl::vertexPositionsMatrix<EigenMatrixX3m>(m0);
584 auto F0 =
585 vcl::triangulatedFaceVertexIndicesMatrix<Eigen::MatrixX3i>(m0, m0BiMap);
586
587 auto V1 = vcl::vertexPositionsMatrix<EigenMatrixX3m>(m1);
588 auto F1 =
589 vcl::triangulatedFaceVertexIndicesMatrix<Eigen::MatrixX3i>(m1, m1BiMap);
590
591 EigenMatrixX3m VR;
592 Eigen::MatrixXi FR;
593 Eigen::VectorXi indices; // mapping indices for birth faces
594
595 bool result = ::igl::copyleft::cgal::mesh_boolean(
596 V0,
597 F0,
598 V1,
599 F1,
600 static_cast<::igl::MeshBooleanType>(op),
601 VR,
602 FR,
603 indices);
604
605 if (!result) {
606 throw std::runtime_error(
607 "Mesh inputs must induce a piecewise constant winding number "
608 "field. Make sure that both the input meshes are watertight "
609 "(closed).");
610 }
611
612 Eigen::MatrixXi FRTri = FR;
613 Eigen::VectorXi indicesTri = indices;
614
615 Eigen::VectorXi outIndices;
616 if constexpr (vcl::PolygonMeshConcept<MeshType>) {
617 FR = detail::rebuildPolygons<ScalarType>(
618 FR,
619 indices,
620 F0.rows(),
621 m0.faceContainerSize(),
622 m0BiMap,
623 m1BiMap,
624 outIndices);
625 indices = outIndices; // to map back to original polygon indices
626 }
627 else {
628 // if one of the meshes has deleted faces, we need to remap the indices
629 // to the original face indices
630 if (m0.faceContainerSize() != m0.faceCount() ||
631 m1.faceContainerSize() != m1.faceCount()) {
632 outIndices.resize(indices.size());
633 uint m0Size = m0.faceContainerSize();
634 for (int i = 0; i < indices.size(); ++i) {
635 int j = indices[i];
636 if (j < F0.rows()) {
637 outIndices[i] = m0BiMap.polygon(j);
638 }
639 else {
640 outIndices[i] = m0Size + m1BiMap.polygon(j - F0.rows());
641 }
642 }
643 indices = outIndices;
644 }
645 }
646
647 auto out = vcl::meshFromMatrices<MeshType>(VR, FR);
648
649 using FNormalType =
650 detail::NormalTypeHelper<FaceType<MeshType>, Point3<ScalarType>>::type;
651 using VNormalType = detail::
652 NormalTypeHelper<VertexType<MeshType>, Point3<ScalarType>>::type;
653
654 detail::transferFaceComponents(
655 m0, m1, indices, out, FNormalType(), Color::Gray, 0.0);
656
657 detail::transferVertexComponents(
658 m0,
659 m1,
660 F0,
661 F1,
662 FRTri,
663 indicesTri,
664 out,
665 VNormalType(),
666 Color::Gray,
667 0.0);
668
669 return out;
670}
