63template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
64void loadMeshes(
65 std::vector<MeshType>& meshes,
66 const std::string& filename,
67 std::vector<MeshInfo>& loadedInfo,
68 const LoadSettings& settings,
69 LogType& log = nullLogger)
70{
71 FileFormat ff = FileInfo::fileFormat(filename);
72
73 loadedInfo.clear();
74
75#ifdef VCLIB_WITH_TINYGLTF
76 if (ff == gltfFileFormat()) {
77 loadGltf(meshes, filename, loadedInfo, settings, log);
78 }
79 else {
80 throw UnknownFileFormatException(ff.extensions().front());
81 }
82#endif
83}
