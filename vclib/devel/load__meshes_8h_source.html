65template<MeshConcept MeshType, LoggerConcept LogType = NullLogger>
66void loadMeshes(
67 std::vector<MeshType>& meshes,
68 const std::string& filename,
69 std::vector<MeshInfo>& loadedInfo,
70 const LoadSettings& settings,
71 LogType& log = nullLogger)
72{
73 FileFormat ff = FileInfo::fileFormat(filename);
74
75 loadedInfo.clear();
76
77#ifdef VCLIB_WITH_TINYGLTF
78 if (ff == gltfFileFormat()) {
79 loadGltf(meshes, filename, loadedInfo, settings, log);
80 }
81 else {
82 throw UnknownFileFormatException(ff.extensions().front());
83 }
84#endif
85}
