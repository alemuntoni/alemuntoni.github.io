124 void init()

139 void show()
140 {
141 init();
142
143 mUpdateRequested = true;
144
145 // Loop runs as long as an update is requested by the application.
146 // In headless mode, this allows fast consecutive renders (no swap
147 // buffer blocking), but can become a busy loop if continuous redraw is
148 // active.
149 while (mUpdateRequested) {
150 mUpdateRequested = false;
151 DerivedRenderApp::WM::paint(derived());
152 }
153 }
