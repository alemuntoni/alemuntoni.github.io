36class DirectionalLightUniforms
37{
38 // just first 3 components are used
39 inline static std::array<float, 4> sDir = {0.0, 0.0, 1.0, 0.0};
40
41 // just first 3 components are used
42 inline static std::array<float, 4> sCol = {1.0, 1.0, 1.0, 1.0};
43
44 inline static StaticUniform sLightDirUniform {
45 "u_lightDirPack",
46 bgfx::UniformType::Vec4};
47 inline static StaticUniform sLightColorUniform {
48 "u_lightColorPack",
49 bgfx::UniformType::Vec4};
50
51public:
52 DirectionalLightUniforms() = delete;
53
58 template<typename S>
59 static void setLight(const vcl::DirectionalLight<S>& light)
60 {
61 sDir[0] = light.direction().x();
62 sDir[1] = light.direction().y();
63 sDir[2] = light.direction().z();
64
65 sCol[0] = light.color().redF();
66 sCol[1] = light.color().greenF();
67 sCol[2] = light.color().blueF();
68 // light color alpha is not used
69 }
70
71 static void bind()
72 {
73 sLightDirUniform.bind(sDir.data());
74 sLightColorUniform.bind(sCol.data());
75 }
76};
