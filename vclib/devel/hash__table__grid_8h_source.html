34template<typename GridType, typename ValueType, bool AllowDuplicates = true>
35class HashTableGrid : public AbstractGrid<GridType, ValueType>
36{
37 static_assert(
38 AllowDuplicates || std::equality_comparable<ValueType>,
39 "Not allowing duplicates in a Spatial Data Structures means that "
40 "ValueType must implement operator==.");
41
42 using AbsGrid = AbstractGrid<GridType, ValueType>;
43
44 friend AbsGrid;

46public:

49private:
50 using MapType = std::unordered_multimap<KeyType, ValueType>;
51 using MapValueType = MapType::value_type;

53 MapType mMap;
54
55public:

57
58 using Iterator = std::unordered_multimap<KeyType, ValueType>::iterator;

102 bool empty() const { return mMap.empty(); }
103
110 bool cellEmpty(const KeyType& k) const
111 {
112 return mMap.find(k) == mMap.end();
113 }
114
120 std::set<KeyType> nonEmptyCells() const
121 {
122 std::set<KeyType> keys;
123 for (const auto& p : mMap)
124 keys.insert(p.first);
125 return keys;
126 }
127
133 std::size_t countInCell(const KeyType& k) const { return mMap.count(k); }
134
135 auto valuesInCell(this auto&& self, const KeyType& k)
136 {
137 return self.mMap.equal_range(k);
138 }
139
140 void clear() { mMap.clear(); }
141
142 bool eraseAllInCell(const KeyType& k)
143 {
144 std::pair<Iterator, Iterator> range = mMap.equal_range(k);
145 if (range != mMap.end()) {
146 mMap.erase(range.first, range.second);
147 return true;
148 }
149 return false;
150 }
151
152 void eraseInSphere(const Sphere<typename GridType::ScalarType>& s)
153 {
154 auto toDelVec = this->valuesInSphere(s);
155 for (auto& it : toDelVec)
156 mMap.erase(it);
157 }
158
159 auto begin(this auto&& self) { return self.mMap.begin(); }
160
161 auto end(this auto&& self) { return self.mMap.end(); }
162
163private:
164 bool insertInCell(const KeyType& k, const ValueType& v)
165 {
166 if constexpr (AllowDuplicates) {
167 mMap.emplace(k, v);
168 return true;
169 }
170 else {
171 auto range = mMap.equal_range(k);
172 bool found = false;
173 for (Iterator ci = range.first; ci != range.second && !found;
174 ++ci) {
175 if (ci->second == v) {
176 found = true;
177 }
178 }
179 if (!found)
180 mMap.emplace(k, v);
181 return !found;
182 }
183 }
184
185 bool eraseInCell(const KeyType& k, const ValueType& v)
186 {
187 bool found = false;
188
189 std::pair<Iterator, Iterator> range = mMap.equal_range(k);
190 for (Iterator ci = range.first; ci != range.second; ++ci) {
191 if (ci->second == v) {
192 found = true;
193 mMap.erase(ci);
194 if constexpr (!AllowDuplicates) {
195 return true;
196 }
197 }
198 }
199 return found;
200 }
201};
