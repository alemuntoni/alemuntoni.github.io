158 std::pair<Iterator, Iterator> range = mMap.equal_range(k);

181private:
182 bool insertInCell(const KeyType& k, const ValueType& v)
183 {
184 if constexpr (AllowDuplicates) {
185 mMap.emplace(k, v);
186 return true;
187 }
188 else {
189 auto range = mMap.equal_range(k);
190 bool found = false;
191 for (Iterator ci = range.first; ci != range.second && !found;
192 ++ci) {
193 if (ci->second == v) {
194 found = true;
195 }
196 }
197 if (!found)
198 mMap.emplace(k, v);
199 return !found;
200 }
201 }
202
203 bool eraseInCell(const KeyType& k, const ValueType& v)
204 {
205 bool found = false;
206
207 std::pair<Iterator, Iterator> range = mMap.equal_range(k);
208 for (Iterator ci = range.first; ci != range.second; ++ci) {
209 if (ci->second == v) {
210 found = true;
211 mMap.erase(ci);
212 if constexpr (!AllowDuplicates) {
213 return true;
214 }
215 }
216 }
217 return found;
218 }
