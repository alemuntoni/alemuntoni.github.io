36template<MeshConcept MeshType, uint ELEM_ID>
37uint deleteElementsIf(MeshType& m, Range auto&& elemFilterRng)
38{
39 uint count = 0;
40
41 if constexpr (MeshType::template hasContainerOf<ELEM_ID>()) {
42 std::vector<uint> refIndices(
43 m.template containerSize<ELEM_ID>(), UINT_NULL);
44
45 for (const auto& [e, filter] :
46 std::views::zip(m.template elements<ELEM_ID>(), elemFilterRng)) {
47 if (filter) {
48 m.template deleteElement<ELEM_ID>(m.index(e));
49 count++;
50 }
51 else {
52 refIndices[m.index(e)] = m.index(e);
53 }
54 }
55
56 if (count > 0) {
57 m.template updateReferences<ELEM_ID>(refIndices);
58 }
59 }
60
61 return count;
62}

64template<MeshConcept MeshType, uint ELEM_ID>
65uint deleteElementsIf(
66 MeshType& m,
67 const std::function<bool(
68 const typename MeshType::template ElementType<ELEM_ID>&)>& elemFilter)
69{
70 auto view =
71 m.template elements<ELEM_ID>() | std::views::transform(elemFilter);
72
73 return deleteElementsIf<MeshType, ELEM_ID>(m, view);
74}
