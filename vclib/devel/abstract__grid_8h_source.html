19/*
20 * Developer Documentation
21 * A class that derives from an AbstractGrid must store, in some way, an
22 * association between a GridCell position (a Point of unsigned integers) and
23 * elements of type ValueType. Each cell may contain more than one element
24 * value, and each element value may be stored in more than one cell if it
25 * intersects with more than one cell.
26 *
27 * This class uses C++23 explicit object parameters ("deducing this")
28 * for static polymorphism, avoiding virtual member functions without
29 * requiring the CRTP pattern.
30 *
31 * What you need to do is to declare your Grid Data Structure as follows:
32 *
33 * template<typename GridType, typename ValueType>
34 * class MyDSGrid :
35 * public AbstractGrid<GridType, ValueType>
36 * { ... };
37 *
38 * Make sure that your class:
39 * - declares the AbstractGrid class as friend (if one of the following member
40 * functions is
41 * protected/private);
42 * - implements iterators Iterator and ConstIterator that iterate over pairs of
43 * <KeyType, ValueType> (see below for more details)
44 * - implements the following member functions:
45 * - valuesInCell(this auto&& self, const KeyType& k) -> returns a pair
46 * of begin-end iterators that will allow to cycle over the elements
47 * contained in a cell;
48 * - bool insertInCell(const KeyType&, const ValueType&) -> inserts the
49 * element in a Grid Cell. The returned type tells if the element has been
50 * inserted (in some implementations, insertion could not happen, e.g. when
51 * duplicates are not allowed).
52 * - bool eraseInCell(const KeyType&, const ValueType&) -> erases all the
53 * element equal to the given element from the given Grid Cell. The returned
54 * type tells if the element has been erased from the given cell.
55 * - end(this auto&& self) -> member function that returns the end iterator
56 *
57 * Iterators of your class must:
58 * - Iterate over pairs of <KeyType, ValueType&>:
59 * - operator*() returns a pair having first member that is the key, and
60 * second member that is a reference to the value
61 * - operator->() return a pointer to the same pair
62 * - operator+() and operator+(int), that moves the iterator in the same cell if
63 * there are more values in it, or moves in the next cell-element
64 *
65 * You can reimplement in your derived class all the member functions
66 * implemented in this class that may take advantage of a speedup depending on
67 * the internal storage. You can also hide all the member functions implemented
68 * in this class that do not apply to your derived class.
69 */

176 bool insert(this auto&& self, const ValueType& v)
177 {
178 if constexpr (ElementConcept<VT>) {
179 if (!v.isValid()) {
180 return false;
181 }
182 }
183
184 const VT* vv = addressOfObj(v);
185
186 // if vv is a valid pointer (ValueType, or ValueType* if ValueType is
187 // not a pointer)
188 if (vv) {
189 // first and last cell where insert (could be the same)
190 KeyType bmin, bmax;
191
192 // if ValueType is Point, Point*, Vertex, Vertex*
193 if constexpr (PointConcept<VT> || VertexConcept<VT>) {
194 typename GridType::PointType p;
195 if constexpr (PointConcept<VT>)
196 p = *vv;
197 else
198 p = vv->position();
199 bmin = bmax = self.cell(p);
200 }
201 else { // else, call the boundingBox function
202 // bounding box of value
203 typename GridType::BBoxType bb = boundingBox(*vv);
204
205 bmin = self.cell(bb.min()); // first cell where insert
206 bmax = self.cell(bb.max()); // last cell where insert
207 }
208
209 bool ins = false;
210
211 // custom intersection function between cell and value
212 if (self.mIntersectsFun) {
213 for (const auto& cell : self.cells(bmin, bmax)) {
214 if (self.mIntersectsFun(
215 self.cellBox(cell), dereferencePtr(v))) {
216 ins |= self.insertInCell(cell, v);
217 }
218 }
219 }
220 else {
221 for (const auto& cell : self.cells(bmin, bmax)) {
222 ins |= self.insertInCell(cell, v);
223 }
224 }
225 return ins;
226 }
227 return false;
228 }

339 auto valuesInSphere(
340 this auto&& self,
341 const Sphere<typename GridType::ScalarType>& s)
342 {
343 using Iter =
344 std::decay_t<decltype(self.valuesInCell(KeyType {}).first)>;
345 using IterSet = std::set<Iter, IterComparator<Iter>>;
346
347 // will use this set only if the value type is not a point -- that is
348 // when the value can occupy more than one single cell. We therefore
349 // push the results in a set in order to avoid possible duplicates
350 IterSet valuesSet;
351
352 std::vector<Iter> resVec;
353
354 // interval of cells containing the sphere
355 KeyType first = self.cell(s.center() - s.radius());
356 KeyType last = self.cell(s.center() + s.radius());
357
358 // for each cell in the interval
359 for (const KeyType& c : self.cells(first, last)) {
360 // p is a pair of iterators
361 const auto& p = self.valuesInCell(c);
362 // for each value contained in the cell
363 for (auto it = p.first; it != p.second; ++it) {
364 if (self.valueIsInSpehere(it, s)) {
365 if constexpr (!PointConcept<VT> && !VertexConcept<VT>) {
366 valuesSet.insert(it);
367 }
368 else {
369 resVec.push_back(it);
370 }
371 }
372 }
373 }
374
375 // if the valuetype is a point (or vertex), we already pushed in resVec
376 // - faster than using set
377 if constexpr (!PointConcept<VT> && !VertexConcept<VT>) {
378 resVec = std::vector<Iter>(valuesSet.begin(), valuesSet.end());
379 }
380 return resVec;
381 }

409 // closest queries
410 template<typename QueryValueType>
411 auto closestValue(
412 this auto&& self,
413 const QueryValueType& qv,
414 QueryBoundedDistFunction<QueryValueType> distFunction,
415 typename GridType::ScalarType& dist)
416 {
417 using ScalarType = GridType::ScalarType;
418 using PointType = GridType::PointType;
419 using ResType = decltype(self.end());
420
421 using QVT = RemoveCVRefAndPointer<QueryValueType>;
422 const QVT* qvv = addressOfObj(qv);
423 ResType result = self.end();
424
425 if (qvv) {
426 typename GridType::ScalarType maxDist = dist;
427
428 const ScalarType cellDiag = self.cellDiagonal();
429
430 // bbox of query value
431 typename GridType::BBoxType bb = boundingBox(*qvv);
432
433 ScalarType centerDist = cellDiag;
434 PointType center = bb.center();
435
436 // we first look just on the cells where the query value lies
437
438 // here, we will store also the looking interval where we need to
439 // look
440 Boxui currentIntervalBox;
441 Boxui lastIntervalBox = currentIntervalBox;
442 // first cell where look for closest
443 currentIntervalBox.add(self.cell(bb.min()));
444 // last cell where look for closest
445 currentIntervalBox.add(self.cell(bb.max()));
446
447 // looking just on cells where query lies
448 ScalarType tmp = cellDiag;
449 result =
450 self.closestInCells(qv, tmp, currentIntervalBox, distFunction);
451
452 // we have found (maybe) the closest value contained in the cell(s)
453 // where the query value lies (if the cells were empty, we did not
454 // found nothing).
455
456 // if we found a value, we update the dist, which becames the
457 // max dist value. We will use it for the final search of the
458 // closest value
459 if (result != self.end()) {
460 dist = tmp;
461 centerDist = dist;
462 }
463 else {
464 // we did not find any value in the cells where the query value
465 // lies. We need to look in the neighborhood cells and increase
466 // the looking distance until we find a value or we reach the
467 // max distance
468
469 bool end = false;
470
471 do {
472 lastIntervalBox = currentIntervalBox;
473 currentIntervalBox.add(self.cell(center - centerDist));
474 currentIntervalBox.add(self.cell(center + centerDist));
475
476 result = self.closestInCells(
477 qv,
478 dist,
479 currentIntervalBox,
480 distFunction,
481 lastIntervalBox);
482
483 end = result != self.end();
484 end |= (centerDist > maxDist);
485 end |=
486 (center - centerDist < self.min() &&
487 center + centerDist > self.max());
488
489 // update the centerDist for the next loop (after computing
490 // the end loop condition!!)
491 centerDist += cellDiag;
492 } while (!end);
493 }
494
495 if (result != self.end()) {
496 // last check: look in all the cells inside the sphere of radius
497 // dist, in case there is a closest value
498 currentIntervalBox.add(self.cell(center - dist));
499 currentIntervalBox.add(self.cell(center + dist));
500 auto r = self.closestInCells(
501 qv,
502 dist,
503 currentIntervalBox,
504 distFunction,
505 lastIntervalBox);
506 if (r != self.end()) {
507 result = r;
508 }
509 }
510 }
511
512 return result;
513 }

568 template<typename QueryValueType>
569 auto kClosestValues(
570 this auto&& self,
571 const QueryValueType& qv,
572 uint n,
573 QueryDistFunction<QueryValueType> distFunction)
574 {
575 using ResIt = decltype(self.end());
576 using ResType = std::vector<ResIt>;
577 // KClosest Types
578 using KClosestPairType =
579 std::pair<typename GridType::ScalarType, ResIt>;
580 using KClosestSet = std::
581 set<KClosestPairType, DistIterPairComparator<KClosestPairType>>;
582
583 Boxui ignore; // will contain the interval of cells already visited
584
585 KClosestSet set =
586 self.valuesInCellNeighborhood(qv, n, distFunction, ignore);
587
588 auto it = set.size() >= n ? std::next(set.begin(), n - 1) : set.end();
589 // if we didn't found n values, it means that there aren't n values in
590 // the grid - nothing to do
591 if (it != set.end()) {
592 using QVT = RemoveCVRefAndPointer<QueryValueType>;
593 const QVT* qvv = addressOfObj(qv);
594
595 // bbox of query value
596 typename GridType::BBoxType bb = boundingBox(*qvv);
597 // we need to be sure that there are no values that are closest
598 // w.r.t. the n-th that we have already found by looking in the cell
599 // neighborhood. we extend the bb with the distance of the n-th
600 // closest found value
601 bb.min() -= it->first;
602 bb.max() += it->first;
603
604 // and we look in all of these cells
605 Boxui currentIntervalBox;
606 // first cell where look for closest
607 currentIntervalBox.add(self.cell(bb.min()));
608 // last cell where look for closest
609 currentIntervalBox.add(self.cell(bb.max()));
610
611 // for all the cells in the current interval box
612 for (const KeyType& c : self.cells(
613 currentIntervalBox.min(), currentIntervalBox.max())) {
614 if (!ignore.isInside(c)) {
615 // get the values of the cell c
616 const auto& p = self.valuesInCell(c);
617 // for each value contained in the cell c
618 for (auto it = p.first; it != p.second; ++it) {
619 auto tmp = distFunction(qv, dereferencePtr(it->second));
620 set.insert(std::make_pair(tmp, it));
621 }
622 }
623 }
624 }
625
626 ResType vec;
627 // if there are more than n values in the set, we will return n values,
628 // otherwise set.size()
629 uint retNValues = std::min((uint) set.size(), n);
630 vec.reserve(retNValues);
631 it = set.begin();
632 for (uint i = 0; i < retNValues; i++) {
633 vec.push_back(it->second);
634 it++;
635 }
636 return vec;
637 }

822 template<typename Iterator>
823 bool valueIsInSpehere(
824 const Iterator& it,
825 const Sphere<typename GridType::ScalarType>& s) const
826 {
827 using PointType = GridType::PointType;
828
829 if constexpr (ElementConcept<VT>) {
830 if (!it->second.isValid()) {
831 return false;
832 }
833 }
834
835 const VT* vv = addressOfObj(it->second);
836
837 bool test = false;
838 if constexpr (PointConcept<VT>) {
839 test = vv && s.isInside(*vv);
840 }
841 else if constexpr (VertexConcept<VT>) {
842 test = vv && s.isInside(vv->position());
843 }
844 else { // check if the bbox of the value intersects the sphere
845 test = vv && s.intersects(boundingBox(*vv));
846 }
847 return test;
848 }

855 template<typename QueryValueType>
856 auto closestInCells(
857 this auto&& self,
858 const QueryValueType& qv,
859 typename GridType::ScalarType& dist,
860 const Boxui& interval,
861 QueryBoundedDistFunction<QueryValueType> distFunction,
862 const Boxui& ignore = Boxui())
863 {
864 auto res = self.end();
865
866 // for each cell in the interval
867 for (const KeyType& c : self.cells(interval.min(), interval.max())) {
868 if (!ignore.isInsideStrict(c)) {
869 // p is a pair of iterators
870 const auto& p = self.valuesInCell(c);
871 // for each value contained in the cell
872 for (auto it = p.first; it != p.second; ++it) {
873 auto tmp =
874 distFunction(qv, dereferencePtr(it->second), dist);
875 if (tmp < dist) {
876 dist = tmp;
877 res = it;
878 }
879 }
880 }
881 }
882 return res;
883 }

885 template<typename QueryValueType>
886 auto valuesInCellNeighborhood(
887 this auto&& self,
888 const QueryValueType& qv,
889 uint n,
890 QueryDistFunction<QueryValueType> distFunction,
891 Boxui& ignore)
892 {
893 // types used for K closest neighbors queries
894 using ResIt = decltype(self.end());
895 using KClosestPairType =
896 std::pair<typename GridType::ScalarType, ResIt>;
897 using KClosestSet = std::
898 set<KClosestPairType, DistIterPairComparator<KClosestPairType>>;
899
900 KClosestSet res;
901
902 using QVT = RemoveCVRefAndPointer<QueryValueType>;
903 const QVT* qvv = addressOfObj(qv);
904
905 if (qvv) {
906 Boxui currentIntervalBox;
907 // bbox of query value
908 typename GridType::BBoxType bb = boundingBox(*qvv);
909 // first cell where look for closest
910 currentIntervalBox.add(self.cell(bb.min()));
911 // last cell where look for closest
912 currentIntervalBox.add(self.cell(bb.max()));
913
914 ignore.setNull();
915 while (res.size() < n && currentIntervalBox != ignore) {
916 // for each cell in the interval
917 for (const KeyType& c : self.cells(
918 currentIntervalBox.min(), currentIntervalBox.max())) {
919 if (!ignore.isInside(c)) {
920 const auto& p = self.valuesInCell(c);
921
922 // for each value contained in the cell
923 for (auto it = p.first; it != p.second; ++it) {
924 auto tmp =
925 distFunction(qv, dereferencePtr(it->second));
926 res.emplace(tmp, it);
927 }
928 }
929 }
930 ignore = currentIntervalBox;
931 for (uint i = 0; i < currentIntervalBox.min().DIM; ++i) {
932 if (currentIntervalBox.min()(i) != 0)
933 currentIntervalBox.min()(i)--;
934 if (currentIntervalBox.max()(i) != self.cellCount(i) - 1)
935 currentIntervalBox.max()(i)++;
936 }
937 }
938 }
939
940 return res;
941 }
